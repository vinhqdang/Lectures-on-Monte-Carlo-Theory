\section{Notation and Intuition}
%%=============================================================================

\begin{frame}{Notation You'll Need}
\begin{block}{Symbols used throughout this chapter}
\begin{itemize}
  \item $\Hsp$, $\Ksp$: real Hilbert spaces (from Ch.~2); $\R = ]-\infty,+\infty[$, and we work with functions valued in $\;]-\infty,+\infty] = \R \cup \{+\infty\}$ or $[-\infty,+\infty]$.
  \item $f:\Hsp \to [-\infty,+\infty]$: an \emph{extended-real-valued} function; $+\infty$ means ``infeasible'' / ``infinite cost.''
  \item $\dom f = \{x \in \Hsp : f(x) < +\infty\}$: the \emph{domain} of $f$ (where $f$ is not automatically excluded).
  \item $f$ is \emph{proper} if $f(x) > -\infty$ for every $x$ and $\dom f \neq \varnothing$ (i.e., $f$ never says ``$-\infty$'' and is somewhere finite).
  \item $\epi f = \{(x,\eta) \in \Hsp\times\R : \eta \geq f(x)\}$: the \emph{epigraph}, the set of points lying on or above the graph of $f$.
  \item $\lev_{\leq \xi} f = \{x \in \Hsp : f(x) \leq \xi\}$: the \emph{sublevel set} at height $\xi \in \R$; similarly $\lev_{<\xi} f$.
  \item $\iota_C$: the \emph{indicator function} of a set $C$ ($0$ on $C$, $+\infty$ off $C$) --- turns ``convex on $C$'' into ``convex plus indicator.''
  \item $\alpha \in\, ]0,1[$: a mixing weight; $\alpha x + (1-\alpha) y$ is a point on the open segment between $x$ and $y$.
  \item $B(x_0;\rho)$: the closed (or open, per context) ball of radius $\rho$ centered at $x_0$.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Intuition First: What Does ``Convex Function'' Mean?}
\begin{alertblock}{Plain-language definition}
A function $f$ is \textbf{convex} if: \emph{the straight line segment (chord) joining any two points on the graph of $f$ lies on or above the graph.}
\end{alertblock}

\begin{block}{Concrete example: $f(x) = x^2$}
Take $x=-1.5$ and $y=1$. The chord joining $(-1.5,\,2.25)$ and $(1,\,1)$ should stay above the parabola between them. Check the midpoint-ish combination $\alpha = 0.4$:
\[
  \alpha x + (1-\alpha)y = 0.4(-1.5) + 0.6(1) = -0.6+0.6 = 0.0
\]
\[
  f(0.0) = 0.00 \qquad\text{vs.}\qquad \alpha f(x)+(1-\alpha)f(y) = 0.4(2.25)+0.6(1) = 1.5
\]
Since $0.00 \leq 1.5$, the inequality holds at this point --- and in fact $x^2$ satisfies it everywhere, so $x^2$ is convex. This is exactly Figure~1 below.
\end{block}
\begin{center}
\includegraphics[width=0.42\linewidth]{fig_chord_convex.pdf}
\end{center}
\end{frame}

\begin{frame}{Intuition: What Goes Wrong for Non-Convex Functions}
\begin{block}{A function that fails the chord test}
Let $g(x) = x^4 - 3x^2$ (a ``double well'' with two dips). Take $x=-1.6$, $y=1.6$, $\alpha=0.5$:
\[
  \alpha x + (1-\alpha) y = 0.5(-1.6)+0.5(1.6) = 0, \qquad g(0) = 0
\]
\[
  \alpha g(x) + (1-\alpha) g(y) = g(-1.6) = 1.6^4 - 3(1.6)^2 = 6.5536 - 7.68 = -1.1264
\]
Here $g(0) = 0 > -1.1264 = \alpha g(x)+(1-\alpha)g(y)$: the graph value at the midpoint is \emph{above} the chord value, i.e., the chord dips \emph{below} the graph. This single violation is enough to make $g$ non-convex.
\end{block}
\begin{center}
\includegraphics[width=0.42\linewidth]{fig_chord_nonconvex.pdf}
\end{center}
\end{frame}

\section{8.1 Basic Properties and Examples}
%%=============================================================================

\begin{frame}{The Formal Definition of Convexity}
\begin{block}{Convex function (formal statement, cf.\ Corollary 8.12(iii))}
Let $f: \Hsp \to\, ]-\infty,+\infty]$. Then $f$ is \textbf{convex} if
\[
  (\forall x \in \Hsp)(\forall y \in \Hsp)(\forall \alpha \in\, ]0,1[\,)\qquad
  f\bigl(\alpha x + (1-\alpha) y\bigr) \leq \alpha f(x) + (1-\alpha) f(y).
\]
Here $\alpha$ is a weight in $]0,1[$, $\alpha x+(1-\alpha)y$ is the point that is $\alpha$-of-the-way from $y$ to $x$ on the segment $[x,y]$, and the inequality says: \emph{the value of $f$ at a mixture of two points is at most the same mixture of the two values.}
\end{block}

\begin{block}{Convex on a set $C \subset \dom f$}
\[
(\forall x \in C)(\forall y \in C)(\forall \alpha \in\, ]0,1[\,)\quad
f\bigl(\alpha x+(1-\alpha)y\bigr) \leq \alpha f(x)+(1-\alpha)f(y). \tag{8.4}
\]
And \emph{strictly convex on $C$} strengthens $\leq$ to $<$ whenever $x \neq y$:
\[
x\neq y \;\Rightarrow\; f\bigl(\alpha x+(1-\alpha)y\bigr) < \alpha f(x)+(1-\alpha) f(y). \tag{8.5}
\]
(Definition 8.7 is the case $C = \dom f$.) \textbf{Remark 8.8:} $f+\iota_C$ is convex $\iff$ $C$ is convex and $f$ is convex on $C$.
\end{block}
\end{frame}

\begin{frame}{Proposition 8.4: Convexity via the Epigraph}
\begin{block}{Proposition 8.4}
Let $f:\Hsp \to [-\infty,+\infty]$. Then $f$ is convex \emph{if and only if} $\epi f$ is a convex subset of $\Hsp\times\R$.
\end{block}
\begin{center}
\includegraphics[width=0.5\linewidth]{fig_epigraph.pdf}
\end{center}
{\footnotesize A function $f:\Hsp\to[-\infty,+\infty]$ is convex exactly when the shaded region $\epi f$ above its graph is a convex set.}
\end{frame}

\begin{frame}{Step-by-Step Proof of Proposition 8.4}
\textbf{($\Rightarrow$) Assume $f$ convex.} Let $(x,\xi), (y,\eta) \in \epi f$ (so $\xi \geq f(x)$, $\eta\geq f(y)$) and $\alpha\in\,]0,1[$. Convexity gives
\[
  f(\alpha x+(1-\alpha)y) \leq \alpha f(x)+(1-\alpha)f(y) \leq \alpha\xi+(1-\alpha)\eta,
\]
so $\alpha(x,\xi)+(1-\alpha)(y,\eta) \in \epi f$: $\epi f$ is convex.

\medskip
\textbf{($\Leftarrow$) Assume $\epi f$ convex.} Fix $x,y\in\dom f$. Since $(x,f(x))\in\epi f$ and $(y,f(y))\in\epi f$ trivially (taking $\xi=f(x)$, $\eta=f(y)$), convexity of $\epi f$ gives $\alpha(x,f(x))+(1-\alpha)(y,f(y)) \in \epi f$, i.e.\ exactly $f(\alpha x+(1-\alpha)y)\leq \alpha f(x)+(1-\alpha) f(y)$.

\begin{exampleblock}{Numeric check for $f(x)=x^2$: is $(x,\xi)=(-1.5,\,3)$ in $\epi f$?}
$f(-1.5) = 2.25 \leq 3$, so yes, $(-1.5,3)\in\epi f$. Take $(y,\eta)=(1,\,1.5)\in\epi f$ (since $f(1)=1\leq1.5$) and $\alpha=0.4$: the mixture is $(-0.6+0.6,\ 0.4(3)+0.6(1.5)) = (0,\,2.1)$, and indeed $f(0)=0\leq 2.1$. \checkmark
\end{exampleblock}
\end{frame}

\begin{frame}{Immediate Consequences: Sublevel Sets and Direct Sums}
\begin{block}{Corollary 8.5}
Let $f:\Hsp\to[-\infty,+\infty]$ be convex. Then for every $\xi\in\R$, $\lev_{\leq\xi} f$ is convex.
\end{block}
\begin{block}{Proposition 8.6 (direct sums)}
Let $(\Hsp_i)_{i\in I}$ be real Hilbert spaces, $\Hsp = \bigoplus_{i\in I}\Hsp_i$, and $f_i:\Hsp_i\to\,]-\infty,+\infty]$ convex for each $i$. Suppose $I$ is finite or $\inf_{i\in I} f_i \geq 0$. Then $\bigoplus_{i\in I} f_i$ is convex.
\end{block}

\begin{exampleblock}{Numeric check: $\lev_{\leq 4} (x^2)$}
$\lev_{\leq 4}(x^2) = \{x : x^2 \leq 4\} = [-2,2]$, a convex (interval) set, as guaranteed by Corollary 8.5. Contrast: for the non-convex $g(x)=x^4-3x^2$, $\lev_{\leq -1}(g) \approx [-1.53,-0.80]\cup[0.80,1.53]$ --- \emph{not} an interval, consistent with $g$ being non-convex.
\end{exampleblock}
\end{frame}

\begin{frame}{Strict Convexity: Examples 8.9--8.10}
\begin{block}{Example 8.9: $\norm{\cdot}$ is convex but not strictly convex}
Convexity of the norm is the triangle inequality in disguise. But if $\Hsp\neq\{0\}$, take any $x\neq 0$ and $\alpha\in\,]0,1[$: since $\|\alpha x+(1-\alpha)0\| = \alpha\|x\| = \alpha\|x\|+(1-\alpha)\|0\|$, equality (not strict inequality) holds, so $\norm{\cdot}$ is \emph{not} strictly convex.
\end{block}
\begin{block}{Example 8.10: $\norm{\cdot}^2$ is strictly convex}
This follows from Corollary 2.15 (a strict-convexity identity for the squared norm coming from the parallelogram law).
\end{block}

\begin{exampleblock}{Numeric check in $\R$: $x=2,\,y=-1,\,\alpha=0.5$}
$\norm{\cdot}$: $|\alpha x+(1-\alpha)y| = |0.5| = 0.5$ vs.\ $\alpha|x|+(1-\alpha)|y| = 1+0.5=1.5$: strict, \emph{but} taking $x=2,y=0$ gives equality $1=1$, confirming non-strictness in general.\\
$\norm{\cdot}^2$: $(0.5)^2=0.25$ vs.\ $0.5(4)+0.5(1)=2.5$: strict inequality $0.25<2.5$, and one can check equality \emph{never} occurs unless $x=y$.
\end{exampleblock}
\end{frame}

\begin{frame}{Jensen's Inequality: Proposition 8.11}
\begin{block}{Proposition 8.11 (finite-sum form of convexity)}
$f:\Hsp\to[-\infty,+\infty]$ is convex $\iff$ for every finite family $(\alpha_i)_{i\in I}$ in $]0,1[$ with $\sum_{i\in I}\alpha_i = 1$ and every finite family $(x_i)_{i\in I}$ in $\dom f$,
\[
  f\Bigl(\sum_{i\in I}\alpha_i x_i\Bigr) \leq \sum_{i\in I}\alpha_i f(x_i). \tag{8.6}
\]
\end{block}
Here $(\alpha_i)$ are weights summing to 1 (a probability vector) and $\sum_i \alpha_i x_i$ is a \emph{convex combination} of the points $x_i$: this is Jensen's inequality for finitely many points, generalizing the two-point ($|I|=2$) definition.

\begin{block}{Corollary 8.12: three equivalent faces of convexity}
(i) $f$ convex; (ii) the Jensen inequality (8.6) for all finite convex combinations; (iii) the two-point inequality $f(\alpha x+(1-\alpha)y)\leq\alpha f(x)+(1-\alpha)f(y)$ for all $x,y\in\Hsp$, $\alpha\in\,]0,1[$.
\end{block}
\end{frame}

\begin{frame}{Numeric Verification of Jensen's Inequality}
\begin{exampleblock}{$f(x)=x^2$, three points, weights $(0.2,0.3,0.5)$}
Take $x_1=-2,\,x_2=1,\,x_3=3$ and $\alpha_1=0.2,\,\alpha_2=0.3,\,\alpha_3=0.5$ (they sum to $1$). Then
\[
  \sum_i \alpha_i x_i = 0.2(-2)+0.3(1)+0.5(3) = -0.4+0.3+1.5 = 1.4,
\]
\[
  f\Bigl(\sum_i \alpha_i x_i\Bigr) = 1.4^2 = 1.96,
\]
\[
  \sum_i \alpha_i f(x_i) = 0.2(4)+0.3(1)+0.5(9) = 0.8+0.3+4.5 = 5.6.
\]
Since $1.96 \leq 5.6$, Jensen's inequality holds. \checkmark
\end{exampleblock}

\begin{block}{Proposition 8.13: when does equality hold in the strictly convex case?}
If $f$ is strictly convex, equality $f(\sum_i\alpha_i x_i) = \sum_i \alpha_i f(x_i)$ holds \emph{if and only if} $\{x_i\}_{i\in I}$ is a singleton (all the $x_i$ coincide). The proof is by induction on the number of points $m=|I|$, peeling off one point at a time and using the two-point strict inequality.
\end{block}
\end{frame}

\begin{frame}[fragile]{Python (1/2): Verifying Jensen's Inequality Numerically}
\begin{lstlisting}[caption={Check the convexity inequality for several functions and random convex combinations}]
import numpy as np

def check_jensen(f, xs, alphas, tol=1e-9):
    xs, alphas = np.array(xs, float), np.array(alphas, float)
    assert abs(alphas.sum() - 1.0) < 1e-9
    lhs = f(np.dot(alphas, xs))
    rhs = np.dot(alphas, [f(x) for x in xs])
    return lhs, rhs, lhs <= rhs + tol

f_convex   = lambda x: x**2                # Example 8.10 (strictly convex)
f_nonconv  = lambda x: x**4 - 3*x**2       # double well, NOT convex

xs, alphas = [-2.0, 1.0, 3.0], [0.2, 0.3, 0.5]
lhs, rhs, ok = check_jensen(f_convex, xs, alphas)
print(f"x^2      : f(sum a_i x_i)={lhs:.3f}  sum a_i f(x_i)={rhs:.3f}  ok={ok}")

xs2, alphas2 = [-1.6, 1.6], [0.5, 0.5]
lhs2, rhs2, ok2 = check_jensen(f_nonconv, xs2, alphas2)
print(f"x^4-3x^2 : f(sum a_i x_i)={lhs2:.3f}  sum a_i f(x_i)={rhs2:.3f}  ok={ok2}")
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Python (2/2): Output and Random Stress Test}
\begin{lstlisting}[caption={Expected output, then a Monte-Carlo style stress test over many random combinations}]
# Expected output:
# x^2      : f(sum a_i x_i)=1.960  sum a_i f(x_i)=5.600  ok=True
# x^4-3x^2 : f(sum a_i x_i)=0.000  sum a_i f(x_i)=-1.126  ok=False

rng = np.random.default_rng(0)
violations = 0
for _ in range(200000):
    m = rng.integers(2, 6)
    xs = rng.uniform(-5, 5, size=m)
    a = rng.dirichlet(np.ones(m))       # random weights summing to 1
    lhs, rhs, ok = check_jensen(f_convex, xs, a)
    violations += (not ok)
print("violations for x^2 over 200000 trials:", violations)   # -> 0
\end{lstlisting}
Running this confirms $0$ violations for $f(x)=x^2$ across $200{,}000$ random convex combinations, while $f(x)=x^4-3x^2$ fails immediately --- exactly what Corollary 8.12 predicts.
\end{frame}

\begin{frame}{A Derivative Test for Convexity: Proposition 8.14}
\begin{block}{Proposition 8.14}
Let $\phi:\R\to\,]-\infty,+\infty]$ be proper and differentiable on a nonempty open interval $I\subset \dom\phi$.
\begin{itemize}
  \item[(i)] If $\phi'$ is increasing on $I$, then $\phi$ is convex on $I$.
  \item[(ii)] If $\phi'$ is strictly increasing on $I$, then $\phi$ is strictly convex on $I$.
\end{itemize}
\end{block}
Here $\phi'$ increasing means the slope of the tangent line never decreases as you move right --- exactly the picture of a function curving upward.

\begin{block}{Proof sketch}
Fix $x,y\in I$, $\alpha\in\,]0,1[$. Define $\psi(z) = \alpha\phi(x)+(1-\alpha)\phi(z)-\phi(\alpha x+(1-\alpha)z)$. Then $\psi'(z) = (1-\alpha)\bigl(\phi'(z)-\phi'(\alpha x+(1-\alpha)z)\bigr)$ and $\psi'(x)=0$. Since $\phi'$ is increasing, $\psi'(z)\leq 0$ for $z<x$ and $\psi'(z)\geq0$ for $z>x$: $\psi$ attains its \emph{minimum} on $I$ at $z=x$. Hence $\psi(y)\geq\psi(x)=0$, i.e.\ $\phi(\alpha x+(1-\alpha)y)\leq \alpha\phi(x)+(1-\alpha)\phi(y)$. The strict case is analogous with strict inequalities.
\end{block}
\end{frame}

\begin{frame}{Numeric Check of Proposition 8.14 and Example 8.15}
\begin{exampleblock}{$\phi(x)=x^2$: $\phi'(x)=2x$ is strictly increasing on $\R$}
At $x=-1$: $\phi'(-1)=-2$. At $x=3$: $\phi'(3)=6$. Since $-2 < 6$ and $-1<3$, $\phi'$ is increasing here (and everywhere), confirming $x^2$ is strictly convex by Prop.\ 8.14(ii) --- matching what we already saw directly.
\end{exampleblock}

\begin{block}{Example 8.15: integrating an increasing function gives a convex function}
Let $\psi:\R\to\R$ be increasing, $\alpha\in\R$, and
\[
  \phi(x) = \begin{cases} \displaystyle\int_\alpha^x \psi(t)\,dt, & x \geq \alpha,\\[4pt] +\infty, & \text{otherwise.}\end{cases}\tag{8.11}
\]
Then $\phi$ is convex (strictly convex if $\psi$ is strictly increasing), because $\phi'=\psi$ is increasing on $]\alpha,+\infty[$ by the Fundamental Theorem of Calculus, and Prop.~8.14 applies; adding $\iota_{[\alpha,+\infty[}$ preserves convexity (Remark 8.8).
\end{block}
\end{frame}

\section{8.2 Convexity-Preserving Operations}
%%=============================================================================

\begin{frame}{Building New Convex Functions from Old Ones}
\begin{alertblock}{Big idea}
Convexity is preserved by many natural operations: taking suprema, adding, scaling by positive constants, pointwise limits, composing with affine maps, and composing with increasing convex functions. This lets us build a large catalogue of convex functions from a few basic ones.
\end{alertblock}
\begin{block}{Proposition 8.16 (supremum)}
Let $(f_i)_{i\in I}$ be a family of convex functions $\Hsp \to [-\infty,+\infty]$. Then $\sup_{i\in I} f_i$ is convex. \emph{Proof idea:} $\epi(\sup_i f_i) = \bigcap_i \epi f_i$, an intersection of convex sets, hence convex (Prop.~8.4 + Ex.~3.2(iv)).
\end{block}
\begin{block}{Proposition 8.17 (sums and positive scalars) and 8.18 (pointwise limits)}
The set of convex functions $\Hsp\to\,]-\infty,+\infty]$ is closed under addition and multiplication by $\lambda>0$ (Prop.~8.17). If $(f_n)_{n\in\mathbb N}$ are convex and $f_n\to f$ pointwise, then $f=\lim f_n$ is convex (Prop.~8.18): pass the two-point inequality from Corollary 8.12(iii) to the limit.
\end{block}
\end{frame}

\begin{frame}{Example 8.19: A Convex, Lipschitz ``Limsup Distance'' Function}
\begin{block}{Example 8.19}
Let $(z_n)_{n\in\mathbb N}$ be a bounded sequence in $\Hsp$. Then
\[
  f:\Hsp\to\R : x \mapsto \varlimsup_n \norm{x-z_n}
\]
is convex and Lipschitz continuous with constant $1$.
\end{block}
\textbf{Proof idea.} Set $f_n = \sup\{\norm{\cdot-z_k} : k\geq n\}$: each $f_n$ is convex by Example 8.9 (norm is convex) and Proposition 8.16 (sup of convex). The sequence $(f_n)$ decreases pointwise to $f=\inf_n f_n$, so $f$ is convex by Proposition 8.18. For the Lipschitz bound: $\norm{x-z_n}\leq\norm{x-y}+\norm{y-z_n}$ gives $\varlimsup_n\norm{x-z_n} \leq \norm{x-y}+\varlimsup_n\norm{y-z_n}$, i.e.\ $f(x)-f(y)\leq\norm{x-y}$; symmetrically $f(y)-f(x)\leq\norm{y-x}$, so $|f(x)-f(y)|\leq\norm{x-y}$.

\begin{block}{Composition results: Propositions 8.20--8.21}
\textbf{Prop.\ 8.20:} If $L:\Hsp\to\Ksp$ is affine and $g:\Ksp\to\,]-\infty,+\infty]$ convex, then $g\circ L$ is convex. \textbf{Prop.\ 8.21:} If $f$ is convex, $\phi:\R\to\,]-\infty,+\infty]$ is convex and increasing on $C=\conv(\R\cap\ran f)$, then $\phi\circ f$ is convex.
\end{block}
\end{frame}

\begin{frame}{Examples 8.22--8.23: Radial Integrals and Powers of the Norm}
\begin{block}{Example 8.22}
Let $\psi:\R\to\R$ be continuous, increasing, with $\psi(0)\geq 0$, and set
\[
  f:\Hsp\to\R: x\mapsto \int_0^{\norm{x}}\psi(t)\,dt. \tag{8.13}
\]
Then $f$ is convex; if $\psi$ is strictly increasing, $f$ is strictly convex. \emph{Idea:} write $f=\phi\circ\norm{\cdot}$ for the increasing convex $\phi$ of Example 8.15 (with $\alpha=0$), then apply Prop.~8.21.
\end{block}
\begin{block}{Example 8.23}
For $p\in\,]1,+\infty[$, $\norm{\cdot}^p$ is strictly convex, and
\[
  (\forall x,y\in\Hsp)\quad \norm{x+y}^p \leq 2^{p-1}\bigl(\norm{x}^p+\norm{y}^p\bigr). \tag{8.15}
\]
\emph{Proof:} apply Example 8.22 with $\psi(t)=pt^{p-1}$ for $t\geq0$ (and $\psi(t)=t$ for $t<0$), which is continuous, strictly increasing, $\psi(0)=0$.
\end{block}
\begin{exampleblock}{Numeric check, $p=2$, $x=(3,4)$, $y=(1,-2)$ in $\R^2$}
$\norm{x+y}^2 = \norm{(4,2)}^2 = 20$. Bound: $2^{1}(\norm{x}^2+\norm{y}^2) = 2(25+5)=60$. Indeed $20\leq 60$. \checkmark
\end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Python: Convexity-Preserving Operations in Action}
\begin{lstlisting}[caption={Sup of convex functions, sums, and the norm-power inequality (8.15)}]
import numpy as np

# Proposition 8.16: sup of convex quadratics is convex (piecewise, still convex)
f1 = lambda x: (x - 1)**2
f2 = lambda x: (x + 2)**2
f_sup = lambda x: max(f1(x), f2(x))

def is_convex_on_grid(f, lo=-5, hi=5, n=4000, trials=3000, rng=None):
    rng = rng or np.random.default_rng(1)
    xs = rng.uniform(lo, hi, size=(trials, 2))
    a = rng.uniform(0, 1, size=trials)
    lhs = np.array([f(a[i]*xs[i,0] + (1-a[i])*xs[i,1]) for i in range(trials)])
    rhs = np.array([a[i]*f(xs[i,0]) + (1-a[i])*f(xs[i,1]) for i in range(trials)])
    return np.all(lhs <= rhs + 1e-9)

print("sup(f1,f2) convex?  ", is_convex_on_grid(f_sup))       # -> True

# Example 8.23: ||x+y||^p <= 2^(p-1) (||x||^p + ||y||^p), p = 2
x, y = np.array([3.0, 4.0]), np.array([1.0, -2.0])
p = 2
lhs = np.linalg.norm(x + y) ** p
rhs = 2 ** (p - 1) * (np.linalg.norm(x) ** p + np.linalg.norm(y) ** p)
print(f"||x+y||^p = {lhs:.2f}  <=  2^(p-1)(...)={rhs:.2f} : {lhs <= rhs}")
\end{lstlisting}
\end{frame}

\begin{frame}{Integral Functionals and the Perspective Function}
\begin{block}{Proposition 8.24 (integral functionals)}
Let $(\Omega,\mathcal F,\mu)$ be a measure space, $\mathsf H$ a separable Hilbert space, $\Hsp = L^2((\Omega,\mathcal F,\mu);\mathsf H)$, and $\varphi:\mathsf H\to\,]-\infty,+\infty]$ measurable and convex with an integrable minorant. Then
\[
  f:\Hsp\to\,]-\infty,+\infty] : x\mapsto \int_\Omega \varphi(x(\omega))\,\mu(d\omega) \tag{8.17}
\]
is convex. \emph{Idea:} apply Corollary 8.12 pointwise ($\mu$-a.e.) and integrate the inequality.
\end{block}
\begin{block}{Proposition 8.25 (perspective function)}
Let $\varphi:\Hsp\to\,]-\infty,+\infty]$ be convex. Its \emph{perspective} is
\[
  f:\R\times\Hsp\to\,]-\infty,+\infty] : (\xi,x)\mapsto \begin{cases}\xi\,\varphi(x/\xi), & \xi>0,\\ +\infty, & \text{otherwise.}\end{cases}\tag{8.18}
\]
Then $\epi f = \cone(\{1\}\times\epi\varphi)$ and $f$ is convex. This ``rescale-then-multiply'' construction is the engine behind every divergence on the next slides.
\end{block}
\end{frame}

\begin{frame}{Csisz\'ar Divergences: A Family Built from Perspective Functions}
\begin{block}{Example 8.26: Csisz\'ar $\phi$-divergence}
Let $\Hsp=\R^N$, $I=\{1,\dots,N\}$, $\phi:\R\to\,]-\infty,+\infty]$ convex. For $x=(\xi_i)_{i\in I}$, $y=(\eta_i)_{i\in I}$,
\[
  d_\phi(x,y) = \begin{cases} \displaystyle\sum_{i\in I} \eta_i\,\phi(\xi_i/\eta_i), & y \in \R_{++}^N,\\ +\infty, & \text{otherwise.}\end{cases}\tag{8.20}
\]
Then $d_\phi:\Hsp\times\Hsp\to\,]-\infty,+\infty]$ is convex (combine Prop.~8.25 applied coordinatewise with Prop.~8.6 for direct sums). Different choices of the convex generator $\phi$ give the classical statistical divergences below.
\end{block}
\begin{block}{Example 8.27: Kullback--Leibler divergence}
Taking $\phi(t) = t\ln t$ ($t>0$; $+\infty$ else, convex by Prop.~8.14(ii)) gives
\[
  f(x,y) = \sum_{i\in I} \xi_i \ln(\xi_i/\eta_i), \quad x,y\in\R_{++}^N, \tag{8.21}
\]
the KL divergence; it is convex on $\R_{++}^N\times\R_{++}^N$.
\end{block}
\end{frame}

\begin{frame}{More Divergences from the Same Recipe}
\begin{block}{Example 8.28: Jeffreys divergence}
$\phi(t) = (t-1)\ln t$ ($t>0$) gives $f(x,y) = \sum_i (\xi_i-\eta_i)(\ln\xi_i-\ln\eta_i)$, convex.
\end{block}
\begin{block}{Example 8.29--8.30: Pearson and Hellinger divergences}
$\phi(t)=|t-1|^2$ gives Pearson's $f(x,y)=\sum_i |\xi_i-\eta_i|^2/\eta_i$.\\
$\phi(t) = |\sqrt t - 1|^2$ ($t>0$) gives Hellinger's $f(x,y) = \sum_i |\sqrt{\xi_i}-\sqrt{\eta_i}|^2$.\\
Both $\phi$'s are (strictly) convex by Prop.~8.14(ii), so both $f$'s are convex by Example 8.26.
\end{block}
\begin{block}{Example 8.31: R\'enyi divergence}
For $\alpha\in\,]1,+\infty[$, $\phi(t)=t^\alpha$ ($t>0$) gives $f(x,y) = \sum_i \xi_i^\alpha \eta_i^{1-\alpha}$, convex (by Prop.~8.14(ii) or Example 8.23).
\end{block}
\begin{exampleblock}{Numeric check: KL term at $\xi=2,\eta=1$}
$\phi(2/1)\cdot 1 = 2\ln 2 \approx 1.386 \geq 0$, consistent with $t\ln t \geq -1/e$ for all $t>0$ and $=0$ at $t=1$ (the divergence vanishes only when $x=y$).
\end{exampleblock}
\end{frame}

\begin{frame}{Adjoints, Reciprocals, and the Marginal Function}
\begin{block}{Examples 8.32--8.34}
\textbf{8.32:} For a convex $C\subset\Hsp$, $f(\xi,x) = \xi$ if $\xi>0,\ x\in\xi C$, else $+\infty$, is convex (apply Prop.~8.25 to $1+\iota_C$).\\
\textbf{8.33 (adjoint):} For convex $\phi:\R\to\,]-\infty,+\infty]$, its \emph{adjoint} $\phi^\circ(\xi)=\xi\phi(1/\xi)$ for $\xi>0$ (else $+\infty$) is convex --- it is the perspective of $\phi$ evaluated at $x=1$.\\
\textbf{8.34:} $\phi(\xi)=1/\xi$ for $\xi>0$ (else $+\infty$) is convex, since $\phi = (|\cdot|^2)^\circ$.
\end{block}
\begin{block}{Proposition 8.35: the marginal (partial infimum) function}
Let $\Ksp$ be a Hilbert space and $F:\Hsp\times\Ksp\to\,]-\infty,+\infty]$ convex. Then
\[
  f:\Hsp\to[-\infty,+\infty] : x\mapsto \inf F(x,\Ksp) \tag{8.34}
\]
is convex. \emph{Proof idea:} for $x_1,x_2\in\dom f$ pick $y_1,y_2$ nearly achieving the infima, use convexity of $F$ on $(x_1,y_1),(x_2,y_2)$, then let the near-infima tighten.
\end{block}
\end{frame}

\begin{frame}{Example 8.36: The Minkowski Gauge Is Convex}
\begin{block}{Minkowski gauge}
Let $C\subset\Hsp$ be convex with $0\in C$. The \emph{gauge} of $C$ is
\[
  m_C:\Hsp\to\,]-\infty,+\infty] : x\mapsto \inf\{\xi\in\R_{++} : x\in\xi C\}. \tag{8.36}
\]
Then $m_C$ is convex, $m_C(0)=0$,
\[
  (\forall x\in\Hsp)(\forall\lambda\in\R_{++})\quad m_C(\lambda x) = \lambda\,m_C(x), \tag{8.37}
\]
and $x\in\lambda C$ whenever $\lambda > m_C(x)$. \emph{Idea:} $m_C$ is the marginal function of $f(\xi,x)$ from Example 8.32, so Prop.~8.35 applies.
\end{block}
\begin{exampleblock}{Numeric check: $C = [-1,1]\subset\R$}
$m_C(x) = |x|$. E.g., $m_C(3) = \inf\{\xi>0 : 3\in\xi[-1,1]\} = \inf\{\xi>0:\xi\geq3\}=3=|3|$. \checkmark. And $m_C(2x)=m_C(6)=6=2\cdot m_C(3)$, confirming positive homogeneity (8.37).
\end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Python: Divergences and the Minkowski Gauge}
\begin{lstlisting}[caption={KL divergence convexity check and the Minkowski gauge of an interval}]
import numpy as np

\section{8.3 Topological Properties}

\begin{frame}{From Boundedness to Lipschitz Continuity: Proposition 8.37}
\begin{alertblock}{Big idea}
For convex functions, \emph{local boundedness} near a point is already enough to force strong regularity: continuity, even local Lipschitz continuity. This is special to convexity --- an arbitrary function bounded near a point need not be continuous there.
\end{alertblock}
\begin{block}{Proposition 8.37}
Let $f:\Hsp\to\,]-\infty,+\infty]$ be proper and convex, $x_0\in\dom f$, $\rho\in\R_{++}$.
\begin{itemize}
  \item[(i)] If $\eta = \sup f(B(x_0;\rho)) <+\infty$, then for every $\alpha\in\,]0,1[$ and $x\in B(x_0;\alpha\rho)$: $|f(x)-f(x_0)| \leq \alpha(\eta - f(x_0))$.
  \item[(ii)] If $\delta = \mathrm{diam}\,f(B(x_0;2\rho)) < +\infty$, then $f$ is Lipschitz continuous relative to $B(x_0;\rho)$ with constant $\delta/\rho$.
\end{itemize}
\end{block}
{\footnotesize Proof idea for (i): write $x$ as a convex combination of $x_0$ and a point in $B(x_0;\rho)$, apply the two-point convexity inequality both ways to sandwich $f(x)-f(x_0)$. Part (ii) similarly compares $f(x),f(y)$ via an auxiliary point $z$ chosen so that $x\in[y,z]$.}
\end{frame}

\begin{frame}{Theorem 8.38: The Continuity Equivalence Theorem}
\begin{block}{Theorem 8.38}
Let $f:\Hsp\to\,]-\infty,+\infty]$ be proper and convex, $x_0\in\dom f$. TFAE:
\begin{itemize}
  \item[(i)] $f$ is locally Lipschitz continuous near $x_0$.
  \item[(ii)] $f$ is continuous at $x_0$.
  \item[(iii)] $f$ is bounded (above and below) on a neighborhood of $x_0$.
  \item[(iv)] $f$ is bounded \emph{above} on a neighborhood of $x_0$.
\end{itemize}
Moreover, if one of these holds, $f$ is locally Lipschitz continuous on all of $\intr\dom f$.
\end{block}
\textbf{Why this is remarkable:} for general functions, being bounded above near a point says almost nothing about continuity. For \emph{convex} functions, (iv) $\Rightarrow$ (i) --- the weakest-looking condition (bounded above only) already upgrades all the way to local Lipschitz continuity, and moreover the good behavior automatically spreads to the \emph{whole interior} of the domain.
\begin{block}{Proof chain}
(i)$\Rightarrow$(ii)$\Rightarrow$(iii)$\Rightarrow$(iv): trivial. (iv)$\Rightarrow$(ii): Prop.~8.37(i). (iii)$\Rightarrow$(i): Prop.~8.37(ii). Spreading to $\intr\dom f$: for any $x\in\intr\dom f\setminus\{x_0\}$, construct via a convex combination a ball around $x$ on which $f$ is bounded above, then reapply (iv)$\Leftrightarrow$(i) at $x$.
\end{block}
\end{frame}

\begin{frame}{Corollaries 8.39--8.41: When Is a Convex Function Automatically Continuous?}
\begin{block}{Corollary 8.39}
$\cont f = \intr\dom f$ if any of: (i) $f$ bounded above on some neighborhood; (ii) $f$ lower semicontinuous; (iii) $\Hsp$ finite-dimensional.
\end{block}
\begin{block}{Corollary 8.40}
If $\Hsp$ is finite-dimensional and $f:\Hsp\to\R$ is convex, then $f$ is continuous (everywhere!). This is a striking finite-dimensional phenomenon: convexity alone, with no boundedness or semicontinuity assumption, forces continuity.
\end{block}
\begin{block}{Corollary 8.41}
If $\Hsp$ is finite-dimensional, $f$ proper convex, and $C$ is a nonempty closed bounded subset of $\ri\dom f$ (relative interior), then $f$ is Lipschitz continuous relative to $C$. \emph{Idea:} cover the compact $C$ by finitely many balls on which local Lipschitz constants exist (Cor.~8.39(iii) + Thm.~8.38), then take the max constant.
\end{block}
\end{frame}

\begin{frame}{Example 8.42: Why Infinite Dimensions Are Different}
\begin{block}{A pathological linear functional}
Suppose $\Hsp$ is infinite-dimensional and $f:\Hsp\to\R$ is a linear functional that is \emph{everywhere discontinuous} (such functionals exist; Example 2.27 gives a construction using a Hamel basis / unbounded coordinate functional). Let $C=\lev_{\leq 1} f$, $H=\{x : f(x)=0\}$.
\end{block}
\begin{itemize}
  \item[(i)] $f,-f$ are convex, $\intr\dom f = \Hsp$, but $\cont f = \varnothing$.
  \item[(ii)] $f$ is neither lower nor upper semicontinuous.
  \item[(iii)] $C$ is convex, $0\in\core C$, but $\intr C = \varnothing$.
  \item[(iv)] $f$ is unbounded above and below on \emph{every} nonempty open subset of $\Hsp$.
  \item[(v)] $H$ is a hyperplane that is \emph{dense} in $\Hsp$.
\end{itemize}
\textbf{Lesson:} Corollary 8.40 (continuity for free) is a genuinely finite-dimensional fact --- in infinite dimensions, convexity alone does not force continuity, because linear functionals can already be pathological.
\end{frame}

\begin{frame}{Example 8.43 and the Running Example: The Huber Function}
\begin{block}{Example 8.43: gauge continuity when $0\in\intr C$}
If $C\subset\Hsp$ is convex with $0\in\intr C$, then $\dom m_C = \Hsp$ and $m_C$ is continuous. \emph{Idea:} $0\in\intr C$ gives $\varepsilon x\in C$ for small $\varepsilon>0$ so $m_C(x)\leq1/\varepsilon<\infty$ everywhere; then apply Cor.~8.39(i), since $m_C$ is bounded on the neighborhood $C$ of $0$ (values in $[0,1]$).
\end{block}
\begin{block}{Example 8.44: the Huber function ($\rho\in\R_{++}$)}
\[
  f:\R\to\R : x\mapsto \begin{cases} \rho|x| - \rho^2/2, & |x|>\rho,\\ |x|^2/2, & |x|\leq\rho.\end{cases}\tag{8.47}
\]
$f$ is continuous and convex: convexity from Prop.~8.14(i) ($f'$ is increasing --- it is $x$ on $[-\rho,\rho]$ and $\pm\rho$ outside, a continuous non-decreasing function), continuity from Corollary 8.40 ($\R$ is finite-dimensional).
\end{block}

\begin{frame}[fragile]{Python: Huber Function -- Convexity and Continuity Checks}
\begin{lstlisting}[caption={Verify the Huber function is convex (Jensen) and continuous (no jumps) at the transition rho}]
import numpy as np

rho = 1.0
def huber(x):
    x = np.asarray(x, float)
    return np.where(np.abs(x) > rho, rho*np.abs(x) - rho**2/2, x**2/2)

# Continuity check at the "seam" x = rho
eps = 1e-6
left, right = huber(rho - eps), huber(rho + eps)
print(f"huber(rho-eps)={float(left):.8f}  huber(rho+eps)={float(right):.8f}")
# -> both ~= 0.5, matching |rho|^2/2 = 0.5: no jump

# Jensen check with two points straddling the seam
x1, x2, alpha = -2.0, 1.5, 0.4
xm = alpha*x1 + (1-alpha)*x2
lhs = huber(xm)
rhs = alpha*huber(x1) + (1-alpha)*huber(x2)
print(f"huber(midpoint)={float(lhs):.4f}  vs  mixture={float(rhs):.4f}  "
      f"convex_ok={bool(lhs <= rhs + 1e-9)}")
\end{lstlisting}
Running this shows the left/right values agree at the seam $x=\rho$ (continuity) and the Jensen inequality holds even when the two points lie in \emph{different} pieces of the definition (convexity of the ``glued'' function).
\end{frame}

\begin{frame}{Interior of the Epigraph and Sublevel Sets: Propositions 8.45--8.46}
\begin{block}{Proposition 8.45}
Let $f:\Hsp\to\,]-\infty,+\infty]$ be proper and convex, bounded above on a neighborhood of some $x\in\dom f$. Then $\intr\epi f \neq \varnothing$. \emph{Idea:} the point $(x,f(x)+\rho)$ for suitable $\rho$ has a whole ball around it inside $\epi f$, using the same estimate as Theorem 8.38.
\end{block}
\begin{block}{Proposition 8.46}
Let $f:\Hsp\to[-\infty,+\infty]$.
\begin{itemize}
  \item[(i)] If $f$ is upper semicontinuous at $x_0$ with $f(x_0)<0$, then $x_0\in\intr\lev_{\leq0} f$.
  \item[(ii)] If $f$ is convex and $f(x_0)<0$ for some $x_0$, then $\intr\lev_{\leq0} f \subset \lev_{<0} f$.
\end{itemize}
\end{block}
\begin{block}{Corollary 8.47}
If $f$ is convex, $\lev_{<0} f\neq\varnothing$, and one of: (i) $f$ upper semicontinuous on $\lev_{<0}f$; (ii) $f$ lower semicontinuous and $\dom f$ open; (iii) $\Hsp$ finite-dimensional and $\dom f$ open --- then $\intr\lev_{\leq0} f = \lev_{<0} f$: the interior of the closed sublevel set equals the strict sublevel set.
\end{block}
\end{frame>

\begin{frame}{Chapter 8 Summary}
\begin{block}{What convex functions are (8.1)}
Defined by the chord inequality $f(\alpha x+(1-\alpha)y)\leq \alpha f(x)+(1-\alpha)f(y)$; equivalently, $\epi f$ is a convex set (Prop.~8.4); Jensen's inequality (Prop.~8.11) extends this to finite convex combinations; a differentiable $\phi:\R\to\R$ is convex iff $\phi'$ is increasing (Prop.~8.14).
\end{block}
\begin{block}{How to build convex functions (8.2)}
Sups, sums, positive scalar multiples, pointwise limits, affine precomposition, increasing-convex postcomposition, integral functionals, perspective functions, and the Minkowski gauge all preserve convexity --- this generates norms, powers of norms, and all the classical statistical divergences (KL, Jeffreys, Pearson, Hellinger, R\'enyi).
\end{block}
\begin{block}{Why convexity is topologically powerful (8.3)}
Local boundedness above already forces local Lipschitz continuity (Thm.~8.38); in finite dimensions, every real-valued convex function is automatically continuous (Cor.~8.40) --- a fact that fails in infinite dimensions (Example 8.42).
\end{block}
\end{frame}

Chapter 9 specializes to \emph{lower semicontinuous} convex functions --- the class $\Gamma(\Hsp)$ --- where the theory becomes most powerful, including the existence of continuous affine minorants obtained by projecting onto the epigraph.
\end{frame}
