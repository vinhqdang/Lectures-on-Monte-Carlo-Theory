\section{Notation You'll Need}
%==============================================================

\begin{frame}[allowframebreaks]{Notation You'll Need / Recap}

  \textbf{Ambient space.} $\mHH$ is a real Hilbert space with inner
  product $\inner{\cdot}{\cdot}$ and norm $\norm{\cdot}$.

  \textbf{Extended-real-valued functions.} $f:\mHH\to\left]-\infty,+\infty\right]$.
  \begin{itemize}
    \item $\dom f=\{x\in\mHH: f(x)<+\infty\}$ (the \emph{domain}).
    \item $f$ is \emph{proper} if $\dom f\neq\emptyset$ (it never
          takes the value $-\infty$, and is finite somewhere).
    \item $f\in\GamZero$ means: $f$ is proper, convex, and lower
          semicontinuous (lsc) -- the standing assumption in most
          existence results (recap from Chapters 8--9).
    \item $\iC$ is the \emph{indicator function} of a set $C$: $0$ on
          $C$, $+\infty$ off $C$. Minimizing $f+\iC$ over $\mHH$ is
          the same as minimizing $f$ over $C$.
  \end{itemize}

  \textbf{Level sets.} For $\xi\in\R$,
  $\lev_{\le\xi}f=\{x\in\mHH: f(x)\le\xi\}$ is the \emph{lower level
  set} at height $\xi$.

  \textbf{Balls.} $B(x;\rho)=\{y\in\mHH:\norm{y-x}\le\rho\}$.

  \textbf{Quasiconvexity (recap, Ch.\ 10).} $f$ is \emph{quasiconvex}
  if every lower level set $\lev_{\le\xi}f$ is convex; \emph{strictly
  quasiconvex} strengthens this so that no two distinct points on the
  same level set can have their midpoint at a strictly lower level
  (this rules out flat regions below the top of a level set).

  \textbf{Strict / strong convexity (recap, Ch.\ 8--10).} $f$ is
  \emph{strictly convex} if
  $f(\alpha x+(1-\alpha)y)<\alpha f(x)+(1-\alpha)f(y)$ for all
  $x\neq y$ in $\dom f$ and $\alpha\in\left]0,1\right[$. $f$ is
  \emph{$\beta$-strongly convex} ($\beta>0$) if $f-\tfrac\beta2\norm{\cdot}^2$
  is convex -- strictly stronger than strict convexity, and it is
  what buys us quantitative (Lipschitz-type) control on minimizing
  sequences, not just uniqueness.

  \framebreak

  \textbf{What this chapter is about.} Given $f:\mHH\to\left]-\infty,+\infty\right]$
  and a subset $C\subseteq\mHH$, we study the problem
  \[
    \text{minimize } f(x) \text{ subject to } x\in C.
  \]
  Three separate questions organize the chapter:
  \begin{enumerate}
    \item \textbf{Value:} what is $\inf f(C)$, the greatest lower
          bound on $f$ over $C$ (Section 11.1)?
    \item \textbf{Attainment:} is there an actual point $x\in C$
          with $f(x)=\inf f(C)$ -- a \emph{minimizer} (Section 11.2),
          and if so is it \emph{unique} (Section 11.3) or does it
          \emph{exist at all} (Section 11.4)?
    \item \textbf{Approximation:} if no minimizer is at hand yet, how
          do \emph{minimizing sequences} (sequences whose values
          approach the infimum) behave -- do they converge, and to
          what (Section 11.5)?
  \end{enumerate}

  \textbf{Running numeric example (preview).} Throughout, we will
  track $f:\R\to\R:x\mapsto (x-3)^2+1$, minimized first over all of
  $\R$, then over the restricted convex set $C=[0,2]$. This one
  concrete function will illustrate every abstract statement below.

\end{frame}

\section{11.1 Infima and Suprema}
%==============================================================

\begin{frame}{11.1 Intuition: Infimum vs.\ Minimum}

  \textbf{Plain language, first.} For $f:\mHH\to\left]-\infty,+\infty\right]$
  and $C\subseteq\mHH$, write $f(C)=\{f(x):x\in C\}\subseteq\left]-\infty,+\infty\right]$,
  a set of extended real numbers.

  \begin{block}{The infimum -- a number, not necessarily a point}
    $\inf f(C)$ is the \textbf{greatest lower bound} of the set of
    values $f(C)$. It always exists in $[-\infty,+\infty]$ (by
    convention $\inf\emptyset=+\infty$). It only tells you \emph{how
    low $f$ can get} on $C$ -- it says nothing about whether some
    actual $x\in C$ achieves that value.
  \end{block}

  \begin{alertblock}{The key distinction}
    \textbf{Not every infimum is attained.} E.g.\ $f(x)=e^{-x}$ on
    $\R$ has $\inf f(\R)=0$, but $f(x)>0$ for every $x\in\R$: no
    point achieves the value $0$. A \textbf{minimizer} is a
    \emph{point} $x\in C$ with $f(x)=\inf f(C)$ exactly -- when a
    minimizer exists, we write $\min f(C)$ instead of $\inf f(C)$,
    signaling attainment. Section 11.2 defines minimizers precisely;
    Sections 11.3--11.4 ask when they are unique / exist at all.
  \end{alertblock}

  \textbf{Dually}, $\sup f(C)$ is the least upper bound of $f(C)$;
  everything below has an obvious ``upper'' mirror image, so we focus
  on infima (minimization), which is the book's convention.

\end{frame}

\begin{frame}{11.1 A Continuity Fact About $\inf f(C)$}

  \textbf{Why this matters.} It is tempting to think that enlarging
  $C$ to its closure $\overline C$ can only \emph{decrease} the
  infimum (more points to search over). The following shows that,
  for a convex $f$, this cannot happen once $C$ has nonempty
  interior meeting $\dom f$: closing up $C$ does not help.

  \begin{block}{Proposition 11.1(iv)}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be proper and convex,
    let $C\subseteq\mHH$, and suppose $\dom f\cap\intr C\neq\emptyset$.
    Then
    \[
      \inf f(\overline C) = \inf f(C).
    \]
  \end{block}

  \textbf{Proof.} $\overline C\supseteq C$ gives
  $\inf f(\overline C)\le\inf f(C)$ trivially (a bigger feasible set
  can only make the infimum smaller or equal). For the reverse: fix
  $x_0\in\overline C$, $x_1\in\dom f\cap\intr C$, and set
  $x_\alpha=(1-\alpha)x_0+\alpha x_1$ for $\alpha\in\left]0,1\right[$.
  Convexity along the segment plus continuity of $f$ relative to its
  domain near $x_1\in\intr C$ (via Chapter 9's continuity results)
  give $f(x_0)=\lim_{\alpha\downarrow0}f(x_\alpha)\ge\inf f(C)$, since
  each $x_\alpha\in C$ for small $\alpha$. As $x_0\in\overline C$ was
  arbitrary, $\inf f(\overline C)\ge\inf f(C)$. \hfill$\blacksquare$

\end{frame}

\begin{frame}{11.1 Example 11.2: Support Functions Live in $\Gamma_0$}

  \textbf{Why this example.} It shows the infimum/supremum machinery
  of this section in action: a support function is literally defined
  as a supremum, and turns out to automatically inherit convexity and
  lower semicontinuity.

  \begin{block}{Support function (recap, Ch.\ 7) and Example 11.2}
    For $C\subseteq\mHH$ nonempty, the support function is
    $\sigma_C(x)=\sup_{y\in C}\inner{x}{y}$. Then:
    \begin{itemize}
      \item $\sigma_C$ is \emph{sublinear} (positively homogeneous
            and subadditive) and belongs to $\Gamma_0(\mHH)$;
      \item $\sigma_C=\sigma_{\overline{\mathrm{conv}}\,C}$ (support
            functions cannot distinguish $C$ from its closed convex
            hull), and $\dom\sigma_C=\operatorname{bar}\overline{\mathrm{conv}}\,C$
            (the \emph{barrier cone});
      \item if $C$ is bounded, $\sigma_C$ is real-valued and
            continuous on all of $\mHH$.
    \end{itemize}
  \end{block}

  \textbf{Why it works.} $\sigma_C$ is a supremum of the family of
  continuous linear (hence convex, lsc) functionals
  $(\inner{x}{\cdot})_{x\in C}$; a supremum of convex lsc functions is
  again convex and lsc (Ch.\ 9), and $\sigma_C(0)=0$ pins down that it
  is sublinear rather than just convex.

\end{frame}

\section{11.2 Minimizers}
%==============================================================

\begin{frame}{11.2 Intuition: Local vs.\ Global for Convex Functions}

  \textbf{The punchline of this section, in one sentence.} For convex
  functions, there is no such thing as a ``merely local'' minimum that
  fails to be global -- convexity kills the usual calculus worry about
  getting stuck in a shallow local dip.

  \textbf{Why this is special.} For a general (nonconvex) function,
  local minimizers abound and are usually not global. Convexity
  prevents this: any point that looks locally optimal (best in some
  small ball around it) must in fact be optimal everywhere. This is
  what makes convex minimization tractable.

\end{frame}

\begin{frame}{11.2 Definition 11.3: Minimizers, Argmin, Local Minimizers}

  \begin{block}{Definition 11.3}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be proper and let
    $x\in\mHH$. Then $x$ is a \emph{(global) minimizer} of $f$ if
    $f(x)=\inf f(\mHH)$, i.e.\ $f(x)=\min f(\mHH)\in\R$. The set of
    minimizers of $f$ is denoted $\Argmin f$. If $\Argmin f$ is a
    singleton, its unique element is denoted $\argmin_{x\in\mHH}f(x)$.

    Now let $C\subseteq\mHH$ with $C\cap\dom f\neq\emptyset$. A
    \emph{minimizer of $f$ over $C$} is a minimizer of $f+\iC$; the
    set of such minimizers is $\Argmin_C f$. If there exists
    $\rho\in\Rpp$ such that $x$ is a minimizer of $f$ over
    $B(x;\rho)$, then $x$ is a \emph{local minimizer} of $f$.
  \end{block}

  \begin{itemize}
    \item $\Argmin f$: the (possibly empty, possibly non-singleton)
          set of \emph{global} minimizers.
    \item $\Argmin_C f$: replace ``everywhere'' by ``on $C$'' -- via
          the indicator trick $f+\iC$, constrained minimization is
          just unconstrained minimization of a modified function.
    \item Local minimizer: optimal only within some small ball -- a
          priori a weaker notion than global minimizer.
  \end{itemize}

\end{frame}

\begin{frame}{11.2 Proposition 11.4: Local $\Rightarrow$ Global (Convex Case)}

  \begin{block}{Proposition 11.4}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be proper and convex.
    Then every local minimizer of $f$ is a (global) minimizer.
  \end{block}

  \textbf{Step-by-step proof.}
  \begin{enumerate}
    \item Take $x\in\mHH$ and $\rho\in\Rpp$ such that
          $f(x)=\min f(B(x;\rho))$ (i.e.\ $x$ is a local minimizer,
          witnessed by the ball of radius $\rho$).
    \item Fix any $y\in\dom f\setminus B(x;\rho)$ (a candidate point
          outside the ball -- if no such $y$ exists, $x$ is trivially
          global). Set $\alpha=1-\rho/\norm{x-y}\in\left]0,1\right[$
          and $z=\alpha x+(1-\alpha)y$.
    \item \textbf{Key geometric fact:} $z\in B(x;\rho)$. Indeed
          $\norm{z-x}=(1-\alpha)\norm{y-x}=\rho$, so $z$ sits exactly
          on the boundary of the ball -- reachable from $x$ along the
          segment to $y$, but not further than $\rho$.
    \item By convexity of $f$ along the segment from $x$ to $y$,
          \[
            f(x)\le f(z)=f(\alpha x+(1-\alpha)y)\le\alpha f(x)+(1-\alpha)f(y).
          \]
          The first inequality is local optimality at $x$ (since
          $z\in B(x;\rho)$); the second is convexity.
    \item Rearranging: $(1-\alpha)f(x)\le(1-\alpha)f(y)$, and since
          $1-\alpha>0$, we get $f(x)\le f(y)$.
    \item Since $y\in\dom f\setminus B(x;\rho)$ was arbitrary, and
          $f(x)\le f(\cdot)$ already holds trivially inside
          $B(x;\rho)$ by local optimality, $f(x)\le f(y)$ for
          \emph{every} $y\in\dom f$: $x$ is a global minimizer.
          \hfill$\blacksquare$
  \end{enumerate}

\end{frame}

\begin{frame}{11.2 Proposition 11.5: Interior Minimizers Over $C$ Are Global}

  \begin{block}{Proposition 11.5}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be proper and convex,
    and let $C\subseteq\mHH$. Suppose that $x$ is a minimizer of $f$
    over $C$ such that $x\in\intr C$. Then $x$ is a minimizer of $f$.
  \end{block}

  \textbf{Proof.} Since $x\in\intr C$, there exists $\rho\in\Rpp$
  with $B(x;\rho)\subseteq C$. Because $x$ minimizes $f$ over $C$,
  $f(x)=\inf f(C)\le\inf f(B(x;\rho))$, and since
  $B(x;\rho)\subseteq C$ the reverse inequality also holds, so
  $f(x)=\inf f(B(x;\rho))$: $x$ is a \emph{local} minimizer of $f$.
  Proposition 11.4 then upgrades this to a global minimizer.
  \hfill$\blacksquare$

  \vspace{0.3em}
  \textbf{Reading it.} This is exactly why, in the running example
  below, checking whether the constrained minimizer over $C=[0,2]$
  lies in the \emph{interior} of $C$ tells you immediately whether it
  is also the unconstrained minimizer.

\end{frame}

\section{Running Example, Part I: Minimizing $(x-3)^2+1$}
%==============================================================

\begin{frame}{Running Example: Setup}

  \textbf{The function.} Let $\mHH=\R$ and
  \[
    f:\R\to\R:\ x\mapsto (x-3)^2+1.
  \]
  This $f$ is real-valued, continuous, and \emph{strongly convex}
  with constant $\beta=2$ (since $f-\tfrac12(2)\norm{\cdot}^2$
  reduces to an affine function, which is convex).

  \begin{block}{Minimizing over $\mHH=\R$ (unconstrained)}
    Completing the square is already done: $f(x)\ge 1$ for all
    $x\in\R$, with equality iff $x=3$. Hence
    \[
      \Argmin f=\{3\},\qquad \argmin_{x\in\R}f(x)=3,\qquad \min f(\R)=1.
    \]
  \end{block}

  \textbf{Now restrict.} Let $C=[0,2]\subset\R$: closed, convex,
  bounded, and $3\notin C$. What is $\Argmin_C f$?

  \begin{itemize}
    \item On $C=[0,2]$, $f$ is \emph{strictly decreasing} (since the
          vertex $x=3$ lies to the right of all of $C$), so the
          minimum over $C$ is attained at the \emph{right endpoint}.
    \item $x_C=2$, $f(x_C)=(2-3)^2+1=2$. So
          $\Argmin_C f=\{2\}$, $\min_{x\in C}f(x)=2$.
    \item Since $x_C=2\in\bdry C$ (\emph{not} $\intr C=\left]0,2\right[$),
          Proposition 11.5 does \emph{not} apply -- consistent with
          $x_C=2\neq 3=\argmin_{\R}f$: the constrained minimizer is
          genuinely different from the unconstrained one.
  \end{itemize}

\end{frame}

\begin{frame}{Running Example: Picture}
  \begin{center}
    \includegraphics[width=0.72\textwidth]{fig_quadratic_running_example}
  \end{center}
  \footnotesize Minimizing $f(x)=(x-3)^2+1$ over $\R$ gives

\end{frame}

\section{11.3 Uniqueness of Minimizers}
%==============================================================

\begin{frame}{11.3 Intuition: What Forces a Minimizer to Be Unique?}

  \textbf{The idea.} A convex function can have a whole flat
  ``valley'' of minimizers (e.g.\ $g(x)=\max\{0,|x|-1\}$ has
  $\Argmin g=[-1,1]$). Uniqueness requires ruling out flatness at the
  bottom: either the function itself must be strictly convex, or the
  feasible set must be strictly convex so it cannot contain a whole
  segment of near-optimal points.

  \begin{alertblock}{Two independent routes to uniqueness (this section)}
    \begin{enumerate}
      \item Make $f$ \emph{strictly convex} (or $f+\iC$ strictly
            quasiconvex) -- Proposition 11.8(i), Corollary 11.9.
      \item Keep $f$ merely convex, but make the constraint set $C$
            \emph{strictly convex} and require the interior of $C$ to
            miss all unconstrained minimizers -- Proposition 11.8(ii).
    \end{enumerate}
  \end{alertblock}

\end{frame}

\begin{frame}{11.3 Propositions 11.6--11.7: Building Blocks}

  \begin{block}{Proposition 11.6}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be proper and
    quasiconvex. Then $\Argmin f$ is convex.
  \end{block}
  \textbf{Why:} $\Argmin f$ is precisely the lowest-height lower
  level set of $f$, and quasiconvexity says \emph{every} lower level
  set is convex.

  \begin{block}{Proposition 11.7 (even, convex $\phi:\R\to\left]-\infty,+\infty\right]$)}
    Suppose $\phi$ is proper, even ($\phi(-\xi)=\phi(\xi)$), and
    convex. Then: (i) $0\in\Argmin\phi$; (ii) $\phi$ is increasing on
    $\Rp$; (iii) $\phi$ is decreasing on $\R_-$. If moreover $\phi$
    vanishes \emph{only} at $0$, then (iv) $\phi$ is strictly
    increasing on $\Rp$ and (v) strictly decreasing on $\R_-$.
  \end{block}

  \textbf{Why:} Symmetry plus convexity forces $\phi(0)\le\tfrac12\phi(\xi)+\tfrac12\phi(-\xi)=\phi(\xi)$
  for every $\xi$, giving (i); a convex combination argument
  ($\alpha=\xi/\eta$ for $0<\xi<\eta$) then gives monotonicity, and
  strictness upgrades to strict monotonicity once $0$ is the
  \emph{only} zero of $\phi$. This is exactly the mechanism behind
  why $\norm{\cdot}^2$ (even, strictly convex, vanishing only at $0$)
  has a unique, strictly increasing-away-from-origin radial profile.

\end{frame}

\begin{frame}{11.3 Proposition 11.8: At Most One Minimizer Over $C$}

  \begin{block}{Proposition 11.8}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be quasiconvex and let
    $C$ be a convex subset of $\mHH$ with $C\cap\dom f\neq\emptyset$.
    Suppose one of the following holds:
    \begin{enumerate}
      \item[(i)] $f+\iC$ is \emph{strictly} quasiconvex.
      \item[(ii)] $f$ is convex, $(\intr C)\cap\Argmin f=\emptyset$,
            and $C$ is \emph{strictly convex}, i.e.\
            \[
              (\forall x\in C)(\forall y\in C)\quad x\neq y\ \Rightarrow\
              \frac{x+y}{2}\in\intr C.
            \]
    \end{enumerate}
    Then $f$ has at most one minimizer over $C$.
  \end{block}

  \textbf{Proof sketch.} Assume $C$ is not a singleton, set
  $\mu=\inf f(C)$, and suppose toward a contradiction that $x\neq y$
  in $C\cap\dom f$ both achieve $f(x)=f(y)=\mu$. Since $x,y$ lie in
  the convex set $C\cap\lev_{\le\mu}f$, so does the midpoint
  $z=(x+y)/2$, hence $f(z)=\mu$ too.
  \begin{itemize}
    \item[(i)] Strict quasiconvexity of $f+\iC$ forces
          $\mu=f(z)<\max\{f(x),f(y)\}=\mu$ -- impossible.
    \item[(ii)] Strict convexity of $C$ gives $z\in\intr C$, and
          Proposition 11.5 upgrades $z$ (a minimizer over $C$) to a
          global minimizer of $f$: $z\in(\intr C)\cap\Argmin f$,
          contradicting the hypothesis that this set is empty.
  \end{itemize}
  Either way, no two distinct minimizers can exist. \hfill$\blacksquare$

\end{frame}

\begin{frame}{11.3 Corollary 11.9: The Case $C=\mHH$}

  \begin{block}{Corollary 11.9}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be proper and
    \emph{strictly convex}. Then $f$ has at most one minimizer.
  \end{block}

  \textbf{Proof.} A strictly convex function is automatically
  strictly quasiconvex, so Proposition 11.8(i) applies with $C=\mHH$
  (where $\iC\equiv0$, so $f+\iC=f$ is strictly quasiconvex).
  \hfill$\blacksquare$

  \vspace{0.5em}
  \textbf{Checking the running example.} $f(x)=(x-3)^2+1$ is strictly
  convex on $\R$ (indeed strongly convex), so Corollary 11.9
  guarantees \emph{at most one} minimizer -- consistent with
  $\Argmin f=\{3\}$ found earlier. Uniqueness alone, however, says
  nothing about \emph{existence}; that is Section 11.4's job.

\end{frame}

\begin{frame}{11.3 Picture: Strict Convexity Is What Buys Uniqueness}
  \begin{center}
    \includegraphics[width=0.85\textwidth]{fig_unique_vs_flat}
  \end{center}
  \footnotesize Left: $f(x)=(x-3)^2+1$ is strictly convex, so
  Corollary 11.9 applies -- exactly one minimizer, $x=3$. Right: a
  merely convex (not strictly convex) function can be perfectly flat
  --- and hence constant at its minimum value --- along an entire
  interval, giving infinitely many minimizers. The distinction is not
  cosmetic: it is precisely strict convexity that Corollary 11.9
  needs.
\end{frame}

\section{11.4 Existence of Minimizers}
%==============================================================

\begin{frame}{11.4 Intuition: Why Might a Minimizer Fail to Exist?}

  \textbf{Two failure modes to keep in mind.}
  \begin{itemize}
    \item \textbf{Escape to infinity:} $f(x)=e^{-x}$ on $\R$ has
          $\inf f=0$ but no minimizer -- the near-optimal points march
          off to $+\infty$ forever.
    \item \textbf{Boundary not included:} $f(x)=x$ on the open
          interval $\left]0,1\right[$ has $\inf f=0$, not attained,
          because the infimizing point $x=0$ was excluded from the
          (non-closed) domain.
    \end{itemize}

  \textbf{The fix, informally.} Existence theorems in this section
  combine three ingredients that jointly rule out both failure modes:
  \begin{enumerate}
    \item \emph{Lower semicontinuity} -- no sudden downward jumps
          that would let the infimum ``hide'' just below a
          discontinuity;
    \item \emph{(quasi)convexity} -- turns weak convergence into a
          genuine tool via weak lower semicontinuity (Ch.\ 10);
    \item \emph{Boundedness of the relevant lower level set}
          (via \emph{coercivity}, or a bounded feasible set) --
          rules out escape to infinity, and combined with closedness
          and convexity gives \emph{weak compactness} (Ch.\ 3), on
          which a continuous (here: lsc) function attains its
          infimum.
  \end{enumerate}

\end{frame}

\begin{frame}{11.4 Theorem 11.10: The Master Existence Theorem}

  \begin{alertblock}{Theorem 11.10}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be lower
    semicontinuous and quasiconvex, and let $C$ be a closed convex
    subset of $\mHH$ such that, for some $\xi\in\R$,
    $C\cap\lev_{\le\xi}f$ is nonempty and bounded. Then $f$ has a
    minimizer over $C$.
  \end{alertblock}

  \textbf{Step-by-step proof.}
  \begin{enumerate}
    \item Since $f$ is lsc and quasiconvex, Proposition 10.25 (Ch.\
          10) upgrades this to \emph{weak} lower semicontinuity of
          $f$: whenever $x_n\rightharpoonup x$ weakly,
          $f(x)\le\varliminf f(x_n)$.
    \item Set $D=C\cap\lev_{\le\xi}f$. Since $C$ is closed convex and
          $\lev_{\le\xi}f$ is closed (by lsc of $f$) and convex (by
          quasiconvexity of $f$), $D$ is closed and convex; by
          hypothesis $D$ is also nonempty and bounded.
    \item By Theorem 3.37 (weak compactness of bounded closed convex
          sets in a Hilbert space), $D$ is \emph{weakly compact}.
    \item Minimizing $f$ over $C$ is \emph{equivalent} to minimizing
          $f$ over $D$: any point of $C$ with $f$-value $\le\xi$ lies
          in $D$, and any point outside $D$ either leaves $C$ or has
          $f$-value $>\xi\ge\inf f(D)$, so cannot be a minimizer over
          $C$ once $D\neq\emptyset$.
    \item A weakly lower semicontinuous function attains its infimum
          on a nonempty weakly compact set (Lemma 2.30 together with
          the Weierstrass-type Theorem 1.29 applied in $\mHH^{\mathrm{weak}}$).
          Hence $f$ attains $\inf f(D)=\inf f(C)$ at some point of
          $D\subseteq C$: a minimizer of $f$ over $C$ exists.
          \hfill$\blacksquare$
  \end{enumerate}

\end{frame}

\begin{frame}{11.4 Coercivity: Definitions 11.11}

  \textbf{Intuition.} Coercivity is exactly the condition that rules
  out ``escape to infinity'': $f$ must become (and stay) large once
  you go far enough from the origin, in every direction.

  \begin{block}{Definition 11.11}
    Let $f:\mHH\to[-\infty,+\infty]$. Then $f$ is \emph{coercive} if
    \[
      \lim_{\norm{x}\to+\infty}f(x)=+\infty,
    \]
    and \emph{supercoercive} if
    \[
      \lim_{\norm{x}\to+\infty}\frac{f(x)}{\norm{x}}=+\infty.
    \]
    By convention $f$ is coercive and supercoercive if $\mHH=\{0\}$.
  \end{block}

  \begin{itemize}
    \item Coercive: $f$ eventually exceeds \emph{every} fixed height
          $\xi$ as $\norm{x}\to\infty$ -- e.g.\ $f(x)=\norm{x}$ is
          coercive but not supercoercive (grows only linearly).
    \item Supercoercive: $f$ grows \emph{faster than linearly} in
          norm -- e.g.\ $f(x)=\norm{x}^2$ is supercoercive. Every
          supercoercive function is coercive, not conversely.
  \end{itemize}

\end{frame}

\begin{frame}{11.4 Propositions 11.12--11.14: Coercivity Toolkit}

  \begin{block}{Proposition 11.12}
    $f:\mHH\to[-\infty,+\infty]$ is coercive $\iff$ all its lower
    level sets $(\lev_{\le\xi}f)_{\xi\in\R}$ are bounded.
  \end{block}
  \textbf{Why:} if some level set is unbounded, you can find a
  sequence with $\norm{x_n}\to\infty$ but $f(x_n)$ staying bounded --
  exactly a failure of coercivity, and conversely.

  \begin{block}{Proposition 11.13 (finite-dimensional refinement)}
    If $\mHH$ is finite-dimensional and $f\in\GamZero$, then $f$ is
    coercive $\iff$ \emph{some single} level set $\lev_{\le\xi}f$ is
    nonempty and bounded (no need to check \emph{all} of them).
  \end{block}
  This uses recession cones (Ch.\ 6): if one bounded level set exists
  but a higher one is unbounded, the recession cone of the higher
  level set would be a nonzero direction along which $f$ stays
  bounded when added to points of the lower level set -- contradicting
  its boundedness. This finite-dimensional shortcut can fail in
  infinite dimensions (Exercise 11.12).

  \begin{block}{Proposition 11.14}
    If $f\in\GamZero$ and $g:\mHH\to\left]-\infty,+\infty\right]$ is
    supercoercive, then $f+g$ is supercoercive. (Uses that $f$ is
    minorized by a continuous affine functional, Theorem 9.20.)
  \end{block}

\end{frame}

\begin{frame}{11.4 Proposition 11.15 and Corollary 11.16}

  \begin{block}{Proposition 11.15}
    Let $f\in\GamZero$ and let $C$ be a closed convex subset of
    $\mHH$ with $C\cap\dom f\neq\emptyset$. If (i) $f$ is coercive,
    or (ii) $C$ is bounded, then $f$ has a minimizer over $C$.
  \end{block}
  \textbf{Why:} pick $x\in C\cap\dom f$; then
  $D=C\cap\lev_{\le f(x)}f$ is nonempty, closed, convex, and bounded
  (by hypothesis (i) or (ii) via Proposition 11.12), so Theorem 11.10
  applies directly.

  \begin{block}{Corollary 11.16}
    Let $f,g\in\GamZero$ with $\dom f\cap\dom g\neq\emptyset$.
    Suppose (i) $f$ is supercoercive, or (ii) $f$ is coercive and $g$
    is bounded below. Then $f+g$ is coercive and has a minimizer over
    $\mHH$. If in addition $f$ or $g$ is strictly convex, then
    $f+g$ has \emph{exactly one} minimizer.
  \end{block}
  \textbf{Why:} case (i) is Proposition 11.14 (supercoercive
  $\Rightarrow$ coercive); case (ii) follows since
  $(f+g)(x)\ge f(x)+\inf g(\mHH)\to+\infty$. Uniqueness then follows
  from Corollary 11.9 since $f+g$ is strictly convex whenever $f$ or
  $g$ is.

\end{frame}

\begin{frame}{11.4 Corollary 11.17: Strong Convexity Gives Existence \emph{and} Uniqueness}

  \begin{alertblock}{Corollary 11.17}
    Let $f\in\GamZero$ be strongly convex. Then $f$ is supercoercive
    and it has \emph{exactly one} minimizer over $\mHH$.
  \end{alertblock}

  \textbf{Step-by-step proof.}
  \begin{enumerate}
    \item By definition of $\beta$-strong convexity, $q=\tfrac12\norm{\cdot}^2$
          and there is $\beta\in\Rpp$ with $f-\beta q$ convex.
    \item Write $f=\beta q+(f-\beta q)$: a sum of the supercoercive
          function $\beta q$ (since $q(x)/\norm{x}=\norm{x}/2\to\infty$)
          and a function $f-\beta q\in\GamZero$ (proper, convex by
          construction, and lsc since $f$ and $q$ both are and $q$ is
          continuous).
    \item Proposition 11.14 then gives: $f$ is supercoercive.
    \item Supercoercive $\Rightarrow$ coercive, so Corollary 11.16(i)
          (with $g\equiv0\in\GamZero$, or directly via Proposition
          11.15(i) since $f\in\GamZero$ is coercive) gives that $f$
          has a minimizer over $\mHH$.
    \item Strong convexity $\Rightarrow$ strict convexity, so
          Corollary 11.9 gives \emph{at most one} minimizer.
    \item Combining Steps 4 and 5: $f$ has \emph{exactly one}
          minimizer. \hfill$\blacksquare$
  \end{enumerate}

  \textbf{Running example check.} $f(x)=(x-3)^2+1$ is strongly
  convex with $\beta=2$ on $\R$: Corollary 11.17 alone -- without any
  hand computation -- already guarantees a unique minimizer exists.
  We found it explicitly above: $\bar x=3$.

\end{frame}

\begin{frame}{11.4 Proposition 11.18: The Asymptotic Center (Overview)}

  \textbf{Why include this.} It is a genuinely useful application of
  the existence \emph{and} uniqueness machinery above, feeding
  directly into fixed-point theory (Fej\'er monotonicity, KM
  iterations from earlier chapters).

  \begin{block}{Proposition 11.18 (Asymptotic center), key points}
    Let $(z_n)_{n\in\mathbb N}$ be bounded in $\mHH$, let $C$ be
    nonempty closed convex, let $T:C\to C$ be nonexpansive, and set
    $f:\mHH\to\R:x\mapsto\varlimsup\norm{x-z_n}^2$. Then:
    \begin{itemize}
      \item $f$ is strongly convex with constant $2$, hence (by
            Cor.\ 11.17 logic) supercoercive;
      \item $f+\iC$ is strongly convex and supercoercive; its unique
            minimizer $z_C$ (over $C$) is the \emph{asymptotic center}
            of $(z_n)_{n\in\mathbb N}$ relative to $C$;
      \item if $z_n\rightharpoonup z$ weakly, then $z_C=P_Cz$ (the
            metric projection of the weak limit);
      \item if $(z_n)_{n\in\mathbb N}$ is Fej\'er monotone w.r.t.\
            $C$, then $P_Cz_n\to z_C$;
      \item if $z_{n+1}=Tz_n$ for all $n$, or if $z_n-Tz_n\to0$, then
            $z_C\in\operatorname{Fix}T$.
    \end{itemize}
  \end{block}

  \textbf{Punchline.} Existence + uniqueness of a minimizer
  (guaranteed abstractly by strong convexity, Cor.\ 11.17) is enough
  to \emph{define} a canonical point (the asymptotic center)
  attached to any bounded sequence -- a genuinely new object built
  entirely from this chapter's tools.

\end{frame}

\begin{frame}{11.4 Corollary 11.19: A Boundedness Equivalence}

  \begin{block}{Corollary 11.19}
    Let $C$ be nonempty closed convex, $T:C\to C$ nonexpansive, and
    for $z_0\in C$ set $z_{n+1}=Tz_n$. The following are equivalent:
    (i) $\operatorname{Fix}T\neq\emptyset$; (ii) for \emph{every}
    $z_0\in C$, $(z_n)_{n\in\mathbb N}$ is bounded; (iii) for
    \emph{some} $z_0\in C$, $(z_n)_{n\in\mathbb N}$ is bounded.
  \end{block}

  \textbf{Why:} (i)$\Rightarrow$(ii) is standard Fej\'er-monotonicity
  theory (Ch.\ 5); (ii)$\Rightarrow$(iii) is trivial; and
  (iii)$\Rightarrow$(i) is exactly Proposition 11.18(vi): if the
  orbit is bounded, the asymptotic center construction manufactures
  an actual fixed point of $T$. This is a clean illustration of how
  an \emph{existence-of-minimizer} statement (Cor.\ 11.17, hidden
  inside Prop.\ 11.18) becomes an \emph{existence-of-fixed-point}
  statement.

\end{frame}

\section{11.5 Minimizing Sequences}
%==============================================================

\begin{frame}{11.5 Intuition: Life Without (Yet) Knowing a Minimizer}

  \textbf{Recall (Definition 1.8).} A \emph{minimizing sequence} of a
  proper $f:\mHH\to\left]-\infty,+\infty\right]$ is any sequence
  $(x_n)_{n\in\mathbb N}$ in $\dom f$ with $f(x_n)\to\inf f(\mHH)$.

  \textbf{Why this matters practically.} Numerical algorithms
  \emph{produce} minimizing sequences (e.g.\ gradient steps,
  proximal iterations) long before -- if ever -- they land on an
  exact minimizer. Section 11.5 asks: under what conditions does a
  minimizing sequence actually \emph{converge}, and does it converge
  to a minimizer?

  \begin{alertblock}{Spoiler: things can go wrong in several ways}
    A minimizing sequence can (a) be unbounded, (b) be bounded but
    have no weak cluster point behave nicely, (c) converge weakly but
    not strongly, or (d) converge to a point that fails to be a
    minimizer at all if the corresponding hypotheses are dropped. The
    examples below (11.23--11.28) are a guided tour of these
    pathologies, and Propositions 11.20, 11.21, 11.29--11.31 are the
    positive results that rule them out under suitable assumptions.
  \end{alertblock}

\end{frame}

\begin{frame}{11.5 Propositions 11.20--11.21: First Positive Results}

  \begin{block}{Proposition 11.20}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be proper and
    coercive. Then every minimizing sequence of $f$ is bounded.
  \end{block}
  \textbf{Why:} immediate from Definition 1.8 (minimizing sequences
  eventually enter every lower level set at height close to
  $\inf f$) plus Proposition 11.12 (coercive $\iff$ bounded level
  sets).

  \begin{block}{Proposition 11.21}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be proper, lsc, and
    quasiconvex, and let $(x_n)_{n\in\mathbb N}$ be a minimizing
    sequence that converges \emph{weakly} to some $x\in\mHH$. Then
    $f(x)=\min f(\mHH)$: the weak limit is automatically a minimizer.
  \end{block}
  \textbf{Why:} Proposition 10.25 (Ch.\ 10) gives weak sequential
  lower semicontinuity of $f$ under these hypotheses, so
  $\inf f(\mHH)\le f(x)\le\varliminf f(x_n)=\inf f(\mHH)$: squeeze.

  \textbf{The catch these two leave open.} Coercivity guarantees
  \emph{boundedness}, not weak \emph{convergence}; and even when a
  minimizing sequence does converge weakly, Remark 11.22 below shows
  the target need not be reached \emph{strongly}.

\end{frame}

\begin{frame}{11.5 Remark 11.22: Strong Convergence Is Delicate}

  \begin{block}{Remark 11.22}
    Let $f\in\GamZero$ and $(x_n)_{n\in\mathbb N}$ in $\dom f$.
    \begin{enumerate}
      \item[(i)] If $(x_n)_{n\in\mathbb N}$ converges \emph{strongly}
            to a minimizer $x$ of $f$, and either $\mHH=\R$ or
            $x\in\intr\dom f$, then $(x_n)_{n\in\mathbb N}$ \emph{is}
            a minimizing sequence (continuity of $f$ near $x$ forces
            $f(x_n)\to f(x)=\inf f(\mHH)$).
      \item[(ii)] Strong convergence to a minimizer does \emph{not}
            by itself imply the sequence is minimizing, in general
            (a counterexample uses the construction behind Example
            9.43).
      \item[(iii)] In \emph{infinite}-dimensional $\mHH$, even weak
            convergence of $(x_n)_{n\in\mathbb N}$ to a minimizer $x$
            with $x\in\intr\dom f$ need not make $(x_n)_{n\in\mathbb N}$
            minimizing: take an orthonormal sequence $x_n$ and
            $f=\norm{\cdot}$; then $x_n\rightharpoonup 0$,
            $f(0)=0=\inf f(\mHH)$, yet $f(x_n)\equiv1$ for every $n$.
    \end{enumerate}
  \end{block}

  \textbf{Moral.} Finite-dimensionality and interior-of-domain
  conditions are doing real work in (i); drop either one and strange
  things can happen, as (ii)--(iii) show concretely.

\end{frame}

\begin{frame}{11.5 A Guided Tour of Pathologies: Examples 11.23--11.26}

  \textbf{Setting for all of these:} $\mHH=\R^2$ unless noted.

  \begin{block}{Example 11.23 -- rate of value convergence vs.\ distance to Argmin}
    $f(\xi_1,\xi_2)=\xi_2^2/\xi_1$ on $\xi_1>0$, $0$ at the origin,
    $+\infty$ otherwise; $\Argmin f=\Rp\times\{0\}$. Taking
    $x_n=((n+1)^{p+2},n+1)$ for fixed $p\ge1$ gives
    $f(x_n)-\min f=O(1/n^p)\to0$ (minimizing), yet the distance from
    $x_n$ to $\Argmin f$ still $\to+\infty$: values can converge to
    the infimum arbitrarily fast while the \emph{iterates} drift
    farther and farther from the minimizer set.
  \end{block}

  \begin{block}{Example 11.24 -- similar drift for a simpler function}
    $f(\xi_1,\xi_2)=\xi_1+\norm{(\xi_1,\xi_2)}$, $\Argmin f=\R_-\times\{0\}$.
    With $x_n=(-n,1)$, $f(x_n)-\min f\to0$ but the distance to
    $\Argmin f$ stays $\equiv1$ (does not even shrink to $0$).
  \end{block}

  \begin{block}{Example 11.26 -- bounded but not weakly convergent}
    $f=\iota_{[-x,x]}$ (for fixed $x\neq0$), $x_n=(-1)^nx$: a bounded
    minimizing sequence that has \emph{two} distinct cluster points
    and hence does not converge weakly at all.
  \end{block}

\end{frame}

\begin{frame}{11.5 Examples 11.25 and 11.27: Weak but Not Strong Convergence}

  \begin{block}{Example 11.25}
    $f(\xi_1,\xi_2)=\xi_1\ln(\xi_1/\xi_2)-\xi_1+\xi_2$ on
    $\R_{++}^2$, $\xi_2$ on $\{0\}\times\Rp$, $+\infty$ otherwise;
    $f\in\GamZero$, $\min f(\mHH)=0$, $\Argmin f=\Rp(1,1)$. Along
    $\varepsilon_n\downarrow0$: $x_n=(\varepsilon_n,\varepsilon_n)\to(0,0)$
    with $f(x_n)\equiv0$ (minimizing); but
    $y_n=(\varepsilon_n,e^{-1/\varepsilon_n})\to(0,0)$ has
    $f(y_n)\to1\neq0$, and $z_n=(\varepsilon_n,e^{-1/\varepsilon_n^2})\to(0,0)$
    has $f(z_n)\to+\infty$ -- \emph{three} sequences converging to the
    \emph{same} point of $\Argmin f$, with wildly different limiting
    values of $f$.
  \end{block}

  \begin{block}{Example 11.27 -- infinite dimensions, orthonormal basis}
    $\mHH$ infinite-dimensional separable, $(x_k)_{k\in\mathbb N}$ an
    orthonormal basis, $\omega_k\downarrow0$, and
    $f(x)=\sum_k\omega_k\inner{x}{x_k}^2$: real-valued, continuous,
    strictly convex, unique minimizer $0$, but \emph{not coercive}
    (rescaling along the basis, $y_n=x_n/\sqrt{\omega_n}$, gives
    $\norm{y_n}\to\infty$ while $f(y_n)\equiv1$). The minimizing
    sequence $x_n$ converges weakly but \emph{not strongly} to $0$.
  \end{block}

\end{frame}

\begin{frame}{11.5 Example 11.28: Alternating Minimization Can Miss the Target}

  \begin{block}{Example 11.28}
    On $\mHH=\R^2$, let
    $f(\xi_1,\xi_2)=\max\{2\xi_1-\xi_2,\,2\xi_2-\xi_1\}+\iota_{\Rp^2}(\xi_1,\xi_2)$.
    Then $f\in\GamZero$, coercive, unique minimizer $\bar x=(0,0)$,
    $\min f(\mHH)=0$. Starting from $x_0\in\Rp\times\Rpp$, define
    $x_{n+1}=(\xi_{1,n+1},\xi_{2,n+1})$ by first minimizing
    $f(\cdot,\xi_{2,n})$ over $\R$ to get $\xi_{1,n+1}$, then
    minimizing $f(\xi_{1,n+1},\cdot)$ over $\R$ to get $\xi_{2,n+1}$
    (\emph{alternating}, coordinate-by-coordinate minimization).
  \end{block}

  \textbf{The surprising outcome.} For every $n\ge1$,
  $x_n=(\xi_{2,0},\xi_{2,0})\neq\bar x$ and
  $f(x_n)=\xi_{2,0}\neq\inf f(\mHH)$: the iterates get \emph{stuck}
  immediately (after one full cycle) at a \emph{non}-optimal point
  and never move again -- alternating minimization, applied naively
  coordinate-by-coordinate to a merely convex (not separately strictly
  convex in a useful sense) function, need not converge to the true
  minimizer, and does not even produce a minimizing sequence here.

  \textbf{Takeaway.} Coercivity + convexity + lsc guarantee a
  minimizer \emph{exists} (Theorem 11.10); they say nothing about
  whether a \emph{particular algorithm} finds it.

\end{frame}

\begin{frame}{11.5 Proposition 11.29: Weak and Strong Convergence Conditions}

  \begin{alertblock}{Proposition 11.29}
    Let $f:\mHH\to\left]-\infty,+\infty\right]$ be proper, lsc, and
    quasiconvex, and let $(x_n)_{n\in\mathbb N}$ be a minimizing
    sequence such that, for some $\xi\in\left]\inf f(\mHH),+\infty\right[$,
    $C=\lev_{\le\xi}f$ is bounded. Then:
    \begin{enumerate}
      \item[(i)] $(x_n)_{n\in\mathbb N}$ has a weak sequential cluster
            point, and every such point is a minimizer of $f$.
      \item[(ii)] If $f+\iC$ is \emph{strictly} quasiconvex, $f$
            possesses a unique minimizer $x$ and $x_n\rightharpoonup x$.
      \item[(iii)] If $f+\iC$ is \emph{uniformly} quasiconvex, $f$
            possesses a unique minimizer $x$ and $x_n\to x$
            (strongly).
    \end{enumerate}
  \end{alertblock}

  \textbf{Proof idea for (iii), the strongest conclusion.} Let $\phi$
  be the modulus of uniform quasiconvexity of $f+\iC$. A direct
  estimate along the segment from $x_n$ to the (already known, by
  (ii)) unique minimizer $x$ gives, for $\alpha\in\left]0,1\right[$,
  \[
    \inf f(\mHH)+\alpha(1-\alpha)\phi(\norm{x_n-x})\ \le\ f\bigl(\alpha x_n+(1-\alpha)x\bigr)\ \le\ \max\{f(x_n),f(x)\}=f(x_n).
  \]
  Since $f(x_n)\to f(x)=\inf f(\mHH)$, this forces
  $\phi(\norm{x_n-x})\to0$, and since $\phi$ vanishes only at $0$ and
  is increasing, $\norm{x_n-x}\to0$. \hfill$\blacksquare$

\end{frame}

\begin{frame}{11.5 Corollary 11.30 and Proposition 11.31}

  \begin{block}{Corollary 11.30 (coercive case, $C=\mHH$)}
    Let $f\in\GamZero$ be coercive and $(x_n)_{n\in\mathbb N}$ a
    minimizing sequence. Then (i) $(x_n)_{n\in\mathbb N}$ has a weak
    cluster point, every one a minimizer; (ii) if $f$ is strictly
    convex, $f$ has a unique minimizer $x$ and $x_n\rightharpoonup x$;
    (iii) if $f$ is uniformly convex on every bounded subset of
    $\dom f$, $f$ has a unique minimizer $x$ and $x_n\to x$.
  \end{block}
  This is Proposition 11.29 specialized to $C=\mHH$, using coercivity
  (Prop.\ 11.12) in place of an assumed bounded level set.

  \begin{block}{Proposition 11.31 (a genuinely different mechanism: uniform convexity of $C$, not $f$)}
    Let $f\in\GamZero$, let $C$ be bounded closed convex with
    $C\cap\dom f\neq\emptyset$ and $C\cap\Argmin f=\emptyset$, and
    suppose $C$ itself is \emph{uniformly convex} (a quantitative
    strengthening of strict convexity of the \emph{set}, not the
    function). Then any minimizing sequence of $f+\iC$ converges
    \emph{strongly} to the (unique) minimizer of $f$ over $C$.
  \end{block}
  \textbf{Why it is different:} here $f$ itself need not be strictly
  or uniformly convex at all -- the strong convergence is bought
  entirely by geometric uniform convexity of the constraint set $C$
  (e.g.\ any closed ball is uniformly convex, Exercise 11.18).

\end{frame}

\begin{frame}[fragile]{Python: Minimizers, Minimizing Sequences, and Coercivity}
\begin{lstlisting}[caption={Numeric check of the running example $f(x)=(x-3)^2+1$}]
import numpy as np

def f(x):
    """Strongly convex running example: f(x) = (x-3)^2 + 1."""
    return (x - 3.0) ** 2 + 1.0

# --- Unconstrained minimizer over R (Section 11.2/11.4) ---
xs = np.linspace(-50, 50, 2_000_001)
i_star = np.argmin(f(xs))
print("argmin over R ~", xs[i_star], " min value ~", f(xs[i_star]))
# argmin over R ~ 3.0   min value ~ 1.0   (exact: x=3, f(x)=1)

# --- Constrained minimizer over C = [0, 2] (Prop. 11.4/11.5) ---
xc = np.linspace(0.0, 2.0, 200_001)
j_star = np.argmin(f(xc))
print("argmin over C=[0,2] ~", xc[j_star], " min value ~", f(xc[j_star]))
# argmin over C=[0,2] ~ 2.0   min value ~ 2.0   (attained at bdry C)

# --- Coercivity check (Definition 11.11 / Proposition 11.12) ---
# f is coercive iff f(x) -> +infty as |x| -> infty, equivalently
# iff every lower level set {x : f(x) <= xi} is bounded.
for xi in [1.5, 5.0, 50.0]:
    level_set = xs[f(xs) <= xi]
    bounded = (level_set.size == 0) or (np.ptp(level_set) < np.inf)
    lo, hi = (level_set.min(), level_set.max()) if level_set.size else (None, None)
    print(f"xi={xi:6.1f}: lev_<=xi f = [{lo}, {hi}]  (bounded: {bounded})")

# --- A minimizing sequence x_n = 3 - 1/n (Definition 1.8, Section 11.5) ---
n = np.arange(1, 11)
xn = 3.0 - 1.0 / n
fn = f(xn)
print("\n n    x_n       f(x_n)      f(x_n) - inf f")
for k in range(len(n)):
    print(f"{n[k]:2d}  {xn[k]: .6f}  {fn[k]: .6f}   {fn[k]-1.0: .6f}")
# f(x_n) decreases monotonically to 1 = inf f(R); x_n -> 3 strongly,
# exactly as Corollary 11.30(iii) predicts for a strongly convex f.
\end{lstlisting}
\end{frame}

  \begin{block}{What we learned}
    \begin{itemize}
      \item \textbf{11.1 Infima and suprema:} $\inf f(C)$ is always
            defined, but need not be attained; for convex $f$ with
            $\dom f\cap\intr C\neq\emptyset$, closing up $C$ does not
            change the infimum (Prop.\ 11.1(iv)).
      \item \textbf{11.2 Minimizers:} a minimizer is a \emph{point}
            achieving the infimum (Def.\ 11.3); for convex $f$, every
            \emph{local} minimizer is automatically \emph{global}
            (Prop.\ 11.4), and an interior-to-$C$ minimizer over $C$
            is a global minimizer of $f$ (Prop.\ 11.5).
      \item \textbf{11.3 Uniqueness:} strict convexity of $f$
            (Cor.\ 11.9), or strict convexity of $C$ together with
            $(\intr C)\cap\Argmin f=\emptyset$ (Prop.\ 11.8(ii)),
            each force at most one minimizer over $C$.
      \item \textbf{11.4 Existence:} lsc + quasiconvex $f$ with some
            bounded, nonempty $C\cap\lev_{\le\xi}f$ guarantees a
            minimizer over $C$ (Theorem 11.10); coercivity is the key
            boundedness-of-level-sets ingredient (Prop.\ 11.12), and
            \emph{strong} convexity buys both existence and
            uniqueness at once, via supercoercivity (Cor.\ 11.17).
      \item \textbf{11.5 Minimizing sequences:} coercivity keeps them
            bounded (Prop.\ 11.20); lsc + quasiconvexity makes weak
            limits automatically minimizers (Prop.\ 11.21); but
            Examples 11.23--11.28 show strong/weak convergence and
            even convergence of the algorithm's iterates to the
            minimizer set can all fail without extra structure
            (strict/uniform convexity of $f$ or of $C$, Props.\
            11.29--11.31).
      \item \textbf{Running example:} $f(x)=(x-3)^2+1$ has unique
            minimizer $3$ (value $1$) over $\R$, and unique minimizer
            $2$ (value $2$, at the boundary) over $C=[0,2]$ -- one
            concrete function witnessing every abstract theorem above.
    \end{itemize}
  \end{block}
