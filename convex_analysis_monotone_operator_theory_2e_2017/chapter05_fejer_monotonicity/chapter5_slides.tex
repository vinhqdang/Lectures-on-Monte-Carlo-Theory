\section{5.1 Fejér Monotone Sequences}

\begin{frame}{Definition: Fejér Monotone Sequences}
    \begin{definition}[Fejér Monotonicity]
        Let $C$ be a nonempty subset of $\mathcal{H}$ and $(x_n)_{n \in \mathbb{N}}$ a sequence in $\mathcal{H}$.
        The sequence $(x_n)_{n \in \mathbb{N}}$ is \textbf{Fejér monotone} with respect to $C$ if
        \[
            (\forall x \in C)(\forall n \in \mathbb{N}) \quad \|x_{n+1} - x\| \leq \|x_n - x\|.
        \]
    \end{definition}

    \vspace{0.3cm}

    \textbf{Intuition}:
    \begin{itemize}
        \item Distance to \emph{every} point in $C$ is decreasing
        \item Sequence approaches the set $C$ uniformly
        \item Powerful convergence tool without explicit limit information
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Example}: Orthonormal sequence $(e_n)$ in infinite-dimensional space is Fejér monotone w.r.t. $\{0\}$.
\end{frame}

\begin{frame}{Key Properties of Fejér Monotone Sequences}
    \begin{theorem}[Proposition 5.4]
        Let $C$ be a nonempty subset of $\mathcal{H}$ and $(x_n)_{n \in \mathbb{N}}$ be Fejér monotone with respect to $C$.
        Then:
        \begin{enumerate}
            \item $(x_n)_{n \in \mathbb{N}}$ is bounded.
            \item For every $x \in C$, $(\|x_n - x\|)_{n \in \mathbb{N}}$ converges.
            \item $(d_C(x_n))_{n \in \mathbb{N}}$ is decreasing and converges.
            \item $\|x_{n+m} - x_n\| \leq 2d_C(x_n)$ for all $m, n \in \mathbb{N}$.
        \end{enumerate}
    \end{theorem}

    \vspace{0.3cm}

    \textbf{Consequence}: Distance function decreases monotonically, making strong convergence possible.
\end{frame}

\begin{frame}{Weak Convergence Result}
    \begin{theorem}[Theorem 5.5]
        Let $C$ be a nonempty subset of $\mathcal{H}$ and $(x_n)_{n \in \mathbb{N}}$ be a sequence in $\mathcal{H}$.
        Suppose that $(x_n)_{n \in \mathbb{N}}$ is Fejér monotone with respect to $C$ and every weak sequential cluster point of $(x_n)_{n \in \mathbb{N}}$ belongs to $C$.
        Then $(x_n)_{n \in \mathbb{N}}$ converges weakly to a point in $C$.
    \end{theorem}

    \vspace{0.3cm}

    \textbf{Proof Idea}:
    \begin{itemize}
        \item Bounded sequence has weak cluster points (Banach-Alaoglu)
        \item All cluster points lie in $C$
        \item Decreasing distance to $C$ forces unique cluster point
    \end{itemize}
\end{frame}

\begin{frame}{Strong Convergence Under Interior Condition}
    \begin{theorem}[Proposition 5.10 - Raik]
        Let $C$ be a subset of $\mathcal{H}$ with $\text{int } C \neq \emptyset$ and $(x_n)_{n \in \mathbb{N}}$ be a sequence in $\mathcal{H}$ that is Fejér monotone with respect to $C$.
        Then $(x_n)_{n \in \mathbb{N}}$ converges strongly to a point in $\mathcal{H}$ and
        \[
            \sum_{n \in \mathbb{N}} \|x_{n+1} - x_n\| < +\infty.
        \]
    \end{theorem}

    \vspace{0.3cm}

    \textbf{Key Insight}: When $C$ has interior, Fejér monotonicity implies strong convergence!
\end{frame}

\begin{frame}{Convergence to Projections}
    \begin{corollary}[Corollary 5.8]
        Let $C$ be a nonempty closed convex subset of $\mathcal{H}$, $(x_n)_{n \in \mathbb{N}}$ be a sequence in $\mathcal{H}$, and $x \in C$. Suppose that $(x_n)_{n \in \mathbb{N}}$ is Fejér monotone with respect to $C$ and $x_n \to x$.
        Then $P_C x_n \to x$.
    \end{corollary}

    \vspace{0.3cm}

    \begin{theorem}[Theorem 5.12 - Linear Convergence]
        Let $C$ be a nonempty closed convex subset of $\mathcal{H}$, $(x_n)_{n \in \mathbb{N}}$ be Fejér monotone with respect to $C$, and $\kappa \in [0, 1[$ such that
        \[
            (\forall n \in \mathbb{N}) \quad d_C(x_{n+1}) \leq \kappa d_C(x_n).
        \]
        Then $(x_n)_{n \in \mathbb{N}}$ converges linearly to a point in $C$:
        \[
            (\forall n \in \mathbb{N}) \quad \|x_n - x\| \leq 2\kappa^n d_C(x_0).
        \]
    \end{theorem}
\end{frame}

\begin{frame}{Affine Subspace Specialization}
    \begin{theorem}[Proposition 5.9]
        Let $C$ be a closed affine subspace of $\mathcal{H}$ and $(x_n)_{n \in \mathbb{N}}$ be a sequence in $\mathcal{H}$ that is Fejér monotone with respect to $C$.
        Then:
        \begin{enumerate}
            \item $(\forall n \in \mathbb{N}) P_C x_n = P_C x_0$.
            \item If every weak sequential cluster point of $(x_n)_{n \in \mathbb{N}}$ belongs to $C$, then $x_n \to P_C x_0$.
        \end{enumerate}
    \end{theorem}

    \vspace{0.3cm}

    \textbf{Proof Sketch}:
    \begin{itemize}
        \item Use Corollary 3.22(i) on affine subspaces
        \item Fejér monotonicity restricts to $C$
        \item Projection is constant along Fejér sequence
    \end{itemize}
\end{frame}

\begin{frame}{Firmly Quasinonexpansive Operators}
    \begin{theorem}[Proposition 5.13]
        Let $(T_n)_{n \in \mathbb{N}}$ be a family of firmly quasinonexpansive operators from $\mathcal{H}$ to $\mathcal{H}$ with $C = \bigcap_{n \in \mathbb{N}} \text{Fix} T_n \neq \emptyset$, and $(\lambda_n)_{n \in \mathbb{N}}$ be a sequence in $]0, 2[$.
        Set $x_0 \in \mathcal{H}$ and
        \[
            (\forall n \in \mathbb{N}) \quad x_{n+1} = x_n + \lambda_n(T_n x_n - x_n).
        \]
        Then:
        \begin{enumerate}
            \item $(\forall x \in C)(\forall n \in \mathbb{N}) \quad \|x_{n+1} - x\|^2 \leq \|x_n - x\|^2 - \lambda_n(2 - \lambda_n)\|T_n x_n - x_n\|^2$.
            \item $(x_n)_{n \in \mathbb{N}}$ is Fejér monotone with respect to $C$.
        \end{enumerate}
    \end{theorem}

    \vspace{0.2cm}

    \textbf{Application}: Projection operators onto closed convex sets are firmly nonexpansive!
\end{frame}

\section{5.2 Krasnosel'skii-Mann Iteration}

\begin{frame}{The Krasnosel'skii-Mann Algorithm}
    \begin{theorem}[Theorem 5.14]
        Let $D$ be a nonempty closed convex subset of $\mathcal{H}$, let $T: D \to D$ be a nonexpansive operator with $\text{Fix} T \neq \emptyset$, and $x_0 \in D$.
        Set
        \[
            (\forall n \in \mathbb{N}) \quad x_{n+1} = T x_n
        \]
        and suppose $x_n - T x_n \to 0$. Then:
        \begin{enumerate}
            \item $(x_n)_{n \in \mathbb{N}}$ converges weakly to a point in $\text{Fix} T$.
            \item If $D = -D$ and $T$ is odd, then $(x_n)_{n \in \mathbb{N}}$ converges strongly to a point in $\text{Fix} T$.
        \end{enumerate}
    \end{theorem}

    \vspace{0.3cm}

    \textbf{Key Issue}: Direct Banach-Picard iteration $x_{n+1} = Tx_n$ may not satisfy $x_n - Tx_n \to 0$ for nonexpansive $T$.
\end{frame}

\begin{frame}{Groetsch's Theorem (Krasnosel'skii-Mann Relaxation)}
    \begin{theorem}[Theorem 5.15 - Groetsch]
        Let $D$ be a nonempty closed convex subset of $\mathcal{H}$, let $T: D \to D$ be a nonexpansive operator with $\text{Fix} T \neq \emptyset$, let $(\lambda_n)_{n \in \mathbb{N}}$ be a sequence in $[0, 1]$ such that $\sum_{n \in \mathbb{N}} \lambda_n(1 - \lambda_n) = +\infty$, and $x_0 \in D$.
        Set
        \[
            (\forall n \in \mathbb{N}) \quad x_{n+1} = x_n + \lambda_n(Tx_n - x_n).
        \]
        Then:
        \begin{enumerate}
            \item $(x_n)_{n \in \mathbb{N}}$ is Fejér monotone with respect to $\text{Fix} T$.
            \item $(Tx_n - x_n)_{n \in \mathbb{N}}$ converges strongly to $0$.
            \item $(x_n)_{n \in \mathbb{N}}$ converges weakly to a point in $\text{Fix} T$.
        \end{enumerate}
    \end{theorem}

    \vspace{0.3cm}

    \textbf{Stepsize Condition}: $\sum \lambda_n(1 - \lambda_n) = \infty$ ensures asymptotic regularity.
\end{frame}

# Find fixed point via KM iteration
import numpy as np

def km_iteration(T, x0, lambda_n, N=100):
    x = x0
    history = [x0]
    residuals = []
    
    for n in range(N):
        Tx = T(x)
        residuals.append(np.linalg.norm(x - Tx))
        
        x = x + lambda_n[n] * (Tx - x)
        history.append(x)
    
    return np.array(history), residuals

# Example: T(x) = 0.6*x + 0.1
T = lambda x: 0.6*x + 0.1
x0 = 2.0
lambda_n = [0.5] * 100

history, residuals = km_iteration(T, x0, lambda_n)

\begin{frame}{Averaged Operators}
    \begin{definition}
        An operator $T: \mathcal{H} \to \mathcal{H}$ is \textbf{$\alpha$-averaged} if there exists a nonexpansive operator $R$ and $\alpha \in ]0, 1[$ such that
        \[
            T = (1 - \alpha) \text{Id} + \alpha R.
        \]
    \end{definition}

    \vspace{0.3cm}

    \begin{theorem}[Proposition 5.16]
        Let $\alpha \in ]0, 1[$, let $T: \mathcal{H} \to \mathcal{H}$ be an $\alpha$-averaged operator with $\text{Fix} T \neq \emptyset$, let $(\lambda_n)_{n \in \mathbb{N}}$ be a sequence in $[0, 1/\alpha]$ such that $\sum_{n \in \mathbb{N}} \lambda_n(1 - \alpha\lambda_n) = +\infty$, and $x_0 \in \mathcal{H}$.
        Then:
        \begin{enumerate}
            \item $(x_n)_{n \in \mathbb{N}}$ is Fejér monotone with respect to $\text{Fix} T$.
            \item $(Tx_n - x_n)_{n \in \mathbb{N}}$ converges strongly to $0$.
            \item $(x_n)_{n \in \mathbb{N}}$ converges weakly to a point in $\text{Fix} T$.
        \end{enumerate}
    \end{theorem}
\end{frame}

\section{5.3 Iterating Compositions of Averaged Operators}

\begin{frame}{String-Averaged Relaxed Projections}
    \begin{example}[Example 5.21]
        Let $(C_i)_{i \in I}$ be a finite family of closed convex sets with $C = \bigcap_{i \in I} C_i \neq \emptyset$.
        For every $i \in I$, let $\beta_i \in [0, 2[$ and set $T_i = (1 - \beta_i)\text{Id} + \beta_i P_{C_i}$.
        Set $x_0 \in \mathcal{H}$ and
        \[
            (\forall n \in \mathbb{N}) \quad x_{n+1} = \sum_{k=1}^p \omega_k T_{i(k,1)} \cdots T_{i(k,m_k)} x_n,
        \]
        with weights $\omega_k > 0$ and $\sum_k \omega_k = 1$.
        Then $(x_n)_{n \in \mathbb{N}}$ converges weakly to a point in $C$.
    \end{example}

    \vspace{0.3cm}

    \textbf{Key Ideas}:
    \begin{itemize}
        \item Combine projections onto multiple sets with relaxation
        \item String-averaged uses compositions
        \item Convex combinations of different orderings
    \end{itemize}
\end{frame}

\begin{frame}{POCS: Projections Onto Convex Sets}
    \begin{theorem}[Theorem 5.23]
        Let $D$ be a nonempty closed convex subset of $\mathcal{H}$, let $m$ be a strictly positive integer, set $I = \{1, \ldots, m\}$, let $(T_i)_{i \in I}$ be a family of nonexpansive operators from $D$ to $D$ with $\text{Fix}(T_1 \cdots T_m) \neq \emptyset$.
        Set $x_0 \in D$ and
        \[
            (\forall n \in \mathbb{N}) \quad x_{n+1} = T_1 \cdots T_m x_n.
        \]
        Then $x_n - T_1 \cdots T_m x_n \to 0$ and there exist $y_1 \in \text{Fix} T_1 \cdots T_m$, etc., such that
        convergence relations hold (see text).
    \end{theorem}

    \vspace{0.3cm}

    \textbf{Application}: POCS for finding intersection $C = \bigcap_i C_i$ via cyclic projections.
\end{frame}

\begin{frame}{Corollary 5.24: POCS Algorithm}
    \begin{corollary}[POCS Algorithm]
        Let $m$ be strictly positive, set $I = \{1, \ldots, m\}$, let $(C_i)_{i \in I}$ be a family of nonempty closed convex subsets of $\mathcal{H}$, let $(P_i)_{i \in I}$ denote their respective projectors, and $x_0 \in \mathcal{H}$.
        Suppose the intersection is nonempty and set
        \[
            (\forall n \in \mathbb{N}) \quad x_{n+1} = P_1 \cdots P_m x_n.
        \]
        Then the sequence converges weakly to the intersection.
    \end{corollary}

    \vspace{0.3cm}

    \textbf{Convergence}: Weak convergence to intersection when it exists.
\end{frame}

\begin{frame}{Bregman and Von Neumann-Halperin Algorithms}
    \begin{corollary}[Corollary 5.26 - Bregman]
        Let $m$ be strictly positive, set $I = \{1, \ldots, m\}$, let $(C_i)_{i \in I}$ be a family of closed convex subsets of $\mathcal{H}$ with $C = \bigcap_i C_i \neq \emptyset$.
        The cyclic projection sequence $x_{n+1} = P_1 \cdots P_m x_n$ converges weakly to a point in $C$.
    \end{corollary}

    \vspace{0.3cm}

    \begin{corollary}[Corollary 5.30 - von Neumann-Halperin]
        Let $m$ be strictly positive, set $I = \{1, \ldots, m\}$, let $(C_i)_{i \in I}$ be a family of closed affine subspaces of $\mathcal{H}$ with $C = \bigcap_i C_i \neq \emptyset$.
        Then $x_n \to P_C x_0$ (strong convergence).
    \end{corollary}

    \vspace{0.3cm}
    
    \textbf{Key Insight}: For affine subspaces, we get strong convergence instead of weak!
\end{frame}

\section{5.4 Quasi-Fejér Monotone Sequences}

\begin{frame}{Definition: Quasi-Fejér Monotonicity}
    \begin{definition}[Definition 5.32]
        Let $C$ be a nonempty subset of $\mathcal{H}$ and $(x_n)_{n \in \mathbb{N}}$ a sequence in $\mathcal{H}$.
        Then $(x_n)_{n \in \mathbb{N}}$ is \textbf{quasi-Fejér monotone} with respect to $C$ if
        \[
            (\forall x \in C)(\exists (\varepsilon_n)_{n \in \mathbb{N}} \in \ell_+^1(\mathbb{N}))(\forall n \in \mathbb{N}) \quad \|x_{n+1} - x\|^2 \leq \|x_n - x\|^2 + \varepsilon_n.
        \]
    \end{definition}

    \vspace{0.3cm}

    \textbf{Key Difference}: Allows small errors $\varepsilon_n$ that sum to a finite value.

    \vspace{0.3cm}

    \begin{theorem}[Theorem 5.33]
        Let $(x_n)_{n \in \mathbb{N}}$ be quasi-Fejér monotone with respect to $C$.
        Then:
        \begin{enumerate}
            \item For every $x \in C$, $(\|x_n - x\|)_{n \in \mathbb{N}}$ converges.
            \item $(x_n)_{n \in \mathbb{N}}$ is bounded.
            \item $(d_C(x_n))_{n \in \mathbb{N}}$ converges.
            \item If every weak cluster point belongs to $C$, then $(x_n)_{n \in \mathbb{N}}$ converges weakly.
        \end{enumerate}
    \end{theorem}
\end{frame}

\begin{frame}{Quasi-Fejér Analysis with Perturbations}
    \begin{theorem}[Proposition 5.34]
        Let $T: \mathcal{H} \to \mathcal{H}$ be nonexpansive with $\text{Fix} T \neq \emptyset$, let $x_0 \in \mathcal{H}$, and let $(\lambda_n)_{n \in \mathbb{N}}$, $(e_n)_{n \in \mathbb{N}}$ satisfy $\sum_{n \in \mathbb{N}} \lambda_n(1 - \lambda_n) = +\infty$ and $\sum_{n \in \mathbb{N}} \lambda_n \|e_n\| < +\infty$.
        Set
        \[
            (\forall n \in \mathbb{N}) \quad x_{n+1} = x_n + \lambda_n(Tx_n + e_n - x_n).
        \]
        Then:
        \begin{enumerate}
            \item $(x_n)_{n \in \mathbb{N}}$ is quasi-Fejér monotone with respect to $\text{Fix} T$.
            \item $(Tx_n - x_n)_{n \in \mathbb{N}}$ converges strongly to $0$.
            \item $(x_n)_{n \in \mathbb{N}}$ converges weakly to a point in $\text{Fix} T$.
        \end{enumerate}
    \end{theorem}

    \vspace{0.3cm}

    \textbf{Robustness}: Algorithm tolerates summable errors in the operator!
\end{frame}

\section{5.5 Nonlinear Ergodic Theorems}

\begin{frame}{Cesàro Averages and Ergodic Theorems}
    \begin{theorem}[Theorem 5.36]
        Let $(x_n)_{n \in \mathbb{N}}$ and $(y_n)_{n \in \mathbb{N}}$ be sequences in $\mathcal{H}$, let $C$ be a nonempty subset, and set $C_m = \text{conv}\{y_k\}_{k \geq m}$.
        Suppose that:
        \begin{enumerate}
            \item $(y_n)_{n \in \mathbb{N}}$ is quasi-Fejér monotone with respect to $C$.
            \item Distance from $x_n$ to Cesàro hulls $C_m$ goes to zero.
            \item Every weak cluster point of $(x_n)_{n \in \mathbb{N}}$ is in $C$.
        \end{enumerate}
        Then $(x_n)_{n \in \mathbb{N}}$ converges weakly to a point in $C$.
    \end{theorem}

    \vspace{0.3cm}

    \textbf{Key Idea}: Combines quasi-Fejér monotonicity with Cesàro averaging.
\end{frame}

\begin{frame}{Corollary 5.37: Cesàro-Averaged Sequences}
    \begin{corollary}[Corollary 5.37]
        Let $(y_n)_{n \in \mathbb{N}}$ be quasi-Fejér monotone with respect to a nonempty subset $C$.
        For every $n \in \mathbb{N}$, let $(\alpha_{n,k})_{0 \leq k \leq n}$ be real numbers in $[0, 1]$ such that
        \[
            (\forall n \in \mathbb{N}) \sum_{k=0}^n \alpha_{n,k} = 1 \quad \text{and} \quad (\forall k \in \mathbb{N}) \lim_{n \to \infty} \alpha_{n,k} = 0.
        \]
        Set
        \[
            (\forall n \in \mathbb{N}) \quad x_n = \sum_{k=0}^n \alpha_{n,k} y_k
        \]
        and suppose every weak cluster point is in $C$.
        Then $(x_n)_{n \in \mathbb{N}}$ converges weakly to a point in $C$.
    \end{corollary}

    \vspace{0.3cm}

    \textbf{Application}: Baillon's nonlinear ergodic theorem uses Cesàro average of iterates.
\end{frame}

\begin{frame}{Baillon's Nonlinear Ergodic Theorem}
    \begin{example}[Example 5.38 - Baillon]
        Let $D$ be a nonempty closed convex subset of $\mathcal{H}$, let $T: D \to D$ be a nonexpansive operator with $\text{Fix} T \neq \emptyset$, and $x_0 \in D$.
        Set
        \[
            (\forall n \in \mathbb{N}) \quad x_n = \frac{1}{n+1} \sum_{k=0}^n T^k x_0.
        \]
        Then $(x_n)_{n \in \mathbb{N}}$ converges weakly to a point in $\text{Fix} T$.
    \end{example}

    \vspace{0.3cm}

    \textbf{Proof Sketch}:
    \begin{enumerate}
        \item Set $C = \text{Fix} T$ and $y_k = T^k x_0$
        \item Show $(y_k)$ is Fejér monotone w.r.t. $C$
        \item Apply Corollary 5.37 with $\alpha_{n,k} = 1/(n+1)$
    \end{enumerate}

    \vspace{0.3cm}

    \textbf{Implication}: Even when iterates diverge, their Cesàro average converges!
\end{frame}

        # Check convergence
        res = (linalg.norm(project_onto_C1(x) - x) +
               linalg.norm(project_onto_C2(x) - x))
        if res < tol:
            break
    
    return x
        \end{lstlisting}

        \column{0.48\textwidth}
        \textbf{Example}:
        \begin{itemize}
            \item $C_1 = \{x : \|x\| \leq 1\}$ (disk)
            \item $C_2 = \{x : x_1 + x_2 \leq 1\}$ (half-space)
            \item Find intersection
        \end{itemize}

        \vspace{0.3cm}

        \textbf{Advantages}:
        \begin{itemize}
            \item Simple to implement
            \item Only needs projections
            \item Weak convergence guaranteed
        \end{itemize}
    \end{columns}
\end{frame}

def solve_linear_system(A, b, x0, 
                         max_iter=100):
    """Solve Ax=b via alternating projections"""
    x = x0
    for n in range(max_iter):
        # Project onto hyperplane Ax=b
        res = A @ x - b
        x = x - (A.T @ res) / (
            linalg.norm(A, 'fro')**2)
        
    return x

\begin{frame}{Practical Convergence Tips}
    \begin{itemize}
        \item \textbf{Krasnosel'skii-Mann}: Use $\lambda_n = 1/2$ (simple and effective)
        \item \textbf{Stepsizes}: If unsure, use $\sum \lambda_n(1-\lambda_n) = \infty$ (e.g., $\lambda_n = 1 - 1/(n+1)$)
        \item \textbf{Stopping Criterion}: Monitor distance to $C$ or residual $\|Tx_n - x_n\|$
        \item \textbf{Multiple Projections}: POCS is robust but may converge slowly
        \item \textbf{Acceleration}: Use relaxation parameters $\beta_i \in [0, 2]$ for projections
        \item \textbf{Perturbations}: Quasi-Fejér allows summable errors (robust to noise)
    \end{itemize}
\end{frame}
