\section{Notation and Intuition}
% =================================================================

\begin{frame}[allowframebreaks]{Notation You'll Need}

  Throughout, $\cH$, $\cK$ are real Hilbert spaces with inner product
  $\inner{\cdot}{\cdot}$ and induced norm $\norm{\cdot}$.

  \begin{itemize}
    \item $\Gamma_0(\cH)$: proper, lower semicontinuous, convex functions
      $\cH\to\left]-\infty,+\infty\right]$ (studied in Ch.~8--9).
    \item $\dom f=\{x\in\cH: f(x)<+\infty\}$; $f$ \textbf{proper} if
      $\dom f\ne\varnothing$ and $f$ never takes the value $-\infty$.
    \item $\epi f=\{(x,\xi)\in\cH\times\R: f(x)\le\xi\}$, the
      \textbf{epigraph} (region on/above the graph).
    \item $\iota_C$: the \textbf{indicator function} of a set $C$
      ($0$ on $C$, $+\infty$ off $C$).
    \item $d_C(x)=\inf_{y\in C}\norm{x-y}$: \textbf{distance} to $C$;
      $P_C$: the (nearest-point) \textbf{projection} onto a closed convex
      $C$.
    \item $\tau_y f = f(\cdot - y)$: \textbf{translation} of $f$ by $y$.
    \item $B(x;\rho)$: closed ball of radius $\rho$ centred at $x$.
    \item $f$ is \textbf{coercive} if $f(x)\to+\infty$ as
      $\norm{x}\to+\infty$; \textbf{supercoercive} if
      $f(x)/\norm{x}\to+\infty$.
    \item $f$ is \textbf{$\beta$-Lipschitz} if $|f(x)-f(y)|\le\beta\norm{x-y}$
      for all $x,y$.
    \item $\Argmin f=\{x\in\cH: f(x)=\inf f(\cH)\}$; $\Fix T=\{x:T(x)=x\}$.
    \item $T:\cH\to\cH$ is \textbf{firmly nonexpansive} if
      $\norm{Tx-Ty}^2\le\inner{x-y}{Tx-Ty}$ for all $x,y$ (stronger than
      nonexpansive; recall Ch.~4).
    \item \textbf{New in this chapter:} $f\co g$ (infimal convolution),
      $\me{\gamma}$ (Moreau envelope of parameter $\gamma$), $\Prox_f$
      (proximity operator), $L\triangleright f$ (infimal postcomposition).
  \end{itemize}
\end{frame}

\begin{frame}{Big Picture: What Is This Chapter About?}
  \textbf{Infimal convolution -- the intuition.}
  Given two functions $f,g$ on $\cH$, their infimal convolution
  \emph{blends} them: at each point $x$, you split $x=y+(x-y)$ optimally
  between the two functions and take the best combined cost. It is the
  \textbf{functional analogue of adding two sets/operators in parallel}
  -- exactly as the sum $C+D$ of two sets is built from all ways of
  splitting a point between $C$ and $D$, $f\co g$ is built from all ways
  of splitting the argument between $f$ and $g$.

  \bigskip
  \textbf{Moreau envelope / proximity operator -- the intuition.}
  Take a possibly \emph{kinked}, nonsmooth convex function $f$ (e.g.\
  $f(x)=|x|$) and infimal-convolve it with a scaled squared distance
  $\frac{1}{2\gamma}\norm{\cdot}^2$. The result, the \textbf{Moreau
  envelope} $\megamma$, is always \emph{smooth} (Fr\'echet
  differentiable!) and has the \emph{same} minimizers as $f$. The point
  $y$ that achieves the minimum in this blend is the \textbf{proximity
  operator} $\Prox_{\gamma f}(x)$: it \emph{nudges $x$ toward minimizing
  $f$ while staying close to $x$}, trading off cost-of-$f$ against
  distance-from-$x$. This is the single most important construction for
  the rest of the book (proximal algorithms, Ch.~23--30).
\end{frame}

\section{12.1 Definition and Basic Facts}
% =================================================================

\begin{frame}[allowframebreaks]{12.1 Infimal Convolution: Definition}

  \textbf{Intuition first.} To evaluate $(f\co g)(x)$, imagine
  distributing the point $x$ between two ``workers'' $f$ and $g$: worker
  $f$ handles a piece $y$, worker $g$ handles the remaining piece $x-y$,
  each pays its own cost, and you choose the split $y$ that minimizes the
  \emph{total} cost. This is exactly how resistors in parallel combine,
  or how two convex constraints/penalties can be merged into one.

  \begin{block}{Infimal convolution (12.1)/(12.3)}
    Let $f,g:\cH\to\left]-\infty,+\infty\right]$. The infimal convolution
    of $f$ and $g$ is
    $$f\co g:\cH\to[-\infty,+\infty]:x\mapsto\inf_{y\in\cH}\bigl(f(y)+g(x-y)\bigr)
      =\inf_{\substack{(u,v)\in\cH\times\cH\\u+v=x}}\bigl(f(u)+g(v)\bigr).$$
    \emph{Symbols:} $x$ is the point being evaluated; $y$ (equivalently
    $u$) ranges over all ways to split $x$; $g(x-y)$ (equivalently
    $g(v)$) is the cost paid by the complementary piece.
    $f\co g$ is \textbf{exact at $x$} if the infimum is \emph{attained}
    (a genuine minimum); it is \textbf{exact}, written $f\mathbin{\ovoerline{\co}} g$
    in the book's notation, if it is exact at every point of its domain.
  \end{block}

  \framebreak

  \textbf{Examples.} $\iota_{\{y\}}\co f=\tau_y f$ (Example 12.4):
  convolving with a point mass just translates. More importantly,
  $d_C=\iota_C\co\norm{\cdot}$ (Example 12.2/Corollary 12.12): the
  distance function to a set $C$ \emph{is} an infimal convolution of the
  indicator of $C$ with the norm -- splitting $x$ between ``being in
  $C$'' (cost $0$ or $+\infty$) and ``distance travelled.''

  \begin{block}{Continuous affine minorant, Definition 12.5}
    $f$ possesses a continuous affine minorant with slope $u\in\cH$ if
    $f-\inner{\cdot}{u}$ is bounded below, i.e.\ $f\ge\inner{\cdot}{u}+\eta$
    for some $\eta\in\R$.
  \end{block}
  This technical condition rules out functions that are unbounded below
  along every affine direction; it is exactly what keeps $f\co g$ from
  collapsing to $-\infty$ identically.
\end{frame}

\begin{frame}[allowframebreaks]{12.1 Proposition 12.6: Basic Properties}

  \begin{block}{Proposition 12.6}
    Let $f,g,h:\cH\to\left]-\infty,+\infty\right]$. Then:
    \begin{enumerate}
      \item[(i)] If $f,g$ possess continuous affine minorants with the
        \emph{same} slope $u$, then $f\co g$ possesses a continuous
        affine minorant with slope $u$, and $-\infty\notin(f\co g)(\cH)$.
      \item[(ii)] $\dom(f\co g)=\dom f+\dom g$.
      \item[(iii)] $f\co g=g\co f$ \quad (\textbf{commutativity}).
      \item[(iv)] If $f,g,h$ share a common-slope minorant, then
        $f\co(g\co h)=(f\co g)\co h$ \quad (\textbf{associativity}).
    \end{enumerate}
  \end{block}

  \textbf{Why the minorant hypothesis matters.}
  \begin{block}{Example 12.7 (minorant assumption cannot be dropped)}
    Let $f:\R\to\R:\xi\mapsto\xi$ and $g=-f$. Then $f\co g\equiv-\infty$
    everywhere: the split $y\to+\infty,\,x-y\to-\infty$ makes
    $f(y)+g(x-y)=y-(x-y)\to-\infty$ with no lower bound.
  \end{block}
  Here $f$ and $g$ have continuous affine minorants with \emph{opposite}
  slopes ($u=1$ and $u=-1$), not the same slope -- Proposition 12.6(i)
  correctly does not apply.

  \framebreak

  \begin{block}{Proposition 12.8: Epigraphs}
    \begin{enumerate}
      \item[(i)] $\epi f+\epi g\subset\epi(f\co g)$ always.
      \item[(ii)] If $f\co g$ is exact ($f\co g=f\mathbin{\ovoerline\co}g$), then
        $\epi(f\co g)=\epi f+\epi g$ (equality).
    \end{enumerate}
  \end{block}
  \textbf{Reading it:} adding two epigraphs (as subsets of
  $\cH\times\R$, i.e.\ combining a point of one graph with a point of the
  other and stacking their heights) always sits \emph{inside} the
  epigraph of the infimal convolution; the two agree exactly when the
  splitting infimum is actually achieved. Example 12.13 (ahead) shows
  this containment can be strict.
\end{frame}

\section{12.2 Infimal Convolution of Convex Functions}
% =================================================================

\begin{frame}[allowframebreaks]{12.2 Convexity Is Preserved}

  \begin{block}{Proposition 12.11}
    If $f$ and $g$ are convex, then $f\co g$ is convex.
  \end{block}
  \emph{Proof idea:} write $F(x,y)=f(y)+g(x-y)$, a jointly convex
  function on $\cH\times\cH$, and observe $f\co g$ is its
  \textbf{marginal function} (infimum over one block of variables);
  marginals of jointly convex functions are convex (Prop.~8.35).

  \begin{block}{Corollary 12.12}
    For $C\subset\cH$ convex, $d_C=\iota_C\co\norm{\cdot}$ is convex.
  \end{block}

  \textbf{A cautionary example -- things can go wrong even in
  $\Gamma_0(\cH)$.}
  \begin{block}{Example 12.13}
    $f(\xi)=1/\xi$ for $\xi>0$, $+\infty$ otherwise; $g=f^{\vee}$ (the
    reflection $g(\xi)=f(-\xi)$). Both are in $\Gamma_0(\R)$, yet
    $f\co g\equiv0$ and is \emph{nowhere exact}; moreover the sum of
    epigraphs $C+D$ (with $C=\epi f$, $D=\epi g$) is the \emph{open}
    upper half-plane, so $\iota_{C+D}$ is \emph{not} lower
    semicontinuous. So exactness and lower semicontinuity of $f\co g$
    are \emph{not} automatic, even for nice $f,g\in\Gamma_0(\cH)$.
  \end{block}

  \framebreak

  \textbf{When does it behave well?}
  \begin{block}{Proposition 12.14}
    Let $f,g\in\Gamma_0(\cH)$ and suppose \textbf{either}
    \begin{enumerate}
      \item[(i)] $f$ is supercoercive, \textbf{or}
      \item[(ii)] $f$ is coercive and $g$ is bounded below.
    \end{enumerate}
    Then $f\co g=f\mathbin{\ovoerline\co}g\in\Gamma_0(\cH)$: the infimal
    convolution is \emph{exact} and again proper, convex, lsc.
  \end{block}
  Intuition: coercivity of $f$ prevents the minimizing sequence $y_n$
  in the splitting infimum from ``running off to infinity,'' which is
  exactly the failure mode behind Example 12.13.

  \framebreak

  \textbf{The central example: convolving with a power of the norm.}
  This is the bridge to Sections 12.3--12.4.
  \begin{block}{Proposition 12.9 / (12.6)}
    Let $f$ be proper, $p\in[1,+\infty[$, and for $\gamma\in\R_{++}$ set
    $$g_\gamma=f\co\Bigl(\frac{1}{\gamma p}\norm{\cdot}^p\Bigr).$$
    Then for every $\gamma$ and every $x\in\cH$: $\dom g_\gamma=\cH$;
    for $\mu>\gamma$, $\inf f(\cH)\le g_\mu(x)\le g_\gamma(x)\le f(x)$;
    $\inf g_\gamma(\cH)=\inf f(\cH)$; $g_\mu(x)\downarrow\inf f(\cH)$ as
    $\mu\uparrow+\infty$; and $g_\gamma$ is bounded above on every ball.
  \end{block}
  \emph{Reading it:} $g_\gamma$ is always a real-valued, everywhere-defined
  ``softened'' version of $f$ sandwiched between $\inf f(\cH)$ and $f$
  itself, and it \emph{relaxes down towards $\inf f$} as $\gamma\to\infty$
  while \emph{tightening back up to $f$} as $\gamma\to0$.

  \framebreak

  \begin{block}{Proposition 12.15 (the case $p>1$)}
    Let $f\in\Gamma_0(\cH)$, $\gamma\in\R_{++}$, $p\in\left]1,+\infty\right[$.
    Then
    $$f\co\Bigl(\frac{1}{\gamma p}\norm{\cdot}^p\Bigr):\cH\to\left]-\infty,+\infty\right]:
      x\mapsto\inf_{y\in\cH}\Bigl(f(y)+\frac{1}{\gamma p}\norm{x-y}^p\Bigr)$$
    is convex, \textbf{real-valued}, \textbf{continuous}, and
    \textbf{exact}. Moreover the infimum is \textbf{uniquely attained}
    for every $x$.
  \end{block}
  This is because $\frac{1}{\gamma p}\norm{\cdot}^p$ is supercoercive and
  strictly convex for $p>1$, so Proposition 12.14(i) gives exactness and
  strict convexity gives uniqueness of the minimizer. The two headline
  special cases are $p=1$ (Section 12.3) and $p=2$ (Section 12.4).
\end{frame}

\section{12.3 Pasch--Hausdorff Envelope}
% =================================================================

\begin{frame}[allowframebreaks]{12.3 Pasch--Hausdorff Envelope: $p=1$}

  \textbf{Intuition.} Convolving $f$ with $\beta\norm{\cdot}$ produces
  the \emph{largest $\beta$-Lipschitz function that still lies below
  $f$} -- the best Lipschitz approximation from below, analogous to
  taking the tightest cone of slope $\beta$ that still fits under the
  graph of $f$ everywhere.

  \begin{block}{Definition 12.16}
    Let $f:\cH\to\left]-\infty,+\infty\right]$ and $\beta\in\R_+$. The
    \textbf{$\beta$-Pasch--Hausdorff envelope} of $f$ is
    $f\co(\beta\norm{\cdot})$.
  \end{block}
  Being the case $p=1$ of (12.6), it inherits all of Proposition 12.9's
  properties automatically.

  \begin{block}{Proposition 12.17}
    Let $f$ be proper, $\beta\in\R_+$, $g$ its $\beta$-Pasch--Hausdorff
    envelope. Exactly one of the following holds:
    \begin{enumerate}
      \item[(i)] $f$ has a $\beta$-Lipschitz minorant, and $g$ \emph{is}
        the \textbf{largest} such minorant of $f$;
      \item[(ii)] $f$ has no $\beta$-Lipschitz minorant, and $g\equiv-\infty$.
    \end{enumerate}
  \end{block}

  \framebreak

  \textbf{Consequences.}
  \begin{block}{Corollary 12.18}
    If $f:\cH\to\R$ is $\beta$-Lipschitz for some $\beta\in\R_{++}$, then
    $f$ is its own $\beta$-Pasch--Hausdorff envelope (it cannot be
    improved -- it already \emph{is} the best $\beta$-Lipschitz
    minorant of itself).
  \end{block}
  \begin{block}{Corollary 12.19 (Lipschitz extension)}
    Let $C\subset\cH$ be nonempty, $h:C\to\R$ be $\beta$-Lipschitz on
    $C$, and extend $h$ to $\cH$ by $+\infty$ outside $C$. Then the
    $\beta$-Pasch--Hausdorff envelope of this extension is a
    \textbf{$\beta$-Lipschitz continuous extension of $h$ to all of
    $\cH$}.
  \end{block}
  This is a constructive, infimal-convolution proof of the
  McShane/Kirszbraun-type Lipschitz extension theorem: any Lipschitz
  function on a subset extends to a Lipschitz function on the whole
  space, with the \emph{same} constant.
\end{frame}

\section{12.4 Moreau Envelope and Proximity Operator}
% =================================================================

\begin{frame}[allowframebreaks]{12.4 Moreau Envelope: $p=2$}

  \textbf{Intuition (restated precisely).} The case $p=2$ of (12.6) --
  convolving $f$ with a scaled \emph{squared} distance -- is the single
  most useful construction in the book: it \textbf{smooths} $f$ without
  changing where its minimizers live, and its infimum-achieving point
  defines an operator (the proximity operator) that is the building
  block of essentially every algorithm in later chapters.

  \begin{block}{Definition 12.20: Moreau envelope}
    Let $f:\cH\to\left]-\infty,+\infty\right]$, $\gamma\in\R_{++}$. The
    \textbf{Moreau envelope} of $f$ of parameter $\gamma$ is
    $$\megamma=f\co\Bigl(\frac{1}{2\gamma}\norm{\cdot}^2\Bigr).$$
    \emph{Symbols:} $\gamma$ is the smoothing parameter (larger $\gamma$
    $\Rightarrow$ more smoothing, flatter envelope); $\frac{1}{2\gamma}
    \norm{\cdot}^2$ is the quadratic ``proximity penalty.''
  \end{block}

  \begin{block}{Example 12.21}
    For $C\subset\cH$: $\me{\gamma}\iota_C=(2\gamma)^{-1}d_C^2$ -- the
    Moreau envelope of an indicator is (a scaled) squared distance
    function.
  \end{block}

  \framebreak

  \textbf{Composing envelopes (semigroup-type property).}
  \begin{block}{Proposition 12.22}
    Let $f:\cH\to\left]-\infty,+\infty\right]$, $\gamma,\mu\in\R_{++}$.
    \begin{enumerate}
      \item[(i)] $\me{\mu}(\gamma f)=\gamma\bigl(\me{\gamma\mu}f\bigr)$.
      \item[(ii)] $\me{\gamma}(\me{\mu}f)=\me{(\gamma+\mu)}f$.
    \end{enumerate}
  \end{block}
  Part (ii) says: taking the Moreau envelope of a Moreau envelope, with
  parameters $\gamma$ then $\mu$, is the \emph{same} as taking a single
  Moreau envelope with parameter $\gamma+\mu$ -- envelope parameters
  \emph{add} under composition, exactly like a heat-semigroup /
  diffusion-time parameter.

  \framebreak

  \begin{block}{Definition 12.23: Proximity operator}
    Let $f\in\Gamma_0(\cH)$, $x\in\cH$. $\Prox_f x$ is the \textbf{unique}
    point in $\cH$ satisfying
    $$\me{1}f(x)=\min_{y\in\cH}\Bigl(f(y)+\tfrac12\norm{x-y}^2\Bigr)
       =f(\Prox_f x)+\tfrac12\norm{x-\Prox_f x}^2.$$
    $\Prox_f:\cH\to\cH$ is the \textbf{proximity operator} (or
    \emph{proximal mapping}) of $f$. \emph{Symbols:} $x$ is the input
    point; $\Prox_f x$ is the unique minimizer trading off ``low $f$''
    against ``close to $x$.''
  \end{block}

  \begin{block}{Remark 12.24 (general $\gamma$)}
    $$\megamma(x)=f(\Prox_{\gamma f}x)+\frac{1}{2\gamma}\norm{x-\Prox_{\gamma f}x}^2.$$
  \end{block}

  \begin{block}{Example 12.25}
    For $C$ nonempty closed convex, $\Prox_{\iota_C}=P_C$: the proximity
    operator of an indicator function \emph{is} the metric projection.
    (The proximity operator generalizes projection from sets to convex
    functions.)
  \end{block}
\end{frame}

\begin{frame}[allowframebreaks]{12.4 Step-by-Step: Moreau Envelope of $f=|\cdot|$}

  \textbf{Goal.} Derive $\Prox_{\gamma f}$ and $\megamma$ explicitly for
  the running example $f:\R\to\R:x\mapsto|x|$, the classic
  \textbf{soft-thresholding} case (Exercise 12.10(i)).

  \textbf{Step 1 (set up the minimization).} By Definition 12.23/Remark
  12.24, for fixed $x\in\R,\gamma>0$ we must minimize over $y\in\R$:
  $$\varphi(y)=|y|+\frac{1}{2\gamma}(x-y)^2.$$

  \textbf{Step 2 (differentiate on each smooth piece).} For $y>0$,
  $\varphi'(y)=1-\frac{x-y}{\gamma}$; setting this to $0$ gives
  $y=x-\gamma$, valid (i.e.\ $y>0$) exactly when $x>\gamma$.
  For $y<0$, $\varphi'(y)=-1-\frac{x-y}{\gamma}$; setting this to $0$
  gives $y=x+\gamma$, valid ($y<0$) exactly when $x<-\gamma$.

  \framebreak

  \textbf{Step 3 (the flat/kink case).} For $|x|\le\gamma$, check the
  subdifferential optimality condition at $y=0$:
  $0\in\partial|0|-\tfrac1\gamma(x-0)=[-1,1]-x/\gamma$, i.e.\ we need
  $x/\gamma\in[-1,1]$, i.e.\ $|x|\le\gamma$ -- exactly our case. So the
  minimizer is $y^\ast=0$.

  \textbf{Step 4 (assemble the proximity operator).}
  $$\boxed{\Prox_{\gamma|\cdot|}(x)=
     \begin{cases}
       x-\gamma, & x>\gamma,\\
       0, & |x|\le\gamma,\\
       x+\gamma, & x<-\gamma,
     \end{cases}
     \;=\;\mathrm{sign}(x)\,\max(|x|-\gamma,0).}$$
  This is exactly the \textbf{soft-thresholding operator} $S_\gamma$
  ubiquitous in sparse signal processing (LASSO, wavelet denoising).

  \framebreak

  \textbf{Step 5 (plug back in for the Moreau envelope value).}
  \begin{itemize}
    \item If $|x|\le\gamma$: $y^\ast=0$, so
      $\megamma(x)=0+\frac{1}{2\gamma}x^2=\dfrac{x^2}{2\gamma}$.
    \item If $x>\gamma$: $y^\ast=x-\gamma$, so
      $\megamma(x)=(x-\gamma)+\frac{1}{2\gamma}\gamma^2=x-\dfrac{\gamma}{2}$.
    \item If $x<-\gamma$: by symmetry, $\megamma(x)=-x-\dfrac{\gamma}{2}=|x|-\dfrac{\gamma}{2}$.
  \end{itemize}
  $$\boxed{\megamma(x)=
      \begin{cases}
        \dfrac{x^2}{2\gamma}, & |x|\le\gamma,\\[4pt]
        |x|-\dfrac{\gamma}{2}, & |x|>\gamma.
      \end{cases}}$$
  This is exactly the \textbf{Huber function} of parameter $\gamma$: a
  parabola near $0$ that transitions smoothly (matching value \emph{and}
  slope at $|x|=\gamma$) into linear growth away from $0$ -- a smoothed,
  everywhere-differentiable stand-in for $|x|$.
\end{frame}

\begin{frame}{12.4 Running Example: Numeric Evaluation}
  \textbf{Concrete values} for $f(x)=|x|$, using
  $\Prox_{\gamma|\cdot|}(x)=\mathrm{sign}(x)\max(|x|-\gamma,0)$ and
  $\megamma(x)$ from the boxed formulas above.

  \begin{center}
  \begin{tabular}{c c c c c}
    \toprule
    $\gamma$ & $x$ & regime & $\Prox_{\gamma|\cdot|}(x)$ & $\megamma(x)$ \\
    \midrule
    $1$ & $0.5$ & $|x|\le\gamma$   & $0$   & $0.25/2=0.125$ \\
    $1$ & $1$   & $|x|=\gamma$     & $0$   & $1/2=0.5$ \\
    $1$ & $3$   & $|x|>\gamma$     & $2$   & $3-0.5=2.5$ \\
    $2$ & $0.5$ & $|x|\le\gamma$   & $0$   & $0.25/4=0.0625$ \\
    $2$ & $1$   & $|x|\le\gamma$   & $0$   & $1/4=0.25$ \\
    $2$ & $3$   & $|x|>\gamma$     & $1$   & $3-1=2$ \\
    \bottomrule
  \end{tabular}
  \end{center}

  \bigskip
  \textbf{Sanity checks (Definition 12.23's identity).}
  For $\gamma=1,x=3$: $f(\Prox)+\tfrac12\norm{x-\Prox}^2
  =|2|+\tfrac12(3-2)^2=2+0.5=2.5=\megamma(3)$. \checkmark\\
  For $\gamma=2,x=3$: $f(1)+\tfrac1{2\cdot2}(3-1)^2=1+\tfrac{4}{4}=2=\megamma(3)$. \checkmark
\end{frame}

  gives more smoothing/flattening. Right: the corresponding proximity
  operators $\Prox_{\gamma|\cdot|}$ (soft-thresholding), each a
  flat-then-shifted-identity map with dead-zone $[-\gamma,\gamma]$.}

\begin{frame}[fragile]{Python: Moreau Envelope and Proximity Operator of $|\cdot|$}

import numpy as np

def prox_abs(x, gamma):
    """Prox_{gamma |.|}(x) = soft-thresholding operator S_gamma(x)."""
    return np.sign(x) * np.maximum(np.abs(x) - gamma, 0.0)

def moreau_envelope_abs(x, gamma):
    """Moreau envelope ^gamma f of f = |.|  (Huber function)."""
    x = np.asarray(x, dtype=float)
    small = np.abs(x) <= gamma
    return np.where(small, x**2 / (2.0 * gamma), np.abs(x) - gamma / 2.0)

# Reproduce the numeric table exactly:
for gamma in (1.0, 2.0):
    for x in (0.5, 1.0, 3.0):
        p = prox_abs(x, gamma)
        env = moreau_envelope_abs(np.array([x]), gamma)[0]
        # cross-check against the defining identity f(p) + ||x-p||^2/(2g)
        check = abs(p) + (x - p) ** 2 / (2 * gamma)
        print(f"gamma={gamma:.0f}  x={x:.1f}  Prox={p:.1f}  "
              f"^gamma f(x)={env:.4f}  (check {check:.4f})")

\begin{frame}[allowframebreaks]{12.4 Proposition 12.26: Variational Characterization}

  This is the workhorse characterization of the proximity operator,
  used to prove almost everything else in the section.

  \begin{block}{Proposition 12.26}
    Let $f\in\Gamma_0(\cH)$, $x,p\in\cH$. Then
    $$p=\Prox_f x \quad\Longleftrightarrow\quad
      (\forall y\in\cH)\quad \inner{y-p}{x-p}+f(p)\le f(y).$$
  \end{block}
  \emph{Reading it:} $p$ is the proximal point of $x$ iff, from $p$'s
  vantage point, moving to \emph{any} other point $y$ can only help $f$
  by less than the inner-product ``penalty'' $\inner{y-p}{x-p}$ picks up
  -- a first-order optimality condition for the defining minimization,
  \emph{without} needing differentiability of $f$.

  \framebreak

  \begin{block}{Proposition 12.27}
    Let $\varphi(\gamma)=f(\Prox_{\gamma f}x)$. Then $\varphi$ is
    \textbf{decreasing} in $\gamma$, $\sup\varphi(\R_{++})\le f(x)$, and
    $$\norm{x-\Prox_{\gamma f}x}^2+\gamma f(\Prox_{\gamma f}x)\le\gamma f(x).$$
  \end{block}
  As the smoothing parameter $\gamma$ grows, the proximal point is
  allowed to travel further from $x$ to chase a lower value of $f$.

  \begin{block}{Proposition 12.28}
    $\Prox_f$ and $\operatorname{Id}-\Prox_f$ are both \textbf{firmly
    nonexpansive}.
  \end{block}
  This connects straight back to Ch.~4: proximity operators are exactly
  the kind of well-behaved (firmly nonexpansive) maps whose fixed-point
  iterations converge nicely -- the reason proximal algorithms work.

  \begin{block}{Proposition 12.29}
    $\Fix\Prox_f=\Argmin f$ (12.27).
  \end{block}
  Fixed points of the proximity operator are \emph{exactly} the
  minimizers of $f$ -- this is what lets one minimize $f$ by iterating
  $\Prox_f$ (proximal-point algorithm, Ch.~28).
\end{frame}

\begin{frame}[allowframebreaks]{12.4 Proposition 12.30: Differentiability of the Moreau Envelope}

  This is the key structural payoff of the whole construction: the
  Moreau envelope \emph{always} smooths, even when $f$ itself is
  nowhere differentiable.

  \begin{block}{Proposition 12.30}
    Let $f\in\Gamma_0(\cH)$, $\gamma\in\R_{++}$. Then
    $\megamma:\cH\to\R$ is \textbf{Fr\'echet differentiable} on all of
    $\cH$, with gradient
    $$\nabla(\megamma)=\gamma^{-1}\bigl(\operatorname{Id}-\Prox_{\gamma f}\bigr),$$
    and this gradient is \textbf{$\gamma^{-1}$-Lipschitz continuous}.
  \end{block}
  \emph{Symbols:} $\operatorname{Id}-\Prox_{\gamma f}$ is the ``residual''
  map $x\mapsto x-\Prox_{\gamma f}x$; dividing by $\gamma$ gives exactly
  the gradient. \emph{Reading it:} the direction of steepest ascent of
  $\megamma$ at $x$ points from the proximal point $\Prox_{\gamma f}x$
  back to $x$, scaled by $1/\gamma$ -- and this gradient map is Lipschitz
  (hence $\megamma$ is $C^{1,1}$), regardless of how nonsmooth $f$ was.

  \textbf{Check against our running example.} For $f=|\cdot|$, $\gamma=1$,
  $x=3$: $\Prox_{1\cdot|\cdot|}(3)=2$, so
  $\nabla(\me1f)(3)=(3-2)/1=1$. Directly differentiating
  $\me1f(x)=x-1/2$ on $x>1$ gives slope $1$. \checkmark

  \framebreak

  \begin{block}{Corollary 12.31}
    For $C$ nonempty closed convex, $d_C^2$ is Fr\'echet differentiable
    with $\nabla d_C^2=2(\operatorname{Id}-P_C)$.
  \end{block}
  (Apply Proposition 12.30 to $f=\iota_C$, $\gamma=1/2$, using
  $\me{1/2}\iota_C=d_C^2$ and $\Prox_{(1/2)\iota_C}=P_C$.)

  \begin{block}{Proposition 12.32}
    For $K$ a nonempty closed convex cone and $q=\tfrac12\norm{\cdot}^2$:
    $\nabla(q\circ P_K)=P_K$.
  \end{block}

  \framebreak

  \textbf{Limiting behaviour as $\gamma\to0$ or $\gamma\to\infty$.}
  \begin{block}{Proposition 12.33}
    Let $f\in\Gamma_0(\cH)$, $x\in\cH$. The nets $(\megamma(x))_\gamma$
    and $(f(\Prox_{\gamma f}x))_\gamma$ are decreasing in $\gamma$, and:
    \begin{enumerate}
      \item[(i)] $\megamma(x)\downarrow\inf f(\cH)$ and
        $f(\Prox_{\gamma f}x)\downarrow\inf f(\cH)$ as $\gamma\uparrow+\infty$.
      \item[(ii)] $\megamma(x)\uparrow f(x)$ as $\gamma\downarrow0$.
      \item[(iii)] If $x\in\dom f$: $f(\Prox_{\gamma f}x)\uparrow f(x)$
        and $\gamma^{-1}\norm{x-\Prox_{\gamma f}x}^2\to0$ as
        $\gamma\downarrow0$.
    \end{enumerate}
  \end{block}
  \emph{Reading it:} as $\gamma\downarrow0$, the envelope ``un-smooths''
  back to $f$ itself (no smoothing at all); as $\gamma\uparrow\infty$,
  the envelope flattens all the way down to the global infimum -- the
  two extremes recover, respectively, $f$ and its relaxed floor.
\end{frame}

\section{12.5 Infimal Postcomposition}
% =================================================================

\begin{frame}[allowframebreaks]{12.5 Infimal Postcomposition}

  \textbf{Intuition.} Infimal convolution split a point \emph{additively}
  between two functions; infimal postcomposition generalizes this to
  splitting along a \emph{linear map} $L$: to evaluate the pushed-forward
  function at $y$, minimize $f$ over the entire preimage (fiber)
  $L^{-1}\{y\}$.

  \begin{block}{Definition 12.34}
    Let $f:\cH\to[-\infty,+\infty]$, $\cK$ a real Hilbert space,
    $L:\cH\to\cK$. The \textbf{infimal postcomposition} of $f$ by $L$ is
    $$L\triangleright f:\cK\to[-\infty,+\infty]:y\mapsto
       \inf_{\substack{x\in\cH\\Lx=y}}f(x)=\inf f(L^{-1}\{y\}),$$
    exact at $y$ if this infimum is a genuine minimum, i.e.\
    $(\exists x\in\cH)$ $Lx=y$ and $(L\triangleright f)(y)=f(x)\in\left]-\infty,+\infty\right]$;
    $L\triangleright f$ is \textbf{exact} if exact at every point of its
    domain (then written $L\triangleright f$ without the bar-notation
    distinction).
  \end{block}

  \framebreak

  \begin{block}{Example 12.35: latent group lasso}
    Partition/cover index set $I=\{1,\dots,N\}$ by groups $(I_k)_{k\in K}$
    (possibly overlapping), let $L:\bigoplus_kx_k\mapsto\sum_kx_k$ sum the
    group-blocks back into $\cK=\R^N$, and let
    $f(\boldsymbol x)=\sum_{k\in K}\norm{x_k}_p$. Then the
    \textbf{latent group lasso penalty} is exactly $L\triangleright f$:
    $$\norm{y}_{\mathrm{lgl}}=\inf_{\substack{\boldsymbol x\in\cH\\ \sum_kx_k=y}}\sum_{k\in K}\norm{x_k}_p.$$
    When the groups are pairwise disjoint, this reduces to the ordinary
    group lasso penalty. This is a genuinely modern (sparse-statistics)
    application built directly on the machinery of this chapter.
  \end{block}

  \framebreak

  \begin{block}{Proposition 12.36}
    \begin{enumerate}
      \item[(i)] $\dom(L\triangleright f)=L(\dom f)$.
      \item[(ii)] If $f$ is convex and $L$ is affine, $L\triangleright f$
        is convex.
    \end{enumerate}
  \end{block}

  \textbf{Closing the loop: infimal convolution is a special case.}
  \begin{block}{Proposition 12.37}
    Let $f,g:\cH\to\left]-\infty,+\infty\right]$ and set
    $L:\cH\times\cH\to\cH:(x,y)\mapsto x+y$ (the addition map). Then
    $$f\co g=L\triangleright(f\oplus g),$$
    where $f\oplus g:(x,y)\mapsto f(x)+g(y)$ is the (separable) direct
    sum. \emph{Reading it:} infimal convolution is literally infimal
    postcomposition of the direct sum $f\oplus g$ by the addition map --
    everything in Sections 12.1--12.4 is a special case of Section 12.5's
    single, more general construction.
  \end{block}
\end{frame}

\section{Summary}
% =================================================================

\begin{frame}{Chapter 12 Summary}
  \begin{itemize}
    \item \textbf{Infimal convolution} $f\co g(x)=\inf_y(f(y)+g(x-y))$:
      the functional analogue of parallel combination; convex when $f,g$
      are (Prop.~12.11); needs common-slope minorants to avoid
      $-\infty$ (Prop.~12.6); exact under coercivity assumptions
      (Prop.~12.14).
    \item \textbf{Convolving with $\frac{1}{\gamma p}\norm{\cdot}^p$}
      (Prop.~12.9, 12.15) always yields something real-valued,
      continuous; for $p>1$, exact with a \emph{unique} minimizer.
    \item \textbf{Pasch--Hausdorff envelope} ($p=1$): the largest
      $\beta$-Lipschitz minorant of $f$; gives a Lipschitz extension
      theorem (Cor.~12.19).
    \item \textbf{Moreau envelope} $\megamma=f\co\frac{1}{2\gamma}
      \norm{\cdot}^2$ ($p=2$): always real-valued, convex, \emph{Fr\'echet
      differentiable} with $\gamma^{-1}$-Lipschitz gradient
      $\nabla\megamma=\gamma^{-1}(\operatorname{Id}-\Prox_{\gamma f})$
      (Prop.~12.30) -- smooths any $f\in\Gamma_0(\cH)$ without moving its
      minimizers ($\Fix\Prox_f=\Argmin f$, Prop.~12.29).
    \item \textbf{Proximity operator} $\Prox_f$: unique minimizer of
      $f(y)+\tfrac12\norm{x-y}^2$; firmly nonexpansive (Prop.~12.28);
      generalizes projection ($\Prox_{\iota_C}=P_C$); for $f=|\cdot|$ it
      is exactly \textbf{soft-thresholding}
      $\Prox_{\gamma|\cdot|}(x)=\mathrm{sign}(x)\max(|x|-\gamma,0)$.
    \item \textbf{Infimal postcomposition} $L\triangleright f$
      generalizes infimal convolution to splitting along a linear map;
      powers modern penalties like the latent group lasso.
    \item \textbf{Why this matters for the rest of the book:} the
      Moreau envelope/proximity operator pair is the foundation for
      proximal algorithms (Ch.~23, 28), resolvents of monotone operators
      (Ch.~23), and splitting methods for nonsmooth convex
      minimization -- referenced throughout the rest of this book and in
      companion texts on KM iterations, E-values, and Bayesian
      computation.
  \end{itemize}
\end{frame}
