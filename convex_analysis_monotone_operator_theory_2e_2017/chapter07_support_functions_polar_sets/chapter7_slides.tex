\section{Notation and Motivation}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}[allowframebreaks]{Notation You'll Need}

  \begin{block}{The Ambient Space}
    Throughout, $\Hil$ is a real \textbf{Hilbert space} with inner product
    $\ip{\cdot}{\cdot}$ and norm $\|\cdot\|$. All figures and numeric examples
    in this deck take $\Hil = \R^2$ (or $\R$), so you can picture every set as
    a region in the plane and every $u \in \Hil$ as an arrow from the origin.
  \end{block}

  \begin{block}{Carried Over from Earlier Chapters}
    \begin{itemize}
      \item $C \subset \Hil$: an arbitrary subset (often closed and convex).
      \item $\conv C$: the \textbf{convex hull} of $C$ (smallest convex set
        containing $C$); $\closure{\conv}\,C$ is its closure, the
        \textbf{closed convex hull}.
      \item $\bdry C$, $\Int C$: the boundary and interior of $C$.
      \item $N_C x$: the \textbf{normal cone} to $C$ at $x$ (Chapter 6),
        $N_Cx = \{u \in \Hil : \sup\ip{C-x}{u} \le 0\}$ for $x \in C$, and
        $N_Cx = \varnothing$ for $x \notin C$.
      \item $P_C$: the \textbf{metric projection} onto a closed convex $C$.
      \item $K^\ominus$: the \textbf{polar cone} of a cone $K$ (Chapter 6),
        $K^\ominus = \{u \in \Hil : \sup\ip{K}{u}\le 0\}$; $V^\perp$: the
        \textbf{orthogonal complement} of a linear subspace $V$.
      \item $\mathrm{lev}_{\le 1} f = \{u : f(u) \le 1\}$: the sublevel set of
        a function $f$ at height $1$.
    \end{itemize}
  \end{block}

  \framebreak

  \begin{block}{New in This Chapter}
    \begin{itemize}
      \item $\spts C$: the set of \textbf{support points} of $C$ (Section 7.1).
      \item $\sigma_C : \Hil \to [-\infty,+\infty]$: the \textbf{support
        function} of $C$ (Section 7.2), $\sigma_C(u) = \sup\ip{C}{u}$.
      \item $C^\odot$: the \textbf{polar set} of $C$ (Section 7.3),
        $C^\odot = \{u \in \Hil : (\forall x \in C)\ \ip{x}{u} \le 1\}$.
      \item $\|\cdot\|_*$: the \textbf{dual norm} of a norm $\|\cdot\|$ on a
        finite-dimensional space.
      \item $\ell^2(\N)$, $\ell^2_+(\N)$: square-summable real sequences, and
        those with all nonnegative terms (a closed convex cone with empty
        interior in $\ell^2(\N)$).
    \end{itemize}
  \end{block}

  \begin{alertblock}{The Running Examples (used throughout this deck)}
    Two shapes anchor every computation below:
    \[
      C_{\mathrm{disk}} = \{x \in \R^2 : \|x\| \le 1\} \quad (\text{closed
      unit disk}), \qquad C_{\mathrm{sq}} = [-1,1]^2 \quad (\text{closed unit
      square}).
    \]
    We repeatedly compute $\sigma_{C}(u)$ and $C^\odot$ for these two sets, so
    the abstract definitions always have concrete numbers attached.
  \end{alertblock}

\end{frame}

\begin{frame}{Why This Chapter Matters}

  \begin{block}{The Big Picture}
    A closed convex set $C$ can be described in two dual ways: as a
    \emph{collection of points}, or as an \emph{intersection of half-spaces}
    that pin it down from the outside. The \textbf{support function} is the
    bridge between these two pictures: it packages, for every direction $u$,
    the single number ``how far can I go into $C$ if I walk in direction
    $u$?'' Knowing $\sigma_C(u)$ for \emph{every} direction $u$ turns out to
    completely determine $C$ (when $C$ is closed and convex) --- this is the
    content of Corollary~7.12 below.
  \end{block}

  \begin{block}{Plain-English Preview}
    \begin{itemize}
      \item \textbf{Support point}: a point of $C$ that sits on some
        supporting hyperplane --- a point you can ``touch'' from outside
        with a flat wall that leaves all of $C$ on one side.
      \item \textbf{Support function} $\sigma_C(u)$: the maximum value of
        $\ip{x}{u}$ over $x \in C$ --- the signed distance (scaled by
        $\|u\|$) you can travel into $C$ along direction $u$.
      \item \textbf{Polar set} $C^\odot$: the set of directions $u$ for which
        this maximum never exceeds $1$ --- a dual object that, remarkably,
        recovers $C$ itself when you take the polar twice (the
        \emph{bipolar theorem}, Theorem~7.18).
    \end{itemize}
  \end{block}

\end{frame}

\section{7.1 Support Points}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}[allowframebreaks]{Definition 7.1: Support Points and Supporting Hyperplanes}

  \begin{block}{Definition 7.1}
    Let $C$ be a nonempty subset of $\Hil$, let $x \in C$, and suppose that
    $u \in \Hil \smallsetminus \{0\}$. If
    \[
      \sup \ip{C}{u} \le \ip{x}{u}, \tag{7.1}
    \]
    then $\{y \in \Hil : \ip{y}{u} = \ip{x}{u}\}$ is a \textbf{supporting
    hyperplane} of $C$ at $x$, and $x$ is a \textbf{support point} of $C$
    with \textbf{normal vector} $u$. The set of support points of $C$ is
    denoted $\spts C$, and the closure of $\spts C$ by $\closure{\spts}\,C$.
  \end{block}

  \begin{block}{Plain-English Intuition}
    Inequality (7.1) says: every point of $C$ scores at most as high as $x$
    itself when you measure ``progress in direction $u$'' by the inner
    product $\ip{\cdot}{u}$. Geometrically, the hyperplane through $x$
    perpendicular to $u$ is a flat wall that touches $C$ at (at least) the
    point $x$, with the whole set $C$ lying on one side. A \textbf{support
    point} is exactly a point where such a wall can be placed --- $x$ is
    ``exposed'' to the outside along direction $u$.
  \end{block}

  \framebreak

  \begin{block}{Proposition 7.2}
    Let $C$ and $D$ be nonempty subsets of $\Hil$ such that $C \subset D$.
    Then $C \cap \spts D \subset \spts C = C \cap \spts \closure{C}$.
  \end{block}

  \begin{block}{Intuition}
    If $x$ supports the bigger set $D$, it certainly supports the smaller set
    $C \subset D$ (the same wall still keeps all of $C$ on one side, since
    $C \subset D$). And a point of $C$ supports $C$ exactly when it supports
    the closure $\closure{C}$ --- taking closure does not create or destroy
    support points already inside $C$.
  \end{block}

  \begin{block}{Proposition 7.3 (Support Points via the Normal Cone)}
    Let $C$ be a nonempty convex subset of $\Hil$. Then
    \[
      \spts C = \{x \in C : N_C x \smallsetminus \{0\} \ne \varnothing\}
        = N_C^{-1}(\Hil \smallsetminus \{0\}). \tag{7.2}
    \]
  \end{block}

  \begin{block}{Intuition}
    Recall $N_Cx$ collects the directions $u$ for which the supporting
    inequality (7.1) holds at $x$ (Chapter 6). Proposition~7.3 simply
    restates Definition~7.1 in that language: $x$ is a support point exactly
    when its normal cone contains \emph{some} nonzero vector, i.e., when $x$
    admits at least one outward-pointing normal direction along which a
    supporting hyperplane can be erected.
  \end{block}

\end{frame}

\begin{frame}[allowframebreaks]{Theorem 7.4 (Bishop--Phelps) and Consequences}

  \begin{block}{Theorem 7.4 (Bishop--Phelps)}
    Let $C$ be a nonempty closed convex subset of $\Hil$. Then
    \[
      \spts C = P_C(\Hil \smallsetminus C) \quad \text{and} \quad
      \closure{\spts}\,C = \bdry C. \tag{7.3}
    \]
  \end{block}

  \begin{block}{Plain-English Intuition}
    The first identity says: \textbf{every} support point of $C$ arises as
    the metric projection of \emph{some} outside point onto $C$, and
    \emph{every} projection of an outside point is a support point (with
    normal vector pointing from $x$ back toward the source point, by the
    variational characterization of $P_C$). The second identity is the
    headline result: although a boundary point of $C$ need not itself be a
    support point (Example~7.7 below shows this can fail), the support
    points are always \textbf{dense} in the boundary --- you can approximate
    any boundary point arbitrarily well by genuine support points.
  \end{block}

  \framebreak

  \begin{block}{Proposition 7.5}
    Let $C$ be a convex subset of $\Hil$ such that $\Int C \ne \varnothing$.
    Then $\bdry C \subset \spts \closure{C}$ and
    $C \cap \bdry C \subset \spts C$.
  \end{block}

  \begin{block}{Corollary 7.6}
    Let $C$ be a nonempty closed convex subset of $\Hil$ and suppose that one
    of the following holds: (i)~$\Int C \ne \varnothing$; (ii)~$C$ is a closed
    affine subspace; (iii)~$\Hil$ is finite-dimensional. Then
    $\spts C = \bdry C$.
  \end{block}

  \begin{block}{Why This Matters}
    Corollary~7.6(iii) is the case you will use most in practice: \textbf{in
    finite dimensions, every closed convex set has exactly the support
    points you would naively expect} --- all of $\bdry C$, no more, no less.
    The subtleties in Theorem~7.4 and Example~7.7 (next slide) are genuinely
    infinite-dimensional phenomena.
  \end{block}

\end{frame}

\begin{frame}{Example 7.7: A Boundary Point That Is Not a Support Point}

  \begin{exampleblock}{Example 7.7}
    Let $\Hil = \ell^2(\N)$ and $C = \ell^2_+(\N)$, a nonempty closed convex
    cone with \emph{empty interior}. For $x = (\xi_k)_{k\in\N} \in \Hil$, the
    projection is $P_Cx = (\max\{\xi_k,0\})_{k\in\N}$. By Theorem~7.4,
    $x = (\xi_k)_{k\in\N} \in C$ is a support point of $C$ if and only if
    $\{k \in \N : \xi_k = 0\} \ne \varnothing$ (i.e., $x$ has \emph{at least
    one} zero coordinate). Therefore
    \[
      \Big(\tfrac{1}{2^k}\Big)_{k\in\N} \in \bdry C \smallsetminus \spts C:
    \]
    this sequence has every coordinate strictly positive, so it lies on the
    boundary of the cone (it is not an interior point, since $\Int C =
    \varnothing$ here) but it is \emph{not} a support point, because no
    coordinate is exactly zero.
  \end{exampleblock}

  \begin{block}{Takeaway}
    This is exactly the phenomenon Corollary~7.6 rules out in finite
    dimensions: with $\Int C = \varnothing$ and $\Hil$ infinite-dimensional,
    none of conditions (i)--(iii) of Corollary~7.6 hold, and indeed
    $\spts C \subsetneq \bdry C$ strictly. \textbf{A boundary point of a
    closed convex set need not be a support point}, even when $C$ is a
    closed convex cone.
  \end{block}

\end{frame}

\section{7.2 Support Functions}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{Definition 7.8: The Support Function}

  \begin{block}{Definition 7.8}
    Let $C$ be a subset of $\Hil$. The \textbf{support function} of $C$ is
    \[
      \sigma_C : \Hil \to [-\infty,+\infty] : u \mapsto \sup \ip{C}{u}.
      \tag{7.4}
    \]
  \end{block}

  \begin{block}{Plain-English Intuition}
    Fix a direction $u$. Among all points $x \in C$, look at the ``score''
    $\ip{x}{u}$ --- how far $x$ projects along $u$ (scaled by $\|u\|$). The
    support function $\sigma_C(u)$ records the \emph{best possible score}:
    how far into direction $u$ the set $C$ reaches. If $C$ is empty,
    $\sigma_C \equiv -\infty$ by convention (the sup of the empty set). If
    $C$ is unbounded in direction $u$, $\sigma_C(u) = +\infty$. Otherwise
    $\sigma_C(u)$ is a genuine finite real number for that $u$.
  \end{block}

  \begin{alertblock}{Why This Function Fully Characterizes Closed Convex Sets}
    Knowing $\sigma_C(u)$ for \emph{every} $u \in \Hil$ tells you, for every
    direction, the exact half-space $\{x : \ip{x}{u} \le \sigma_C(u)\}$ that
    must contain $C$. Intersecting all of these half-spaces recovers $C$
    exactly when $C$ is closed and convex (Corollary~7.12) --- so $\sigma_C$
    is a complete, direction-indexed ``fingerprint'' of $C$.
  \end{alertblock}

\end{frame}

\begin{frame}{Example 7.9: Support Function of an Interval in $\R$}

  \begin{exampleblock}{Example 7.9}
    Let $\Omega$ be a nonempty subset of $\R$, set $\underline\omega = \inf
    \Omega$ and $\overline\omega = \sup \Omega$. Then
    \[
      (\forall \xi \in \R) \quad \sigma_\Omega(\xi) =
      \begin{cases}
        \underline\omega\,\xi, & \text{if } \xi < 0; \\
        0, & \text{if } \xi = 0; \\
        \overline\omega\,\xi, & \text{if } \xi > 0.
      \end{cases}
      \tag{7.5}
    \]
  \end{exampleblock}

  \begin{block}{Intuition}
    In $\R$, ``direction $u=\xi$'' just means ``positive'' ($\xi>0$) or
    ``negative'' ($\xi<0$). Walking in the positive direction, the farthest
    you can score is $\overline\omega$ (the supremum of $\Omega$) times the
    scale $\xi$; walking in the negative direction, the farthest score comes
    from the \emph{infimum} $\underline\omega$, again scaled by $\xi$ (note
    $\xi<0$, so a very negative $\underline\omega$ can still give a large
    positive product). At $\xi=0$ every point scores $0$, so the sup is $0$.
    For instance, if $\Omega = [-2,5]$, then $\sigma_\Omega(3) = 5\cdot 3=15$
    and $\sigma_\Omega(-3) = (-2)(-3) = 6$.
  \end{block}

\end{frame}

\begin{frame}{Step-by-Step: The Support Function of the Unit Disk}

  \begin{block}{Example 7.10, Specialized: $C = $ Closed Unit Disk in $\R^2$}
    Let $C_{\mathrm{disk}} = \{x \in \R^2 : \|x\| \le 1\}$. We claim
    $\sigma_{C_{\mathrm{disk}}}(u) = \|u\|$ for every $u \in \R^2$: the
    support function of the unit ball of a norm is the \textbf{dual norm},
    and the Euclidean norm is (famously) self-dual.
  \end{block}

  \begin{block}{Derivation, by Cauchy--Schwarz}
    For any $x$ with $\|x\|\le 1$ and any $u \in \R^2$,
    \[
      \ip{x}{u} \le \|x\|\,\|u\| \le \|u\|,
    \]
    with equality throughout when $x = u/\|u\|$ (a unit vector, so it lies in
    $C_{\mathrm{disk}}$, and it is aligned exactly with $u$). Hence the
    supremum over $x \in C_{\mathrm{disk}}$ is attained and equals $\|u\|$:
    \[
      \sigma_{C_{\mathrm{disk}}}(u) = \sup_{\|x\|\le 1} \ip{x}{u} = \|u\|.
    \]
  \end{block}

  \begin{exampleblock}{Concrete Numbers}
    Take $u = (3,4)$, so $\|u\| = \sqrt{3^2+4^2} = 5$. The maximizer is
    $x^\ast = u/\|u\| = (0.6,\,0.8) \in C_{\mathrm{disk}}$, and indeed
    $\ip{x^\ast}{u} = 0.6\cdot 3 + 0.8 \cdot 4 = 1.8+3.2 = 5 = \|u\|$. So
    $\sigma_{C_{\mathrm{disk}}}(3,4) = 5$.
  \end{exampleblock}

\end{frame}

  \centering

\end{frame}

\begin{frame}[allowframebreaks]{Proposition 7.11 and Corollary 7.12: Half-Spaces Reconstruct $C$}

  \begin{block}{Proposition 7.11}
    Let $C$ be a subset of $\Hil$ and set
    \[
      (\forall u \in \Hil) \quad H_u = \{x \in \Hil : \ip{x}{u} \le
      \sigma_C(u)\}. \tag{7.6}
    \]
    Then $\closure{\conv}\,C = \bigcap_{u \in \Hil} H_u$.
  \end{block}

  \begin{block}{Intuition}
    Each $H_u$ is a closed half-space forced to contain $C$ (that is exactly
    what $\sigma_C(u)$ measures). Intersecting \emph{all} of them over every
    possible direction $u$ squeezes down as tightly as possible around $C$
    from every side at once --- and Proposition~7.11 says this intersection
    lands \emph{exactly} on the closed convex hull of $C$, no tighter and no
    looser.
  \end{block}

  \begin{block}{Corollary 7.12}
    Let $C$ be a closed convex subset of $\Hil$. Then $C$ is the intersection
    of all the closed half-spaces of $\Hil$ that contain $C$.
  \end{block}

  \begin{block}{Why This Matters}
    If $C$ is already closed and convex, $\closure{\conv}\,C = C$, so
    Proposition~7.11 collapses to Corollary~7.12: \textbf{a closed convex set
    is completely determined by the collection of half-spaces containing
    it}, i.e., by its support function alone. This is the formal statement
    behind the ``fingerprint'' claim on the Definition~7.8 slide.
  \end{block}

\end{frame}

\begin{frame}{Proposition 7.13: Support Function Is Blind to Taking Convex Hulls}

  \begin{block}{Proposition 7.13}
    Let $C$ be a subset of $\Hil$. Then
    \[
      \sigma_C = \sigma_{\conv C} = \sigma_{\closure{\conv}\,C}.
    \]
  \end{block}

  \begin{block}{Intuition}
    Taking convex combinations of points in $C$ can never beat the best
    individual point of $C$ when scoring $\ip{\cdot}{u}$ (a weighted average
    of scores is at most the largest score being averaged), so
    $\sigma_{\conv C} \le \sigma_C$; and clearly $\sigma_C \le
    \sigma_{\conv C}$ since $C \subset \conv C$. Passing to the closure adds
    only limit points, whose scores are themselves limits of scores already
    achieved in $\conv C$, so the supremum does not change either. In short:
    \textbf{the support function only sees the ``outer shape''} of $C$, and
    is indifferent to filling in interior points or boundary limit points.
  \end{block}

\end{frame}

\section{7.3 Polar Sets}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{Definition 7.14: The Polar Set}

  \begin{block}{Definition 7.14}
    Let $C$ be a subset of $\Hil$. The \textbf{polar set} of $C$ is
    \[
      C^\odot = \mathrm{lev}_{\le 1}\, \sigma_C = \{u \in \Hil :
      (\forall x \in C)\ \ip{x}{u} \le 1\}. \tag{7.7}
    \]
  \end{block}

  \begin{block}{Plain-English Intuition}
    $C^\odot$ collects exactly the directions $u$ for which the support
    function $\sigma_C(u)$ does not exceed $1$. Equivalently: $u \in
    C^\odot$ means the half-space $\{x : \ip{x}{u}\le 1\}$ --- the ``unit''
    half-space in direction $u$ --- is wide enough to still contain all of
    $C$. \textbf{The polar set is a dual, direction-indexed shadow of $C$}:
    large $C$ (reaching far in many directions) tends to force $C^\odot$ to
    be small (few directions with $\sigma_C(u)\le 1$), and vice versa.
  \end{block}

  \begin{block}{Consistency Check on $C_{\mathrm{disk}}$}
    Since $\sigma_{C_{\mathrm{disk}}}(u) = \|u\|$, we get
    $C_{\mathrm{disk}}^\odot = \{u : \|u\| \le 1\} = C_{\mathrm{disk}}$: the
    closed unit disk is \textbf{self-polar} (Exercise~7.7 in the book).
  \end{block}

\end{frame}

\begin{frame}{Example 7.15: The Polar of a Cone Is Its Polar Cone}

  \begin{exampleblock}{Example 7.15}
    Let $K$ be a cone in $\Hil$. Then $K^\odot = K^\ominus$ (the polar cone
    from Chapter 6). In particular, if $K$ is a linear subspace, then
    $K^\odot = K^\ominus = K^\perp$.
  \end{exampleblock}

  \begin{block}{Intuition}
    For a cone $K$, if some $x \in K$ scored $\ip{x}{u} > 0$ for a candidate
    normal $u$, then scaling $x$ up (still inside $K$, since $K$ is a cone)
    would make $\ip{x}{u}$ arbitrarily large, violating the ``$\le 1$''
    bound in (7.7). So membership in $C^\odot$ for a cone collapses to the
    strictly stronger condition $\sup\ip{K}{u} \le 0$, which is exactly the
    defining condition of the polar cone $K^\ominus$ from Chapter~6. For a
    subspace $V$, nonnegativity in both directions forces equality, i.e.\
    orthogonality, recovering the familiar $V^\ominus = V^\perp$.
  \end{block}

\end{frame}

\begin{frame}{Running Numeric Example: The Square and Its Polar Diamond}

  \begin{exampleblock}{Setup}
    Let $C_{\mathrm{sq}} = [-1,1]^2$, the closed unit ball of the sup-norm
    $\|\cdot\|_\infty$. By Example~7.10 (dual norms), its support function is
    the \emph{dual} norm of $\|\cdot\|_\infty$, namely the $\ell^1$ norm:
    \[
      \sigma_{C_{\mathrm{sq}}}(u) = |u_1| + |u_2| = \|u\|_1.
    \]
  \end{exampleblock}

  \begin{block}{Two Numeric Checks of $\sigma_{C_{\mathrm{sq}}}$}
    For $u=(1,1)$: $\sigma_{C_{\mathrm{sq}}}(1,1) = \sup_{x \in C_{\mathrm{sq}}}
    (x_1+x_2) = 1+1 = 2$, attained at $x^\ast=(1,1)$; and indeed
    $\|(1,1)\|_1 = 2$. For $u=(2,-1)$: the maximizer is $x^\ast=(1,-1)$,
    giving $\sigma_{C_{\mathrm{sq}}}(2,-1) = 2\cdot 1 + (-1)(-1) = 3 =
    \|(2,-1)\|_1$.
  \end{block}

  \begin{block}{The Polar Set: From $\sigma_{C_{\mathrm{sq}}}$ to $C_{\mathrm{sq}}^\odot$}
    By (7.7), $C_{\mathrm{sq}}^\odot = \{u : \|u\|_1 \le 1\}$, the closed
    unit ball of $\|\cdot\|_1$ --- a diamond with vertices $(\pm 1,0)$,
    $(0,\pm1)$. Check $u=(1,1)$: $\|(1,1)\|_1=2>1$, so $(1,1) \notin
    C_{\mathrm{sq}}^\odot$ (consistent with $\sigma_{C_{\mathrm{sq}}}(1,1)=2>1$).
    Check $u=(0.5,0.5)$: $\|(0.5,0.5)\|_1=1$, so this point sits exactly on
    $\bdry\, C_{\mathrm{sq}}^\odot$.
  \end{block}

\end{frame}

\begin{frame}[allowframebreaks]{Proposition 7.16: Basic Properties of the Polar}

  \begin{block}{Proposition 7.16}
    Let $C$ and $D$ be subsets of $\Hil$. Then:
    \begin{enumerate}
      \item[(i)] $C^\perp \subset C^\ominus \subset C^\odot$.
      \item[(ii)] $0 \in C^\odot$, and $C^\odot$ is closed and convex.
      \item[(iii)] $C \cup \{0\} \subset C^{\odot\odot}$.
      \item[(iv)] If $C \subset D$, then $D^\odot \subset C^\odot$ and
        $C^{\odot\odot} \subset D^{\odot\odot}$.
      \item[(v)] $C^{\odot\odot\odot\odot} = C^\odot$.
      \item[(vi)] $(\closure{\conv}\,C)^\odot = C^\odot$.
    \end{enumerate}
  \end{block}

  \begin{block}{Intuition, Item by Item}
    (ii): $u=0$ trivially satisfies $\ip{x}{0}=0\le 1$ for every $x$, and
    $C^\odot$ is an intersection of closed half-spaces (one per $x \in C$),
    hence closed and convex automatically. (iii): applying the defining
    inequality of $C^{\odot\odot}$ to elements of $C^\odot$ shows every point
    of $C$ (and $0$) satisfies it. (iv): \textbf{polarity reverses
    inclusions} --- a bigger set imposes more constraints, shrinking its
    polar; this is the same ``duality flips order'' pattern seen for polar
    cones in Chapter~6. (vi): matches Proposition~7.13 --- the polar, like
    the support function it is built from, cannot distinguish $C$ from its
    closed convex hull.
  \end{block}

  \framebreak

  \begin{block}{Remark 7.17 (Specializing to Cones)}
    Combining Example~7.15 with Proposition~7.16, for an arbitrary cone $K$:
    $K \subset K^{\ominus\ominus}$, $K^{\ominus\ominus\ominus} =
    K^\ominus$, and $(\conv K)^\ominus = K^\ominus$. For a linear subspace
    $V$: $V \subset V^{\perp\perp}$, $V^{\perp\perp\perp}=V^\perp$, and
    $\closure{V}^\perp = V^\perp$.
  \end{block}

\end{frame}

\begin{frame}{Theorem 7.18 (The Bipolar Theorem) and Its Corollary}

  \begin{block}{Theorem 7.18 (Bipolar Theorem)}
    Let $C$ be a subset of $\Hil$. Then
    \[
      C^{\odot\odot} = \closure{\conv}\,(C \cup \{0\}).
    \]
  \end{block}

  \begin{block}{Plain-English Intuition}
    Polarizing twice does not quite return $C$ in general --- it returns
    the \emph{smallest closed convex set containing $C$ and the origin}.
    Any ``non-convexity,'' ``non-closedness,'' or ``missing the origin'' in
    $C$ gets repaired by the double-polar operation, but nothing beyond
    that changes. This is the exact convex-analytic analogue of the
    biorthogonal-complement fact $V^{\perp\perp}=\closure{V}$ for subspaces.
  \end{block}

  \begin{block}{Corollary 7.19}
    Let $C$ be a subset of $\Hil$. Then: (i) $C$ is closed, convex, and
    contains the origin $\iff C^{\odot\odot}=C$; (ii) $C$ is a nonempty
    closed convex cone $\iff C^{\ominus\ominus}=C$; (iii) $C$ is a closed
    linear subspace $\iff C^{\perp\perp}=C$.
  \end{block}

\end{frame}

\begin{frame}{Bipolar Theorem, Checked on the Running Examples}

  \begin{exampleblock}{$C_{\mathrm{sq}} = [-1,1]^2$}
    $C_{\mathrm{sq}}$ is already closed, convex, and contains $0$, so
    Corollary~7.19(i) predicts $C_{\mathrm{sq}}^{\odot\odot} =
    C_{\mathrm{sq}}$. Indeed, we computed $C_{\mathrm{sq}}^\odot = \{u :
    \|u\|_1 \le 1\}$ (the $\ell^1$ diamond); polarizing once more returns
    the sup-norm ball, since the dual of $\|\cdot\|_1$ is $\|\cdot\|_\infty$:
    $C_{\mathrm{sq}}^{\odot\odot} = \{x : \|x\|_\infty \le 1\} =
    C_{\mathrm{sq}}$.
  \end{exampleblock}

  \begin{exampleblock}{$C_{\mathrm{disk}} = B(0;1)$}
    Self-polar, so trivially $C_{\mathrm{disk}}^{\odot\odot} =
    C_{\mathrm{disk}}^\odot = C_{\mathrm{disk}}$, again consistent with
    Corollary~7.19(i).
  \end{exampleblock}

  \begin{block}{A Non-Convex Sanity Check}
    Let $C = \{(1,0),(0,1)\}$ (two points, not convex, not containing $0$).
    Then $\sigma_C(u) = \max\{u_1,u_2\}$, so $C^\odot = \{u : u_1 \le 1
    \text{ and } u_2 \le 1\}$, and one finds $C^{\odot\odot} =
    \closure{\conv}(C \cup \{0\})$: the triangle with vertices $(0,0)$,
    $(1,0)$, $(0,1)$ --- exactly Theorem~7.18's prediction, and visibly
    larger than $C$ itself.
  \end{block}

\end{frame}

\section{Python Verification}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}[fragile]{Python: Support Functions and Polar Sets, Numerically}

The code below computes $\sigma_C(u)$ by brute-force maximization over a fine
sample of each set's boundary, verifies the disk formula
$\sigma_{C_{\mathrm{disk}}}(u)=\|u\|$ and the square formula
$\sigma_{C_{\mathrm{sq}}}(u)=\|u\|_1$, and checks numerically that
$C_{\mathrm{sq}}^{\odot\odot}=C_{\mathrm{sq}}$ (Corollary~7.19(i)) by
sampling many directions.

\begin{lstlisting}[style=pycode]
import numpy as np
rng = np.random.default_rng(0)

# --- sigma_C via brute-force sampling of C's boundary/interior -----------
def sigma_disk(u):                       # closed unit disk
    return np.linalg.norm(u)             # exact: sigma(u) = ||u||

def sigma_square(u):                     # C_sq = [-1,1]^2
    return abs(u[0]) + abs(u[1])         # exact: sigma(u) = ||u||_1

# check against brute-force sup over many sampled points of each set
disk_pts = np.array([[np.cos(t), np.sin(t)]
                      for t in np.linspace(0, 2*np.pi, 4000)])
sq_pts = rng.uniform(-1, 1, size=(20000, 2))
sq_pts = np.vstack([sq_pts, [[1,1],[1,-1],[-1,1],[-1,-1]]])  # incl. corners

max_err_disk, max_err_sq = 0.0, 0.0
for _ in range(200):
    u = rng.uniform(-3, 3, size=2)
    bf_disk = np.max(disk_pts @ u)
    bf_sq = np.max(sq_pts @ u)
    max_err_disk = max(max_err_disk, abs(bf_disk - sigma_disk(u)))
    max_err_sq = max(max_err_sq, abs(bf_sq - sigma_square(u)))

print(f"max |brute-force - formula| on disk:   {max_err_disk:.5f}")
print(f"max |brute-force - formula| on square: {max_err_sq:.5f}")

# --- bipolar check: C_sq^(odd-odd) == C_sq -------------------------------
def in_polar_of_square(u, n_dirs=400):
    thetas = np.linspace(0, 2*np.pi, n_dirs)
    for t in thetas:
        x = np.array([np.cos(t), np.sin(t)])   # sample C_sq^o boundary dirs
        pass
    return None  # C_sq^o = {v : ||v||_1 <= 1}; test membership directly

def bipolar_square_contains(x, n_samples=2000):
    # x in C_sq^(oo)  <=>  <x|v> <= 1 for all v in C_sq^o = {||v||_1<=1}
    thetas = rng.uniform(0, 2*np.pi, n_samples)
    vs = np.array([[np.cos(t), np.sin(t)] for t in thetas])
    vs = vs / np.abs(vs).sum(axis=1, keepdims=True)  # project onto l1 ball bdry
    return np.all(vs @ x <= 1 + 1e-9)

test_pts = [(1,1), (1,-1), (0.999,0.999), (1.001,1.001)]
for x in test_pts:
    print(f"x={x}: in C_sq^(oo)? {bipolar_square_contains(np.array(x))}"
          f"  (expect True for ||x||_inf<=1, False for (1.001,1.001))")
\end{lstlisting}

\end{frame}

\begin{frame}[allowframebreaks]{Chapter Summary}

  \begin{block}{The Three Big Objects}
    \centering
    \begin{tabular}{lp{6.6cm}}
      \toprule
      \textbf{Object} & \textbf{Role} \\
      \midrule
      $\spts C$ & support points; dense in $\bdry C$ always
        (Bishop--Phelps), equal to $\bdry C$ in finite dimensions \\
      $\sigma_C(u) = \sup\ip{C}{u}$ & support function; a complete,
        direction-indexed fingerprint of $\closure{\conv}\,C$ \\
      $C^\odot = \{u : \sup\ip{C}{u}\le 1\}$ & polar set; dual shadow of
        $C$, closed and convex by construction \\
      \bottomrule
    \end{tabular}
  \end{block}

  \begin{block}{The Key Identities}
    \begin{itemize}
      \item $\closure{\conv}\,C = \bigcap_{u} \{x : \ip{x}{u} \le
        \sigma_C(u)\}$ (Proposition~7.11): half-spaces reconstruct the
        closed convex hull.
      \item $\sigma_C = \sigma_{\conv C} = \sigma_{\closure{\conv}\,C}$
        (Proposition~7.13): the support function is blind to convexifying
        or closing $C$.
      \item $C^{\odot\odot} = \closure{\conv}\,(C\cup\{0\})$
        (Theorem~7.18, the bipolar theorem): double polarity repairs
        non-convexity, non-closedness, and a missing origin --- nothing
        else.
      \item For a cone $K$: $K^\odot = K^\ominus$ (Example~7.15); for a
        subspace $V$: $V^\odot=V^\ominus=V^\perp$.
    \end{itemize}
  \end{block}

  \framebreak

  \begin{block}{Running Examples, Side by Side}

      $[-1,1]^2$ (unit square) & $\|u\|_1$ & $\{u:\|u\|_1\le 1\}$ (diamond) \\
      \bottomrule
