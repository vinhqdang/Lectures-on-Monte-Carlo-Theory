\section{1.1 Sets and Basic Notation}

\begin{frame}[allowframebreaks]
\frametitle{Sets and Basic Notation}

\begin{itemize}
\item A \textbf{set} is a collection of distinct objects
\item Nonempty sets: $A, B, C, \ldots$
\item Power set: $2^A$ is the family of all subsets of $A$
\item Set operations: union $A \cup B$, intersection $A \cap B$, complement $A^c$
\item Cartesian product: $A \times B = \{(a,b) : a \in A, b \in B\}$
\end{itemize}

\vspace{0.5cm}

\textbf{Key Definitions:}
\begin{enumerate}
\item A \textbf{line segment} between $x, y \in \mathbb{X}$ consists of points of the form:
$$[x, y] = \{(1-\alpha)x + \alpha y \mid 0 \leq \alpha \leq 1\}$$

\item Related notations:
\begin{align*}
]x, y] &= \{(1-\alpha)x + \alpha y \mid 0 < \alpha \leq 1\} \\
[x, y[ &= \{(1-\alpha)x + \alpha y \mid 0 \leq \alpha < 1\} \\
]x, y[ &= \{(1-\alpha)x + \alpha y \mid 0 < \alpha < 1\}
\end{align*}
\end{enumerate}

\end{frame}

% Section 2: Operators
\section{1.2 Operators}

\begin{frame}[allowframebreaks]
\frametitle{Single-Valued and Set-Valued Operators}

\textbf{Definitions:}
\begin{enumerate}
\item \textbf{Single-valued operator}: $T: X \to Y$ maps every point $x \in X$ to exactly one point $Tx \in Y$

\item \textbf{Set-valued operator}: $A: X \to 2^Y$ maps every point $x \in X$ to a set $Ax \subseteq Y$

\item An operator can be \textbf{single-valued but not everywhere defined}:
$$A: X \to 2^Y; \quad x \mapsto \begin{cases} \{Tx\}, & \text{if } x \in D; \\ \emptyset, & \text{otherwise} \end{cases}$$

where $D \subseteq X$
\end{enumerate}

\framebreak

\textbf{Graph and Inverse:}
\begin{itemize}
\item The \textbf{graph} of $A$ is:
$$\text{gra } A = \{(x, y) \in X \times Y : y \in Ax\}$$

\item For $C \subseteq X$: 
$$A(C) = \bigcup_{x \in C} Ax$$

\item \textbf{Composition} $B \circ A$ (where $B: Y \to 2^Z$):
$$B \circ A: X \to 2^Z; \quad x \mapsto B(Ax) = \bigcup_{y \in Ax} By$$
\end{itemize}

\framebreak

\textbf{Domain, Range, and Inverse:}
\begin{itemize}
\item \textbf{Domain}: $\text{dom } A = \{x \in X : Ax \neq \emptyset\}$
\item \textbf{Range}: $\text{ran } A = A(X)$
\item \textbf{Inverse} $A^{-1}$: $\text{gra } A^{-1} = \{(y,x) \in Y \times X : (x,y) \in \text{gra } A\}$

Key properties:
\begin{align*}
\text{dom } A^{-1} &= \text{ran } A \\
\text{ran } A^{-1} &= \text{dom } A
\end{align*}
\end{itemize}

\end{frame}

\begin{frame}[fragile]
\frametitle{Python Example: Operators}

% Section 3: Order
\section{1.3 Order}

\begin{frame}[allowframebreaks]
\frametitle{Ordered Sets}

\textbf{Binary Relations:} Let $A$ be nonempty and $\preceq$ a binary relation on $A \times A$.

\textbf{Properties:}
\begin{enumerate}
\item \textbf{Reflexive}: $(\forall a \in A) \quad a \preceq a$
\item \textbf{Transitive}: $(\forall a,b,c \in A) \quad a \preceq b \text{ and } b \preceq c \Rightarrow a \preceq c$
\item \textbf{Antisymmetric}: $(\forall a,b \in A) \quad a \preceq b \text{ and } b \preceq a \Rightarrow a = b$
\item \textbf{Connex}: $(\forall a,b \in A) \quad a \preceq b \text{ or } b \preceq a$
\end{enumerate}

\framebreak

\textbf{Ordered Set Hierarchies:}

\begin{table}[h]
\centering
\begin{tabular}{|c|c|c|}
\hline
\textbf{Type} & \textbf{Requirements} & \textbf{Properties} \\
\hline
Directed Set & (1), (2) & Partial order with universal bounds \\
Partially Ordered Set & (1), (2), (3) & Transitive, antisymmetric relation \\
Totally Ordered Set & (1), (2), (3), (4) & Every two elements comparable \\
\hline
\end{tabular}
\end{table}

\vspace{0.5cm}

\textbf{Bounds:} For $B \subseteq A$:
\begin{itemize}
\item \textbf{Upper bound}: $b \in B$ is an upper bound if $(\forall a \in B) \, a \preceq b$
\item \textbf{Maximal element}: $a \in A$ is maximal if $(\forall c \in A) \, a \preceq c \Rightarrow c = a$
\end{itemize}

\framebreak

\textbf{Zorn's Lemma (Fact 1.1):}
\begin{theorem}
Let $A$ be a partially ordered set such that every chain in $A$ has an upper bound. Then $A$ contains a maximal element.
\end{theorem}

\vspace{0.3cm}

This fundamental result is equivalent to the axiom of choice and has important applications in functional analysis and optimization.

\end{frame}

% Section 4: Nets
\section{1.4 Nets}

\begin{frame}[allowframebreaks]
\frametitle{Nets and Generalized Sequences}

\textbf{Motivation:} Sequences indexed by $\mathbb{N}$ are insufficient for characterizing topological properties in general spaces. We use \textbf{nets} instead.

\textbf{Definition:} A \textbf{net} (or \textbf{generalized sequence}) in $X$ indexed by $A$ is an operator $(x_a)_{a \in A}$.

\textbf{Key property:} $A$ is a \textbf{directed set}, meaning:
\begin{itemize}
\item Reflexive: $a \preceq a$ for all $a \in A$
\item Transitive: if $a \preceq b$ and $b \preceq c$, then $a \preceq c$
\item Directed: for any $a, b \in A$, there exists $c \in A$ with $a \preceq c$ and $b \preceq c$
\end{itemize}

\framebreak

\textbf{Convergence of Nets:} A net $(x_a)_{a \in A}$ converges to $x \in X$ if:
$$(\forall V \in \mathcal{V}(x))(\exists c \in A)(\forall a \in A) \, a \succeq c \Rightarrow x_a \in V$$

\textbf{Subnet and Subsequence:}
\begin{itemize}
\item A net $(y_b)_{b \in B}$ is a \textbf{subnet} of $(x_a)_{a \in A}$ via $k: B \to A$ if:
$$(\forall b \in B) \quad y_b = x_{k(b)}$$
with the property: $(\forall a \in A)(\exists d \in B)(\forall b \in B) \, b \succeq d \Rightarrow k(b) \succeq a$

\item A \textbf{sequence} is a net indexed by $\mathbb{N}$
\end{itemize}

\framebreak

\centering
\includegraphics[width=0.8\textwidth]{net_vs_sequence.pdf}
\caption{Nets generalize sequences to arbitrary topological spaces}
\end{figure}

\end{frame}

% Section 5: The Extended Real Line
\section{1.5 The Extended Real Line}

\begin{frame}[allowframebreaks]
\frametitle{The Extended Real Line}

\textbf{Definition:} The \textbf{extended real line} is:
$$[-\infty, +\infty] = \mathbb{R} \cup \{-\infty\} \cup \{+\infty\}$$

with the order extended by:
$$(\forall \xi \in \mathbb{R}) \quad -\infty < \xi < +\infty$$

\framebreak

\textbf{Arithmetic Rules (Remark 1.3):}
\begin{itemize}
\item $+\infty + (+\infty) = +\infty$, \quad $-\infty + (-\infty) = -\infty$
\item $+\infty + \xi = +\infty$ for all $\xi > -\infty$
\item $-\infty + \xi = -\infty$ for all $\xi < +\infty$
\item Expressions like $+\infty - \infty$, $0 \cdot \infty$, $\infty/\infty$ are \textbf{undefined}
\item For $\alpha > 0$: $\alpha \cdot (+\infty) = +\infty$, \quad $\alpha \cdot (-\infty) = -\infty$
\item For $\alpha < 0$: $\alpha \cdot (+\infty) = -\infty$, \quad $\alpha \cdot (-\infty) = +\infty$
\end{itemize}

\framebreak

\begin{figure}
\centering
\includegraphics[width=0.9\textwidth]{extended_real_line.pdf}
\caption{The extended real line: $[-\infty, +\infty]$}
\end{figure}

\framebreak

\textbf{Infimum and Supremum:} For $S \subseteq [-\infty, +\infty]$:
\begin{itemize}
\item $\inf S$ is the greatest lower bound: $(\forall a \in S) \, \inf S \leq a$
\item $\sup S$ is the least upper bound: $(\forall a \in S) \, a \leq \sup S$
\item Special conventions: $\inf \emptyset = +\infty$, $\sup \emptyset = -\infty$
\end{itemize}

\end{frame}

% Section 6: Functions
\section{1.6 Functions}

\begin{frame}[allowframebreaks]
\frametitle{Functions with Values in the Extended Real Line}

\textbf{Definition:} Let $f: X \to [-\infty, +\infty]$. 

The \textbf{domain} of $f$ is:
$$\text{dom } f = \{x \in X : f(x) < +\infty\}$$

\textbf{Key Sets Associated with $f$:}

\begin{align*}
\text{gra } f &= \{(x, \xi) \in X \times \mathbb{R} : f(x) = \xi\} \\
\text{epi } f &= \{(x, \xi) \in X \times \mathbb{R} : f(x) \leq \xi\} \\
\text{lev}_\xi f &= \{x \in X : f(x) \leq \xi\} \\
\text{lev}_{<\xi} f &= \{x \in X : f(x) < \xi\}
\end{align*}

\framebreak

\textbf{Function Properties:}
\begin{itemize}
\item $f$ is \textbf{proper} if $-\infty \notin f(X)$ and $\text{dom } f \neq \emptyset$
\item $f$ is \textbf{increasing} (or monotone) if: $(\forall \xi, \eta \in D) \, \xi < \eta \Rightarrow f(\xi) \leq f(\eta)$
\item $f$ is \textbf{strictly increasing} if: $(\forall \xi, \eta \in D) \, \xi < \eta \Rightarrow f(\xi) < f(\eta)$
\end{itemize}

\framebreak

\begin{figure}
\centering
\includegraphics[width=0.95\textwidth]{function_concepts.pdf}
\caption{Graph, epigraph, and level sets of a function}
\end{figure}

\framebreak

\textbf{Lemma 1.6:} For a family of functions $\{f_i\}_{i \in I}$ from $X$ to $[-\infty, +\infty]$:

\begin{align*}
\text{epi } \left(\sup_{i \in I} f_i\right) &= \bigcap_{i \in I} \text{epi } f_i \\
\text{epi } \left(\min_{i \in I} f_i\right) &= \bigcup_{i \in I} \text{epi } f_i \quad (I \text{ finite})
\end{align*}

\end{frame}

% Section 7: Topological Spaces
\section{1.7 Topological Spaces}

\begin{frame}[allowframebreaks]
\frametitle{Topological Spaces and Basic Concepts}

\textbf{Definition:} A \textbf{topology} on $X$ is a family $\mathcal{T}$ of subsets of $X$ such that:
\begin{enumerate}
\item $\emptyset, X \in \mathcal{T}$
\item Arbitrary unions of sets in $\mathcal{T}$ are in $\mathcal{T}$
\item Finite intersections of sets in $\mathcal{T}$ are in $\mathcal{T}$
\end{enumerate}

\textbf{Key Definitions:}
\begin{itemize}
\item \textbf{Open set}: an element of $\mathcal{T}$
\item \textbf{Closed set}: complement of an open set
\item \textbf{Neighborhood} of $x$: any open set containing $x$
\item $\mathcal{V}(x)$: collection of all neighborhoods of $x$
\end{itemize}

\framebreak

\textbf{Closure and Interior:}
\begin{itemize}
\item \textbf{Closure} $\overline{C}$: smallest closed set containing $C$
\item \textbf{Interior} $C^\circ$: largest open set contained in $C$
\item $C$ is closed iff $C = \overline{C}$
\item $C$ is open iff $C = C^\circ$
\end{itemize}

\framebreak

\begin{figure}
\centering
\includegraphics[width=0.85\textwidth]{closure_interior.pdf}
\caption{Closure and interior of sets in topological spaces}
\end{figure}

\framebreak

\textbf{Hausdorff Spaces:} A space $X$ is \textbf{Hausdorff} if:
$$(\forall x, y \in X, x \neq y)(\exists U, V \in \mathcal{V}(x), \mathcal{V}(y)) \quad U \cap V = \emptyset$$

\vspace{0.3cm}

\textbf{Compactness:} A set $C$ in a topological space $X$ is \textbf{compact} if every open cover has a finite subcover.

\framebreak

\begin{figure}
\centering
\includegraphics[width=0.85\textwidth]{compactness.pdf}
\caption{Compact vs. non-compact sets}
\end{figure}

\end{frame}

% Section 8: Two-Point Compactification
\section{1.8 Two-Point Compactification of $\mathbb{R}$}

\begin{frame}[allowframebreaks]
\frametitle{Two-Point Compactification}

\textbf{The Extended Real Line as a Compact Space:} The extended real line $[-\infty, +\infty]$ equipped with the topology having as a base:
\begin{itemize}
\item Open real intervals $(a, b)$
\item Intervals of the form $[-\infty, \xi)$ and $(\xi, +\infty]$
\end{itemize}

is a \textbf{compact Hausdorff space}.

\framebreak

\textbf{Fact 1.15:} Let $(\xi_a)_{a \in A}$ be a net in $[-\infty, +\infty]$. Then:

\begin{enumerate}
\item The nets $\left(\inf_{b \in A, a \preceq b} \xi_b\right)_{a \in A}$ and $\left(\sup_{b \in A, a \preceq b} \xi_b\right)_{a \in A}$ converge to $\underline{\lim} \xi_a$ and $\overline{\lim} \xi_a$, respectively.

\item The net $(\xi_a)_{a \in A}$ possesses subnets converging to $\liminf \xi_a$ and $\limsup \xi_a$.

\item The net $(\xi_a)_{a \in A}$ converges iff $\liminf \xi_a = \limsup \xi_a$.
\end{enumerate}

\framebreak

\textbf{Lemma 1.16:} For nets $(\xi_a)_{a \in A}$ and $(\eta_a)_{a \in A}$ in $[-\infty, +\infty]$:
$$\underline{\lim} \xi_a + \underline{\lim} \eta_a \leq \underline{\lim}(\xi_a + \eta_a) \leq \overline{\lim}(\xi_a + \eta_a) \leq \overline{\lim} \xi_a + \overline{\lim} \eta_a$$

\end{frame}

% Section 9: Lower Semicontinuity
\section{1.10 Lower Semicontinuity}

\begin{frame}[allowframebreaks]
\frametitle{Lower Semicontinuity}

\textbf{Definition 1.21:} Let $X$ be a Hausdorff space and $f: X \to [-\infty, +\infty]$.

$f$ is \textbf{lower semicontinuous} at $x \in X$ if:
$$(\forall \xi \in ]-\infty, f(x)[)(\exists V \in \mathcal{V}(x)) \quad f(V) \subset [\xi, +\infty]$$

Equivalently, $f$ is lower semicontinuous at $x$ if:
$$\liminf_{y \to x} f(y) \geq f(x)$$

\framebreak

\textbf{Characterizations (Lemma 1.23):}

The following are equivalent:
\begin{enumerate}
\item $f$ is lower semicontinuous on $X$
\item $(\forall \xi \in \mathbb{R}) \quad \text{lev}_\xi f$ is closed
\item The epigraph $\text{epi } f$ is closed
\item For every $x \in X$: $f(x) = \sup_{(x_a)_{a \in A} \to x} \liminf f(x_a)$
\end{enumerate}

\framebreak

\begin{figure}
\centering
\includegraphics[width=0.85\textwidth]{lower_semicontinuous.pdf}
\caption{Lower semicontinuous function: $\liminf_{y \to x} f(y) = f(x)$}
\end{figure}

\framebreak

\textbf{Example 1.22:} The function
$$f: \mathbb{R} \to [-\infty, +\infty]; \quad f(x) = \begin{cases} 1/x, & \text{if } x > 0; \\ +\infty, & \text{otherwise} \end{cases}$$

is continuous on $\mathbb{R}$ with $\text{dom } f = \mathbb{R}_{++}$ being a proper open subset.

\end{frame}

% Section 10: Sequential Topological Notions
\section{1.11 Sequential Topological Notions}

\begin{frame}[allowframebreaks]
\frametitle{Sequential Topological Notions}

\textbf{Definition 1.33:} A subset $C$ of a Hausdorff space $X$ is \textbf{sequentially compact} if every sequence in $C$ has a sequential cluster point in $C$.

\vspace{0.3cm}

\textbf{Note:} Sequences are not sufficient to describe all topological properties in general spaces. For instance:
\begin{itemize}
\item A set can be closed but not sequentially closed
\item Sequences may not adequately describe limit behavior in non-metrizable spaces
\item Nets are the proper generalization
\end{itemize}

\framebreak

\textbf{Sequential Continuity:} A function $f: X \to Y$ is \textbf{sequentially continuous} at $x$ if:
$$(\forall \text{ sequences } (x_n)_{n \in \mathbb{N}} \to x) \quad f(x_n) \to f(x)$$

\vspace{0.3cm}

\textbf{Sequential Lower Semicontinuity:} Replace nets with sequences in the characterization:
$$f(x) = \sup_{\substack{(x_n)_{n \in \mathbb{N}} \to x}} \liminf_{n \to \infty} f(x_n)$$

\framebreak

\textbf{Key Point:} In metrizable spaces (e.g., $\mathbb{R}^n$), sequential notions coincide with topological ones. However, for general topological spaces, nets provide the proper framework.

\end{frame}

% Key Results and Examples
\section{Summary and Examples}

\begin{frame}[allowframebreaks]
\frametitle{Key Results Summary}

\textbf{Major Theorems in Chapter 1:}

\begin{enumerate}
\item \textbf{Zorn's Lemma (Fact 1.1):} Every partially ordered set in which every chain has an upper bound contains a maximal element.

\item \textbf{Compactness Characterization (Fact 1.11):} A set $C$ in a Hausdorff space is compact iff every open cover has a finite subcover.

\item \textbf{Net Convergence (Fact 1.19):} If $(x_a)_{a \in A}$ converges to $x$ and $(x_{k(b)})_{b \in B}$ is a subnet, then $x_{k(b)} \to x$.

\item \textbf{Lower Semicontinuity (Lemma 1.23):} Equivalence between continuity characterizations via neighborhoods and via epigraph closedness.

\item \textbf{Two-Point Compactification (Fact 1.15):} The extended real line $[-\infty, +\infty]$ is compact, and every net in it has convergent subnets.
\end{enumerate}

\framebreak

\textbf{Computational Example: Function Analysis in Python}

\begin{lstlisting}[language=Python]
import numpy as np
import matplotlib.pyplot as plt

# Define a lower semicontinuous function
def f(x):
    return 0.3 * x**2 - 0.5 * x + 0.5

# Check level sets and epigraph
x = np.linspace(-2, 3, 100)
y = f(x)

# Compute infimum and supremum
xi = 1.0  # Height level
level_set = x[y <= xi]  # {x : f(x) <= xi}

print(f"Level set at xi={xi}: [{level_set.min():.2f}, {level_set.max():.2f}]")
print(f"Infimum of f: {y.min():.4f}")
print(f"Supremum of f: {y.max():.4f}")

# Verify lower semicontinuity numerically
x0 = 0.5
eps = 0.01
neighborhood = x[np.abs(x - x0) < eps]
f_neighborhood = f(neighborhood)

assert f(x0) <= np.min(f_neighborhood), "Lower semicontinuity violated!"
print("Function is lower semicontinuous at x0 = 0.5")
\end{lstlisting}
