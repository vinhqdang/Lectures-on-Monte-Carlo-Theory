\section{Notation You'll Need}

\begin{frame}[allowframebreaks]{Notation You'll Need: From Chapter 1 to Hilbert Spaces}

  \begin{block}{Quick Recap from Chapter 1 (Topology and Nets)}
    Chapter 1 built the topological vocabulary we now reuse without
    re-deriving it: a \textbf{net} $(x_a)_{a \in A}$ is a generalization of a
    sequence indexed by a directed set, used because in infinite-dimensional
    spaces sequences alone do not always suffice to characterize closures,
    continuity, and compactness. A set is \textbf{closed} if it contains the
    limits of all its convergent nets; it is \textbf{compact} if every net in
    it has a subnet converging to a point of the set (equivalently, every
    open cover has a finite subcover). We also assume familiarity with
    \textbf{Zorn's lemma} and basic normed-space language (norm, Banach
    space, continuity, Lipschitz continuity).
  \end{block}

  \begin{alertblock}{What is a Hilbert Space? (Plain English First)}
    A \textbf{Hilbert space} is a vector space equipped with a notion of
    \emph{angle and length} --- called an \textbf{inner product} --- that is
    \textbf{complete}, meaning it has no ``missing limit points'': every
    sequence that looks like it should converge (a Cauchy sequence) actually
    does converge to a point already in the space. Think of it as
    $\R^n$ generalized to possibly \emph{infinite} dimensions, keeping every
    piece of Euclidean geometry (Pythagoras, orthogonality, projections)
    intact.
  \end{alertblock}

  \framebreak

  \begin{block}{Definition: Inner Product and Hilbert Space}
    Let $\Hil$ be a real vector space. An \textbf{inner product} (or
    \emph{scalar product}) is a map $\ip{\cdot}{\cdot} : \Hil \times \Hil \to
    \R$ that is bilinear, symmetric ($\ip{x}{y} = \ip{y}{x}$), and positive
    definite ($\ip{x}{x} > 0$ for $x \ne 0$). It induces the \textbf{norm}
    $\|x\| = \sqrt{\ip{x}{x}}$ and the \textbf{distance} $d(x,y) = \|x-y\|$.
    $\Hil$ is a \textbf{(real) Hilbert space} if, in addition, $(\Hil, d)$ is
    \textbf{complete}. Two vectors $x, y$ are \textbf{orthogonal}, written
    $x \perp y$, if $\ip{x}{y} = 0$ --- the abstract version of ``perpendicular.''
  \end{block}

  \begin{exampleblock}{$\R^n$ as the Motivating Example}
    In $\Hil = \R^n$, $\ip{x}{y} = \sum_{i=1}^n \xi_i \eta_i$ (the familiar dot
    product), $\|x\| = \sqrt{\sum_i \xi_i^2}$ is Euclidean length, and
    completeness is just the fact that $\R^n$ has no ``holes.'' Everything in
    this chapter is designed so that it reduces to ordinary Euclidean
    geometry when $\Hil = \R^n$, while remaining valid when $\Hil$ is
    infinite-dimensional (e.g., a space of functions or random variables).
  \end{exampleblock}

  \begin{alertblock}{The Running Numeric Example (used throughout this deck)}
    Throughout, we instantiate every identity with the concrete vectors
    \[
      x = (3, 4) \in \R^2, \qquad y = (1, -2) \in \R^2.
    \]
    We have $\|x\| = \sqrt{9+16} = 5$, $\|y\| = \sqrt{1+4} = \sqrt{5} \approx
    2.236$, and $\ip{x}{y} = 3(1) + 4(-2) = 3 - 8 = -5$.
  \end{alertblock}

\end{frame}

\section{2.1 Notation and Examples}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{Section 2.1: Why a Catalogue of Examples?}
  \begin{block}{Intuition}
    Before manipulating abstract Hilbert spaces, it helps to know the
    concrete spaces that occur constantly in applications: finite direct
    sums, sequence spaces $\ell^2$, matrix spaces, $L^2$ function spaces, and
    spaces of random variables. All of them are instances of one recipe:
    take an inner product, check completeness, done. This section is a
    reference catalogue you will return to throughout the book.
  \end{block}

  \begin{block}{Hilbert Direct Sum (Example 2.1)}
    Let $I$ be a nonempty (index) set and $(\Hil_i, \|\cdot\|_i)_{i \in I}$ a
    family of real Hilbert spaces. The \textbf{Hilbert direct sum} is
    \[
      \bigoplus_{i \in I} \Hil_i = \Big\{ \bm{x} = (x_i)_{i\in I} \in
      \bigtimes_{i \in I} \Hil_i \ \Big|\ \sum_{i \in I} \|x_i\|_i^2 < +\infty
      \Big\},
    \]
    with coordinatewise addition/scalar multiplication and scalar product
    $(\bm{x}, \bm{y}) \mapsto \sum_{i \in I} \ip{x_i}{y_i}_i$. Here
    $\bigtimes_{i\in I}\Hil_i$ is the ordinary Cartesian product; the extra
    condition $\sum_i \|x_i\|_i^2 < +\infty$ is what keeps the norm finite
    when $I$ is infinite.
  \end{block}
\end{frame}

\begin{frame}[allowframebreaks]{Example Catalogue: $\ell^2(I)$, $\R^N$, Matrix Spaces}

  \begin{block}{$\ell^2(I)$ and $\R^N$ (Examples 2.2--2.3)}
    If every $\Hil_i$ is the Euclidean line $\R$, the direct sum becomes
    \[
      \ell^2(I) = \bigoplus_{i \in I} \R, \qquad
      \ip{x}{y} = \sum_{i \in I} \xi_i \eta_i,
    \]
    the space of square-summable families indexed by $I$. The
    \textbf{standard unit vectors} $(e_i)_{i \in I}$ are defined by
    $e_i(j) = 1$ if $j=i$ and $0$ otherwise --- the same role played by the
    standard basis vectors of $\R^n$. When $I = \{1,\dots,N\}$, $\ell^2(I)$
    is exactly the familiar Euclidean space $\R^N$.
  \end{block}

  \begin{block}{Matrix Spaces (Examples 2.4--2.5)}
    For strictly positive integers $M, N$: $\R^{M \times N}$ is the Hilbert
    space of real $M \times N$ matrices with scalar product
    $(A,B) \mapsto \tra(A^{\mathsf T}B)$ ($\tra$ is the trace), inducing the
    \textbf{Frobenius norm} $\|\cdot\|_{\mathsf F}$. $\mathbb S^N$ is the
    subspace of $\R^{N\times N}$ consisting of all \textbf{symmetric}
    matrices.
  \end{block}

  \framebreak

  \begin{block}{$L^p$ and $L^2$ Spaces (Examples 2.6--2.8)}
    Let $(\Omega, \mathcal F, \mu)$ be a (positive) measure space. A property
    holds \textbf{$\mu$-almost everywhere} ($\mu$-a.e.) if it holds outside a
    set of measure zero. For a separable real Hilbert space
    $(\mathsf H, \ip{\cdot}{\cdot}_{\mathsf H})$ and $p \in [1,+\infty[$,
    $L^p((\Omega,\mathcal F,\mu);\mathsf H)$ is the space of (equivalence
    classes of) Borel measurable $x : \Omega \to \mathsf H$ with
    $\int_\Omega \|x(\omega)\|_{\mathsf H}^p \, \mu(d\omega) < +\infty$; for
    $p=2$ it is a real Hilbert space with
    \[
      (x,y) \mapsto \int_\Omega \ip{x(\omega)}{y(\omega)}_{\mathsf H}\,\mu(d\omega).
    \]
    Taking $\mathsf H = \R$ gives the classical Lebesgue space $L^2(\Omega,
    \mathcal F,\mu)$. Taking $\Omega = [0,T]$ with Lebesgue measure gives
    $L^2([0,T];\mathsf H)$, with scalar product $(x,y)\mapsto \int_0^T
    \ip{x(t)}{y(t)}_{\mathsf H}\,dt$.
  \end{block}

  \framebreak

  \begin{block}{Random Variables (Example 2.9)}
    A \textbf{probability space} is a measure space $(\Omega,\mathcal F,\mathsf
    P)$ with $\mathsf P(\Omega)=1$; a property holding $\mathsf P$-a.e.\ is
    said to hold \textbf{almost surely} (a.s.). A \textbf{random variable}
    is a measurable $X : \Omega \to \R$, with expected value $\mathsf E X =
    \int_\Omega X(\omega)\,\mathsf P(d\omega)$ when it exists. Then
    \[
      L^2(\Omega,\mathcal F,\mathsf P) = \big\{ X \text{ r.v.} \mid \mathsf
      E|X|^2 < +\infty \big\}
    \]
    is the Hilbert space of finite-second-moment random variables, with
    scalar product $(X,Y) \mapsto \mathsf E(XY)$ --- the space in which all
    of probability-theoretic $L^2$ (variance, covariance, orthogonality of
    uncorrelated variables) lives.
  \end{block}

  \begin{block}{Sobolev Space $W^{1,2}$ (Example 2.10)}
    For $T \in \R_{++}$ and $y \in L^2([0,T];\mathsf H)$, the antiderivative
    $x(t) = x(0) + \int_0^t y(s)\,ds$ is a.e.\ differentiable with $x' = y$
    a.e.\ Then $W^{1,2}([0,T];\mathsf H) = \{ x \in L^2([0,T];\mathsf H) \mid
    x' \in L^2([0,T];\mathsf H)\}$, with scalar product $(x,y) \mapsto
    \int_0^T \ip{x(t)}{y(t)}_{\mathsf H}\,dt + \int_0^T
    \ip{x'(t)}{y'(t)}_{\mathsf H}\,dt$ --- functions that are themselves
    square-integrable \emph{and} whose derivative is too.
  \end{block}

\end{frame}

\begin{frame}[fragile]{Python: Verifying the Running Example's Basic Numbers}

We instantiate the Hilbert space $\R^2$ (dot product) and check the norms
and inner product of the running example $x=(3,4)$, $y=(1,-2)$ that we will

\end{lstlisting}

\end{frame}

\section{2.2 Basic Identities and Inequalities}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{Cauchy--Schwarz: Intuition First}
  \begin{block}{Intuition}
    In $\R^n$, $\ip{x}{y} = \|x\|\|y\|\cos\theta$ where $\theta$ is the angle
    between $x$ and $y$. Since $|\cos\theta| \le 1$ always, we immediately
    get $|\ip{x}{y}| \le \|x\|\|y\|$. The Cauchy--Schwarz inequality is
    exactly this fact, stated without ever mentioning ``angle'' explicitly
    (which is essential once $\Hil$ is infinite-dimensional and there is no
    a priori geometric picture). Equality is the case $\cos\theta = \pm 1$,
    i.e., $x$ and $y$ are parallel.
  \end{block}

  \begin{block}{Fact 2.11 (Cauchy--Schwarz)}
    Let $x, y \in \Hil$. Then
    \[
      |\ip{x}{y}| \le \|x\|\,\|y\|.
    \]
    Moreover, $\ip{x}{y} = \|x\|\,\|y\| \iff (\exists\, \alpha \in \R_+)\
    x = \alpha y \text{ or } y = \alpha x$.
    \\[2pt]
    \textit{Here $\R_+ = [0,+\infty[$ and the condition means $x,y$ point in
    the same direction (are nonnegative multiples of one another).}
  \end{block}

  \begin{exampleblock}{Running Example}
    $|\ip{x}{y}| = |-5| = 5$ and $\|x\|\|y\| = 5 \times \sqrt5 \approx
    11.18$. Indeed $5 \le 11.18$, and since $-5 \ne 11.18$ the vectors are
    \emph{not} positive multiples of each other (their angle is obtuse,
    consistent with $\ip{x}{y}<0$).
  \end{exampleblock}
\end{frame}

\begin{frame}{Lemma 2.12: The Core Identities --- Intuition}
  \begin{block}{Intuition}
    These four identities are all algebraic consequences of expanding
    $\ip{x\pm y}{x\pm y}$, exactly as you would expand $(a\pm b)^2$ in
    ordinary algebra. Geometrically, (ii) says: in a parallelogram with
    sides $x,y$, the sum of the squares of the two diagonals equals twice
    the sum of the squares of the two sides (a classical Euclidean
    theorem). (iii) says the inner product is fully recoverable from norms
    alone --- ``length determines angle.'' (iv) is a three-point refinement
    used later for convexity arguments.
  \end{block}

  \begin{block}{Lemma 2.12}
    Let $x,y,z \in \Hil$. Then:
    \begin{align*}
      \text{(i)}\quad & \|x+y\|^2 = \|x\|^2 + 2\ip{x}{y} + \|y\|^2. \\
      \text{(ii) \emph{Parallelogram identity:}}\quad
        & \|x+y\|^2 + \|x-y\|^2 = 2\|x\|^2 + 2\|y\|^2. \\
      \text{(iii) \emph{Polarization identity:}}\quad
        & 4\ip{x}{y} = \|x+y\|^2 - \|x-y\|^2. \\
      \text{(iv) \emph{Apollonius's identity:}}\quad
        & \|x-y\|^2 = 2\|z-x\|^2 + 2\|z-y\|^2 - 4\|z-(x+y)/2\|^2.
    \end{align*}
  \end{block}
\end{frame}

\begin{frame}{Step-by-Step Proof: Parallelogram and Polarization Identities}
  \begin{block}{Proof of (i)}
    Just expand using bilinearity and symmetry of $\ip{\cdot}{\cdot}$:
    \[
      \|x+y\|^2 = \ip{x+y}{x+y} = \ip{x}{x} + \ip{x}{y} + \ip{y}{x} + \ip{y}{y}
      = \|x\|^2 + 2\ip{x}{y} + \|y\|^2.
    \]
  \end{block}

  \begin{block}{Deriving (ii) and (iii) from (i)}
    Replacing $y$ by $-y$ in (i) gives
    \[
      \|x-y\|^2 = \|x\|^2 - 2\ip{x}{y} + \|y\|^2. \tag{$\ast$}
    \]
    \textbf{Adding} (i) and ($\ast$): the cross terms $\pm 2\ip{x}{y}$
    cancel their opposite-sign partner only in the \emph{difference}; in the
    \emph{sum} they double the $\|x\|^2,\|y\|^2$ terms and the
    $\ip{x}{y}$ terms cancel:
    \[
      \|x+y\|^2 + \|x-y\|^2 = 2\|x\|^2 + 2\|y\|^2 \quad \text{(ii)}.
    \]
    \textbf{Subtracting} ($\ast$) from (i): the $\|x\|^2,\|y\|^2$ terms cancel
    and the $\ip{x}{y}$ terms add:
    \[
      \|x+y\|^2 - \|x-y\|^2 = 4\ip{x}{y} \quad \text{(iii)}.
    \]
  \end{block}

  \begin{block}{Proof of (iv)}
    Apply (ii) to the two points $u = (z-x)/2$ and $v=(z-y)/2$: note
    $u+v = z - (x+y)/2$ and $u - v = (y-x)/2$, so $\|u+v\|^2+\|u-v\|^2 =
    2\|u\|^2+2\|v\|^2$ becomes, after multiplying through by $4$,
    exactly $4\|z-(x+y)/2\|^2 + \|x-y\|^2 = 2\|z-x\|^2+2\|z-y\|^2$, which
    rearranges to (iv). $\blacksquare$
  \end{block}
\end{frame}

\begin{frame}{Numeric Verification of the Parallelogram Law in $\R^2$}
  \begin{exampleblock}{Running Example: $x=(3,4)$, $y=(1,-2)$}
    \begin{align*}
      x+y &= (4,2), & \|x+y\|^2 &= 16+4 = 20,\\
      x-y &= (2,6), & \|x-y\|^2 &= 4+36 = 40,\\
      \text{LHS} &= \|x+y\|^2+\|x-y\|^2 = 20+40 &&= 60,\\
      \text{RHS} &= 2\|x\|^2+2\|y\|^2 = 2(25)+2(5) &&= 60. \qquad\checkmark
    \end{align*}
    Polarization check: $4\ip{x}{y} = 20 - 40 = -20 \Rightarrow \ip{x}{y} =
    -5$, matching the direct computation $3(1)+4(-2)=-5$. $\checkmark$
  \end{exampleblock}

\end{frame}

\begin{frame}[fragile]{Python: Parallelogram, Polarization, and Apollonius --- Numeric Check}

\begin{lstlisting}[style=pycode]
import numpy as np
rng = np.random.default_rng(1)

x = np.array([3.0, 4.0])
y = np.array([1.0, -2.0])
z = np.array([0.0, 0.0])   # arbitrary third point for Apollonius

lhs_pll = np.dot(x+y, x+y) + np.dot(x-y, x-y)
rhs_pll = 2*np.dot(x, x) + 2*np.dot(y, y)
print("Parallelogram:", lhs_pll, "==", rhs_pll)         # 60 == 60

polarization = (np.dot(x+y, x+y) - np.dot(x-y, x-y)) / 4
print("Polarization <x|y> recovered:", polarization,      # -5.0
      " direct <x|y> =", np.dot(x, y))                     # -5.0

lhs_apo = np.dot(x-y, x-y)
rhs_apo = (2*np.dot(z-x, z-x) + 2*np.dot(z-y, z-y)
           - 4*np.dot(z-(x+y)/2, z-(x+y)/2))
print("Apollonius:", lhs_apo, "==", rhs_apo)               # 40 == 40

# Random stress-test over many (x,y,z) triples in R^5
ok = True
for _ in range(10000):
    xr, yr, zr = rng.normal(size=(3, 5))
    ok &= np.isclose(np.dot(xr+yr, xr+yr)+np.dot(xr-yr, xr-yr),
                      2*np.dot(xr, xr)+2*np.dot(yr, yr))
print("Parallelogram holds for 10000 random R^5 triples:", bool(ok))
\end{lstlisting}
\end{frame}

\begin{frame}{Lemmas 2.13--2.14: Orthogonality and Barycenters}
  \begin{block}{Intuition for Lemma 2.13}
    Item (i) says $\ip{x}{y}\le 0$ (the angle between $x,y$ is obtuse or
    right) exactly when moving from $x$ \emph{towards} $-y$ (i.e., to
    $x-\alpha y$ for small $\alpha>0$) never decreases the length $\|x\|$ ---
    a calculus-free way to say ``$y$ is not a descent direction for
    $\|\cdot\|^2$ at $x$.'' Item (ii) is the same idea specialized to
    genuine orthogonality, allowing $\alpha$ to range over all of $[-1,1]$.
  \end{block}

  \begin{block}{Lemma 2.13}
    Let $x,y \in \Hil$. Then:
    \begin{align*}
      \text{(i)}\ & \ip{x}{y}\le 0 \iff (\forall \alpha\in\R_+)\ \|x\|\le
        \|x-\alpha y\| \iff (\forall \alpha\in[0,1])\ \|x\|\le\|x-\alpha y\|.\\
      \text{(ii)}\ & x\perp y \iff (\forall \alpha\in\R)\ \|x\|\le\|x-\alpha
        y\| \iff (\forall \alpha\in[-1,1])\ \|x\|\le\|x-\alpha y\|.
    \end{align*}
  \end{block}

  \begin{block}{Lemma 2.14: Barycentric (Jensen-Type) Identity}
    For finite families $(x_i)_{i\in I},(u_i)_{i\in I}$ in $\Hil$ and
    weights $(\alpha_i)_{i\in I}$ in $\R$ with $\sum_i \alpha_i = 1$:
    \[
      \Big\|\sum_{i\in I}\alpha_i x_i\Big\|^2 + \tfrac12\sum_{i\in
      I}\sum_{j\in I}\alpha_i\alpha_j\|x_i-x_j\|^2 = \sum_{i\in
      I}\alpha_i\|x_i\|^2.
    \]
    \textit{Reading it: the squared norm of a weighted barycenter, plus a
    ``spread'' correction term, exactly recovers the weighted average of
    the squared norms --- the abstract Hilbert-space analogue of the
    variance decomposition $\mathsf E[X^2]=(\mathsf E X)^2+\mathrm{Var}(X)$.}
  \end{block}
\end{frame}

\begin{frame}{Uniform/Strict Convexity and the Hardy--Littlewood--P\'olya Inequality}
  \begin{block}{Corollaries 2.15--2.16: Hilbert Spaces are Uniformly and Strictly Convex}
    For $x,y\in\Hil$, $\alpha\in\R$:
    \[
      \|\alpha x+(1-\alpha)y\|^2+\alpha(1-\alpha)\|x-y\|^2 =
      \alpha\|x\|^2+(1-\alpha)\|y\|^2.
    \]
    Consequently, if $x\ne y$, $\|x\|=\|y\|$, and $\alpha\in\,]0,1[$, then
    $\|\alpha x+(1-\alpha)y\| < \|x\|$: \emph{the midpoint (or any strict
    convex combination) of two distinct equal-norm points is strictly
    shorter} --- the geometric heart of why balls in Hilbert space are
    ``round'' with no flat spots, which later drives uniqueness of best
    approximations (Chapter 3).
  \end{block}

  \begin{block}{Fact 2.18 (Hardy--Littlewood--P\'olya)}
    Let $x,y\in\R^N$ and let $x_\downarrow, y_\downarrow$ be their
    \emph{rearrangement vectors} with entries sorted in decreasing order.
    Then
    \[
      \ip{x}{y} \le \ip{x_\downarrow}{y_\downarrow},
    \]
    with equality iff some permutation matrix $P$ satisfies $Px=x_\downarrow$
    and $Py=y_\downarrow$. \textit{Intuition: pairing up the largest entries
    of $x$ with the largest entries of $y$ maximizes the dot product ---
    ``sorted alignment beats any other pairing.''}
  \end{block}
\end{frame}

\section{2.3 Linear Operators and Functionals}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{Bounded Operators: Intuition and Definitions}
  \begin{block}{Intuition}
    A linear map between normed spaces is continuous exactly when it cannot
    stretch \emph{any} vector by an unbounded factor; that worst-case
    stretching factor is the \textbf{operator norm}. This mirrors how, for
    matrices, the largest singular value tells you the maximum possible
    amplification of any input vector.
  \end{block}

  \begin{block}{Definition: $\mathcal B(\mathcal X,\mathcal Y)$ and Operator Norm}
    For real normed spaces $\mathcal X,\mathcal Y$,
    \[
      \mathcal B(\mathcal X,\mathcal Y) = \{T:\mathcal X\to\mathcal Y \mid T
      \text{ linear and continuous}\}, \qquad \mathcal B(\mathcal X) =
      \mathcal B(\mathcal X,\mathcal X),
    \]
    equipped with the norm $\|T\| = \sup_{\|x\|\le 1}\|Tx\|$. $\mathcal
    B(\mathcal X,\mathcal Y)$ is a normed space, and a Banach space if
    $\mathcal Y$ is.
  \end{block}

  \begin{exampleblock}{Example 2.19: Spectral Norm}
    If $A\in\R^{M\times N}$, then $A\in\mathcal B(\R^N,\R^M)$ and its
    operator norm $\|A\|_2$ (the \textbf{spectral norm}) is its largest
    singular value; moreover $\|A\|_2\le\|A\|_{\mathsf F}$ (the Frobenius
    norm bounds the spectral norm).
  \end{exampleblock}
\end{frame}

\begin{frame}{Continuity, Bilinear Maps, and the Uniform Boundedness Principle}
  \begin{block}{Fact 2.20 and Fact 2.21}
    A linear $T:\mathcal X\to\mathcal Y$ is continuous at \emph{one} point
    iff it is \textbf{Lipschitz continuous} everywhere (linearity turns
    ``continuous somewhere'' into ``uniformly controlled everywhere''). A
    bilinear $T:\mathcal X\oplus\mathcal Y\to\mathcal Z$ is continuous iff
    $(\exists\beta\in\R_+)(\forall x,y)\ \|T(x,y)\|\le\beta\|x\|\|y\|$ ---
    the two-variable analogue of Lipschitz continuity.
  \end{block}

  \begin{block}{Lemma 2.22 (Uniform Boundedness Principle / Banach--Steinhaus)}
    Let $\mathcal X$ be a real Banach space, $\mathcal Y$ a real normed
    space, and $(T_i)_{i\in I}$ operators in $\mathcal B(\mathcal
    X,\mathcal Y)$ that are \textbf{pointwise bounded}, i.e.,
    $(\forall x)\sup_{i\in I}\|T_ix\|<+\infty$. Then $(T_i)_{i\in I}$ is
    \textbf{uniformly bounded}: $\sup_{i\in I}\|T_i\|<+\infty$.
    \textit{Intuition: if a family of continuous linear maps never blows up
    any single input, completeness of $\mathcal X$ forces them to share one
    common operator-norm bound --- no input can be a ``hidden'' witness of
    unbounded growth.}
  \end{block}
\end{frame}

\begin{frame}{Monotone Operators and the Riesz--Fr\'echet Representation}
  \begin{block}{Definition 2.23: Monotone Linear Operators}
    Let $T:\Hil\to\Hil$ be linear and $\alpha\in\R_{++}$. $T$ is:
    \begin{itemize}
      \item \textbf{monotone} if $(\forall x\in\Hil)\ \ip{Tx}{x}\ge 0$;
      \item \textbf{strictly monotone} if $(\forall x\ne 0)\ \ip{Tx}{x}>0$;
      \item \textbf{$\alpha$-strongly monotone} if $(\forall x)\
        \ip{Tx}{x}\ge\alpha\|x\|^2$.
    \end{itemize}
    \textit{Intuition: $\ip{Tx}{x}\ge 0$ says $T$ never points ``backwards''
    relative to its input --- the operator-theoretic generalization of a
    nonnegative real number (indeed for $\Hil=\R$, $T$ monotone just means
    $T$ is multiplication by a nonnegative scalar). This is the seed
    concept for the entire theory of monotone operators developed later in
    the book.}
  \end{block}

  \begin{block}{Fact 2.24 (Riesz--Fr\'echet Representation)}
    Let $f\in\mathcal B(\Hil,\R)$ (a continuous linear functional). Then
    there exists a \emph{unique} $u\in\Hil$ such that $(\forall x\in\Hil)\
    f(x)=\ip{x}{u}$, and $\|f\|=\|u\|$. \textit{Every continuous linear
    functional ``is'' a vector, via the inner product} --- this identifies
    $\Hil$ with its own continuous dual $\Hil^*$.
  \end{block}
\end{frame}

\begin{frame}{Adjoint Operators, Hyperplanes, and Half-Spaces}
  \begin{block}{Adjoint Operator}
    If $\Kil$ is a real Hilbert space and $T\in\mathcal B(\Hil,\Kil)$, the
    \textbf{adjoint} of $T$ is the unique $T^*\in\mathcal B(\Kil,\Hil)$
    satisfying $(\forall x\in\Hil)(\forall y\in\Kil)\ \ip{Tx}{y} =
    \ip{x}{T^*y}$. \textit{This generalizes the matrix transpose:} for
    $A\in\R^{M\times N}$, $A^*=A^{\mathsf T}$.
  \end{block}

  \begin{block}{Fact 2.25 (Selected Adjoint Properties)}
    With $\ker T = \{x\in\Hil\mid Tx=0\}$: (i) $T^{**}=T$; (ii)
    $\|T^*\|=\|T\|=\sqrt{\|T^*T\|}$; (iv) $(\ker T)^\perp =
    \overline{\ran}\,T^*$; (v) $(\ran T)^\perp = \ker T^*$. \textit{These
    are the abstract analogues of the familiar matrix facts
    $\mathrm{row}(A)^\perp = \mathrm{null}(A)$ and $\mathrm{col}(A)^\perp =
    \mathrm{null}(A^{\mathsf T})$.}
  \end{block}

  \begin{block}{Hyperplanes and Half-Spaces (via Riesz--Fr\'echet)}
    For $u\in\Hil\setminus\{0\}$, $\eta\in\R$: the \textbf{closed
    hyperplane} $\{x\mid \ip{x}{u}=\eta\}$; the \textbf{closed half-space}
    with \emph{outer normal} $u$ is $\{x\mid \ip{x}{u}\le\eta\}$; the
    \textbf{open half-space} is $\{x\mid\ip{x}{u}<\eta\}$. The distance to
    the hyperplane $C=\{x\mid\ip{x}{u}=\eta\}$ is
    \[
      d_C(x) = \frac{|\ip{x}{u}-\eta|}{\|u\|}.
    \]
    \textit{This is exactly the point-to-plane distance formula from
    analytic geometry, now valid in any Hilbert space.}
  \end{block}
\end{frame}

\begin{frame}[allowframebreaks]{A Discontinuous Linear Functional (Example 2.27)}
  \begin{block}{Why This Matters}
    Fact 2.24 shows every \emph{continuous} linear functional comes from a
    vector via the inner product. Without continuity this fails badly:
    using a \textbf{Hamel basis} (a purely algebraic, uncountable basis
    that exists by Zorn's lemma when $\Hil$ is infinite-dimensional), one
    can build a linear functional that is discontinuous \emph{everywhere}
    and whose zero set (a hyperplane) is \emph{dense} in $\Hil$ rather than
    closed. This is a cautionary example: linearity alone buys you very
    little without continuity.
  \end{block}

  \begin{block}{Sketch of the Construction}
    Gram--Schmidt produces an orthonormal set $B=\{e_n\}_{n\in\N}$ and an
    uncountable set $C=\{c_a\}_{a\in A}$ with $B\cup C$ a Hamel basis of
    $\Hil$. Define $f(x) = \sum_n \xi_n$ where $x = \sum_n \xi_n e_n +
    \sum_a \eta_a c_a$ (a finite sum). Taking $\alpha_n=1/(n+1)$ and
    $x_n=\sum_{k=0}^n \alpha_k e_k$, one shows $(x_n)$ converges strongly to
    some $z\in\Hil$ while $f(x_n)\to+\infty$, so $f$ is discontinuous at
    $z$ --- and by Fact 2.20, discontinuous everywhere. The zero set $C =
    \{f=0\}$ is then a hyperplane that is \emph{not} closed (in fact dense,
    as shown later in Example 8.42).
  \end{block}
\end{frame}

\begin{frame}[fragile]{Python: Operator Norm and Adjoint via a Concrete Matrix}

We compute the operator (spectral) norm, the Frobenius norm, and the
adjoint (transpose) for a concrete $A \in \R^{2\times 3}$, verifying
$\|A\|_2 \le \|A\|_{\mathsf F}$ (Example 2.19) and $\ip{Ax}{y}=\ip{x}{A^*y}$
(the defining property of the adjoint).

\begin{lstlisting}[style=pycode]
import numpy as np
rng = np.random.default_rng(2)

A = np.array([[2.0, 0.0, 1.0],
              [1.0, 3.0, -1.0]])   # A in R^{2x3}, i.e. B(R^3, R^2)

sing_vals = np.linalg.svd(A, compute_uv=False)
spectral_norm = sing_vals.max()                 # ||A||_2
frobenius_norm = np.linalg.norm(A, 'fro')        # ||A||_F

print("Singular values:", sing_vals)
print(f"||A||_2 = {spectral_norm:.4f}  <=  ||A||_F = {frobenius_norm:.4f} ?",
      spectral_norm <= frobenius_norm + 1e-9)

# Adjoint check: <Ax|y>_{R^2} == <x|A^T y>_{R^3} for random x,y
x = rng.normal(size=3)
y = rng.normal(size=2)
lhs = np.dot(A @ x, y)
rhs = np.dot(x, A.T @ y)
print(f"<Ax|y> = {lhs:.6f},  <x|A*y> = {rhs:.6f},  equal? {np.isclose(lhs, rhs)}")
\end{lstlisting}
\end{frame}

\section{2.4 Strong and Weak Topologies}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{Strong Topology vs. Weak Topology: Intuition}
  \begin{block}{Strong (Norm) Topology}
    The metric topology of $(\Hil,d)$ is the \textbf{strong topology} (or
    \emph{norm topology}): a net $(x_a)_{a\in A}$ converges strongly to $x$,
    written $x_a\to x$, if $\|x_a-x\|\to 0$. Used without modifiers, all
    topological notions in $\Hil$ (closedness, compactness, etc.) refer to
    this topology --- exactly the topology of ``ordinary distance-based
    convergence'' you know from $\R^n$.
  \end{block}

  \begin{block}{Definition 2.29: The Weak Topology}
    The family of all finite intersections of \textbf{open half-spaces} of
    $\Hil$ forms the base of the \textbf{weak topology}, denoted
    $\Hil^{\mathrm{weak}}$. \textit{Intuition: instead of insisting a point
    is trapped inside a small \emph{ball} (which needs control of every
    coordinate/direction at once, hopeless in infinite dimensions), the
    weak topology only insists a point is trapped on the correct side of
    finitely many hyperplanes at a time --- a much coarser, easier-to-satisfy
    notion of ``closeness.''}
  \end{block}

  \begin{block}{Weak Convergence of Nets}
    A net $(x_a)_{a\in A}$ \textbf{converges weakly} to $x$, written
    $x_a\wto x$, if $(\forall u\in\Hil)\ \ip{x_a}{u}\to\ip{x}{u}$.
    \textit{Every ``linear measurement'' of $x_a$ converges to the same
    measurement of $x$, even though $x_a$ itself may never get close to $x$
    in norm.}
  \end{block}
\end{frame}

\begin{frame}{Lemma 2.30: The Weak Topology is Hausdorff --- Step-by-Step Proof}
  \begin{block}{Why This Matters}
    Hausdorff means distinct points can always be topologically separated;
    without it, weak limits would not even be unique. The proof is a clean
    illustration of how half-space neighborhoods are built from the inner
    product.
  \end{block}

  \begin{block}{Proof}
    Suppose $x\ne y$ in $\Hil$. Set $u = x-y$ (a nonzero vector, since
    $x\ne y$) and $w=(x+y)/2$ (their midpoint). Define
    \[
      U = \{z\in\Hil \mid \ip{z-w}{u}>0\}, \qquad
      V = \{z\in\Hil \mid \ip{z-w}{u}<0\}.
    \]
    \textbf{Step 1:} $U,V$ are open half-spaces (hence weakly open), since
    $z\mapsto\ip{z-w}{u}$ is a continuous affine functional.
    \textbf{Step 2:} $x\in U$, because $\ip{x-w}{u} = \ip{(x-y)/2}{x-y} =
    \tfrac12\|x-y\|^2 > 0$. Likewise $y\in V$, since $\ip{y-w}{u} =
    -\tfrac12\|x-y\|^2<0$.
    \textbf{Step 3:} $U\cap V=\varnothing$ trivially, since a number cannot
    be simultaneously $>0$ and $<0$. Hence $U,V$ are disjoint weak
    neighborhoods of $x,y$ respectively, proving $\Hil^{\mathrm{weak}}$ is
    Hausdorff. $\blacksquare$
  \end{block}
\end{frame}

\begin{frame}{Example 2.32: An Orthonormal Sequence Converges Weakly but Not Strongly}
  \begin{block}{Setup and Intuition}
    Suppose $\Hil$ is infinite-dimensional and $(x_n)_{n\in\N}$ is an
    \textbf{orthonormal sequence}: $\|x_n\|=1$ for all $n$ and
    $\ip{x_n}{x_m}=0$ for $n\ne m$. Intuitively, each $x_n$ ``points in a
    brand-new direction,'' so from the point of view of any \emph{fixed}
    linear measurement $u$, only finitely much of $u$'s ``energy'' can align
    with each $x_n$ before it is used up --- forcing $\ip{x_n}{u}\to 0$. But
    the vectors themselves never approach each other or $0$ in length.
  \end{block}

  \begin{block}{Step-by-Step Proof}
    \textbf{Step 1 (Bessel's inequality):} For any $u\in\Hil$,
    $\sum_{n\in\N}|\ip{x_n}{u}|^2 \le \|u\|^2 < +\infty$.
    \textbf{Step 2:} A convergent series has terms $\to 0$, so
    $\ip{x_n}{u}\to 0$ for \emph{every} fixed $u$. By definition, this says
    $x_n\wto 0$.
    \textbf{Step 3:} However $\|x_n\|\equiv 1 \ne 0$, so $x_n\not\to 0$
    strongly. In fact $(x_n)$ has \emph{no} Cauchy subsequence at all: for
    $n\ne m$, $\|x_n-x_m\|^2 = \|x_n\|^2+\|x_m\|^2 = 2$ (Pythagoras, using
    orthogonality), a fixed positive distance that never shrinks.
    \textbf{Conclusion:} the unit sphere $\{x\mid\|x\|=1\}$ is (strongly)
    closed but \emph{not} weakly sequentially closed, since it contains
    every $x_n$ but not their weak limit $0$. $\blacksquare$
  \end{block}
\end{frame}

  \end{figure}

    Left: each standard basis vector $e_n$ places its one unit of ``mass''

  \end{block}
\end{frame}

\begin{frame}[fragile]{Python: Numerically Confirming Example 2.32}

\begin{lstlisting}[style=pycode]
import numpy as np

N_dim = 500                       # truncated ambient dimension (stand-in for infinity)
u = 1.0 / np.arange(1, N_dim + 1)  # fixed u = (1, 1/2, 1/3, ...) in ell^2(N)
print("||u|| =", np.linalg.norm(u))   # finite: u is in ell^2

def e(n, dim=N_dim):
    v = np.zeros(dim)
    v[n - 1] = 1.0
    return v

# Weak convergence indicator: <e_n | u> = u_n -> 0
for n in [1, 5, 20, 100, 300]:
    inner = np.dot(e(n), u)
    print(f"n={n:4d}:  <e_n|u> = {inner:.6f}   ||e_n|| = {np.linalg.norm(e(n)):.1f}")

# No Cauchy subsequence: pairwise distances are all sqrt(2)
dists = [np.linalg.norm(e(n) - e(m)) for n, m in [(1, 2), (3, 50), (10, 400)]]
print("pairwise ||e_n - e_m|| for distinct n,m:", dists)   # all sqrt(2) = 1.4142...
\end{lstlisting}
\end{frame}

\begin{frame}{Compactness in the Weak Topology}
  \begin{block}{Fact 2.33: Four Equivalent Characterizations of Finite Dimension}
    The following are equivalent: (i) $\Hil$ is finite-dimensional; (ii)
    the closed unit ball $B(0;1)$ is (strongly) compact; (iii) the weak
    topology coincides with the strong topology; (iv) the weak topology is
    metrizable. \textit{In infinite dimensions, none of these hold --- the
    weak topology is a genuinely different, coarser structure.}
  \end{block}

  \begin{block}{Fact 2.34 (Banach--Alaoglu--Bourbaki)}
    The closed unit ball $B(0;1)$ of $\Hil$ is \textbf{weakly compact} ---
    \emph{always}, regardless of dimension. \textit{This is the payoff of
    weakening the topology: we trade away metrizability/finite-dimensional
    intuition (Fact 2.33) but recover compactness of bounded sets, which is
    indispensable for existence proofs (e.g., minimizers of convex
    functions) throughout the rest of the book.}
  \end{block}

  \begin{block}{Fact 2.35, Lemma 2.36, Fact 2.37 (Eberlein--\v Smulian)}
    The weak topology on $B(0;1)$ is metrizable iff $\Hil$ is separable
    (Fact 2.35). A set $C\subset\Hil$ is weakly compact iff it is weakly
    closed and bounded (Lemma 2.36). Weak compactness and weak
    \emph{sequential} compactness coincide (Fact 2.37) --- so despite the
    weak topology often not being metrizable, sequences still suffice to
    test weak compactness.
  \end{block}
\end{frame}

\begin{frame}{Consequences: Corollary 2.38, Lemma 2.39, Lemma 2.41}
  \begin{block}{Corollary 2.38}
    For $C\subset\Hil$, TFAE: (i) $C$ weakly compact; (ii) $C$ weakly
    sequentially compact; (iii) $C$ weakly closed and bounded. (Combine
    Lemma 2.36 and Fact 2.37.)
  \end{block}

  \begin{block}{Lemma 2.39}
    A \textbf{bounded} set $C$ is weakly closed iff it is weakly
    sequentially closed. \textit{Boundedness is essential here} ---
    Remark 2.40 notes that without it, weakly sequentially closed sets
    need not be weakly closed (Example 3.33 later in the book).
  \end{block}

  \begin{block}{Lemma 2.41}
    If $\Kil$ is a real Hilbert space and $T:\Hil\to\Kil$ is a
    \textbf{continuous affine} operator, then $T$ is \textbf{weakly
    continuous}. \textit{Proof idea:} write $T=L+T0$ with $L$ linear and
    continuous; for a net $x_a\wto x$ and any $y\in\Kil$, $\ip{Lx_a}{y} =
    \ip{x_a}{L^*y}\to\ip{x}{L^*y}=\ip{Lx}{y}$, i.e., $Lx_a\wto Lx$, so
    $Tx_a = T0+Lx_a \wto T0+Lx = Tx$.
  \end{block}
\end{frame}

\section{2.5 Weak Convergence of Sequences}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{Lemma 2.42: The Norm is Weakly Lower Semicontinuous --- Proof}
  \begin{block}{Intuition}
    Weak convergence can ``lose length'': a weak limit can be
    \emph{shorter} than what the sequence's lengths suggest (as in Example
    2.32, where $\|x_n\|\equiv1$ but the weak limit $0$ has norm $0 < 1$).
    What can never happen is the reverse --- the limit \emph{surprising you
    by being longer}. This one-sided control is \textbf{weak lower
    semicontinuity} of the norm.
  \end{block}

  \begin{block}{Lemma 2.42}
    For every net $(x_a)_{a\in A}$ in $\Hil$ and every $x\in\Hil$:
    \[
      x_a\wto x \quad\Rightarrow\quad \|x\|\le\varliminf\|x_a\|.
    \]
  \end{block}

  \begin{block}{Proof}
    Suppose $x_a \wto x$. By definition of weak convergence (taking
    $u=x$), $\ip{x_a}{x}\to\ip{x}{x}=\|x\|^2$. By Cauchy--Schwarz,
    $|\ip{x_a}{x}|\le\|x_a\|\,\|x\|$. Hence
    \[
      \|x\|^2 = \lim \ip{x_a}{x} = \varliminf \ip{x_a}{x} \le \varliminf
      \big(\|x_a\|\,\|x\|\big) = \big(\varliminf\|x_a\|\big)\|x\|.
    \]
    If $x=0$ the conclusion is trivial ($0\le\varliminf\|x_a\|$). Otherwise
    divide both sides by $\|x\|>0$ to get $\|x\|\le\varliminf\|x_a\|$.
    $\blacksquare$
  \end{block}
\end{frame}

\begin{frame}{Lemmas 2.43--2.44: Joint Continuity Along Mixed Nets}
  \begin{block}{Intuition}
    A bilinear operator that is norm-continuous stays continuous when one
    argument converges weakly and the other strongly, \emph{provided} the
    weakly-converging net stays bounded. Boundedness compensates for the
    weaker mode of convergence --- this is the recurring pattern throughout
    weak-convergence analysis: weak convergence alone is fragile, but
    ``weak + bounded'' behaves almost as well as strong convergence.
  \end{block}

  \begin{block}{Lemma 2.43}
    Let $\mathcal G,\Kil$ be real Hilbert spaces and $T:\Hil\oplus\mathcal
    G\to\Kil$ bilinear with $\|T(x,u)\|\le\beta\|x\|\|u\|$ for some
    $\beta\in\R_+$. If $(x_a)$ is \textbf{bounded} with $x_a\wto x$, and
    $u_a\to u$ (strongly), then $T(x_a,u_a)\wto T(x,u)$.
  \end{block}

  \begin{block}{Lemma 2.44 (Special Case: the Inner Product Itself)}
    If $(x_a)$ is bounded with $x_a\wto x$ and $u_a\to u$ strongly, then
    $\ip{x_a}{u_a}\to\ip{x}{u}$ (as an ordinary limit in $\R$, i.e., apply
    Lemma 2.43 with $\Kil=\R$, $T=\ip{\cdot}{\cdot}$).
  \end{block}
\end{frame}

\begin{frame}{Lemmas 2.45--2.46: Weakly Convergent Subsequences and Uniqueness}
  \begin{block}{Lemma 2.45}
    Every \textbf{bounded} sequence $(x_n)_{n\in\N}$ in $\Hil$ possesses a
    \textbf{weakly convergent subsequence}. \textit{Proof idea:} let $\rho =
    \sup_n\|x_n\|$; $B(0;\rho)$ is weakly sequentially compact by
    Banach--Alaoglu--Bourbaki (Fact 2.34) together with Eberlein--\v Smulian
    (Fact 2.37), and $(x_n)$ lies inside it.
  \end{block}

  \begin{block}{Lemma 2.46}
    A sequence $(x_n)_{n\in\N}$ \textbf{converges weakly} iff it is
    \textbf{bounded} and has \textbf{at most one} weak sequential cluster
    point. \textit{Intuition: this is the weak-topology analogue of the
    familiar fact that a bounded real sequence converges iff it has a
    unique cluster point --- boundedness guarantees \emph{some} cluster
    point exists (Lemma 2.45); uniqueness then pins down which one is the
    limit.}
  \end{block}

  \begin{block}{Lemma 2.47}
    If, for every $x$ in a set $C$, $(\|x_n-x\|)_n$ converges and every weak
    sequential cluster point of $(x_n)$ lies in $C$, then $(x_n)$ converges
    weakly to a point of $C$. \textit{A convenient sufficient condition for
    weak convergence used heavily in fixed-point/Fej\'er-monotonicity
    arguments later in the book (Chapter 5).}
  \end{block}
\end{frame}

\begin{frame}{Propositions 2.48--2.50: Relating Sequences to Orthonormal Systems}
  \begin{block}{Proposition 2.48}
    If $(y_n)$ is orthonormal and $x_n - y_n\to 0$ (strongly), then
    $x_n\wto 0$. \textit{Follows immediately from Example 2.32, since
    $y_n\wto 0$ and subtracting a strongly-vanishing perturbation does not
    change the weak limit.}
  \end{block}

  \begin{block}{Proposition 2.49 (Partial Converse)}
    If $x_n\wto 0$ and $\|x_n\|\equiv 1$, then some subsequence
    $(x_{k_n})$ and \emph{some} orthonormal sequence $(y_n)$ satisfy
    $x_{k_n}-y_n\to 0$. \textit{So every unit-norm, weakly-null sequence
    ``looks like'' an orthonormal sequence up to a strongly-vanishing
    correction, along a subsequence --- a converse to Example 2.32 /
    Proposition 2.48.}
  \end{block}

  \begin{block}{Proposition 2.50}
    If $\overline{\spn}\{e_i\}_{i\in I}=\Hil$ (a \textbf{total} or
    \emph{spanning} family), then $x_n\wto x$ iff $(x_n)$ is bounded and
    $\ip{x_n}{e_i}\to\ip{x}{e_i}$ for every single $i\in I$. \textit{Weak
    convergence is testable on any spanning/total family of test vectors,
    not just ``all of $\Hil$'' --- a major practical simplification, e.g.\
    testing against a countable orthonormal basis suffices in a separable
    Hilbert space.}
  \end{block}
\end{frame}

\begin{frame}{Lemma 2.51 and Corollary 2.52: When Weak Becomes Strong}
  \begin{block}{Lemma 2.51}
    \begin{itemize}
      \item[(i)] $x_n\wto x$ and $\varlimsup\|x_n\|\le\|x\|$
        $\iff$ $x_n\to x$ (strongly).
      \item[(ii)] If $\Hil$ is \textbf{finite-dimensional}, $x_n\wto x
        \iff x_n\to x$ (weak and strong topologies coincide, as already
        seen in Fact 2.33).
      \item[(iii)] If $x_n\wto x$ and $u_n\to u$, then $\ip{x_n}{u_n}\to
        \ip{x}{u}$.
    \end{itemize}
  \end{block}

  \begin{block}{Corollary 2.52 (Key Characterization of Strong Convergence)}
    \[
      x_n \to x \quad\iff\quad \big[\, x_n\wto x \ \text{ and }\ \|x_n\|\to
      \|x\| \,\big].
    \]
    \textit{Intuition: weak convergence already gets every ``linear
    measurement'' right; what it can miss is the overall length (recall
    Example 2.32, where lengths stayed at 1 while the weak limit had length
    0). Once you additionally know the lengths converge to the right
    value, there is no room left for the vectors to differ, and weak
    convergence upgrades to strong convergence.}
  \end{block}
\end{frame}

\begin{frame}{Step-by-Step Proof of Corollary 2.52, with Numeric Check}
  \begin{block}{Proof ($\Leftarrow$, the nontrivial direction)}
    Suppose $x_n\wto x$ and $\|x_n\|\to\|x\|$. Expand, using bilinearity:
    \[
      \|x_n-x\|^2 = \|x_n\|^2 - 2\ip{x_n}{x} + \|x\|^2.
    \]
    As $n\to\infty$: $\|x_n\|^2\to\|x\|^2$ (given), and $\ip{x_n}{x}\to
    \ip{x}{x}=\|x\|^2$ (definition of $x_n\wto x$, testing against $u=x$).
    Hence
    \[
      \|x_n-x\|^2 \to \|x\|^2 - 2\|x\|^2+\|x\|^2 = 0,
    \]
    i.e., $x_n\to x$ strongly. ($\Rightarrow$ is immediate: strong
    convergence implies weak convergence and $\|x_n\|\to\|x\|$ by
    continuity of the norm.) $\blacksquare$
  \end{block}

  \begin{exampleblock}{Contrast with Example 2.32, Numerically}
    For the orthonormal sequence $(e_n)$: $e_n\wto 0$ but $\|e_n\|\equiv 1
    \not\to \|0\|=0$ --- the norm-convergence hypothesis of Corollary 2.52
    \emph{fails}, which is exactly why $e_n\not\to 0$ strongly. This shows
    the hypothesis $\|x_n\|\to\|x\|$ cannot be dropped.
  \end{exampleblock}
\end{frame}

# A sequence in R^3 converging strongly to x_star: weak convergence in
# finite dimensions coincides with strong convergence (Lemma 2.51(ii)),

x_star = np.array([1.0, -1.0, 2.0])

strong_err = [np.linalg.norm(v - x_star) for v in xn]

print("n=399: weak-test error   =", weak_err[-1])   # -> 0
print("n=399: ||x_n||-||x|| err =", norm_err[-1])    # -> 0
print("n=399: strong ||x_n-x|| =", strong_err[-1])   # -> 0, as Cor. 2.52 predicts

N = 300
u_test = 1.0 / np.arange(1, N + 1)
for n in [10, 100, 300]:
    e_n = np.zeros(N); e_n[n - 1] = 1.0

\begin{frame}{Lemma 2.53: Ostrowski's Theorem in Finite Dimensions}
  \begin{block}{Lemma 2.53}
    Suppose $\Hil$ is finite-dimensional and $(x_n)_{n\in\N}$ is a
    \textbf{bounded} sequence with $x_n - x_{n+1}\to 0$. Then the set of
    \textbf{cluster points} of $(x_n)$ is \textbf{compact and connected}.
    \textit{Intuition: if consecutive terms get arbitrarily close together,
    the sequence cannot ``jump'' between two far-apart cluster regions
    without leaving a trail of intermediate points --- so the cluster set
    cannot have gaps (this is Ostrowski's classical theorem, Theorem 1.49,
    specialized to $\Hil$).} This is used later to analyze the asymptotic
    behavior of fixed-point iterations, where consecutive iterates
    naturally satisfy $x_n-x_{n+1}\to 0$.
  \end{block}
\end{frame}

\begin{frame}{G\^ateaux and Fr\'echet Derivatives: Intuition}
  \begin{block}{Intuition}
    Both notions generalize the single-variable derivative $f'(x) =
    \lim_{h\to0}(f(x+h)-f(x))/h$ to Hilbert-space-valued maps.
    \textbf{G\^ateaux differentiability} only asks that the directional
    limit exists \emph{separately} along each fixed direction $y$ (a
    ``one-line-at-a-time'' derivative). \textbf{Fr\'echet differentiability}
    is stronger: it asks for one single linear approximation that works
    \emph{uniformly} in every direction at once, exactly like the ordinary
    multivariable derivative in $\R^n$. Fr\'echet $\Rightarrow$ G\^ateaux,
    but not conversely.
  \end{block}

  \begin{block}{Definition 2.54 (G\^ateaux Derivative)}
    Let $C\subset\Hil$, $T:C\to\Kil$, and $x\in C$ such that every segment
    from $x$ stays in $C$ for small steps. $T$ is \textbf{G\^ateaux
    differentiable} at $x$ if there is $\mathsf DT(x)\in\mathcal
    B(\Hil,\Kil)$ with
    \[
      (\forall y\in\Hil)\quad \mathsf DT(x)y = \lim_{\alpha\downarrow 0}
      \frac{T(x+\alpha y)-T(x)}{\alpha}.
    \]
    \textit{$\mathsf DT(x)$: the G\^ateaux derivative, a continuous linear
    operator that best approximates $T$'s change along each direction $y$.}
  \end{block>
\end{frame}
