\section{Introduction: Subliner and Perspective Functions}

\begin{frame}{Perspective and Recession Functions (Examples 10.5-10.6)}
\begin{definition}
Let $\varphi: \mathcal{H} \to ]-\infty, +\infty]$ be convex and define the \textbf{perspective function}:
\[
f: \mathbb{R} \times \mathcal{H} \to ]-\infty, +\infty] : (\xi, x) \mapsto \begin{cases}
\xi\varphi(x/\xi), & \text{if } \xi > 0; \\
+\infty, & \text{otherwise}
\end{cases}
\]
Then $f$ is \textbf{sublinear}.
\end{definition}

\textbf{Key insight:} The perspective function preserves and extends convexity by adding the scale parameter $\xi$.

\begin{corollary}
For a convex function $f \in \Gamma_0(\mathcal{H})$, its \textbf{recession function}
\[
(\text{rec } f)(y) = \liminf_{x \to +\infty} \frac{f(x + \alpha y) - f(x)}{\alpha}
\]
is sublinear.
\end{corollary}
\end{frame}

\begin{frame}{Example: Perspective of Log Function}
\begin{itemize}
\item Original function: $\varphi(x) = -\log(x)$ (convex for $x > 0$)
\item Perspective: $f(\xi, x) = \xi \varphi(x/\xi) = -\xi \log(x/\xi) = -\xi[\log x - \log \xi] = \xi\log\xi - \xi\log x$
\item Properties:
  \begin{itemize}
  \item Positively homogeneous: $f(t\xi, tx) = tf(\xi, x)$ for $t > 0$
  \item Convex in both arguments
  \item Used in information geometry and convex optimization
  \end{itemize}
\end{itemize}

\end{frame}

\section{Uniform and Strong Convexity}

\begin{frame}{Definition 10.7: Uniform and Strong Convexity}
Let $f: \mathcal{H} \to ]-\infty, +\infty]$ be proper and $\phi: \mathbb{R}_+ \to [0, +\infty]$ increasing, vanishing only at 0.

\begin{definition}
$f$ is \textbf{uniformly convex} with modulus $\phi$ if:
\[
(\forall x \in \text{dom } f)(\forall y \in \text{dom } f)(\forall \alpha \in [0,1])
\]
\[
f(\alpha x + (1-\alpha)y) + \alpha(1-\alpha)\phi(\|x-y\|) \leq \alpha f(x) + (1-\alpha)f(y)
\]
\end{definition}

\begin{definition}
If $\phi(t) = (\beta/2) \cdot t^2$ for some $\beta \in \mathbb{R}_{++}$, then $f$ is \textbf{strongly convex} with constant $\beta$ (or modulus of strong convexity).
\end{definition}

\textbf{Hierarchy:} Strong convexity $\Rightarrow$ Uniform convexity $\Rightarrow$ Strict convexity $\Rightarrow$ Convexity
\end{frame}

\begin{frame}{Proposition 10.8: Characterizing Strong Convexity}
\begin{proposition}
Let $f: \mathcal{H} \to ]-\infty, +\infty]$ be proper and $\beta \in \mathbb{R}_{++}$. Then $f$ is strongly convex with constant $\beta$ if and only if $f - (\beta/2)\|\cdot\|^2$ is convex.
\end{proposition}

\textbf{Proof sketch:}
\begin{itemize}
\item Define $g(x) = f(x) - (\beta/2)\|x\|^2$
\item Show that strong convexity of $f$ is equivalent to convexity of $g$
\item Direct consequence of Corollary 2.15
\end{itemize}

\textbf{Example 10.9:}
\begin{itemize}
\item $\|\cdot\|$ is sublinear but not strictly convex (hence not uniformly convex)
\item $\|\cdot\|^2$ is strongly convex with constant $\beta = 2$ but NOT positively homogeneous
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Computing Exact Moduli: Example 10.10}
\begin{example}
Let $\phi: \mathbb{R}_+ \to \mathbb{R}_+$ be increasing and $C \subseteq \mathcal{H}$ bounded and convex. Set
\[
f: \mathcal{H} \to \mathbb{R} : x \mapsto \int_0^{\|x\|} \phi(t) \, dt
\]
Then $f$ is uniformly convex on $C$ with modulus determined by $\phi$.
\end{example}

\textbf{Example applications:}
\begin{itemize}
\item $p \in ]1, +\infty[$ and $\phi(t) = t^{p-1}$: then $\|\cdot\|_p^p$ is uniformly convex
\item Particular case $p = 2$: $\phi(t) = t$ gives $\|\cdot\|_2^2$ strongly convex with $\beta = 2$
\item Higher $p$: stronger uniform convexity
\end{itemize}

\includegraphics[width=0.6\textwidth]{norm_convexity.pdf}
\end{frame}

\begin{frame}{Definition 10.11: Exact Modulus of Convexity}
\begin{definition}
Let $f: \mathcal{H} \to ]-\infty, +\infty]$ be proper and convex. The \textbf{exact modulus of convexity} is:
\[
\varphi: \mathbb{R}_+ \to [0, +\infty]
\]
\[
t \mapsto \inf_{\substack{x, y \in \text{dom } f \\ \|x-y\|=t \\ \alpha \in [0,1]}} \left( \frac{\alpha f(x) + (1-\alpha)f(y) - f(\alpha x + (1-\alpha)y)}{\alpha(1-\alpha)} \right)
\]
\end{definition}

\textbf{Interpretation:}
\begin{itemize}
\item Measures the minimum ``gap'' between secant line and function
\item Quantifies how much $f$ curves away from linear interpolation
\item $\varphi(0) = 0$ always (by definition)
\item Larger $\varphi$ means more curvature (stronger convexity)
\end{itemize}

\centering
\includegraphics[width=0.6\textwidth]{modulus_of_convexity.pdf}
\end{frame}

\begin{frame}[allowframebreaks]{Proposition 10.12: Properties of Modulus}
\begin{proposition}
Let $f: \mathcal{H} \to ]-\infty, +\infty]$ be proper and convex, with exact modulus $\varphi$. Then:
\[
(\forall t \in \mathbb{R}_+)(\forall \gamma \in [1, +\infty[) \quad \varphi(\gamma t) \geq \gamma^2 \varphi(t).
\]
Moreover, $\varphi$ is increasing.
\end{proposition}

\textbf{Proof strategy:}
\begin{enumerate}
\item Establish that $\varphi(0) = 0$ and $\varphi(t) > 0$ for $t > 0$
\item For $\gamma \in ]1, 2]$: Use specific choice of $\alpha$ and $\gamma$ to show inequality
\item For $\gamma \in [2, +\infty[$: Write as product of factors in $]1, 2]$ and apply repeatedly
\item Monotonicity follows from definition
\end{enumerate}

\begin{corollary}
$f$ is uniformly convex if and only if $\varphi$ vanishes only at 0; in this case, $f$ is uniformly convex with modulus $\varphi$.
\end{corollary}
\end{frame}

\begin{frame}[allowframebreaks]{Corollary 10.13 \& 10.14: Relationship to Moduli}
\begin{corollary}[10.13]
Let $f: \mathcal{H} \to ]-\infty, +\infty]$ be proper and convex, with exact modulus $\varphi$. Then $f$ is uniformly convex if and only if $\varphi$ vanishes only at 0; in this case, $f$ is uniformly convex with modulus $\varphi$.
\end{corollary}

\begin{proposition}[10.14]
Let $f: \mathcal{H} \to ]-\infty, +\infty]$ be proper and convex with exact modulus $\varphi$, and define
\[
\psi: \mathbb{R}_+ \to [0, +\infty] : t \mapsto \inf_{\substack{x, y \in \text{dom } f \\ \|x-y\|=t}} \left( \frac{1}{2}f(x) + \frac{1}{2}f(y) - f\left(\frac{1}{2}x + \frac{1}{2}y\right) \right)
\]
Then: (i) $2\psi \leq \varphi \leq 4\psi$, and (ii) $f$ is uniformly convex iff $\psi$ vanishes only at 0.
\end{proposition}

\framebreak

\textbf{Key relationship:} The midpoint modulus $\psi$ captures uniform convexity but is easier to compute than the full modulus $\varphi$.

\textbf{Bounds interpretation:}
\begin{itemize}
\item $\varphi \leq 4\psi$: The full gap is at most 4 times the midpoint gap
\item $2\psi \leq \varphi$: The midpoint gap is at least half the full gap
\item These bounds allow approximation of uniform convexity
\end{itemize}
\end{frame}

\begin{frame}{Proposition 10.15: Powers of Uniformly Convex Functions}
\begin{proposition}
Let $g: \mathcal{H} \to \mathbb{R}_+$ be uniformly convex with exact modulus $\varphi$, and $p \in [1, +\infty[$. Then $g^p$ is uniformly convex, with exact modulus $\chi$ satisfying:
\[
\chi \geq 2^{1-2p} \min\{p^{2^{1-p}}, 1-2^{-p}\} \varphi^p.
\]
\end{proposition}

\textbf{Application to norms:}
\begin{itemize}
\item For $g(x) = \|x\|$, which is uniformly convex with $\varphi(t) = $ determined by $\|\cdot\|$
\item Then $\|x\|^p$ is also uniformly convex
\item Example: $\|x\|_2^p$ is uniformly convex for all $p \in [1, +\infty[$
\item Stronger convexity for $p > 2$
\end{itemize}

\textbf{Takeaway:} Powers of uniformly convex functions inherit uniform convexity with quantifiable bounds.
\end{frame}

\begin{frame}{Example 10.16: Norm Powers are Uniformly Convex}
\begin{example}
For $p \in [2, +\infty[$, the function $\|\cdot\|^p$ is uniformly convex.
\end{example}

\textbf{Proof:}
\begin{itemize}
\item By Example 10.9(ii), $\|\cdot\|^2$ is strongly convex with constant 2
\item Hence $\|\cdot\|^2$ is uniformly convex
\item Since $p/2 \in [1, +\infty[$, Proposition 10.15 gives that $(\|\cdot\|^2)^{p/2} = \|\cdot\|^p$ is uniformly convex
\end{itemize}

\textbf{Partial order:}
\[
p \in [1, 2) \Rightarrow \text{not uniformly convex}
\]
\[
p = 2 \Rightarrow \text{strongly convex}
\]
\[
p > 2 \Rightarrow \text{uniformly convex (but not strongly)}
\]

\centering
\includegraphics[width=0.6\textwidth]{convexity_variants.pdf}
\end{frame}

\begin{frame}{Proposition 10.17: Uniform Convexity on Compact Sets}
\begin{proposition}
Let $f: \mathcal{H} \to ]-\infty, +\infty]$ be proper and convex, and let $C$ be a nonempty compact convex subset of $\text{dom } f$ such that $f$ is strictly convex on $C$ and $f|_C$ is continuous. Then $f$ is uniformly convex on $C$.
\end{proposition}

\textbf{Significance:}
\begin{itemize}
\item Compactness + strict convexity + continuity $\Rightarrow$ uniform convexity
\item Useful for finite-dimensional optimization
\item Shows that locally, smooth strictly convex functions behave uniformly
\end{itemize}

\textbf{Example 10.19:} Construction showing hypotheses matter
\begin{itemize}
\item $\mathcal{H} = \mathbb{R}^2$: $f(\xi, \eta) = \frac{\eta^2}{2\xi} + \xi^2$ (if $\xi > 0, \eta \in \mathbb{R}$)
\item Is strictly convex on $C = B(0; \rho) \cap \text{dom } f$
\item But NOT uniformly convex on $C$ (fails at boundary)
\end{itemize}
\end{frame}

\begin{frame}{Corollary 10.18: Finite-Dimensional Case}
\begin{corollary}
Suppose $\mathcal{H}$ is finite-dimensional, let $f: \mathcal{H} \to \mathbb{R}$ be strictly convex, and let $C$ be a nonempty bounded closed convex subset of $\mathcal{H}$. Then $f$ is uniformly convex on $C$.
\end{corollary}

\textbf{Proof:} Combine Corollary 8.40 (continuity in finite dimensions) and Proposition 10.17.

\textbf{Practical implications:}
\begin{itemize}
\item In finite dimensions: Strict convexity + compactness $\Rightarrow$ Uniform convexity
\item No need to verify the modulus directly
\item Automatic local uniform convexity near interior of domain
\item Important for convergence analysis of optimization algorithms
\end{itemize}
\end{frame}

\section{Quasiconvexity}

\begin{frame}{Definition 10.20: Quasiconvex Functions}
\begin{definition}
Let $f: \mathcal{H} \to ]-\infty, +\infty]$. Then $f$ is \textbf{quasiconvex} if its lower level sets $(\text{lev}_\xi f)_{\xi \in \mathbb{R}}$ are convex.
\end{definition}

\textbf{Equivalently:}
\[
(\forall x, y \in \text{dom } f)(\forall \alpha \in [0,1])
\quad f(\alpha x + (1-\alpha)y) \leq \max\{f(x), f(y)\}
\]

\textbf{Key observations:}
\begin{itemize}
\item Quasiconvexity is WEAKER than convexity
\item Convex $\Rightarrow$ Quasiconvex (converse false)
\item Level sets are more fundamental than epigraph
\item More flexible than convexity for applications
\end{itemize}

\centering
\includegraphics[width=0.7\textwidth]{quasiconvex_vs_convex.pdf}
\end{frame}

\begin{frame}{Examples 10.21-10.23: Quasiconvex Functions}
\begin{example}[10.21]
Every convex function is quasiconvex (direct from Corollary 8.5).
\end{example}

\begin{example}[10.22]
Every increasing or decreasing function $f: \mathbb{R} \to ]-\infty, +\infty]$ is quasiconvex.
\end{example}

\begin{example}[10.23]
If $f: \mathcal{H} \to ]-\infty, +\infty]$ is quasiconvex and $\eta \in \mathbb{R}$, then $\min\{f, \eta\}$ is quasiconvex.
\end{example}

\textbf{Proof (10.23):} For $g = \min\{f, \eta\}$, if $\xi \leq \eta$:
\[
\text{lev}_\xi g = \text{lev}_\xi f \cap \text{lev}_\xi \eta = \text{lev}_\xi f
\]
which is convex (as $f$ is quasiconvex).
\end{frame}

\begin{frame}{Propositions 10.24-10.26: Operations on Quasiconvex Functions}
\begin{proposition}[10.24]
Let $(f_i)_{i \in I}$ be a family of quasiconvex functions from $\mathcal{H}$ to $]-\infty, +\infty]$. Then $\sup_{i \in I} f_i$ is quasiconvex.
\end{proposition}

\begin{proposition}[10.25]
Let $f: \mathcal{H} \to ]-\infty, +\infty]$ be quasiconvex. The following are equivalent:
\begin{enumerate}
\item $f$ is weakly sequentially lower semicontinuous
\item $f$ is sequentially lower semicontinuous
\item $f$ is lower semicontinuous
\item $f$ is weakly lower semicontinuous
\end{enumerate}
\end{proposition}

\begin{proposition}[10.26]
$f: \mathcal{H} \to ]-\infty, +\infty]$ is quasiconvex if and only if
\[
(\forall x, y \in \text{dom } f)(\forall \alpha \in [0,1]) \quad f(\alpha x + (1-\alpha)y) \leq \max\{f(x), f(y)\}
\]
\end{proposition}
\end{frame}

\begin{frame}{Definition 10.27: Strict and Uniform Quasiconvexity}
\begin{definition}
Let $f: \mathcal{H} \to ]-\infty, +\infty]$ be proper, $C \subseteq \text{dom } f$ nonempty, and $\phi: \mathbb{R}_+ \to [0, +\infty]$ increasing and vanishing only at 0. Then $f$ is:

\begin{enumerate}
\item \textbf{Strictly quasiconvex} if
\[
(x \neq y, \alpha \in (0,1)) \Rightarrow f(\alpha x + (1-\alpha)y) < \max\{f(x), f(y)\}
\]

\item \textbf{Uniformly quasiconvex} with modulus $\phi$ if
\[
f(\alpha x + (1-\alpha)y) + \alpha(1-\alpha)\phi(\|x-y\|) \leq \max\{f(x), f(y)\}
\]

\item Strict and uniform versions restricted to $C$ (on $C$)
\end{enumerate}
\end{definition}

\textbf{Hierarchy:}
\[
\text{Uniformly quasiconvex} \Rightarrow \text{Strictly quasiconvex} \Rightarrow \text{Quasiconvex}
\]
\end{frame}

\begin{frame}[fragile]{Examples 10.29-10.32: Illustrating Variants}
\begin{example}[10.29]
A strictly increasing or strictly decreasing proper $f: \mathbb{R} \to ]-\infty, +\infty]$ is strictly quasiconvex.
\end{example}

\begin{example}[10.30]
Let $f: \mathcal{H} \to \mathbb{R}$ and $\phi: \text{ran } f \to \mathbb{R}$ increasing.
\begin{enumerate}
\item If $f$ strictly quasiconvex and $\phi$ strictly increasing, then $\phi \circ f$ strictly quasiconvex
\item If $\phi \circ f$ strictly convex, then $f$ strictly quasiconvex
\end{enumerate}
\end{example}

\begin{lstlisting}[language=Python, basicstyle=\ttfamily\tiny]
# Numerical illustration: Composing functions preserves quasiconvexity
import numpy as np
import matplotlib.pyplot as plt

x = np.linspace(0.1, 5, 100)
f = np.sqrt(x)  # quasiconvex (strictly increasing)
phi = lambda y: np.log(y)  # strictly increasing
phi_of_f = phi(f)  # strictly quasiconvex

plt.plot(x, phi_of_f, 'b-')
plt.title(r'$\phi(f(x)) = \log(\sqrt{x})$ is strictly quasiconvex')
\end{lstlisting}

\end{frame}

\begin{frame}{Examples 10.31-10.32: Subtle Distinctions}
\begin{example}[10.31]
For $p \in \mathbb{R}_{++}$:
\begin{enumerate}
\item $\|\cdot\|^p$ is strictly quasiconvex
\item If $\mathcal{H} \neq \{0\}$ and $p < 1$, then $\|\cdot\|^p$ is NOT convex and NOT uniformly quasiconvex
\end{enumerate}
\end{example}

\begin{example}[10.32]
$f: \mathbb{R} \to \mathbb{R}, x \mapsto \nu x$ where $\nu \in \mathbb{R}_{++}$ is:
\begin{itemize}
\item Convex
\item NOT strictly convex (affine!)
\item Uniformly quasiconvex with modulus $\phi(t) = \nu t$
\end{itemize}
\end{example}

\textbf{Takeaway:}
\begin{itemize}
\item Quasiconvexity variants decouple different notions of convexity
\item Allows modeling functions with monotone behavior
\item Useful in economics, non-convex optimization, and games
\end{itemize}
\end{frame}

\section{Summary and Applications}

\begin{frame}{Hierarchy of Convexity Classes}

\begin{frame}{Summary of Key Results}

\textbf{Uniform Convexity (Section 10.2):}
\begin{itemize}
\item Quantifies how much function curves away from secant lines
\item Exact modulus $\varphi(t)$ measures minimal gap
\item Strong convexity: $\varphi(t) = (\beta/2)t^2$ (quadratic gap)
\item Equivalent to $f - (\beta/2)\|\cdot\|^2$ being convex
\item In finite dimensions: Strict convexity + compactness $\Rightarrow$ Uniform convexity
\end{itemize}

\textbf{Quasiconvexity (Section 10.3):}
\begin{itemize}
\item Defined by convexity of level sets (not epigraph)
\item Weaker than convexity: all convex are quasiconvex, but not vice versa
\item Preserved under: supremum, lower semicontinuity
\item Strict variant: strict inequality for distinct points
\item Uniform variant: adds modulus term for gap control
\end{itemize}

\textbf{Applications:}
\begin{itemize}
\item Uniform convexity $\Rightarrow$ Unique minimizers, fast convergence
\item Quasiconvexity $\Rightarrow$ Non-convex but tractable optimization
\item Monotone and level-set based problems
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Python Implementation: Computing Moduli}

\begin{lstlisting}[language=Python, basicstyle=\ttfamily\small]
import numpy as np
from scipy.optimize import minimize

def exact_modulus_estimation(f, x1, x2, t, num_alpha=50):
    """Estimate exact modulus phi(t) for convex function f"""
    alphas = np.linspace(0, 1, num_alpha)
    gaps = []
    
    for alpha in alphas:
        x_mid = alpha * x1 + (1 - alpha) * x2
        
        # Gap = (af(x1) + (1-a)f(x2) - f(a*x1 + (1-a)*x2)) / (a(1-a))
        if alpha > 0 and alpha < 1:
            numerator = (alpha * f(x1) + (1 - alpha) * f(x2) 
                        - f(x_mid))
            gap = numerator / (alpha * (1 - alpha))
            gaps.append(gap)
    
    # phi(t) = inf over gaps
    return np.min(gaps) if gaps else np.inf

# Example: f(x) = x^2
f = lambda x: x**2
x1, x2 = -1.5, 1.2
t = np.linalg.norm(np.array(x2) - np.array(x1))

phi_estimated = exact_modulus_estimation(f, x1, x2, t)
print(f"Estimated modulus phi({t:.2f}) = {phi_estimated:.4f}")
print(f"For f(x)=x^2, theoretical value = {t**2/4:.4f}")
\end{lstlisting}
\end{frame}
