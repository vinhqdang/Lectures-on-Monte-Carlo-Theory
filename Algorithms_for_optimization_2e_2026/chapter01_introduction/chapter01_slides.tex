\section{History of Optimization}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}[allowframebreaks]{A Brief History of Optimization}
  \begin{block}{Ancient Roots}
    \begin{itemize}
      \item \textbf{Pythagoras / Euclid} ($\sim$300 BC): geometric extremum problems.
      \item \textbf{Archimedes} ($\sim$250 BC): maximizing the area of a polygon inscribed in a circle — a forerunner of \emph{calculus of variations}.
      \item \textbf{Fibonacci} (125–1174 BC): \emph{The Book of the Abacus} popularized the idea that mathematics could solve the world's problems.
    \end{itemize}
  \end{block}

  \framebreak

  \begin{block}{Calculus Era (17th–18th Century)}
    \begin{itemize}
      \item \textbf{Fermat} (1643): introduced the principle that \emph{derivatives vanish at extrema} — first systematic derivative-based method.
      \item \textbf{Newton / Leibniz} (1660s–1680s): differential calculus provided tools to \emph{identify stationary points} of functions.
      \item \textbf{Euler / Lagrange} (1740–1770s): \emph{calculus of variations} — optimizing over functions rather than scalars; Lagrange multipliers for constrained optimization.
    \end{itemize}
  \end{block}

  \framebreak

  \begin{block}{Modern Era}
    \begin{itemize}
      \item \textbf{Cauchy} (1847): steepest-descent method — first numerical optimization algorithm.
      \item \textbf{Kantorovich} (1939): formulated \emph{linear programming} for production planning; awarded Nobel Prize in Economics (1975).
      \item \textbf{Dantzig} (1947): \emph{Simplex algorithm} for linear programming.
      \item \textbf{Kuhn \& Tucker} (1951): KKT conditions for constrained optimization (now standard).
      \item \textbf{Computer age} (1950s–): large-scale optimization via iterative numerical algorithms.
      \item \textbf{Deep learning} (2012–): Watson/ChatGPT trained by optimizing networks with billions of parameters using stochastic gradient descent.
    \end{itemize}
  \end{block}
\end{frame}

\section{The Optimization Process}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{The Optimization Process}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Core Idea}
        Optimization is the process of \textbf{improving a system design}
        by adjusting its parameters to minimize (or maximize) a performance metric,
        subject to constraints.
      \end{block}
      \vspace{0.5em}
      \begin{alertblock}{Engineering Perspective}
        \begin{enumerate}
          \item Specify the \textbf{design problem}.
          \item Choose a \textbf{mathematical formulation}.
          \item Select an \textbf{optimization algorithm}.
          \item \textbf{Evaluate} the design (simulation / experiment).
          \item \textbf{Iterate} until convergence.
        \end{enumerate}
      \end{alertblock}

    \column{0.50\textwidth}
      \includegraphics[width=\linewidth]{fig_opt_process}
      \vspace{0.3em}
      \begin{exampleblock}{Design Examples}
        \begin{itemize}
          \item Aircraft geometry (airfoil parameters)
          \item Bicycle frame structure
          \item Autonomous vehicle controller
          \item Medical image classifier (neural network)
        \end{itemize}
      \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{Optimization Algorithm Selection}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \begin{block}{Key Design Considerations}
        \begin{itemize}
          \item \textbf{Dimensionality}: number of design variables $n$.
          \item \textbf{Gradient availability}: analytic vs.\ numerical vs.\ none.
          \item \textbf{Constraint structure}: unconstrained, bound, general.
          \item \textbf{Budget}: function evaluations may be expensive.
          \item \textbf{Noise}: deterministic vs.\ stochastic objective.
          \item \textbf{Multi-modality}: many local minima?
        \end{itemize}
      \end{block}
      \begin{exampleblock}{No Free Lunch Theorem (informal)}
        No algorithm outperforms all others on every problem.
        Algorithm choice must match problem structure.
      \end{exampleblock}

    \column{0.42\textwidth}
      \begin{block}{Method Families}
        \centering
        \begin{tabular}{ll}
          \toprule
          \textbf{Family} & \textbf{Example} \\
          \midrule
          First-order  & Gradient descent \\
          Second-order & Newton's method \\
          Stochastic   & SGD, Adam \\
          Population   & Genetic algorithms \\
          Surrogate    & Bayesian optim. \\
          Direct       & Nelder-Mead \\
          \bottomrule
        \end{tabular}
      \end{block}
  \end{columns}
\end{frame}

\section{Mathematical Formulation}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{Mathematical Formulation}
  \begin{block}{Standard Form of an Optimization Problem}
    \[
      \min_{\vx \in \mathcal{X}} \; f(\vx)
      \quad \text{subject to} \quad
      g_i(\vx) \leq 0, \; i = 1,\ldots,m,
      \quad
      h_j(\vx) = 0, \; j = 1,\ldots,p
    \]
    \begin{itemize}
      \item $\vx = (x_1, x_2, \ldots, x_n)^\top \in \R^n$ \hfill \textbf{design point} (decision variables)
      \item $f : \R^n \to \R$ \hfill \textbf{objective function} (minimize)
      \item $g_i(\vx) \leq 0$ \hfill \textbf{inequality constraints}
      \item $h_j(\vx) = 0$ \hfill \textbf{equality constraints}
      \item $\mathcal{X}$ \hfill \textbf{feasible set}
    \end{itemize}
  \end{block}
  \begin{columns}[T]
    \column{0.45\textwidth}
      \begin{alertblock}{Feasible Set}
        The \textbf{feasible set} $\mathcal{X}$ is the set of all design points satisfying
        every constraint:
        \[
          \mathcal{X} = \{\vx \mid g_i(\vx)\leq 0,\; h_j(\vx)=0,\;\forall\,i,j\}
        \]
        A \textbf{design point} is \emph{feasible} if $\vx \in \mathcal{X}$.
      \end{alertblock}
    \column{0.52\textwidth}
      \includegraphics[width=\linewidth]{fig_feasible_set}
  \end{columns}
\end{frame}

\begin{frame}{Feasible Set — Numerical Example}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{exampleblock}{Example: 2D Constrained Problem}
        \[
          \min_{x_1,x_2} \; (x_1-2)^2 + (x_2-1)^2
        \]
        subject to:
        \[
          g_1: x_1 + x_2 \leq 2, \quad
          g_2: x_1 \geq 0, \quad
          g_3: x_2 \geq 0
        \]
      \end{exampleblock}
      \vspace{0.5em}
      The \textbf{unconstrained minimum} is $(2,1)$ — \emph{infeasible}
      (violates $g_1$).\\[4pt]
      The \textbf{constrained minimum} is at the boundary of $\mathcal{X}$,
      found by solving the KKT conditions.\\[4pt]
      \textbf{Plugging in boundary point} $(1, 1)$:
      \[
        f(1,1) = (1-2)^2 + (1-1)^2 = 1
      \]
    \column{0.45\textwidth}
      \begin{block}{Design-Point Classification}
        \begin{tabular}{ll}
          \toprule
          Point & Status \\
          \midrule
          $(0, 0)$ & Feasible (corner) \\
          $(1, 1)$ & Feasible (on bound.) \\
          $(2, 1)$ & \textcolor{red}{Infeasible} \\
          $(0.5, 0.5)$ & Feasible (interior) \\
          \bottomrule
        \end{tabular}
      \end{block}
      \begin{alertblock}{Modeling note}
        Constraints can be modeled as:
        \begin{itemize}
          \item \textbf{Inequality}: $g(\vx)\leq 0$
          \item \textbf{Equality}: $h(\vx)=0$
          \item \textbf{Bound}: $x_i \in [l_i, u_i]$
        \end{itemize}
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Mathematical Formulation — Python Example}
  \begin{columns}[T]
    \column{0.54\textwidth}
      \begin{lstlisting}[style=pycode,caption={Constrained optimization with SciPy}]
import numpy as np
from scipy.optimize import minimize

# Objective: f(x) = (x1-2)^2 + (x2-1)^2
def f(x):
    return (x[0]-2)**2 + (x[1]-1)**2

# Constraints: x1+x2 <= 2, x1>=0, x2>=0
constraints = [
    {'type': 'ineq', 'fun': lambda x: 2 - x[0] - x[1]},
]
bounds = [(0, None), (0, None)]

x0 = np.array([0.5, 0.5])   # initial guess
result = minimize(f, x0,
                  method='SLSQP',
                  bounds=bounds,
                  constraints=constraints)

print("Optimal x:", result.x)
print("f(x*)   :", result.fun)
# Optimal x: [1. 1.]
# f(x*)   : 1.0
      \end{lstlisting}

    \column{0.44\textwidth}
      \begin{block}{Key SciPy API}
        \texttt{scipy.optimize.minimize}:
        \begin{itemize}
          \item \texttt{method='SLSQP'}: Sequential Least Squares Programming (handles constraints)
          \item \texttt{bounds}: box constraints per variable
          \item \texttt{constraints}: list of dicts with \texttt{'type'} (\texttt{'eq'}/\texttt{'ineq'}) and \texttt{'fun'}
        \end{itemize}
      \end{block}
      \begin{exampleblock}{Output}
        \texttt{x* = [1.0, 1.0]}\\
        \texttt{f(x*) = 1.0}\\
        Constraint $g_1$: $1+1=2\leq 2$ \checkmark\\
        Active constraint!
      \end{exampleblock}
  \end{columns}
\end{frame}

\section{Minima: Local vs.\ Global}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{Types of Minima}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Global Minimum}
        A point $\vx^*$ is a \textbf{global minimum} if
        \[
          f(\vx^*) \leq f(\vx) \quad \forall\, \vx \in \mathcal{X}
        \]
        \begin{itemize}
          \item \emph{Unique global minimum}: strict inequality $<$.
          \item Finding it is generally NP-hard for non-convex $f$.
        \end{itemize}
      \end{block}

      \vspace{0.4em}

      \begin{block}{Local Minimum}
        A point $\vx^*$ is a \textbf{local minimum} if there exists $\delta > 0$ such that
        \[
          f(\vx^*) \leq f(\vx) \quad \forall\, \vx \in \mathcal{X},\; \|\vx - \vx^*\| < \delta
        \]
        \begin{itemize}
          \item A \textbf{strong local min}: strict inequality.
          \item Every global min is a local min, not vice versa.
        \end{itemize}
      \end{block}

    \column{0.50\textwidth}
      \includegraphics[width=\linewidth]{fig_local_global}
      \vspace{0.3em}
      \begin{exampleblock}{Numerical Example}
        $f(x) = \tfrac{x^3}{3} - x$ has a local minimum at $x=1$
        (not global — unbounded below).
      \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{Strong vs.\ Weak Minima}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \begin{block}{Classification}
        \begin{tabular}{lll}
          \toprule
          \textbf{Type} & \textbf{Condition} & \textbf{Note} \\
          \midrule
          Strong local min & $f(\vx^*) < f(\vx)$, $\vx\neq\vx^*$ & Unique locally \\
          Weak local min   & $f(\vx^*) \leq f(\vx)$            & Plateau possible \\
          Global min       & $f(\vx^*) \leq f(\vx)$, $\forall \vx\in\mathcal{X}$ & Hard to find \\
          \bottomrule
        \end{tabular}
      \end{block}
      \vspace{0.4em}
      \begin{alertblock}{Practical Implication}
        Most optimization algorithms \textbf{converge to local minima}.
        Strategies to escape:
        \begin{itemize}
          \item Multi-start (random restarts)
          \item Simulated annealing / population methods
          \item Convex relaxation
        \end{itemize}
      \end{alertblock}

    \column{0.42\textwidth}
      \begin{exampleblock}{Exercise 1.1 (Book)}
        \textbf{Q}: Give a function with a local min that is not global.\\[4pt]
        \textbf{A}: $f(x) = x^3/3 - x$\\
        Local min at $x=1$: $f(1) = -2/3$.\\
        But $f(x) \to -\infty$ as $x\to-\infty$,\\
        so $x=1$ is not the global minimum.
      \end{exampleblock}
      \vspace{0.4em}
      \begin{exampleblock}{Exercise 1.2 (Book)}
        \textbf{Q}: Minimizers of $f(x) = x^3 - x$?\\[4pt]
        \textbf{A}: No minimum — unbounded below.
      \end{exampleblock}
  \end{columns}
\end{frame}

\section{Optimality Conditions}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}[allowframebreaks]{1.6 Optimality Conditions — Univariate}
  \begin{block}{Setting}
    Consider \textbf{unconstrained} minimization of $f : \R \to \R$.
    Necessary conditions for $x^*$ to be a local minimum:
  \end{block}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{alertblock}{First-Order Necessary Condition (FONC)}
        \[
          f'(x^*) = 0
        \]
        A point satisfying FONC is called a \textbf{stationary point}.
        Not all stationary points are minima (could be max or inflection).
      \end{alertblock}

      \vspace{0.4em}

      \begin{alertblock}{Second-Order Necessary Condition (SONC)}
        \[
          f''(x^*) \geq 0
        \]
        Together with FONC, these are \emph{necessary} — all local minima obey them.
      \end{alertblock}

    \column{0.48\textwidth}
      \begin{block}{Sufficient Condition for Strong Local Min}
        \[
          f'(x^*) = 0 \quad \text{and} \quad f''(x^*) > 0
        \]
        $f''(x^*) > 0$ ensures the stationary point is a bowl (locally convex).
      \end{block}
      \vspace{0.4em}
      \begin{exampleblock}{Key Insight via Taylor Expansion}
        From $f(x^*+h) = f(x^*) + h f'(x^*) + O(h^2)$:
        \[
          f(x^*+h) \geq f(x^*) \implies hf'(x^*) \geq 0
        \]
        \[
          f(x^*-h) \geq f(x^*) \implies hf'(x^*) \leq 0
        \]
        \[
          \therefore \; f'(x^*) = 0
        \]
      \end{exampleblock}
  \end{columns}

  \framebreak

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Taylor Expansion (Eq.\ 1.4–1.8)}
        \begin{align}
          f(x^*+h) &= f(x^*) + hf'(x^*) + O(h^2) \tag{1.4}\\
          f(x^*-h) &= f(x^*) - hf'(x^*) + O(h^2) \tag{1.5}\\
          f(x^*+h)\geq f(x^*) &\implies hf'(x^*)\geq 0 \tag{1.6}\\
          f(x^*-h)\geq f(x^*) &\implies hf'(x^*)\leq 0 \tag{1.7}\\
          &\implies f'(x^*) = 0 \tag{1.8}
        \end{align}
      \end{block}
      \begin{block}{Second-Order (Eq.\ 1.9–1.10)}
        \[
          f(x^*+h) = f(x^*) + \underbrace{hf'(x^*)}_{=0} + \frac{h^2}{2}f''(x^*) + O(h^3) \tag{1.9}
        \]
        \[
          f(x^*+h) \geq f(x^*) \implies \frac{h^2}{2}f''(x^*) \geq 0 \implies f''(x^*)\geq 0 \tag{1.10}
        \]
      \end{block}
    \column{0.50\textwidth}
      \includegraphics[width=\linewidth]{fig_optimality_1d}
      \vspace{0.4em}
      \includegraphics[width=\linewidth]{fig_taylor}
  \end{columns}
\end{frame}

\begin{frame}[allowframebreaks]{1.6 Optimality Conditions — Multivariate}
  \begin{block}{Extension to $f : \R^n \to \R$}
    For unconstrained $\min_{\vx} f(\vx)$, the necessary conditions generalize:
  \end{block}

  \begin{columns}[T]
    \column{0.50\textwidth}
      \begin{alertblock}{First-Order Necessary Condition}
        \[
          \grad f(\vx^*) = \mathbf{0}
        \]
        The \textbf{gradient} at a local minimum is zero.
        \[
          \grad f(\vx) = \left[\frac{\partial f}{\partial x_1}, \ldots, \frac{\partial f}{\partial x_n}\right]^\top
        \]
      \end{alertblock}

      \vspace{0.4em}

      \begin{alertblock}{Second-Order Necessary Condition}
        \[
          \hess f(\vx^*) \succeq 0 \quad \text{(positive semidefinite)}
        \]
        The \textbf{Hessian matrix}:
        \[
          [\hess f]_{ij} = \frac{\partial^2 f}{\partial x_i \partial x_j}
        \]
      \end{alertblock}

    \column{0.48\textwidth}
      \begin{block}{Three Possible Regions at $\grad f = 0$}
        \begin{enumerate}
          \item \textbf{Hill}: $\hess f \prec 0$ (neg.\ def.) — local maximum
          \item \textbf{Saddle}: $\hess f$ indefinite — not a min/max
          \item \textbf{Bowl}: $\hess f \succ 0$ (pos.\ def.) — local minimum
        \end{enumerate}
      \end{block}
      \vspace{0.4em}
      \includegraphics[width=\linewidth]{fig_multivariate_regions}
  \end{columns}

  \framebreak

  \begin{exampleblock}{Example 1.2 (Book): Rosenbrock Function}
    \[
      f(\vx) = (1 - x_1)^2 + 5(x_2 - x_1^2)^2
    \]
    \textbf{Check whether $[1,1]$ satisfies FONC and SONC.}

    \vspace{0.4em}
    \begin{columns}[T]
      \column{0.50\textwidth}
        \textbf{Gradient:}
        \[
          \grad f(\vx) = \begin{bmatrix}
            2(10x_1^3 - 10x_1 x_2 + x_1 - 1) \\
            10(x_2 - x_1^2)
          \end{bmatrix}
        \]
        At $[1,1]$: $\grad f(1,1) = \mathbf{0}$ \checkmark

      \column{0.48\textwidth}
        \textbf{Hessian:}
        \[
          \hess f(\vx) = \begin{bmatrix}
            -20(x_2-x_1^2)+40x_1^2+2 & -20x_1 \\
            -20x_1 & 10
          \end{bmatrix}
        \]
        At $[1,1]$:
        \[
          \hess f(1,1) = \begin{bmatrix} 42 & -20 \\ -20 & 10 \end{bmatrix}
        \]
        Eigenvalues $> 0$ (positive definite) \checkmark\\
        \emph{det} $= 42\cdot10 - (-20)^2 = 20 > 0$ \checkmark
    \end{columns}
  \end{exampleblock}

  \framebreak

          \item Narrow curved valley — \emph{banana-shaped}

          \item Not a minimum!

\begin{frame}[fragile]{Optimality Conditions — Python Verification}
  \begin{columns}[T]
    \column{0.54\textwidth}
      \begin{lstlisting}[style=pycode,caption={Checking optimality conditions (Python)}]
import numpy as np

def rosenbrock(x):
    return (1 - x[0])**2 + 5*(x[1] - x[0]**2)**2

def grad_rosenbrock(x):
    x1, x2 = x
    df_dx1 = 2*(10*x1**3 - 10*x1*x2 + x1 - 1)
    df_dx2 = 10*(x2 - x1**2)
    return np.array([df_dx1, df_dx2])

def hess_rosenbrock(x):
    x1, x2 = x
    h11 = -20*(x2 - x1**2) + 40*x1**2 + 2
    h12 = -20*x1
    h22 = 10
    return np.array([[h11, h12], [h12, h22]])

xstar = np.array([1.0, 1.0])
g = grad_rosenbrock(xstar)
H = hess_rosenbrock(xstar)
eigs = np.linalg.eigvalsh(H)

print("Gradient:", g)         # [0. 0.]
print("Hessian:\n", H)        # [[42 -20][-20 10]]
print("Eigenvalues:", eigs)   # > 0 -> PD
print("FONC satisfied:", np.allclose(g, 0))
print("SONC satisfied:", np.all(eigs >= 0))
      \end{lstlisting}
    \column{0.44\textwidth}
      \begin{block}{Output}
        \texttt{Gradient: [0. 0.]}\\[2pt]
        \texttt{Hessian:}\\
        \texttt{[[42. -20.]}\\
        \texttt{ [-20.  10.]]}\\[2pt]
        \texttt{Eigenvalues: [2.0, 50.0]}\\
        \texttt{(all positive)}\\[4pt]
        \texttt{FONC satisfied: True}\\
        \texttt{SONC satisfied: True}
      \end{block}
      \begin{alertblock}{Conclusion}
        $[1,1]$ is a \textbf{strong local minimum}
        of the Rosenbrock function (in fact, the global minimum
        with $f^* = 0$).
      \end{alertblock}
  \end{columns}

\section{Summary \& Exercises}
%% ══════════════════════════════════════════════════════════════════════════

\begin{frame}{1.8 Summary}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \begin{block}{Key Takeaways}
        \begin{enumerate}
          \item Optimization = finding the best design subject to constraints.
          \item Standard form:
            \[
              \min_{\vx\in\mathcal{X}} f(\vx)
            \]
          \item A \textbf{local minimum} satisfies $f(\vx^*)\leq f(\vx)$ in a neighbourhood.
          \item \textbf{Minima require}: $\grad f = \mathbf{0}$ (FONC) and $\hess f \succeq 0$ (SONC).
          \item Zero gradient does \emph{not} imply a minimum — check the Hessian!
          \item Most algorithms converge to \emph{local} minima; escaping requires
                global strategies.
        \end{enumerate}
      \end{block}

    \column{0.48\textwidth}
      \begin{alertblock}{Core Formulas to Remember}
        \[
          \text{FONC:}\quad f'(x^*)=0 \quad [\text{or}\; \grad f(\vx^*)=\mathbf{0}]
        \]
        \[
          \text{SONC:}\quad f''(x^*)\geq 0 \quad [\text{or}\; \hess f(\vx^*)\succeq 0]
        \]
        \[
          \text{Strong local min:}\quad f'(x^*)=0,\; f''(x^*)>0
        \]
      \end{alertblock}
      \vspace{0.4em}
      \begin{exampleblock}{Applications span}
        Engineering design $\cdot$ Machine learning $\cdot$
        Finance $\cdot$ Medicine $\cdot$ Manufacturing $\cdot$
        Robotics \ldots
      \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}[allowframebreaks]{1.9 Exercises (with Solutions)}
  \begin{block}{Exercise 1.1}
    \textbf{Q}: Give an example of a function with a local minimum that is not a global minimum.\\[4pt]
    \textbf{A}: $f(x) = x^3/3 - x$. Local min at $x=1$ ($f''(1)=2>0$),
    but $f(x)\to-\infty$ as $x\to-\infty$.
  \end{block}

  \begin{block}{Exercise 1.2}
    \textbf{Q}: What is the minimum of $f(x) = x^3 - x$?\\[4pt]
    \textbf{A}: No minimum — the function is unbounded below.
  \end{block}

  \begin{block}{Exercise 1.3}
    \textbf{Q}: Give a function bounded below but with no global optimum on $\mathcal{X}=[0,1]$.\\[4pt]
    \textbf{A}: $f(x) = \begin{cases} |x| & x \neq 0 \\ 1 & x = 0\end{cases}$. The infimum is $0$ but it is not attained.
  \end{block}

  \framebreak

  \begin{block}{Exercise 1.4}
    \textbf{Q}: What are the minima and minimizers of $f(x) = \sin(x)$?\\[4pt]
    \textbf{A}: Minimum value $= -1$, at $x = \tfrac{3}{2}\pi + 2\pi i$ for all integers $i$.
    Infinitely many global minimizers.
  \end{block}

  \begin{block}{Exercise 1.5}
    \textbf{Q}: Does the first-order condition $f'(x)=0$ hold at the optimal solution of a constrained problem?\\[4pt]
    \textbf{A}: \textbf{No}. Counterexample: $\min f(x)=x$ subject to $x\geq 1$. Optimal at $x=1$,
    but $f'(1) = 1 \neq 0$. (Constrained conditions require KKT — Chapter 10.)
  \end{block}

  \begin{block}{Exercise 1.6}
    \textbf{Q}: How many minima does $f(x,y)=x^2+y$ subject to $x>y\geq 1$ have?\\[4pt]
    \textbf{A}: \textbf{None}. $g(x)=x^2$ has infimum at $x\to 1$, $h(y)=y$ increases for $y\geq 1$.
    But $x>y$ (strict), so we can only approach $x=y=1$, never attain it.
  \end{block}

  \framebreak

  \begin{block}{Exercise 1.7}
    \textbf{Q}: How many inflection points does $f(x) = x^3 - 10$ have?\\[4pt]
    \textbf{A}: \textbf{One}. $f''(x) = 6x = 0 \Rightarrow x=0$. Sign of $f''$ changes at $x=0$
    (negative for $x<0$, positive for $x>0$), confirming exactly one inflection point.
  \end{block}

  \vspace{0.5em}
  \begin{exampleblock}{Study Tips}
    \begin{itemize}
      \item Always check both FONC \emph{and} SONC when identifying minima.
      \item For constrained problems, FONC becomes the KKT conditions (Chapter 10).
      \item Use Python's \texttt{scipy.optimize.minimize} with
            \texttt{method='L-BFGS-B'} (unconstrained) or
            \texttt{'SLSQP'} (constrained) as your go-to numerical solver.
    \end{itemize}
  \end{exampleblock}
\end{frame}
