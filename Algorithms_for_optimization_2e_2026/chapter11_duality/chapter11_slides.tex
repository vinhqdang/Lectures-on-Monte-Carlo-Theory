\section{11.1 Dual Problem}
% ============================================================

\begin{frame}{Motivation: Why Duality?}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Constrained minimisation can be transformed into a
              \textbf{dual maximisation} problem.
        \item The dual value is always a \textbf{lower bound} on the
              primal optimal value.
        \item In some cases (strong duality), both problems share the
              same optimal value -- enabling \emph{verification}.
        \item Dual structure underpins many practical algorithms:\\
              interior-point methods, ADMM, distributed optimisation.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \includegraphics[width=\linewidth]{fig_lagrangian_overview.pdf}
    \end{column}
  \end{columns}
\end{frame}

% ----
\begin{frame}{The Generalised Lagrangian}
  \textbf{General constrained problem:}
  \begin{equation}
    \min_{\bx}\ f(\bx) \quad \text{s.t.}\quad \bg(\bx)\le\mathbf{0},
    \quad \bh(\bx)=\mathbf{0} \tag{11.1}
  \end{equation}

  \begin{block}{Generalised Lagrangian (Eq.\ 11.2)}
    \[
      \calL(\bx,\bmu,\blam) = f(\bx) + \sum_i \mu_i g_i(\bx) + \sum_j \lambda_j h_j(\bx)
    \]
    \begin{tabular}{ll}
      $\bx\in\R^n$  & design (primal) variables \\
      $\bmu\ge\mathbf{0}$ & inequality multipliers (dual variables) \\
      $\blam\in\R^p$ & equality multipliers (dual variables) \\
    \end{tabular}
  \end{block}

  \medskip
  $\bmu$ and $\blam$ are called \textbf{Lagrange multipliers} or \textbf{dual variables}.
\end{frame}

% ----
\begin{frame}{Primal and Dual Problems}
  \begin{block}{Primal problem (Eq.\ 11.3)}
    \[
      p^* = \min_{\bx}\; \max_{\bmu\ge\mathbf{0},\,\blam}\; \calL(\bx,\bmu,\blam)
    \]
    The outer \emph{maximisation} sets penalty terms $\to +\infty$ for infeasible $\bx$.
  \end{block}

  \begin{block}{Dual function (Eq.\ 11.4)}
    \[
      \calD(\bmu,\blam) = \min_{\bx}\; \calL(\bx,\bmu,\blam)
    \]
    Always \textbf{concave} in $(\bmu,\blam)$ (lower envelope of affine functions).
  \end{block}

  \begin{block}{Dual problem (Eq.\ 11.5)}
    \[
      d^* = \max_{\bmu\ge\mathbf{0},\,\blam}\; \calD(\bmu,\blam)
    \]
  \end{block}
\end{frame}

% ----
\begin{frame}{Weak and Strong Duality}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{block}{Weak Duality (always holds)}
        \[
          d^* \le p^*
        \]
        The dual value is always a \emph{lower bound} on the primal value.
        The gap $p^*-d^*\ge 0$ is the \textbf{duality gap}.
      \end{block}

      \begin{block}{Strong Duality}
        \[
          d^* = p^*
        \]
        Holds for \textbf{convex} problems satisfying a constraint qualification
        (e.g.\ Slater's condition: there exists a strictly feasible point).
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \includegraphics[width=\linewidth]{fig_duality_gap.pdf}
    \end{column}
  \end{columns}
\end{frame}

% ----
\begin{frame}{Example 11.1: Dual Function as Lower Bound}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \textbf{Problem:}
      \[
        \min_x\ \sin(x) \quad \text{s.t.}\ x^2 \le 1
      \]
      \textbf{Lagrangian:} $\calL(x,\mu)=\sin(x)+\mu(x^2-1)$

      \medskip
      \textbf{Primal:}
      \[
        \min_x\max_{\mu\ge 0}\bigl[\sin(x)+\mu(x^2-1)\bigr]
      \]
      \textbf{Dual:}
      \[
        \max_{\mu\ge 0}\min_x\bigl[\sin(x)+\mu(x^2-1)\bigr]
      \]
      Optimal: $x^*=-1$, $p^*\approx -0.841$.\\
      Each purple curve ($\calL$ at fixed $\mu$) has its
      minimum \emph{below} $p^*$ -- weak duality.\\
      Strong duality holds here by inspection.
    \end{column}
    \begin{column}{0.46\textwidth}
      \includegraphics[width=\linewidth]{fig_dual_lower_bound.pdf}
    \end{column}
  \end{columns}
\end{frame}

% ----
\begin{frame}[allowframebreaks]{Example 11.2: Lagrangian Duality with Equality Constraint}
  \textbf{Problem:}
  \[
    \min_{\bx}\ x_1+x_2+x_1 x_2 \quad \text{s.t.}\ x_1^2+x_2^2=1
  \]
  \textbf{Lagrangian:} $\calL(x_1,x_2,\lambda)=x_1+x_2+x_1 x_2+\lambda(x_1^2+x_2^2-1)$

  \medskip
  \textbf{First-order conditions:}
  \[
    \frac{\partial\calL}{\partial x_1}=1+x_2+2\lambda x_1=0,\quad
    \frac{\partial\calL}{\partial x_2}=1+x_1+2\lambda x_2=0,\quad
    x_1^2+x_2^2-1=0
  \]

  \textbf{Four critical points:}
  \smallskip
  \begin{center}
  \begin{tabular}{cccc}
    \toprule
    $x_1$ & $x_2$ & $\lambda$ & $f(\bx)$ \\
    \midrule
    $-1$ & $0$ & $1/2$ & $-1$ \\
    $0$ & $-1$ & $1/2$ & $-1$ \\
    $\frac{\sqrt{2}+1}{\sqrt{2}+2}$ & $\frac{\sqrt{2}+1}{\sqrt{2}+2}$ &
      $\frac{1}{2}(-1-\sqrt{2})$ & $\approx 1.914$ \\
    $\frac{\sqrt{2}-1}{\sqrt{2}-2}$ & $\frac{\sqrt{2}-1}{\sqrt{2}-2}$ &
      $\frac{1}{2}(-1+\sqrt{2})$ & $\approx -0.914$ \\
    \bottomrule
  \end{tabular}
  \end{center}

  \framebreak

  Optimal solutions: $[-1,0]$ and $[0,-1]$ (value $-1$). \\
  Using substitution $u=x_1+x_2$, $v=x_1-x_2$:

  \[
    \calD(\lambda)=\min_{u,v}\frac{1}{4}(2\lambda+1)u^2+u+\frac{1}{4}(2\lambda-1)v^2-\lambda
  \]

  \begin{columns}
    \begin{column}{0.50\textwidth}
      \[
        \calD(\lambda)=
        \begin{cases}
          -\!\left(\dfrac{1}{2\lambda+1}+\lambda\right) & \lambda\ge\tfrac{1}{2}\\[4pt]
          -\infty & \text{otherwise}
        \end{cases}
      \]
      Strong duality holds: $d^*=-1=p^*$.
    \end{column}
    \begin{column}{0.48\textwidth}
      \includegraphics[width=\linewidth]{fig_dual_function_ex2.pdf}
    \end{column}
  \end{columns}
\end{frame}

\section{11.2 Primal-Dual Methods}
% ============================================================

\begin{frame}{Primal-Dual Interior-Point: Motivation}
  \begin{itemize}
    \item \textbf{Primal-dual methods} simultaneously update primal $\bx$ and dual $\bmu$.
    \item Build on interior-point barrier method for:
          \[
            \min_\bx f(\bx) \quad \text{s.t.}\ \bg(\bx)\le\mathbf{0} \tag{11.7}
          \]
    \item Replace with log-barrier problem:
          \[
            \min_\bx f(\bx)-\frac{1}{\rho}\sum_i\log(-g_i(\bx)) \tag{11.9}
          \]
    \item Setting $\mu_i^*=-1/(\rho g_i(\bx^*))$ recovers Lagrangian KKT:
          \[
            \mathbf{0}=\grad f(\bx^*)+\sum_i\mu_i^*\grad g_i(\bx^*) \tag{11.13}
          \]
    \item \textbf{Duality gap} at iteration: $p^*-\calD(\bmu^*)=m/\rho\;\to 0$ as $\rho\to\infty$.
  \end{itemize}
\end{frame}

% ----
\begin{frame}{Primal-Dual System and Newton Step}
  Seek to satisfy \emph{stationarity} and \emph{complementarity} simultaneously:
  \begin{equation}
    \mathbf{r}(\bx,\bmu,\rho)=
    \begin{bmatrix}
      \grad f(\bx)+\sum_i\mu_i\grad g_i(\bx) \\
      -\mu_1 g_1(\bx)-1/\rho \\
      \vdots \\
      -\mu_m g_m(\bx)-1/\rho
    \end{bmatrix} = \mathbf{0} \tag{11.19}
  \end{equation}

  \begin{block}{Newton (descent) direction}
    \[
      \mathbf{d} = -\mathbf{J}^{-1}\,\mathbf{r}
    \]
    where $\mathbf{J}=\partial\mathbf{r}/\partial(\bx,\bmu)$.
  \end{block}

  \smallskip
  \textbf{Step-size safeguards:}
  \begin{itemize}
    \item Bound $\alpha$ to keep $\bmu>0$: $\alpha\le(1-\varepsilon)\mu_i/(-d_{\mu_i})$ if $d_{\mu_i}<0$.
    \item Reduce $\alpha$ while any $g_i(\bx+\alpha d_x)>0$ (maintain feasibility).
    \item Armijo sufficient decrease on $\|\mathbf{r}\|$.
  \end{itemize}
\end{frame}

% ----
\begin{frame}{Example 11.3: Primal-Dual Descent Direction}
  \textbf{Problem:} $\min_x\ x^2$ s.t.\ $(x-3)^2\le 1$\quad (feasible: $x\in[2,4]$)

  \medskip
  \textbf{Residual vector:}
  \[
    \mathbf{r} =
    \begin{bmatrix}
      2x+2\mu(x-3) \\[4pt]
      -\mu(x-3)^2+\mu-1/\rho
    \end{bmatrix} = \mathbf{0}
  \]

  \textbf{Jacobian:}
  \[
    \mathbf{J} =
    \begin{bmatrix}
      2+2\mu & 2(x-3) \\
      -2\mu(x-3) & 1-(x-3)^2
    \end{bmatrix}
  \]

  \textbf{Descent direction:} $\mathbf{d}=-\mathbf{J}^{-1}\mathbf{r}$, where $d_1$ updates $x$ and $d_2$ updates $\mu$.

  \begin{columns}
    \begin{column}{0.56\textwidth}
      The primal-dual method converges quickly because it
      simultaneously enforces both stationarity and the
      complementary slackness condition.
    \end{column}
    \begin{column}{0.42\textwidth}
      \includegraphics[width=\linewidth]{fig_primal_dual_convergence.pdf}
    \end{column}
  \end{columns}
\end{frame}

% ----
\begin{frame}[fragile]{Python: Primal-Dual Interior-Point (Algorithm 11.2)}
\begin{lstlisting}[style=python]
import numpy as np
from scipy.linalg import solve

def primal_dual_interior_point(f, gradf, hessf, gs, gradgs, hessgs,
                                rho, x, mu, alpha_max=1.0, beta=1e-4,
                                gamma=2.0, eps=0.01, p=0.9,
                                res_min=1e-4, gap_min=1e-4):
    """
    Primal-dual interior point method for:
        minimize f(x)  subject to g_i(x) <= 0  for i in 1..m
    """
    m = len(gs)
    def residual(x, mu, rho):
        r_x = gradf(x) + sum(mu[i]*gradgs[i](x) for i in range(m))
        r_mu = [-mu[i]*gs[i](x) - 1/rho for i in range(m)]
        return np.concatenate([r_x, r_mu])

    res = np.linalg.norm(residual(x, mu, rho))
    gap = sum(-mu[i]*gs[i](x) for i in range(m))

    while res > res_min or gap > gap_min:
        rho *= gamma
        # Build Jacobian and solve Newton system
        n = len(x)
        J = np.zeros((n+m, n+m))
        J[:n, :n] = hessf(x) + sum(mu[i]*hessgs[i](x) for i in range(m))
        for i in range(m):
            J[:n, n+i] = gradgs[i](x)
            J[n+i, :n] = -mu[i]*gradgs[i](x)
            J[n+i, n+i] = -gs[i](x)
        r = residual(x, mu, rho)
        d = -solve(J, r)
        d_x, d_mu = d[:n], d[n:]

        # Step size: keep mu > 0
        alpha = alpha_max
        for i in range(m):
            if d_mu[i] < 0:
                alpha = min(alpha, -(1-eps)*mu[i]/d_mu[i])

        # Keep g(x) <= 0
        while any(gs[i](x + alpha*d_x) > 0 for i in range(m)):
            alpha *= p

        # Armijo condition
        while np.linalg.norm(residual(x+alpha*d_x, mu+alpha*d_mu, rho)) \
              > (1 - beta*alpha)*res:
            alpha *= p

        x = x + alpha*d_x
        mu = mu + alpha*d_mu
        res = np.linalg.norm(residual(x, mu, rho))
        gap = sum(-mu[i]*gs[i](x) for i in range(m))

    return x, mu
\end{lstlisting}
\end{frame}

\section{11.3 Dual Ascent}
% ============================================================

\begin{frame}{Dual Ascent: Alternating Optimisation}
  When strong duality holds, an optimum $(\bx^*,\bmu^*,\blam^*)$ satisfies:
  \[
    \calL(\bx^*,\bmu^*,\blam^*)=\min_\bx \calL(\bx,\bmu^*,\blam^*)
    \quad\text{and}\quad
    \calL(\bx^*,\bmu^*,\blam^*)=\max_{\bmu\ge\mathbf{0},\blam}\calL(\bx^*,\bmu,\blam)
  \]

  \begin{block}{Dual Ascent Algorithm}
    \begin{enumerate}
      \item \textbf{Minimise} Lagrangian over $\bx$ (primal update):
            $\bx^{(k+1)}=\arg\min_\bx \calL(\bx,\bmu^{(k)},\blam^{(k)})$
      \item \textbf{Gradient ascent} on dual variables (dual update):
            \[
              \bmu^{(k+1)}=\max\!\bigl(\mathbf{0},\;\bmu^{(k)}+\alpha\,\bg(\bx^{(k+1)})\bigr)
            \]
            \[
              \blam^{(k+1)}=\blam^{(k)}+\alpha\,\bh(\bx^{(k+1)})
            \]
    \end{enumerate}
    Gradient of $\calD$ w.r.t.\ $\bmu,\blam$ is the constraint violation.
  \end{block}

  \begin{itemize}
    \item Dual function is always \emph{concave} $\Rightarrow$ gradient ascent converges.
    \item Slow convergence for ill-conditioned problems (zig-zagging).
  \end{itemize}
\end{frame}

% ----
\begin{frame}{Dual Ascent: Numerical Example}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \textbf{Problem:} $\min x_1^2+x_2^2$ s.t.\ $x_1+x_2=1$

      Dual function:
      \[
        \calD(\lambda)=-\frac{\lambda^2}{2}-\lambda
      \]
      Maximised at $\lambda^*=-1$ giving $\calD(-1)=\tfrac{1}{2}=p^*$.

      \medskip
      \textbf{Update rule:}
      \[
        x_i^{(k+1)}=-\lambda^{(k)}/2
      \]
      \[
        \lambda^{(k+1)}=\lambda^{(k)}+\alpha\underbrace{(x_1+x_2-1)}_{\text{constraint viol.}}
      \]
    \end{column}
    \begin{column}{0.46\textwidth}
      \includegraphics[width=\linewidth]{fig_dual_ascent.pdf}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Method of Multipliers (Augmented Lagrangian)}
  Dual ascent can converge slowly. Adding a quadratic penalty helps:

  \begin{block}{Augmented Lagrangian}
    \[
      \calL_\rho(\bx,\blam)=f(\bx)+\blam^\top\bh(\bx)+\frac{\rho}{2}\|\bh(\bx)\|_2^2
    \]
  \end{block}

  \begin{block}{Method of Multipliers Updates}
    \[
      \bx^{(k+1)}=\arg\min_\bx \calL_\rho(\bx,\blam^{(k)})
    \]
    \[
      \blam^{(k+1)}=\blam^{(k)}+\rho\,\bh(\bx^{(k+1)})
    \]
  \end{block}

  \begin{columns}
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \item Converges for a fixed $\rho>0$ (unlike plain penalty method).
        \item Quadratic term enforces feasibility more aggressively.
        \item Drawback: cannot easily parallelise the joint $\bx$-minimisation.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \includegraphics[width=\linewidth]{fig_method_of_multipliers.pdf}
    \end{column}
  \end{columns}
\end{frame}

\section{11.4 ADMM}
% ============================================================

\begin{frame}{ADMM: Alternating Direction Method of Multipliers}
  \begin{block}{ADMM Canonical Form}
    \[
      \min_{\bx_1,\bx_2}\; f_1(\bx_1)+f_2(\bx_2)
      \quad\text{s.t.}\quad \bA_1\bx_1+\bA_2\bx_2=\bb \tag{11.30}
    \]
  \end{block}

  Augmented Lagrangian:
  \[
    \calL_\rho(\bx_1,\bx_2,\blam)=f_1(\bx_1)+f_2(\bx_2)+\blam^\top(\bA_1\bx_1+\bA_2\bx_2-\bb)
    +\frac{\rho}{2}\|\bA_1\bx_1+\bA_2\bx_2-\bb\|_2^2
  \]

  \begin{block}{ADMM Updates (Eqs.\ 11.35--11.37)}
    \begin{align*}
      \bx_1^{(k+1)} &= \arg\min_{\bx_1}\;\calL_\rho\!\left(\bx_1,\bx_2^{(k)},\blam^{(k)}\right)\\
      \bx_2^{(k+1)} &= \arg\min_{\bx_2}\;\calL_\rho\!\left(\bx_1^{(k+1)},\bx_2,\blam^{(k)}\right)\\
      \blam^{(k+1)} &= \blam^{(k)}+\rho\!\left(\bA_1\bx_1^{(k+1)}+\bA_2\bx_2^{(k+1)}-\bb\right)
    \end{align*}
  \end{block}

  \textbf{Key:} alternating direction gives ADMM its name and enables efficient parallel computation.
\end{frame}

% ----
\begin{frame}{ADMM Block Diagram \& Convergence}
  \begin{columns}
    \begin{column}{0.50\textwidth}
      \includegraphics[width=\linewidth]{fig_admm_block.pdf}

      \medskip
      \textbf{Stopping criteria:}
      \begin{itemize}
        \item \textbf{Primal residual}: $\mathbf{r}^{(k+1)}=\bA_1\bx_1^{(k+1)}+\bA_2\bx_2^{(k+1)}-\bb$
        \item \textbf{Dual residual}: $\mathbf{s}^{(k+1)}=\rho\bA_2^\top(\bx_2^{(k+1)}-\bx_2^{(k)})$
      \end{itemize}
      Terminate when $\|\mathbf{r}\|<\varepsilon_r$ and $\|\mathbf{s}\|<\varepsilon_s$.
    \end{column}
    \begin{column}{0.48\textwidth}
      \includegraphics[width=\linewidth]{fig_admm_convergence.pdf}
      \begin{itemize}
        \item $\rho$ is increased when $\|\mathbf{r}\|\gg\|\mathbf{s}\|$
        \item Decreases $\rho$ when $\|\mathbf{s}\|\gg\|\mathbf{r}\|$
        \item Self-adaptive $\rho$ improves convergence.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ----
\begin{frame}{Scaled ADMM Form}
  Introduce the \textbf{scaled dual variable} $\bu=(1/\rho)\blam$:
  \begin{align*}
    \calL_\rho(\bx_1,\bx_2,\blam)
    &= f_1(\bx_1)+f_2(\bx_2)+\frac{\rho}{2}\|\bA_1\bx_1+\bA_2\bx_2-\bb+\bu\|_2^2
      -\frac{\rho}{2}\|\bu\|_2^2
  \end{align*}

  \begin{block}{Scaled ADMM Updates (Eqs.\ 11.50--11.52)}
    \[
      \bx_1^{(k+1)} = \arg\min_{\bx_1} f_1(\bx_1)+\frac{\rho}{2}\|\bA_1\bx_1-\bv^{(k)}\|_2^2,
      \quad \bv^{(k)}=-\bA_2\bx_2^{(k)}+\bb-\bu^{(k)}
    \]
    \[
      \bx_2^{(k+1)} = \arg\min_{\bx_2} f_2(\bx_2)+\frac{\rho}{2}\|\bA_2\bx_2-\bw^{(k)}\|_2^2,
      \quad \bw^{(k)}=-\bA_1\bx_1^{(k+1)}+\bb-\bu^{(k)}
    \]
    \[
      \bu^{(k+1)} = \bu^{(k)}+\bA_1\bx_1^{(k+1)}+\bA_2\bx_2^{(k+1)}-\bb
              = \bu^{(0)}+\sum_{j=1}^{k+1}\mathbf{r}^{(j)}
    \]
  \end{block}

  The scaled dual variable is the \textbf{running sum of primal residuals}.
\end{frame}

% ----
\begin{frame}[fragile]{Python: ADMM Core Implementation}
\begin{lstlisting}[style=python]
import numpy as np

def admm(f1_prox, f2_prox, A1, A2, b, rho=1.0, gamma=1.5,
         eps_r=1e-3, eps_s=1e-3, max_iter=200):
    """
    ADMM solver for:  min f1(x1) + f2(x2)  s.t.  A1@x1 + A2@x2 = b

    f1_prox(v, rho):  proximal operator of f1 -- argmin f1(x)+rho/2||x-v||^2
    f2_prox(v, rho):  proximal operator of f2
    """
    n1 = A1.shape[1]; n2 = A2.shape[1]
    x1 = np.zeros(n1); x2 = np.zeros(n2); u = np.zeros(len(b))

    for k in range(max_iter):
        # x1-update: minimise f1(x1) + rho/2 ||A1 x1 - v||^2
        v = -A2 @ x2 + b - u
        x1 = f1_prox(A1, v, rho)      # application-specific solver

        # x2-update: minimise f2(x2) + rho/2 ||A2 x2 - w||^2
        w = -A1 @ x1 + b - u
        x2 = f2_prox(A2, w, rho)

        # dual (scaled) variable update
        r = A1 @ x1 + A2 @ x2 - b    # primal residual
        s = rho * A1.T @ (x2 - x2)   # dual residual (simplified)
        u = u + r

        if np.linalg.norm(r) < eps_r and np.linalg.norm(s) < eps_s:
            print(f"ADMM converged at iteration {k+1}")
            break

        # Adaptive rho (Boyd et al. 2011)
        if np.linalg.norm(r) > 10 * np.linalg.norm(s):
            rho *= gamma
        elif np.linalg.norm(s) > 10 * np.linalg.norm(r):
            rho /= gamma

    return x1, x2
\end{lstlisting}
\end{frame}

\section{11.5 ADMM Applications}
% ============================================================

\begin{frame}{11.5.1 General Constrained Optimisation in ADMM Form}
  \textbf{Any} constrained problem can be cast in ADMM form:
  \[
    \min_\bx f(\bx)\quad\text{s.t.}\;\bx\in\mathcal{X}
    \quad\Longrightarrow\quad
    \min_{\bx_1,\bx_2}\underbrace{f(\bx_1)}_{f_1}+\underbrace{\mathbb{1}_\mathcal{X}(\bx_2)}_{f_2}
    \quad\text{s.t.}\;\bx_1-\bx_2=\mathbf{0}
  \]
  where $\mathbb{1}_\mathcal{X}(\bx_2)=0$ if $\bx_2\in\mathcal{X}$, else $+\infty$.

  \begin{block}{Proximal Minimisation}
    \[
      \arg\min_\bx f(\bx)+\frac{\rho}{2}\|\bx-\bp\|_2^2
    \]
    If $f$ is differentiable, linearising gives a gradient step:
    \[
      \bx\approx \bp-\frac{1}{\rho}\grad f(\bp)
    \]
    The $\bx_2$-update becomes: \textbf{project} $\bx_1^{(k+1)}+\bu^{(k)}$ onto $\mathcal{X}$.
  \end{block}
\end{frame}

% ----
\begin{frame}{11.5.2 Alternating Projections}
  \textbf{Goal:} find $\bx\in\mathcal{X}_1\cap\mathcal{X}_2$

  ADMM form:
  \[
    \min_{\bx_1,\bx_2}\mathbb{1}_{\mathcal{X}_1}(\bx_1)+\mathbb{1}_{\mathcal{X}_2}(\bx_2)
    \quad\text{s.t.}\;\bx_1-\bx_2=\mathbf{0}
  \]

  ADMM updates: each is a \textbf{projection} onto the respective convex set.

  \begin{columns}
    \begin{column}{0.52\textwidth}
      \includegraphics[width=\linewidth]{fig_proximal.pdf}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item \textbf{Von Neumann:} alternate projections directly, no dual variable.
              Slow in narrow canyons.
        \item \textbf{Dykstra (= ADMM):} dual variable $\bu$ acts as \emph{momentum}.
              Faster convergence.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ----
\begin{frame}{11.5.3 Least Absolute Deviations (LAD)}
  \textbf{Problem:} $\min_{\bx}\|\bA\bx-\bb\|_1$

  Cast in ADMM form: $\min_{\bx_1,\bx_2}f_1(\bx_1)+\|\bx_2\|_1$\; s.t.\ $\bA\bx_1-\bx_2=\bb$

  \begin{block}{Soft Thresholding Operator $S_\kappa$}
    \[
      (S_\kappa(\bx))_i=
      \begin{cases}
        x_i+\kappa & x_i<-\kappa\\
        0 & |x_i|\le\kappa\\
        x_i-\kappa & x_i>\kappa
      \end{cases}
      \;=\operatorname{sign}(x_i)\max(|x_i|-\kappa,0)
    \]
  \end{block}

  \begin{columns}
    \begin{column}{0.40\textwidth}
      \textbf{ADMM updates:}
      \begin{align*}
        \bx_1' &= (\bA^\top\bA)^{-1}\bA^\top(\bb+\bx_2-\bu)\\
        \bx_2' &= S_{1/\rho}(\bA\bx_1'-\bb+\bu)\\
        \bu' &= \bu+\bA\bx_1'-\bb-\bx_2'
      \end{align*}
      Cholesky of $\bA^\top\bA$ precomputed once.
    \end{column}
    \begin{column}{0.58\textwidth}
      \includegraphics[width=\linewidth]{fig_soft_threshold.pdf}
    \end{column}
  \end{columns}
\end{frame}

% ----
\begin{frame}{11.5.4 Basis Pursuit (Split Bregman)}
  \textbf{Problem:} $\min_\bx\|\bx\|_1$ s.t.\ $\bA\bx=\bb$ (find sparse solutions)

  ADMM updates (Eqs.\ 11.79--11.82):
  \[
    \bx_1' = \bigl(\bI-\bA^\top(\bA\bA^\top)^{-1}\bA\bigr)(\bx_2-\bu)+\bA^\top(\bA\bA^\top)^{-1}\bb
  \]
  \[
    \bx_2' = S_{1/\rho}(\bx_1'+\bu) \qquad
    \bu' = \bu+\bx_1'-\bx_2'
  \]

  \begin{itemize}
    \item Precompute $\bI-\bA^\top(\bA\bA^\top)^{-1}\bA$ and $\bA^\top(\bA\bA^\top)^{-1}\bb$ once.
    \item Each iteration is cheap: one matrix-vector product + soft thresholding.
    \item This is the \textbf{split Bregman} method.
  \end{itemize}
\end{frame}

% ----
\begin{frame}{11.5.5 Lasso: $L_1$-Regularised Regression}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{Lasso problem:}
      \[
        \min_\bx\;\frac{1}{2}\|\bA\bx-\bb\|_2^2+\lambda\|\bx\|_1 \tag{11.83}
      \]
      $\lambda>0$ biases toward \textbf{sparse} solutions.

      \medskip
      ADMM form: $f_1(\bx_1)=\frac{1}{2}\|\bA\bx_1-\bb\|_2^2$,\; $f_2(\bx_2)=\lambda\|\bx_2\|_1$,
      s.t.\ $\bx_1-\bx_2=\mathbf{0}$.

      \medskip
      \textbf{Updates (Eqs.\ 11.85--11.87):}
      \[
        \bx_1' = (\bA^\top\bA+\rho\bI)^{-1}(\bA^\top\bb+\rho(\bx_2-\bu))
      \]
      \[
        \bx_2' = S_{\lambda/\rho}(\bx_1'+\bu)
      \]
      \[
        \bu' = \bu+\bx_1'-\bx_2'
      \]
      Matrix $\bA^\top\bA+\rho\bI$ always invertible for $\rho>0$.
    \end{column}
    \begin{column}{0.43\textwidth}
      \includegraphics[width=\linewidth]{fig_lasso.pdf}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Python: ADMM-Lasso Implementation (Algorithm 11.7)}
\begin{lstlisting}[style=python]
import numpy as np

def lasso_admm(A, b, lam, x1=None, x2=None, rho=1.0,
               gamma=2.0, eps=1e-3, max_iter=100):
    """
    Solve  min 0.5*||Ax - b||^2 + lam*||x||_1
    via ADMM (Algorithm 11.7 in Python).
    """
    n = A.shape[1]
    if x1 is None: x1 = np.zeros(n)
    if x2 is None: x2 = np.zeros(n)
    u = np.zeros(n)

    AtA = A.T @ A
    Atb = A.T @ b

    def soft(v, kappa):
        return np.sign(v) * np.maximum(np.abs(v) - kappa, 0)

    for k in range(max_iter):
        # x1-update: linear system solve (warm-startable)
        x1 = np.linalg.solve(AtA + rho * np.eye(n),
                              Atb + rho * (x2 - u))
        # x2-update: soft thresholding
        x2 = soft(x1 + u, lam / rho)
        # scaled dual update
        u  = u + x1 - x2

        r = np.linalg.norm(x1 - x2)   # primal residual
        if r < eps:
            break
    return x1   # sparse coefficient vector

# Example usage
np.random.seed(42)
m, n = 50, 200
A = np.random.randn(m, n)
x_true = np.zeros(n); x_true[:5] = [3., -2., 1.5, -1., 2.]
b = A @ x_true + 0.1*np.random.randn(m)
x_hat = lasso_admm(A, b, lam=0.1)
print(f"Support recovery: {np.sum(np.abs(x_hat)>0.1)} nonzeros found")
\end{lstlisting}
\end{frame}

\begin{frame}{Consensus Optimisation}
  \textbf{Goal:} minimise sum of $k$ convex objectives:
  \[
    \min_\bx\; f_1(\bx)+f_2(\bx)+\cdots+f_k(\bx) \tag{11.88}
  \]

  \begin{block}{ADMM Consensus Reformulation}
    \[
      \min_{\bx_1^{(1:k)},\,\bx_2}\;\sum_{i=1}^k f_i(\bx_1^{(i)})
      \quad\text{s.t.}\;\bx_1^{(i)}=\bx_2\quad\forall i
    \]
    All copies of $\bx_1^{(i)}$ must agree with the global $\bx_2$.
  \end{block}

  \textbf{Consensus updates (Eqs.\ 11.90--11.92):}
  \begin{align*}
    \bigl(\bx_1^{(i)}\bigr)' &= \arg\min_{\bx_1^{(i)}} f_i\!\left(\bx_1^{(i)}\right)
      + \bigl(\blam^{(i)}\bigr)^\top\!\left(\bx_1^{(i)}-\bx_2\right)
      + \frac{\rho}{2}\bigl\|\bx_1^{(i)}-\bx_2\bigr\|_2^2
      \quad\text{(parallelisable!)}\\
    \bx_2' &= \frac{1}{k}\sum_{i=1}^k\left[\bigl(\bx_1^{(i)}\bigr)'+\frac{1}{\rho}\blam^{(i)}\right]
      \quad\text{(consensus / averaging step)}\\
    \bigl(\blam^{(i)}\bigr)' &= \blam^{(i)}+\rho\left[\bigl(\bx_1^{(i)}\bigr)'-\bx_2'\right]
  \end{align*}
\end{frame}

% ----
\begin{frame}{Consensus ADMM: Convergence to Mean}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \includegraphics[width=\linewidth]{fig_consensus.pdf}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Key properties:}
      \begin{itemize}
        \item Each $\bx_1^{(i)}$-update can run on a \textbf{separate processor}.
        \item Only $\bx_1^{(i)}$, $\blam^{(i)}$, $\bx_2$, $\rho$ need to be communicated.
        \item The implementation details of $f_i$ need not be shared.
        \item Privacy-preserving applications in federated learning.
      \end{itemize}

      \medskip
      As $\rho\to\infty$: $\bx_2\to\bar{\bx}$ (mean of all designs).
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Python: Consensus ADMM (Algorithm 11.8)}
\begin{lstlisting}[style=python]
import numpy as np
from concurrent.futures import ThreadPoolExecutor

def consensus_admm(objectives, x1_inits, x2_init,
                   rho=1.0, gamma=2.0, eps_r=1e-3, eps_s=1e-3,
                   max_iter=100):
    """
    Consensus ADMM:  min sum_i f_i(x)  (f_i in objectives list)
    Each f_i is a callable returning (value, gradient).
    Uses scipy.optimize.minimize for each x1^(i) update.
    """
    from scipy.optimize import minimize as sci_min
    k = len(objectives)
    x1s = [np.array(xi, dtype=float) for xi in x1_inits]
    x2 = np.array(x2_init, dtype=float)
    lams = [np.zeros_like(x2) for _ in range(k)]

    x1_bar_old = np.mean(x1s, axis=0)

    for it in range(max_iter):
        # Parallel x1^(i) updates
        def update_i(args):
            i, f_i, x1, lam = args
            def obj(x):
                val = f_i(x)
                penalty = lam @ (x - x2) + 0.5*rho*np.sum((x - x2)**2)
                return val + penalty
            res = sci_min(obj, x1, method='L-BFGS-B')
            return res.x

        args = [(i, objectives[i], x1s[i], lams[i]) for i in range(k)]
        with ThreadPoolExecutor() as pool:
            x1s = list(pool.map(update_i, args))

        # Consensus (x2) update
        x2 = np.mean([x1s[i] + lams[i]/rho for i in range(k)], axis=0)

        # Dual (lambda) update
        for i in range(k):
            lams[i] += rho * (x1s[i] - x2)

        # Residuals
        x1_bar = np.mean(x1s, axis=0)
        norm_r = np.sqrt(sum(np.sum((x1s[i]-x1_bar)**2) for i in range(k)))
        norm_s = rho * np.sqrt(k) * np.linalg.norm(x1_bar - x1_bar_old)
        x1_bar_old = x1_bar.copy()

        if norm_r < eps_r and norm_s < eps_s:
            print(f"Consensus ADMM converged at iteration {it+1}")
            break
        rho = rho*gamma if norm_r > 10*norm_s else (rho/gamma if norm_s > 10*norm_r else rho)

    return x2
\end{lstlisting}
\end{frame}

\begin{frame}{11.6.2 General Consensus Optimisation}
  When local vectors $\bx_1^{(i)}$ need only contain a \emph{subset} of $\bx_2$:

  \[
    \min_{\bx_1^{(1:k)},\,\bx_2}\;\sum_{i=1}^k f_i\!\left(\bx_1^{(i)}\right)
    \quad\text{s.t.}\;\bx_1^{(i)}=\bx_2^{(i)}\quad\forall i
  \]

  \begin{block}{General Consensus Updates}
    \begin{align*}
      \bigl(\bx_1^{(i)}\bigr)' &= \arg\min_{\bx_1^{(i)}} f_i(\bx_1^{(i)})
        + \bigl(\blam^{(i)}\bigr)^\top(\bx_1^{(i)}-\bx_2^{(i)})
        + \frac{\rho}{2}\|\bx_1^{(i)}-\bx_2^{(i)}\|_2^2\\[4pt]
      (\bx_2)_g' &= \frac{1}{n_g}\sum_{i=1}^k\sum_j
        \bigl[\bigl(\bx_1^{(i)}\bigr)'_j\text{ where component $j$ maps to $g$}\bigr]\\[4pt]
      \bigl(\blam^{(i)}\bigr)' &= \blam^{(i)}+\rho\left(\bigl(\bx_1^{(i)}\bigr)'-\bigl(\bx_2^{(i)}\bigr)'\right)
    \end{align*}
  \end{block}

  $n_g$ = number of local vectors whose entry corresponds to global component $g$.
\end{frame}

\begin{frame}{Example 11.7: Bakery Placement as General Consensus}
  \textbf{Problem:} Place 3 bakeries ($\mathbf{b}^{(1)},\mathbf{b}^{(2)},\mathbf{b}^{(3)}$) to serve 6 establishments.
  \begin{itemize}
    \item 2 cafes at $\mathbf{e}^{(1)}\!=\![0,0]$, $\mathbf{e}^{(2)}\!=\![0.5,0]$
    \item 2 brunch spots at $\mathbf{e}^{(3)}\!=\![3,2]$, $\mathbf{e}^{(4)}\!=\![0.5,3]$
    \item 2 restaurants at $\mathbf{e}^{(5)}\!=\![3.5,-0.5]$, $\mathbf{e}^{(6)}\!=\![2.5,0.7]$
  \end{itemize}
  Bakery 1 (croissants) serves cafes + brunch; bakery 2 (baguettes) serves brunch + restaurants;
  bakery 3 (dinner rolls) serves restaurants only.

  \medskip
  \textbf{Global variable:} $\bx_2=[b_1^{(1)},b_2^{(1)},b_1^{(2)},b_2^{(2)},b_1^{(3)},b_2^{(3)}]^\top$

  \medskip
  Each establishment optimises a \emph{local} vector that only contains entries for
  its compatible bakeries $\Rightarrow$ true distributed computation.
\end{frame}

\begin{frame}{Chapter 11 Summary}
  \begin{columns}
    \begin{column}{0.50\textwidth}
      \begin{itemize}
        \item \textbf{Dual problem:} always concave; its value $\le p^*$ (weak duality).
        \item \textbf{Strong duality} ($d^*=p^*$): holds for convex problems + CQ.
        \item \textbf{Primal-dual interior-point:} simultaneously drives residual and
              duality gap to zero.
        \item \textbf{Dual ascent:} gradient ascent on concave $\calD$; convergence
              guaranteed but slow.
        \item \textbf{Method of multipliers:} augmented Lagrangian gives faster, fixed-$\rho$
              convergence.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item \textbf{ADMM:} decomposes method of multipliers into alternating updates;
              each sub-problem cheaper.
        \item \textbf{ADMM applications:} alternating projections, LAD, basis pursuit,
              lasso, Huber regression.
        \item \textbf{Consensus:} parallelises ADMM over $k$ processors; only
              averages communicated.
        \item \textbf{General consensus:} each agent sees only relevant components --
              privacy-preserving.
      \end{itemize}
    \end{column}
  \end{columns}

  \medskip
  \begin{center}
  \begin{tabular}{lll}
    \toprule
    Method & Primal update & Dual update \\
    \midrule
    Dual ascent & Lagrangian min & Gradient step \\
    Method of multipliers & Aug.\ Lagrangian min & Multiplier step \\
    ADMM & Two alternating min & Multiplier step \\
    Consensus ADMM & $k$ parallel min & Mean + multiplier \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

\begin{frame}[allowframebreaks]{Selected Exercises}
  \textbf{Exercise 11.1.} Show complementary slackness $\mu_i^* g_i(\bx^*)=0$ holds
  when the duality gap is zero.

  \medskip
  \textit{Key idea:} Zero gap $\Rightarrow d^*=p^*=f(\bx^*)$. Since $\calD(\bmu^*,\blam^*)
  \le\calL(\bx^*,\bmu^*,\blam^*)$ and each term $\mu_i^* g_i(\bx^*)\le 0$,
  the sum must be exactly zero $\Rightarrow$ complementary slackness.

  \bigskip
  \textbf{Exercise 11.2.} Is the dual of the dual equal to the primal?

  \medskip
  \textit{Answer:} Not necessarily. From $d^*\le p^*$ and $dd^*\le d^*$, we get
  $dd^*\le p^*$. If a duality gap exists, $dd^*\ne p^*$.

  \framebreak

  \textbf{Exercise 11.3.} ADMM $\bx_1$-update for quadratic $f_1(\bx_1)=\frac{1}{2}\bx_1^\top\mathbf{Q}\bx_1
  +\mathbf{q}^\top\bx_1+q$ (positive semidefinite $\mathbf{Q}$):
  \[
    (\mathbf{Q}+\rho\bA_1^\top\bA_1)\bx_1' = \rho\bA_1^\top(\bA_2\bx_2-\bb)-\mathbf{q}-\blam
  \]

  \bigskip
  \textbf{Exercise 11.6.} Simplify consensus $\bx_2$-update: let $\bar{\bx}_1=(1/k)\sum_i(\bx_1^{(i)})'$.
  Then:
  \[
    \bx_2' = \bar{\bx}_1 + \frac{1}{\rho}\bar{\blam}
  \]
  This removes the need for an explicit $\bx_2$ variable.
\end{frame}

  \textbf{Practical guidelines:}
  \begin{itemize}
    \item Use adaptive $\rho$: scale up when $\|\mathbf{r}\|\gg\|\mathbf{s}\|$, down otherwise.
    \item Initialise with small $\rho$ and increase geometrically ($\rho\leftarrow\gamma\rho$).
    \item Precompute matrix factorisations when possible (LASSO, basis pursuit).
    \item ADMM typically shows linear convergence; near optimal solutions may need refinement.
  \end{itemize}
