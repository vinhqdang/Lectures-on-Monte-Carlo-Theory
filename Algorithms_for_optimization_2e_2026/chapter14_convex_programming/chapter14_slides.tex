\section{Canonical Form of a DCP}
% ============================================================

\begin{frame}{What is Disciplined Convex Programming?}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{block}{Motivation}
        Convex optimization problems are those with a
        \textbf{convex objective} and \textbf{convex/affine constraints}.\\[6pt]
        General techniques exist to find global optima, but:
        \begin{itemize}
          \item Verifying convexity by hand is tedious and error-prone
          \item We need \emph{automated} methods
        \end{itemize}
      \end{block}
      \vspace{4pt}
      \begin{alertblock}{DCP Definition}
        A \textbf{disciplined convex program (DCP)} is a convex
        optimization problem written in such a way that
        \emph{automated methods can verify its convexity} and
        transcribe it into canonical form for any standard solver.
      \end{alertblock}
    \end{column}
    \begin{column}{0.42\textwidth}
      \includegraphics[width=\textwidth]{fig_dcp_canonical.pdf}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Canonical Form of a DCP}
  \begin{block}{Standard Convex Problem (Eq.\ 14.1)}
    \[
      \underset{\mathbf{x}}{\mathrm{minimize}}\quad f_0(\mathbf{x})
      \qquad
      \text{subject to}\quad
      \begin{cases}
        f_i(\mathbf{x}) \leq 0 & i = 1,\ldots,m\\
        h_j(\mathbf{x}) = 0    & j = 1,\ldots,p
      \end{cases}
    \]
    where $\mathbf{x}\in\mathbb{R}^n$,\ $f_0,f_i$ are \textbf{convex},
    and $h_j$ are \textbf{affine}.
  \end{block}

  \vspace{6pt}
  \begin{block}{DCP Requirements}
    \begin{enumerate}
      \item \textbf{Product-free expressions} — no products of
            non-constant sub-expressions
      \item \textbf{Top-level production rules} — objective must be
            convex; inequality LHS convex, RHS concave
      \item \textbf{Sign rules} — track the sign/domain of
            each sub-expression
      \item \textbf{Composition rules} — curvature propagates
            through function composition
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Running Example (Ex.\ 14.1)}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Problem}
        \[
          \underset{\mathbf{x}}{\mathrm{minimize}}\;
          \|\mathbf{A}\mathbf{x} - \mathbf{b}\|_1
          \qquad
          \text{s.t.}\;
          \|\mathbf{C}\mathbf{x} - \mathbf{d}\|_{1.5} \leq 3
        \]
        with
        \[
          \mathbf{A}=\begin{bmatrix}2&-1\\1&3\end{bmatrix},\quad
          \mathbf{b}=\begin{bmatrix}4\\-5\end{bmatrix}
        \]
        \[
          \mathbf{C}=\begin{bmatrix}0&1\\3&-4\end{bmatrix},\quad
          \mathbf{d}=\begin{bmatrix}-1\\0\end{bmatrix}
        \]
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{DCP Status}
        \begin{itemize}
          \item $\|\cdot\|_1$ is \textcolor{dcpgreen}{\textbf{convex}},
                applied to affine expression $\Rightarrow$ convex
          \item $\|\cdot\|_{1.5}$ is \textcolor{dcpgreen}{\textbf{convex}},
                applied to affine expression $\Rightarrow$ convex
          \item Constraint: convex $\leq$ constant\ \checkmark
          \item Objective: minimize convex\ \checkmark
        \end{itemize}
        \vspace{4pt}
        \textcolor{dcpgreen}{\textbf{This is a valid DCP!}}
      \end{block}
      \begin{alertblock}{Key insight}
        Each function comes from an \emph{atom library} with
        known curvature and monotonicity properties.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Atom Library}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \includegraphics[width=\textwidth]{fig_atom_library.pdf}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{What an Atom Provides}
        Each atom $f$ in the library specifies:
        \begin{itemize}
          \item \textbf{Domain} — where $f$ is defined
          \item \textbf{Curvature} — convex / concave / affine
          \item \textbf{Monotonicity} — nondecreasing / nonincreasing
                (per argument)
          \item \textbf{Range} — bounds on $f$'s output
        \end{itemize}
      \end{block}
      \vspace{4pt}
      \begin{block}{Sets for Atoms}
        Also defined as atoms:
        \begin{itemize}
          \item $\mathbb{R}^n$ — real numbers
          \item Positive orthant $\mathbb{R}^n_+$
          \item Epigraphs/hypographs of functions
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Verification of Convexity}
% ============================================================

\begin{frame}{Verification — Overview}
  \begin{block}{Goal}
    Automatically verify that a given problem is a valid DCP.
  \end{block}

  \vspace{6pt}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{block}{Two-Stage Verification}
        \begin{enumerate}
          \item \textbf{Stage 1 — Product-free check}\\
            All expressions must be product-free\\
            (no products of non-constant sub-expressions)
          \item \textbf{Stage 2 — Rule checking}\\
            Verify top-level, sign, and composition
            rules hold throughout the expression tree
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \includegraphics[width=\textwidth]{fig_verification_pipeline.pdf}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Product-Free Expressions}
  \begin{block}{Definition (Eq.\ 14.5)}
    An expression $e$ is \textbf{product-free} if it can be written as:
    \[
      e = a + \mathbf{b}^\top\mathbf{x} + \sum_{j=1}^{q} c_j f_j(\mathbf{e}_j)
    \]
    where $a\in\mathbb{R}$, $\mathbf{b}\in\mathbb{R}^n$, each $f_j$ is
    from the atom library, and each $\mathbf{e}_j$ is a vector of
    product-free sub-expressions.
  \end{block}

  \vspace{6pt}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Product-free (valid)}
        \begin{itemize}
          \item $3 + 2x_1 + \log(x_2)$
          \item $\max(x_1^2 - 3,\; x_2)$
          \item $\|\mathbf{A}\mathbf{x} - \mathbf{b}\|_2$
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Not product-free (invalid)}
        \begin{itemize}
          \item $x_1 \cdot x_2$\ (product of variables)
          \item $x_1^2 = x_1 \cdot x_1$\ \ (also invalid as literal product)
          \item $\sin(x_1) \cdot \log(x_2)$
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{4pt}
  \begin{block}{Note}
    $x^2$ \emph{can} be expressed through the \texttt{square} atom —
    this is how DCP handles nonlinear terms.
  \end{block}
\end{frame}

\begin{frame}{Sign Rules}
  \begin{block}{Scaling a convex expression}
    If $f(\mathbf{x})$ is convex and $c > 0$:
    \begin{itemize}
      \item $c \cdot f(\mathbf{x})$ is \textbf{convex}
      \item $-c \cdot f(\mathbf{x})$ is \textbf{concave}
      \item $c \cdot f(\mathbf{x})$ where $c < 0$: \textbf{concave}
    \end{itemize}
  \end{block}

  \vspace{4pt}
  \begin{block}{Sum Rules}
    \begin{columns}
      \begin{column}{0.45\textwidth}
        \begin{itemize}
          \item convex $+$ convex $=$ \textbf{convex}
          \item concave $+$ concave $=$ \textbf{concave}
          \item affine $+$ anything $=$ \textbf{same curvature}
        \end{itemize}
      \end{column}
      \begin{column}{0.45\textwidth}
        \begin{itemize}
          \item convex $+$ concave $=$ \textbf{neither}
          \item convex $\times$ positive const $=$ \textbf{convex}
          \item convex $\times$ negative const $=$ \textbf{concave}
        \end{itemize}
      \end{column}
    \end{columns}
  \end{block}

  \vspace{4pt}
  \begin{exampleblock}{Example: $3 + 2\log(1+x_1) - \sin(x_2)$ for $x_1\geq 0$, $x_2\in[-\pi,0]$}
    \begin{itemize}
      \item $\log(1+x_1)$ is concave $\Rightarrow$ $2\log(1+x_1)$ is concave
      \item $\sin(x_2)$ for $x_2\in[-\pi,0]$ is convex
            $\Rightarrow$ $-\sin(x_2)$ is concave
      \item Sum of two concave terms $\Rightarrow$ whole expression is \textbf{concave}
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{Composition Rules}
  \begin{block}{Key Theorem (Eqs.\ 14.3--14.4)}
    For $f(\varepsilon_1,\ldots,\varepsilon_m)$, if $f$ is expected to be
    \textbf{convex}:
    \begin{itemize}
      \item $f$ is affine or convex; and for each argument $\varepsilon_i$:
        \begin{itemize}
          \item $f$ is \emph{nondecreasing} in $\varepsilon_i$ over $\mathcal{R}$,
                and $\varepsilon_i$ is convex; \textbf{or}
          \item $f$ is \emph{nonincreasing} in $\varepsilon_i$ over $\mathcal{R}$,
                and $\varepsilon_i$ is concave; \textbf{or}
          \item $\varepsilon_i$ is affine
        \end{itemize}
    \end{itemize}
    \textit{Mirror rules apply for concave $f$.}
  \end{block}

  \vspace{4pt}
  \begin{block}{Decomposition Consequence (Eq.\ 14.7)}
    If $f$ is convex and nondecreasing:
    \[
      f(g(\mathbf{x})) \leq y \quad\iff\quad
      \begin{cases}f(z) \leq y\\g(\mathbf{x}) \leq z\end{cases}
    \]
    This enables \emph{efficient decomposition} into sub-problems.
  \end{block}
\end{frame}

\begin{frame}{Composition Rule — Examples}
  \begin{columns}
    \begin{column}{0.50\textwidth}
      \begin{exampleblock}{Ex.\ 14.3: $f(g(\mathbf{x})) = 1/x$ where $g = 3+2\log(1+x_1)-\sin x_2$}
        \begin{itemize}
          \item $f(x)=1/x$ is convex \emph{and nonincreasing} for $x>0$
          \item Need: $g$ is \emph{concave} and $g\in[3,\infty)$ (where $f$ is nonincreasing)
          \item Compute range:
          \begin{align*}
            \mathrm{range}(g) &\subseteq 3 + 2\cdot\mathrm{range}(\log(1+x_1)) - \sin x_2\\
            &= [3,3] + [0,\infty) + [0,1] = [3,\infty)
          \end{align*}
          \item $g$ is concave (shown earlier), range $\subseteq[3,\infty)$
          \item $\Rightarrow$ $f(g(\mathbf{x}))$ is \textcolor{dcpgreen}{\textbf{convex}}
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.47\textwidth}
      \includegraphics[width=\textwidth]{fig_composition_max.pdf}
      \vspace{-4pt}
      \begin{block}{\small Ex.\ 14.5: $\max(\varepsilon(x_1), x_2)$}
        \small
        \begin{itemize}
          \item $\varepsilon=x_1$: both affine $\Rightarrow$ \textbf{convex}
          \item $\varepsilon=\sin(x_1)$: neither $\Rightarrow$ \textbf{not convex}
          \item $\varepsilon=x_1^2-3$: convex $\Rightarrow$ \textbf{convex}
          \item $\varepsilon=\log(x_1)-1$: concave $\Rightarrow$ \textbf{not convex}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Top-Level Production Rules}
  \begin{block}{Rules for the Whole Problem}
    \begin{tabular}{lll}
      \toprule
      \textbf{Context} & \textbf{Required curvature} & \textbf{Example}\\
      \midrule
      Minimize $f_0$ & $f_0$ must be \textbf{convex} & $\min\;\|\mathbf{x}\|_2$\\
      Maximize $f_0$ & $f_0$ must be \textbf{concave} & $\max\;\log(\det X)$\\
      Inequality LHS & must be \textbf{convex} & $g(\mathbf{x})\leq 0$\\
      Inequality RHS & must be \textbf{concave} & $0\leq h(\mathbf{x})$\\
      Equality LHS/RHS & must be \textbf{affine} & $\mathbf{A}\mathbf{x}=\mathbf{b}$\\
      \bottomrule
    \end{tabular}
  \end{block}

  \vspace{6pt}
  \begin{alertblock}{Why affine equality?}
    An equality $f(\mathbf{x})=g(\mathbf{x})$ is both an upper
    \emph{and} lower bound, so it requires
    $f$ to be \emph{both} convex and concave — i.e.\ affine.
  \end{alertblock}
\end{frame}

\begin{frame}{Automatic Verification — Expression Tree (Ex.\ 14.6)}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \begin{exampleblock}{Problem}
        \small
        \[
          \underset{x_1,x_2}{\mathrm{minimize}}\;
          \mathrm{pow}(a_1x_1+a_2x_2-b_1,\,2)
        \]
        \[
          \text{s.t.}\quad c_1x_1+c_2x_2=d,\quad
          \sqrt{\min(x_1,x_2)}\geq 1-\log(x_2)
        \]
      \end{exampleblock}
      \begin{block}{Verification Tree (simplified)}
        \small
        \texttt{minimize}\\
        \quad$\hookrightarrow$ \texttt{pow}($\varepsilon$,2) \textcolor{dcpblue}{convex}\\
        \quad\quad$\hookrightarrow$ $-b_1+a_1x_1+a_2x_2$ \textcolor{dcpblue}{affine}\\
        \texttt{subject to}\\
        \quad$\hookrightarrow$ \texttt{=}: affine=affine \checkmark\\
        \quad$\hookrightarrow$ \texttt{$\geq$}: concave$\geq$convex \checkmark\\
        \quad\quad$\hookrightarrow$ $\sqrt{\varepsilon}$ \textcolor{dcpgreen}{concave, nondecreasing}\\
        \quad\quad\quad$\hookrightarrow$ $\min(\varepsilon,\varepsilon)$ \textcolor{dcpgreen}{concave, nondecreasing}\\
        \quad\quad$\hookrightarrow$ $1-\log(\varepsilon)$ \textcolor{dcpred}{convex, nonincreasing}
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \includegraphics[width=\textwidth]{fig_expression_tree.pdf}
      \begin{block}{\small Conditions Discovered}
        \small
        \begin{itemize}
          \item Require $x_2>0$ for $\log(x_2)$
          \item Require $\min(x_1,x_2)\geq 0$ for $\sqrt{\cdot}$
          \item Hence need $x_1\geq 0$ and $x_2>0$
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Canonicalization}
% ============================================================

\begin{frame}{Canonicalization — What and Why?}
  \begin{block}{Definition}
    \textbf{Canonicalization} is the process of converting a DCP into
    \emph{partitioned canonical form} that existing solvers can directly handle.
    It enables dual information (lower bounds, dual certificates, convergence assessment).
  \end{block}

  \vspace{4pt}
  \begin{block}{Partitioned Canonical Form (Eq.\ 14.8)}
    \[
      \underset{\mathbf{x}^{(1)},\ldots,\mathbf{x}^{(k)}}{\mathrm{minimize}}\quad
      d + \sum_{i=1}^{k}(\mathbf{c}^{(i)})^\top\mathbf{x}^{(i)}
    \]
    \[
      \text{subject to}\quad
      \mathbf{x}^{(i)}\in S^{(i)}\;\;(i=1{:}k),\qquad
      \sum_{i=1}^{k}\mathbf{A}^{(i)}\mathbf{x}^{(i)}=\mathbf{b}
    \]
    Each $S^{(i)}$ is a convex set from the atom library.
    Coupling happens only through $\mathbf{A}\mathbf{x}=\mathbf{b}$.
  \end{block}

  \vspace{4pt}
  \begin{block}{Two Steps in Canonicalization}
    \begin{enumerate}
      \item \textbf{Linearization} — introduce new atom variables
      \item \textbf{Graph implementation} — represent atoms via
            their epigraphs/hypographs
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Linearization}
  \begin{block}{Process (Eqs.\ 14.9--14.11)}
    Given a product-free expression
    $a + \mathbf{b}^\top\mathbf{x} + \sum_{j=1}^q c_j f_j(\mathbf{e}_j)$,
    the \textbf{linearized} form introduces atom variables $\mathbf{v}$:
    \[
      a + \mathbf{b}^\top\mathbf{x} + \mathbf{c}^\top\mathbf{v}
      \quad\text{with}\quad
      \begin{cases}
        f_j(\mathbf{e}_j) \leq v_j & \text{if } f_j \text{ is convex}\\
        f_j(\mathbf{e}_j) \geq v_j & \text{if } f_j \text{ is concave}
      \end{cases}
    \]
    Introduces $q$ new variables and $q$ new inequality constraints.
  \end{block}

  \vspace{4pt}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \includegraphics[width=\textwidth]{fig_linearization.pdf}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{alertblock}{Key idea}
        Each atom call $f_j(\cdot)$ is replaced by a
        fresh variable $v_j$ plus a \emph{constraint}
        bounding $v_j$ to the atom's value.\\[4pt]
        This makes all function arguments affine, which is
        what standard solvers require.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Linearization — Numerical Example (Ex.\ 14.8)}
  \begin{exampleblock}{Linearizing: $\mathrm{pow}(3 + 2\log(1+x_1) - \sin(x_2),\,-1)$}
    \textbf{Step 1}: Innermost arguments $1+x_1$ and $x_2$ are affine — no action needed.\\[4pt]
    \textbf{Step 2}: Next level $3 + 2\log(1+x_1) - \sin(x_2)$ is not affine.
    Introduce $v_1, v_2, u_{1,1}$:
    \[
      3 + 2v_1 - v_2 \quad\text{s.t.}\quad
      \log(u_{1,1}) \geq v_1,\quad u_{1,1}=1+x_1,\quad \sin(x_2)\leq v_2
    \]
    \textbf{Step 3}: Top-level $\mathrm{pow}(\cdot,-1)$ introduces $v_3, u_{3,1}$:
    \[
      v_3 \quad\text{s.t.}\quad
      \mathrm{pow}(u_{3,1},-1)\leq v_3,\quad u_{3,1}=3+2v_1-v_2
    \]
    \textbf{Result}: 5 new variables, 5 new constraints — \emph{all functions now
    applied to affine arguments}.
  \end{exampleblock}
\end{frame}

\begin{frame}{Graph Implementations — Absolute Value}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \begin{block}{Graph Expansion: $|x|$ (Eq.\ 14.13)}
        The atom $|x|$ is nonlinear but can be expressed as
        the solution to a \textbf{linear program}:
        \[
          |x| \;\Rightarrow\;
          \begin{cases}
            \displaystyle\min_y\; y\\[4pt]
            \text{s.t.}\; y \geq x,\quad y \geq -x
          \end{cases}
        \]
        This \emph{graph implementation} removes the
        nondifferentiable kink at $x=0$.
      \end{block}

      \begin{exampleblock}{Usage: constraint $|2x+3|\leq 5$}
        \small
        Original: $v=|u|,\; u=2x+3,\; v+s=5,\; s\geq 0$\\
        Expanded: $v\geq u,\; v\geq -u,\; u=2x+3,\; v+s=5,\; s\geq 0$
      \end{exampleblock}
    \end{column}
    \begin{column}{0.45\textwidth}
      \includegraphics[width=\textwidth]{fig_abs_graph.pdf}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Epigraph and Hypograph}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \begin{block}{Definitions (Eq.\ 14.15)}
        \[
          \mathrm{epi}\,f = \{(\mathbf{x},y)\in\mathbb{R}^n\times\mathbb{R}\mid
          y \geq f(\mathbf{x})\}
        \]
        \[
          \mathrm{hypo}\,f = \{(\mathbf{x},y)\in\mathbb{R}^n\times\mathbb{R}\mid
          y \leq f(\mathbf{x})\}
        \]
        \begin{itemize}
          \item $f$ is convex $\iff$ $\mathrm{epi}\,f$ is a convex set
          \item $f$ is concave $\iff$ $\mathrm{hypo}\,f$ is a convex set
        \end{itemize}
      \end{block}

      \begin{block}{Graph Expansion via Epigraph/Hypograph}
        Replace convex atom constraint:
        \[
          f(\mathbf{x}) \leq y \;\iff\; (\mathbf{x},y)\in\mathrm{epi}\,f
        \]
        Replace concave atom constraint:
        \[
          f(\mathbf{x}) \geq y \;\iff\; (\mathbf{x},y)\in\mathrm{hypo}\,f
        \]
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \includegraphics[width=\textwidth]{fig_epi_hypo.pdf}
      \vspace{4pt}
      \small\textit{Blue: epigraph of $x^3$ (convex).
      Red: hypograph of $\log x$ (concave).}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Graph Implementations — Norms (Ex.\ 14.9)}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \begin{block}{$L_1$ and $L_\infty$ Norms as LPs}
        \[
          \|\mathbf{x}\|_1 \;\Rightarrow\;
          \begin{cases}
            \min_{\mathbf{v}}\; \mathbf{1}^\top\mathbf{v}\\[2pt]
            \text{s.t.}\; -\mathbf{v}\leq\mathbf{x}\leq\mathbf{v}
          \end{cases}
        \]
        \[
          \|\mathbf{x}\|_\infty \;\Rightarrow\;
          \begin{cases}
            \min_{v}\; v\\[2pt]
            \text{s.t.}\; -v\mathbf{1}\leq\mathbf{x}\leq v\mathbf{1}
          \end{cases}
        \]
      \end{block}

      \begin{block}{General $L_p$ Norm ($p\geq 1$, Eq.\ 14.18)}
        \[
          \|\mathbf{x}\|_p \;\Rightarrow\;
          \begin{cases}
            \min_{\mathbf{v}}\; \mathbf{1}^\top\mathbf{v}\\[2pt]
            \text{s.t.}\; |x_i|^p\leq v_i\;\;\forall i
          \end{cases}
        \]
        Solved via interior point methods.
      \end{block}

      \begin{block}{$L_2$ norm as Least Squares (Ex.\ 14.10)}
        \[
          \min_\mathbf{x}\;\|\mathbf{A}\mathbf{x}-\mathbf{b}\|_2
          \;\Rightarrow\;
          \min_\mathbf{x}\;\|\mathbf{A}\mathbf{x}-\mathbf{b}\|_2^2
        \]
        Solution: $\mathbf{x}^* = (\mathbf{A}^\top\mathbf{A})^{-1}\mathbf{A}^\top\mathbf{b}$
      \end{block}
    \end{column}

          \item Returns solution values
        \end{enumerate}
      \end{block}

        \begin{itemize}
          \item LP (Linear Programs)
          \item QP / SOCP (Second-order cone)
          \item SDP (Semidefinite programs)
          \item GP (Geometric programs)
          \item MIP (Mixed-integer, partial)
        \end{itemize}

\begin{frame}[fragile]{CVXPY: Portfolio Optimization}
\begin{lstlisting}[caption={Mean-variance portfolio optimization}]
import cvxpy as cp
import numpy as np

np.random.seed(0)
n = 10                          # number of assets
mu  = np.random.randn(n) * 0.1 + 0.05  # expected returns
Sig = np.random.randn(n, n)
Sig = Sig.T @ Sig / n + 0.01 * np.eye(n)  # covariance matrix
gamma = 1.0                     # risk-aversion parameter

w = cp.Variable(n)              # portfolio weights

# DCP: minimize risk - gamma * return
risk       = cp.quad_form(w, Sig)       # convex (PSD matrix)
ret        = mu @ w                     # affine
objective  = cp.Minimize(risk - gamma * ret)
constraints = [cp.sum(w) == 1,          # budget constraint
               w >= 0]                  # long-only

problem = cp.Problem(objective, constraints)
print("Is DCP?", problem.is_dcp())   # True
problem.solve()

print(f"Optimal weights: {np.round(w.value, 3)}")
print(f"Expected return: {(mu @ w.value):.4f}")
print(f"Portfolio risk:  {float(cp.sqrt(risk).value):.4f}")
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{CVXPY: Solving via Interior Point Methods}
  \begin{columns}
    \begin{column}{0.50\textwidth}
      \begin{block}{Interior Point Method for Canonical Form}
        Canonical form enables efficient interior point (barrier) methods.
        For the partitioned problem, the Hessian is block-diagonal:
        \[
          \nabla^2 p_{\mathrm{barrier}}(\mathbf{x}) =
          \begin{bmatrix}
            \nabla^2 p^{(1)}_{\mathrm{barrier}}(\mathbf{x}^{(1)}) & & \\
            & \ddots & \\
            & & \nabla^2 p^{(k)}_{\mathrm{barrier}}(\mathbf{x}^{(k)})
          \end{bmatrix}
        \]
        Block-diagonal inverse is \emph{extremely fast} to compute.
      \end{block}
      \includegraphics[width=\textwidth]{fig_interior_point.pdf}
    \end{column}
    \begin{column}{0.47\textwidth}
\begin{lstlisting}[caption={Specifying solvers in CVXPY}]
import cvxpy as cp
import numpy as np

x = cp.Variable(2)
A = np.eye(2)
b = np.array([1., 2.])

prob = cp.Problem(
    cp.Minimize(cp.norm(A @ x - b, 2)),
    [cp.norm(x, 1) <= 2.0]
)

# Choose specific solver
prob.solve(solver=cp.SCS)       # SCS
prob.solve(solver=cp.ECOS)      # ECOS (IP)
# prob.solve(solver=cp.MOSEK)  # MOSEK (commercial)

print(f"x* = {x.value}")
print(f"Solver used: {prob.solver_stats.solver_name}")
print(f"Solve time:  {prob.solver_stats.solve_time:.4f}s")
\end{lstlisting}

\begin{frame}{Chapter Summary}
  \begin{block}{Key Takeaways}
    \begin{enumerate}
      \item \textbf{DCP} is a framework for writing convex optimization
            problems in a form that automated tools can verify and solve.
      \item \textbf{Verification} proceeds in two stages:
            (1) product-free check, then (2) sign/composition/top-level rules
            applied recursively down the expression tree.
      \item \textbf{Canonicalization} transforms a DCP into partitioned
            canonical form via linearization and graph implementations.
      \item \textbf{Interior point methods} efficiently solve canonical-form
            problems by exploiting block-diagonal Hessian structure.
      \item \textbf{CVXPY} (Python) / Convex.jl (Julia) implement DCP:
            parse, verify, canonicalize, and call a numerical solver.
    \end{enumerate}
  \end{block}

\begin{frame}[allowframebreaks]{Selected Exercises}
  \begin{block}{Exercise 14.1 — Extended-Value Functions}
    Show that if $f$ is convex over a convex set $\mathcal{S}$, then
    $g(\mathbf{x}) = f(\mathbf{x})$ if $\mathbf{x}\in\mathcal{S}$,
    $\infty$ otherwise, is also convex over $\mathbb{R}^n$.\\[4pt]
    \textit{Hint:} Check the three cases: both $\mathbf{x},\mathbf{y}\in\mathcal{S}$;
    exactly one in $\mathcal{S}$; neither in $\mathcal{S}$.
  \end{block}

  \begin{block}{Exercise 14.2 — Squares in DCP}
    Products are not DCP-compliant, but $x^2$ can be incorporated
    by defining a \texttt{square($x$)} atom with:
    curvature = convex, nondecreasing for $x\geq 0$,
    nonincreasing for $x\leq 0$, range $=[0,\infty)$.
  \end{block}

  \begin{block}{Exercise 14.5 — Graph Implementation of $\sqrt{x}$}
    $\sqrt{x}$ is concave, so use its hypograph:
    \[
      \mathrm{hypo}\sqrt{x} = \{(x,y)\mid x\geq 0,\; y\leq\sqrt{x}\}
      = \{(x,y)\mid x\geq 0,\; y^2\leq x\}
    \]
    Graph implementation:
    $\sqrt{x} \;\Rightarrow\; \arg\max_y y \;\;\text{s.t.}\;\; y^2\leq x$\\
    This is differentiable everywhere.
  \end{block}

  \begin{block}{Exercise 14.6 — Canonicalize a Norm Problem}
    \[
      \min_\mathbf{x}\;\|\mathbf{A}\mathbf{x}-\mathbf{b}\|_1
      \quad\text{s.t.}\quad
      \|\mathbf{C}\mathbf{x}-\mathbf{d}\|_{1.5}\leq 3
    \]
    Apply $L_1$ graph expansion: $-\mathbf{y}^{(1)}\leq\mathbf{A}\mathbf{x}-\mathbf{b}\leq\mathbf{y}^{(1)}$.
    Apply $L_p$ expansion: $|(\mathbf{C}\mathbf{x}-\mathbf{d})_i|^{1.5}\leq y_i^{(2)}$, minimize $\mathbf{1}^\top\mathbf{y}^{(2)}+s=3$.
  \end{block}
\end{frame}
