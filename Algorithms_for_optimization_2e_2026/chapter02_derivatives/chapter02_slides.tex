\section{1. Scalar Derivatives}
% =============================================================================

\begin{frame}{Motivation: Why Derivatives?}
  \begin{columns}[T]
    \column{0.60\linewidth}
    \begin{itemize}
      \item Optimization: find design point that \textbf{minimises} (or maximises)
            an objective function.
      \item The \emph{derivative} tells us \textbf{how the function changes} as
            we vary an input.
      \item Knowing the derivative allows us to predict which direction
            improves the objective.
      \item In one dimension: \textbf{derivative} $f'(x)$.\\
            In multiple dimensions: \textbf{gradient} $\grad f(\x)$.
    \end{itemize}
    \column{0.38\linewidth}
    \centering
    \includegraphics[width=\linewidth]{fig_tangent_line.pdf}\\
    {\scriptsize The derivative equals the slope of the tangent line.}
  \end{columns}
\end{frame}

\begin{frame}{Scalar Derivative: Definition}
  \begin{block}{Definition (2.2)}
    The \textbf{derivative} $f'(x)$ of a function $f:\R\to\R$ at $x$ is the rate
    at which $f$ changes:
    \[
      f'(x) = \lim_{h\to 0}\frac{f(x+h)-f(x)}{h}
            = \frac{\Delta f(x)}{\Delta x}
    \]
  \end{block}
  \vspace{0.3em}
  \begin{columns}[T]
    \column{0.55\linewidth}
    \textbf{Interpretations}
    \begin{itemize}
      \item \textbf{Geometric:} slope of the tangent line.
      \item \textbf{Linear approximation:}
        $f(x+\Delta x)\approx f(x)+f'(x)\,\Delta x$
      \item \textbf{Leibniz notation:}
        $f'(x)\equiv \dfrac{df(x)}{dx}$
    \end{itemize}
    \column{0.43\linewidth}
    \textbf{Three equivalent limit forms (eq.\ 2.4):}
    \begin{align*}
      f'(x) &= \lim_{h\to 0}\frac{f(x+h)-f(x)}{h}
             && \text{(forward)}\\[4pt]
             &= \lim_{h\to 0}\frac{f(x+h/2)-f(x-h/2)}{h}
             && \text{(central)}\\[4pt]
             &= \lim_{h\to 0}\frac{f(x)-f(x-h)}{h}
             && \text{(backward)}
    \end{align*}
  \end{columns}
\end{frame}

\begin{frame}{Symbolic Differentiation (Example 2.1)}
  \begin{block}{Chain rule / product rule}
    Given $f(x)=x^2+x/2-\sin(x)/x$, symbolic differentiation yields the exact
    closed-form derivative.
  \end{block}
  \begin{columns}[T]
    \column{0.48\linewidth}
    \textbf{Using SymPy (Python equivalent of Julia's SymEngine):}
    \vspace{0.3em}
    \begin{block}{}
      \ttfamily\small
      \begin{tabbing}
        \texttt{from sympy import symbols, diff, sin}\\
        \texttt{x = symbols('x')}\\
        \texttt{f = x**2 + x/2 - sin(x)/x}\\
        \texttt{df = diff(f, x)}\\
        \texttt{\# Result:}\\
        \texttt{\# 2*x + 1/2 + sin(x)/x**2 - cos(x)/x}
      \end{tabbing}
    \end{block}
    \column{0.48\linewidth}
    \textbf{Result:}
    \[
      f'(x)=2x+\frac{1}{2}+\frac{\sin x}{x^2}-\frac{\cos x}{x}
    \]
    \begin{itemize}
      \item Exact (no numerical error).
      \item May produce \emph{expression swell} for complex functions.
      \item Outside scope of this chapter; provided by \texttt{sympy}.
    \end{itemize}
  \end{columns}
\end{frame}

\section{2. Derivatives in Multiple Dimensions}
% =============================================================================

\begin{frame}{Gradient}
  \begin{block}{Definition --- Gradient (eq.\ 2.5)}
    For $f:\R^n\to\R$, the \textbf{gradient} is the vector of partial derivatives:
    \[
      \grad f(\x) =
      \begin{bmatrix}
        \dfrac{\partial f}{\partial x_1} \\[6pt]
        \vdots \\[4pt]
        \dfrac{\partial f}{\partial x_n}
      \end{bmatrix}
      \in \R^n
    \]
    \begin{center}\small
      $x_i$ — $i$-th component of $\x$;\quad
      $\partial f/\partial x_i$ — partial derivative w.r.t.\ $x_i$.
    \end{center}
  \end{block}
  \begin{itemize}
    \item The gradient points in the direction of \textbf{steepest ascent}.
    \item A \textbf{hyperplane} (tangent) can be used to approximate $f$ near $\x$:
          $f(\x+\Delta\x)\approx f(\x)+\grad f(\x)\T\Delta\x$.
  \end{itemize}
\end{frame}

\begin{frame}{Gradient: Geometric Intuition}
  \begin{columns}[T]
    \column{0.52\linewidth}
    \includegraphics[width=\linewidth]{fig_gradient_contour.pdf}
    \column{0.45\linewidth}
    \textbf{Key observations:}
    \begin{itemize}
      \item $\grad f(\x)$ is \emph{perpendicular} to level curves.
      \item Moving \emph{opposite} the gradient is the steepest descent direction.
      \item For $f(\x)=x_1^2+2x_2^2$:
        \[
          \grad f(\x)=\begin{bmatrix}2x_1\\4x_2\end{bmatrix}
        \]
      \item Gradient-based optimisers use $-\grad f(\x)$ to update $\x$.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Jacobian Matrix}
  \begin{block}{Definition --- Jacobian}
    For $\mathbf{f}:\R^n\to\R^m$, the \textbf{Jacobian} is the $m\times n$ matrix
    of partial derivatives:
    \[
      \mathbf{J}_{ij} = \frac{\partial f_i}{\partial x_j},\qquad
      \mathbf{J} =
      \begin{bmatrix}
        \dfrac{\partial f_1}{\partial x_1} & \cdots & \dfrac{\partial f_1}{\partial x_n}\\[6pt]
        \vdots & \ddots & \vdots\\[4pt]
        \dfrac{\partial f_m}{\partial x_1} & \cdots & \dfrac{\partial f_m}{\partial x_n}
      \end{bmatrix}
    \]
  \end{block}
  \begin{itemize}
    \item For scalar $f$ ($m=1$): Jacobian $=\grad f(\x)\T$ (row vector).
    \item Chain rule for vector functions:
          $\dfrac{d\mathbf{f}(\mathbf{g}(\x))}{d\x}
           = \mathbf{J}_{\mathbf{f}} \cdot \mathbf{J}_{\mathbf{g}}$.
  \end{itemize}
\end{frame}

\begin{frame}{Hessian Matrix}
  \begin{columns}[T]
    \column{0.52\linewidth}
    \begin{block}{Definition --- Hessian (eq.\ 2.8)}
      For $f:\R^n\to\R$, the \textbf{Hessian} is the symmetric $n\times n$ matrix
      of second-order partial derivatives:
      \[
        \mathbf{H}_{ij}=\frac{\partial^2 f}{\partial x_i\,\partial x_j}
      \]
      \[
        \mathbf{H} =
        \begin{bmatrix}
          f_{x_1 x_1} & \cdots & f_{x_1 x_n}\\
          \vdots & \ddots & \vdots\\
          f_{x_n x_1} & \cdots & f_{x_n x_n}
        \end{bmatrix}
      \]
    \end{block}
    \begin{itemize}
      \item Captures \textbf{curvature} of $f$ near $\x$.
      \item Used in second-order optimisers (Newton's method).
      \item Symmetric when $f$ is twice continuously differentiable.
    \end{itemize}
    \column{0.45\linewidth}
    \includegraphics[width=\linewidth]{fig_hessian_curvature.pdf}\\
    {\scriptsize Top: isotropic curvature ($\lambda_1=\lambda_2$).
     Bottom: anisotropic — elongated contours.}
  \end{columns}
\end{frame}

\begin{frame}{Directional Derivative}
  \begin{block}{Definition (eq.\ 2.10)}
    The \textbf{directional derivative} of $f$ at $\x$ in direction
    $\mathbf{s}\in\R^n$ is:
    \[
      \nabla_{\mathbf{s}} f(\x)
        = \lim_{h\to 0}\frac{f(\x + h\mathbf{s}) - f(\x)}{h}
        = \grad f(\x)\T \mathbf{s}
    \]
  \end{block}
  \begin{columns}[T]
    \column{0.48\linewidth}
    \textbf{Interpretation:}
    \begin{itemize}
      \item Rate of change of $f$ along direction $\mathbf{s}$.
      \item If $\|\mathbf{s}\|=1$, it is the \emph{slope} in direction $\mathbf{s}$.
      \item Special case: $\mathbf{s}=\mathbf{e}_i$ gives $\partial f/\partial x_i$.
    \end{itemize}
    \column{0.48\linewidth}
    \textbf{Numerical example:}\\
    $f(x_1,x_2)=x_1^2+x_2^2$, $\x=(1,1)\T$, $\mathbf{s}=(1/\sqrt{2},1/\sqrt{2})\T$:
    \[
      \grad f = (2,2)\T
    \]
    \[
      \nabla_\mathbf{s} f = (2,2)\cdot\tfrac{1}{\sqrt{2}}(1,1)\T = 2\sqrt{2}
    \]
  \end{columns}
\end{frame}

\section{3. Numerical Differentiation}
% =============================================================================

\begin{frame}{Finite Difference Methods: Overview}
  \begin{itemize}
    \item \textbf{Approximate} derivatives by evaluating $f$ at nearby points.
    \item Three methods based on different limit interpretations of eq.~(2.4):
  \end{itemize}
  \vspace{0.5em}
  \begin{block}{Finite Difference Approximations}
    \renewcommand{\arraystretch}{1.8}
    \begin{tabular}{lll}
      \toprule
      \textbf{Method} & \textbf{Formula} & \textbf{Error} \\
      \midrule
      Forward difference  & $\dfrac{f(x+h)-f(x)}{h}$ & $O(h)$ \\
      Central difference  & $\dfrac{f(x+h/2)-f(x-h/2)}{h}$ & $O(h^2)$ \\
      Backward difference & $\dfrac{f(x)-f(x-h)}{h}$ & $O(h)$ \\
      \bottomrule
    \end{tabular}
  \end{block}
  \vspace{0.3em}
  \textbf{Central difference is more accurate} (quadratic vs.\ linear error) but requires two function evaluations.
\end{frame}

\begin{frame}{Forward Difference: Taylor Analysis}
  \begin{block}{Taylor Expansion (eq.\ 2.11--2.13)}
    Expanding $f(x+h)$ around $x$:
    \[
      f(x+h) = f(x) + f'(x)h + \frac{f''(x)}{2!}h^2
                    + \frac{f'''(x)}{3!}h^3 + \cdots
    \]
    Rearranging:
    \[
      f'(x) = \frac{f(x+h)-f(x)}{h} - \frac{f''(x)}{2}h - \cdots
            \approx \frac{f(x+h)-f(x)}{h} + O(h)
    \]
  \end{block}
  \begin{columns}[T]
    \column{0.48\linewidth}
    \textbf{Observations:}
    \begin{itemize}
      \item Error term is $O(h)$ --- linear in step size.
      \item Smaller $h$ $\Rightarrow$ better approximation \emph{in principle}.
      \item \textbf{But:} too small $h$ causes
            \textit{subtractive cancellation} (floating-point loss).
    \end{itemize}
    \column{0.48\linewidth}
    \textbf{Practical tip:} Choose $h \approx \sqrt{\epsilon_{\text{mach}}}
    \approx 10^{-8}$ for forward difference ($\epsilon_{\text{mach}}\approx
    2.2\times10^{-16}$ in \texttt{float64}).
  \end{columns}
\end{frame}

\begin{frame}{Central Difference: Taylor Analysis}
  \begin{block}{Taylor expansions (eq.\ 2.15--2.18)}
    \[
      f(x+h/2) = f(x)+f'(x)\tfrac{h}{2}+\tfrac{f''(x)}{2!}\!\left(\tfrac{h}{2}\right)^2
                  +\tfrac{f'''(x)}{3!}\!\left(\tfrac{h}{2}\right)^3+\cdots
    \]
    \[
      f(x-h/2) = f(x)-f'(x)\tfrac{h}{2}+\tfrac{f''(x)}{2!}\!\left(\tfrac{h}{2}\right)^2
                  -\tfrac{f'''(x)}{3!}\!\left(\tfrac{h}{2}\right)^3+\cdots
    \]
    Subtracting and rearranging:
    \[
      f'(x) \approx \frac{f(x+h/2)-f(x-h/2)}{h} + O(h^2)
    \]
  \end{block}
  \begin{itemize}
    \item Error is $O(h^2)$ --- \textbf{quadratic} accuracy.
    \item Even-order terms cancel $\Rightarrow$ faster convergence.
    \item Optimal step: $h\approx\epsilon_{\text{mach}}^{1/3}\approx 10^{-5}$.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Finite Difference: Python Implementation}
\begin{lstlisting}[caption={Forward, central, and backward difference --- Python (NumPy)}]
import numpy as np

def diff_forward(f, x, h=1e-8):
    """Forward difference: O(h) error."""
    return (f(x + h) - f(x)) / h

def diff_central(f, x, h=1e-5):
    """Central difference: O(h^2) error."""
    return (f(x + h/2) - f(x - h/2)) / h

def diff_backward(f, x, h=1e-8):
    """Backward difference: O(h) error."""
    return (f(x) - f(x - h)) / h

# Gradient for multivariate f: R^n -> R  (forward difference)
def gradient_fd(f, x, h=1e-5):
    """Numerical gradient using central differences."""
    n = len(x)
    g = np.zeros(n)
    for i in range(n):
        xp, xm = x.copy(), x.copy()
        xp[i] += h/2;  xm[i] -= h/2
        g[i] = (f(xp) - f(xm)) / h
    return g

# Running example: f(x) = sin(x^2), x = pi/2
f = lambda x: np.sin(x**2)
x0 = np.pi / 2
true_val = 2*x0 * np.cos(x0**2)
print(f"True:    {true_val:.8f}")
print(f"Forward: {diff_forward(f, x0):.8f}")
print(f"Central: {diff_central(f, x0):.8f}")
\end{lstlisting}
\end{frame}

\begin{frame}{Complex-Step Method}
  \begin{block}{Key Idea (eq.\ 2.19--2.22)}
    Evaluate $f$ at a \textbf{complex} point $x+ih$ where $i=\sqrt{-1}$:
    \[
      f(x+ih) = f(x) + ihf'(x) - \frac{h^2 f''(x)}{2!}
                      - \frac{ih^3 f'''(x)}{3!} + \cdots
    \]
    Taking the imaginary part:
    \[
      \operatorname{Im}(f(x+ih)) = hf'(x) - \frac{h^3 f'''(x)}{3!}+\cdots
    \]
    \[
      \boxed{f'(x) \approx \frac{\operatorname{Im}(f(x+ih))}{h} + O(h^2)}
    \]
  \end{block}
  \begin{itemize}
    \item \textbf{No subtractive cancellation} --- uses imaginary part, not difference.
    \item Optimal $h$: choose $h$ extremely small, e.g., $h=10^{-20}$ is fine.
    \item Real part gives $f(x)$ to within $O(h^2)$:
          $\operatorname{Re}(f(x+ih))\approx f(x)-\tfrac{h^2 f''(x)}{2}+\cdots$
    \item \textbf{Requirement:} $f$ must support complex inputs (analytic $f$).
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Complex-Step Method: Python Code + Example}
\begin{lstlisting}[caption={Complex-step derivative (Algorithm 2.2 equivalent)}]
import numpy as np

def diff_complex(f, x, h=1e-20):
    """Complex-step derivative: Im(f(x + i*h)) / h"""
    return np.imag(f(x + h*1j)) / h

# Example 2.4: f(x) = sin(x^2)
f = lambda x: np.sin(x**2)
x0 = np.pi / 2

# function value and derivative using a single evaluation
v = f(x0 + 1e-20j)
fx   = np.real(v)      # f(x0)
dfdx = np.imag(v) / 1e-20  # f'(x0)

print(f"f(pi/2)  = {fx:.8f}")   # approx 0.62426
print(f"f'(pi/2) = {dfdx:.8f}") # approx -2.45425
# True: pi * cos(pi^2/4)
true_d = np.pi * np.cos(np.pi**2 / 4)
print(f"True     = {true_d:.8f}")
\end{lstlisting}
\textbf{Key output:} Both $f(x)$ and $f'(x)$ from a \emph{single} function evaluation.
\end{frame}

\section{4. Automatic Differentiation}
% =============================================================================

\begin{frame}{Automatic Differentiation: Motivation}
  \begin{itemize}
    \item Numerical finite differences are simple but have accuracy/cost tradeoffs.
    \item Symbolic differentiation can suffer from \textit{expression swell}.
    \item \textbf{Automatic differentiation (AD)} computes \emph{exact} derivatives
          of a function specified as computer code.
    \item Based on the \textbf{chain rule}:
      \[
        \frac{d}{dx}f(g(x)) = \frac{df}{dg}\cdot\frac{dg}{dx}
      \]
    \item A program consists of elementary operations: $+$, $-$, $\times$,
          $\div$, $\sin$, $\exp$, $\ln$, \dots --- each has a known derivative.
    \item The \textbf{computational graph} encodes data flow.
  \end{itemize}
\end{frame}

\begin{frame}{Computational Graph}
  \begin{columns}[T]
    \column{0.54\linewidth}
    \centering
    \includegraphics[width=0.98\linewidth]{fig_comp_graph.pdf}\\
    {\scriptsize Computational graph for $f(a,b)=\ln(ab+\max(a,2))$.}
    \column{0.43\linewidth}
    \textbf{Structure:}
    \begin{itemize}
      \item \textbf{Leaf nodes:} inputs ($a$, $b$) and constants ($2$).
      \item \textbf{Internal nodes:} operations ($\times$, $\max$, $+$, $\ln$).
      \item \textbf{Terminal node:} output ($c_4$).
    \end{itemize}
    \vspace{0.5em}
    \textbf{Intermediate values:}
    \begin{align*}
      c_1 &= a\times b \\
      c_2 &= \max(a,2) \\
      c_3 &= c_1 + c_2 \\
      c_4 &= \ln(c_3) = f(a,b)
    \end{align*}
    Two traversal strategies: \textbf{forward} and \textbf{reverse} accumulation.
  \end{columns}
\end{frame}

\begin{frame}[allowframebreaks]{Forward Accumulation}
  \begin{block}{Forward Mode: Chain Rule (eq.\ 2.31)}
    Traverse the graph from \textbf{inputs to outputs}, propagating
    $(\text{value},\ \dot{c})$ pairs where $\dot{c}_i = \partial c_i/\partial x$:
    \[
      \frac{df}{dx} = \frac{df}{dc_4}\cdot\frac{dc_4}{dc_3}\cdot
                      \left(\frac{dc_3}{dc_2}\cdot\frac{dc_2}{dx}
                           +\frac{dc_3}{dc_1}\cdot\frac{dc_1}{dx}\right)
    \]
  \end{block}

  \textbf{Worked example} (eq.\ 2.26--2.30): $f(a,b)=\ln(ab+\max(a,2))$ at $a=3$, $b=2$,
  computing $\partial f/\partial a$:
  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.48\linewidth}
    \small
    \begin{tabular}{lll}
      \toprule
      \textbf{Node} & \textbf{Value} & \textbf{Derivative $\dot{c}$} \\
      \midrule
      $a$ & $3$ & $\dot{a}=1$ \\
      $b$ & $2$ & $\dot{b}=0$ \\
      $c_1=ab$ & $6$ & $\dot{c}_1=b\dot{a}+a\dot{b}=2$ \\
      $c_2=\max(a,2)$ & $3$ & $\dot{c}_2=\dot{a}=1$ (since $2<a$) \\
      $c_3=c_1+c_2$ & $9$ & $\dot{c}_3=\dot{c}_1+\dot{c}_2=3$ \\
      $c_4=\ln c_3$ & $\ln 9$ & $\dot{c}_4=\dot{c}_3/c_3=1/3$ \\
      \bottomrule
    \end{tabular}
    \column{0.48\linewidth}
    \textbf{Result:}
    \[
      f(3,2)=\ln 9,\quad \frac{\partial f}{\partial a}=\frac{1}{3}
    \]
    \begin{itemize}
      \item \textbf{One forward pass} computes one partial derivative.
      \item For $n$ inputs: requires $n$ passes.
      \item Efficient when \textbf{few inputs} (e.g.\ $n\ll m$).
    \end{itemize}
  \end{columns}

  \framebreak

  \textbf{Dual numbers:} A clean way to implement forward accumulation.
  \begin{block}{Dual Number Arithmetic (eq.\ 2.32--2.33)}
    Introduce abstract quantity $\epsilon$ with $\epsilon^2=0$. A dual number
    $a+b\epsilon$ encodes both value ($a$) and derivative ($b$):
    \begin{align*}
      (a+b\epsilon)+(c+d\epsilon) &= (a+c)+(b+d)\epsilon \\
      (a+b\epsilon)\times(c+d\epsilon) &= (ac)+(ad+bc)\epsilon
    \end{align*}
  \end{block}
  \begin{itemize}
    \item Set initial dual number: $x = x_0 + 1\cdot\epsilon$ to compute $f'(x_0)$.
    \item Propagate through all operations naturally.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Forward AD: Python with Dual Numbers}
\begin{lstlisting}[caption={Dual number class for automatic forward differentiation}]
import numpy as np

class Dual:
    """Dual number: v + d*epsilon where epsilon^2 = 0."""
    def __init__(self, v, d=0.0):
        self.v, self.d = float(v), float(d)

    def __add__(self, other):
        if isinstance(other, Dual):
            return Dual(self.v + other.v, self.d + other.d)
        return Dual(self.v + other, self.d)
    __radd__ = __add__

    def __mul__(self, other):
        if isinstance(other, Dual):
            return Dual(self.v * other.v,
                        self.v * other.d + self.d * other.v)
        return Dual(self.v * other, self.d * other)
    __rmul__ = __mul__

    def log(self):
        return Dual(np.log(self.v), self.d / self.v)

# Example 2.5: f(a,b) = log(a*b + max(a,2)) at a=3, b=2
def f(a, b):
    c1 = a * b
    c2 = Dual(max(a.v, 2), a.d if a.v > 2 else 0.0)
    return (c1 + c2).log()

a = Dual(3.0, 1.0)   # differentiate w.r.t. a
b = Dual(2.0, 0.0)
result = f(a, b)
print(f"f(3,2) = {result.v:.6f}")     # ln(9) ~ 2.1972
print(f"df/da  = {result.d:.6f}")     # 1/3 ~ 0.3333
\end{lstlisting}
\end{frame}

\begin{frame}[allowframebreaks]{Reverse Accumulation (Backpropagation)}
  \begin{block}{Reverse Mode: Chain Rule}
    Traverse the graph from \textbf{outputs to inputs}, propagating
    \textbf{adjoints} (sensitivities) $\bar{c}_i = \partial f/\partial c_i$:
    \[
      \bar{c}_i = \sum_{j:\,j \text{ depends on }i}
                  \bar{c}_j\cdot\frac{\partial c_j}{\partial c_i}
    \]
  \end{block}

  \textbf{Procedure:}
  \begin{enumerate}
    \item \textbf{Forward pass:} compute and store all intermediate values
          $c_i$ (the \emph{tape}).
    \item \textbf{Backward pass:} propagate adjoints from output to inputs
          using stored values.
  \end{enumerate}

  \textbf{Complexity:}
  \begin{itemize}
    \item One backward pass computes \textbf{all $n$ partial derivatives}
          simultaneously.
    \item Cost $\approx$ $3\times$ forward pass.
    \item Ideal when \textbf{few outputs, many inputs} (common in ML!).
  \end{itemize}

  \framebreak

  \textbf{Forward vs.\ Reverse --- when to use which:}
  \vspace{0.5em}
  \begin{block}{}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{lll}
      \toprule
      & \textbf{Forward mode} & \textbf{Reverse mode} \\
      \midrule
      Passes needed & $n$ (one per input) & $m$ (one per output) \\
      Best for & $n \ll m$ (few inputs) & $m \ll n$ (few outputs) \\
      Memory & $O(1)$ & $O(\text{graph size})$ \\
      ML / deep learning & less common & \textbf{backpropagation} \\
      \bottomrule
    \end{tabular}
  \end{block}
  \vspace{0.4em}
  \textbf{In deep learning:} $f:\R^n\to\R$ (scalar loss), $n$ huge
  $\Rightarrow$ reverse mode (backprop) computes full gradient in one pass.
\end{frame}

\begin{frame}[fragile]{Reverse AD with JAX (Python)}
\begin{lstlisting}[caption={Automatic differentiation with JAX (forward + reverse mode)}]
# pip install jax
import jax
import jax.numpy as jnp

# Define function
def f(x):
    return jnp.log(x[0]*x[1] + jnp.maximum(x[0], 2.0))

# Reverse mode: full gradient in one pass
x0 = jnp.array([3.0, 2.0])
grad_f = jax.grad(f)          # returns gradient function
g = grad_f(x0)                # [df/da, df/db]
print(f"df/da = {g[0]:.4f}")  # 0.3333
print(f"df/db = {g[1]:.4f}")  # 0.3333

# Jacobian-vector product (forward mode)
primals, tangents = jax.jvp(f, (x0,), (jnp.array([1.0, 0.0]),))
print(f"JVP (df/da) = {tangents:.4f}")  # 0.3333

# Hessian
hess = jax.hessian(f)(x0)
print("Hessian:\n", hess)
\end{lstlisting}
\end{frame}

\section{5. Regression Gradient}
% =============================================================================

\begin{frame}{Regression Gradient: Motivation}
  \begin{itemize}
    \item Sometimes $f$ is a \textbf{black-box}: no closed form, no AD support.
    \item Idea: estimate the gradient from \emph{multiple function evaluations}
          using linear regression.
    \item Take $m$ perturbation vectors $\Delta\x^{(i)}$ and evaluate
          $f(\x+\Delta\x^{(i)})$.
    \item Fit the best linear model:
          $f(\x+\Delta\x)\approx f(\x)+\g\T\Delta\x$
  \end{itemize}
  \begin{block}{Optimisation problem (eq.\ 2.42)}
    \[
      \min_{\g}\sum_{i=1}^{m}\left(
        f\!\left(\x+\Delta\x^{(i)}\right)
        -\left(f(\x)+\g\T\Delta\x^{(i)}\right)
      \right)^2
    \]
  \end{block}
\end{frame}

\begin{frame}{Regression Gradient: Solution (Algorithm 2.3)}
  \begin{block}{Matrix formulation (eq.\ 2.43--2.46)}
    Define:
    \[
      \Delta\mathbf{X} =
      \begin{bmatrix}(\Delta\x^{(1)})\T\\\vdots\\(\Delta\x^{(m)})\T\end{bmatrix},\quad
      \Delta\mathbf{f} =
      \begin{bmatrix}f(\x+\Delta\x^{(1)})-f(\x)\\\vdots\\f(\x+\Delta\x^{(m)})-f(\x)\end{bmatrix}
    \]
    Problem becomes: $\min_{\g}\|\Delta\mathbf{X}\g-\Delta\mathbf{f}\|^2$\\
    Closed-form solution via \textbf{pseudoinverse}:
    \[
      \boxed{\g = \Delta\mathbf{X}^{+}\Delta\mathbf{f}}
    \]
    where $\Delta\mathbf{X}^{+}$ is the Moore--Penrose pseudoinverse.
  \end{block}
  Perturbations: draw $\Delta\x^{(i)}$ uniformly from a hypersphere of radius $\delta$.
\end{frame}

\begin{frame}[fragile]{Regression Gradient: Python Implementation}
\begin{lstlisting}[caption={Regression gradient (Algorithm 2.3 in Python)}]
import numpy as np

def regression_gradient(f, x, m=20, delta=0.1):
    """
    Estimate gradient of f at x using m perturbations
    drawn uniformly from a hypersphere of radius delta.
    """
    n = len(x)
    fx = f(x)
    # Generate m random unit directions, scaled by delta
    dX = np.random.randn(m, n)
    dX /= np.linalg.norm(dX, axis=1, keepdims=True)  # unit sphere
    dX *= delta                                        # scale to radius delta
    # Evaluate function differences
    df = np.array([f(x + dx) - fx for dx in dX])
    # Least-squares solution: g = dX^+ @ df
    g, *_ = np.linalg.lstsq(dX, df, rcond=None)
    return g

# Test on f(x) = x1^2 + x2^2,  true gradient at (1,1): [2, 2]
f = lambda x: x[0]**2 + x[1]**2
x0 = np.array([1.0, 1.0])
np.random.seed(42)
g_est = regression_gradient(f, x0, m=50, delta=0.01)
print(f"Estimated gradient: {g_est}")   # ~[2.0, 2.0]
print(f"True gradient:      [2.0, 2.0]")
\end{lstlisting}
\end{frame}

\section{6. SPSA}
% =============================================================================

\begin{frame}{Simultaneous Perturbation Stochastic Gradient Approximation}
  \begin{block}{Key Idea (SPSA)}
    Estimate the gradient using only \textbf{2 function evaluations},
    regardless of dimension $n$, by perturbing \emph{all} coordinates
    simultaneously with a random $\pm 1$ direction vector $\mathbf{z}$.
  \end{block}

  \begin{block}{SPSA Gradient Estimate (eq.\ 2.48)}
    For a random vector $\mathbf{z}\in\{-1,+1\}^n$:
    \[
      \nabla_\mathbf{z} f(\x)
        \approx \frac{f(\x+\delta\mathbf{z})-f(\x-\delta\mathbf{z})}{2\delta}
    \]
    The component-wise gradient estimate:
    \[
      \hat{g}_i \approx \frac{f(\x+\delta\mathbf{z})-f(\x-\delta\mathbf{z})}{2\delta}\cdot z_i
    \]
  \end{block}

  \begin{itemize}
    \item Average over $m$ samples: $\hat{\g} = \frac{1}{m}\sum_{k=1}^{m}\hat{\g}^{(k)}$.
    \item Unbiased: $\mathbb{E}[\hat{g}_i] = \partial f/\partial x_i$.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{SPSA: Python Implementation}
\begin{lstlisting}[caption={SPSA gradient estimation (Algorithm 2.4 equivalent)}]
import numpy as np

def spsa_gradient(f, x, delta=0.1, m=10):
    """
    SPSA gradient estimate using m pairs of perturbations.
    Each pair requires only 2 function evaluations.
    Total cost: 2*m function evaluations (vs n+1 for finite diff).
    """
    n = len(x)
    g = np.zeros(n)
    for _ in range(m):
        # Random +/-1 perturbation direction
        z = np.random.choice([-1.0, 1.0], size=n)
        fp = f(x + delta * z)
        fm = f(x - delta * z)
        g += ((fp - fm) / (2 * delta)) * z
    return g / m

# Example: f(x,y) = x^2 + y^2 + 0.5*x*y
f = lambda x: x[0]**2 + x[1]**2 + 0.5*x[0]*x[1]
x0 = np.array([1.0, 2.0])
# True gradient: [2*x0 + 0.5*y0, 2*y0 + 0.5*x0] = [3.0, 4.5]
np.random.seed(0)
g_spsa = spsa_gradient(f, x0, delta=0.01, m=500)
print(f"SPSA estimate: {g_spsa}")
print(f"True gradient: [3.0, 4.5]")
\end{lstlisting}
\end{frame}

\section{7. Summary \& Exercises}

\begin{frame}{Chapter Summary}
  \begin{columns}[T]
    \column{0.48\linewidth}
    \begin{block}{Core Concepts}
      \begin{itemize}
        \item \textbf{Derivative} $f'(x)$: instantaneous rate of change.
        \item \textbf{Gradient} $\grad f(\x)$: steepest ascent direction.
        \item \textbf{Jacobian}: first-order multivariate derivatives.
        \item \textbf{Hessian}: second-order; captures curvature.
        \item \textbf{Directional derivative}: rate along a vector.
      \end{itemize}
    \end{block}
    \column{0.48\linewidth}
    \begin{block}{Differentiation Methods}
      \begin{itemize}
        \item \textbf{Symbolic}: exact but may have expression swell.
        \item \textbf{Forward diff}: $O(h)$ error, 2 evaluations.
        \item \textbf{Central diff}: $O(h^2)$ error, 2 evaluations.
        \item \textbf{Complex-step}: $O(h^2)$, no cancellation.
        \item \textbf{Forward AD}: exact, $n$ passes for $n$ inputs.
        \item \textbf{Reverse AD}: exact, 1 backward pass, all derivatives.
        \item \textbf{Regression gradient}: black-box, $m+1$ evals.
        \item \textbf{SPSA}: black-box, $2m$ evals, scalable.
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}[allowframebreaks]{Summary Bullet Points (Section 2.7)}
  \begin{itemize}
    \item Derivatives are useful in optimisation because they provide information
          about how to change a given point to \textbf{improve} the objective function.
    \item For multivariate functions, the \textbf{gradient}, \textbf{Hessian}, and
          \textbf{directional derivative} are the key derivative concepts for directing
          the search for an optimum.
    \item One approach to numerical differentiation includes \textbf{finite difference
          approximations}: forward, central, and backward methods.
    \item The \textbf{complex-step method} can eliminate subtractive cancellation error
          when taking small steps, resulting in high-quality gradient estimates.
    \item \textbf{Automatic differentiation}: forward and reverse accumulation on
          computational graphs give exact numerical derivatives at code level.
    \item The \textbf{regression gradient} uses linear regression to find the best
          estimate of the gradient with respect to a set of sampled perturbations
          and their function evaluations.
    \item \textbf{SPSA} allows estimation of gradients for very large objective
          functions using only 2 function evaluations per iteration.
  \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Selected Exercises}
  \textbf{Exercise 2.1.} Approximate the Hessian from the gradient using forward differences.
  \begin{block}{Solution}
    \[
      H_{ij} = \frac{\partial^2 f}{\partial x_i\,\partial x_j}
             \approx \frac{[\grad f(\x+h\mathbf{e}_j)]_i - [\grad f(\x)]_i}{h}
    \]
    Repeat for each column $j=1,\ldots,n$.
  \end{block}

  \textbf{Exercise 2.2.} What is a drawback of central differences?
  \begin{block}{Solution}
    Central difference requires \textbf{two function evaluations} per dimension
    (versus one for forward/backward), doubling cost.
  \end{block}

  \textbf{Exercise 2.3.} Find the dominant term in $f'(x)=\frac{1}{x}+e^x-\frac{1}{x^2}$
  for $x$ close to zero.
  \begin{block}{Solution}
    For $x<1$: $\frac{1}{x}>1$, so $\frac{1}{x^2}>\frac{1}{x}\gg e^x$.
    The term $-\frac{1}{x^2}$ dominates.
  \end{block}

  \framebreak

  \textbf{Exercise 2.4.} If $f(3+ih)=2+4ih$, what is $f'(3)$?
  \begin{block}{Solution}
    Complex-step: $f'(3)\approx\operatorname{Im}(f(3+ih))/h = 4h/h = 4$.
  \end{block}

  \textbf{Exercise 2.6.} Combine forward and backward differences to estimate $f''(x)$.
  \begin{block}{Solution}
    \[
      f''(x) \approx \frac{f'(x+h/2)-f'(x-h/2)}{h}
              \approx \frac{f(x+h)-2f(x)+f(x-h)}{h^2}
    \]
    Three function evaluations; $O(h^2)$ accuracy.
  \end{block}

  \textbf{Exercise 2.5.} Computational graph for $f(x,y)=\sin(x+y^2)$; forward accumulation
  at $(x,y)=(1,1)$ for $\partial f/\partial y$:
  \begin{center}
    \small
    $y\xrightarrow{(\cdot)^2} c_1=1$;\quad
    $x\to c_2=x=1$;\quad
    $c_1+c_2 \to c_3=2$;\quad
    $\sin(c_3)\to c_4=\sin 2$\\
    $\dot{y}=1$;\quad $\dot{c}_1=2y\cdot\dot{y}=2$;\quad $\dot{c}_2=0$;\quad
    $\dot{c}_3=2$;\quad $\dot{c}_4=2\cos 2$
  \end{center}
  $\Rightarrow$ $\partial f/\partial y = 2\cos 2 \approx -0.832$.
\end{frame}
