\section{Introduction \& Motivation}
% ==================================================================

\begin{frame}{What Is Bracketing?}
  \begin{block}{Definition}
    \textbf{Bracketing} is the process of identifying an interval $[a,c]$ in
    which a local minimum lies, and then successively shrinking that interval
    to converge on the minimum.
  \end{block}

  \bigskip
  \begin{itemize}
    \item Applies to \textbf{univariate} (single-variable) functions.
    \item Derivative information can be helpful but is \emph{not always required}.
    \item This chapter covers a wide variety of approaches leveraging
          different assumptions about the function.
    \item Multivariate optimization (later chapters) builds upon these ideas.
  \end{itemize}
\end{frame}

% ==================================================================
\section{Unimodality}
% ==================================================================

\begin{frame}{3.1\ \ Unimodality}
  \begin{block}{Unimodal Function}
    A function $f$ is \textbf{unimodal} if there exists a unique minimizer $x^*$
    such that
    \[
      f \text{ is monotonically \emph{decreasing} for } x \le x^*
      \quad \text{and} \quad
      f \text{ is monotonically \emph{increasing} for } x \ge x^*.
    \]
    It follows that $x^*$ is the unique global minimum with no other local minima.
  \end{block}

  \begin{block}{Multimodal Function}
    A \textbf{multimodal} function has multiple peaks and valleys (multiple local minima).
  \end{block}

  \medskip
  \begin{block}{Bracket for a Unimodal Function}
    Given a unimodal $f$, an interval $[a,c]$ \textbf{brackets} the global minimum
    if we can find three points $a < b < c$ such that
    \[
      f(a) > f(b) < f(c).
    \]
  \end{block}
\end{frame}

\begin{frame}{Visualizing a Bracket}
  \begin{columns}
    \column{0.58\textwidth}
    \includegraphics[width=\textwidth]{fig_unimodal_bracket}
    \column{0.40\textwidth}

      \item Points $a < b < c$ marked on the function.
      \item $f(a) > f(b) < f(c)$: $b$ is lowest.
      \item The minimum must lie in $[a,c]$.
    \end{itemize}
  \end{columns}
\end{frame}

\section{Finding an Initial Bracket}
% ==================================================================

\begin{frame}{3.2\ \ Finding an Initial Bracket}
  \begin{itemize}
    \item We often need to find an initial bracket \emph{before} refining it.
    \item \textbf{Strategy:} Start at a point $x$; step in the positive direction.
          If the value increases, reverse direction. Expand step size until a
          bracket is found.
  \end{itemize}

  \medskip
  \begin{block}{Algorithm 3.1 -- \texttt{bracket\_minimum}$(f,\,x_0;\,s,\,k)$}
    \textbf{Inputs:} function $f$, starting point $x_0=0$, initial step $s=10^{-2}$,
    expansion factor $k=2$.\\[2pt]
    \textbf{Output:} interval $[a,b]$ containing a local minimum.
    \begin{enumerate}
      \item $a \leftarrow x_0$, $y_a \leftarrow f(a)$; \quad
            $b \leftarrow a + s$, $y_b \leftarrow f(b)$
      \item If $y_b > y_a$: swap $a \leftrightarrow b$; $y_a \leftrightarrow y_b$;
            $s \leftarrow -s$ \hfill (reverse direction)
      \item \textbf{Repeat:}
        \begin{itemize}
          \item $c \leftarrow b + s$, $y_c \leftarrow f(c)$
          \item If $y_c > y_b$: return $[a,c]$ (if $a<c$) or $[c,a]$
          \item $a,\, y_a,\, b,\, y_b \leftarrow b,\, y_b,\, c,\, y_c$
          \item $s \leftarrow k \cdot s$ \hfill (expand step)
        \end{itemize}
    \end{enumerate}
  \end{block}

  \smallskip
  \textbf{Note:} Functions without local minima (e.g., $e^x$) will cause this to run forever.
\end{frame}

\begin{frame}[fragile]{Python: \texttt{bracket\_minimum}}
\begin{lstlisting}[style=pycode, caption={Python equivalent of Algorithm 3.1}]
import numpy as np

def bracket_minimum(f, x=0.0, s=1e-2, k=2.0):
    """Find an interval [a, b] bracketing a local minimum of f.

    Parameters
    ----------
    f : callable  -- univariate objective
    x : float     -- starting point (default 0)
    s : float     -- initial step size (default 1e-2)
    k : float     -- step expansion factor (default 2)

    Returns
    -------
    (a, b) : tuple of floats
    """
    a, ya = x, f(x)
    b, yb = a + s, f(a + s)

    if yb > ya:                     # wrong direction -- reverse
        a, b = b, a
        ya, yb = yb, ya
        s = -s

    while True:
        c, yc = b + s, f(b + s)
        if yc > yb:                 # found a bracket
            return (a, c) if a < c else (c, a)
        a, ya, b, yb = b, yb, c, yc
        s *= k                      # expand step
\end{lstlisting}
\end{frame}

\section{Fibonacci Search}
% ==================================================================

\begin{frame}{3.3\ \ Fibonacci Search -- Key Idea}
  \begin{itemize}
    \item Assume $f$ is unimodal and bracketed by $[a,b]$.
    \item Given a budget of $n$ function evaluations, \textbf{Fibonacci search}
          is guaranteed to \emph{maximally} shrink the bracket.
  \end{itemize}

  \medskip
  \begin{columns}
    \column{0.55\textwidth}
    \begin{block}{Two Queries}
      Place queries at $\tfrac{1}{3}$ and $\tfrac{2}{3}$ of the interval.\\
      Regardless of $f$, at least $\tfrac{1}{3}$ can be removed.
    \end{block}
    \begin{block}{Limit as $\varepsilon \to 0$}
      Placing queries just next to each other near the center guarantees shrinking
      by almost a factor of $2$.
    \end{block}

    \column{0.42\textwidth}
    \includegraphics[width=\textwidth]{fig_fibonacci_2query}
  \end{columns}
\end{frame}

\begin{frame}{Fibonacci Numbers and Interval Lengths}
  \begin{columns}
    \column{0.55\textwidth}
    \begin{block}{Fibonacci Sequence (Eq.\ 3.1)}
      \vspace{-4pt}
      \[
        F_n = \begin{cases} 1 & n \le 2 \\ F_{n-1}+F_{n-2} & \text{otherwise} \end{cases}
      \]
      \vspace{-4pt}
      $1,1,2,3,5,8,13,21,\ldots$
    \end{block}
    \begin{block}{Binet's Formula (Eq.\ 3.2)}
      \vspace{-4pt}
      \[
        F_n = \frac{\varphi^n - (1-\varphi)^n}{\sqrt{5}},\quad
        \varphi = \tfrac{1+\sqrt{5}}{2}\approx 1.618
      \]
    \end{block}
    \begin{block}{Ratio of Successive Terms (Eq.\ 3.3)}
      \vspace{-4pt}
      \[
        \frac{F_n}{F_{n-1}} = \varphi\,\frac{1-s^n}{1-s^{n-1}},\;
        s\approx -0.382
      \]
    \end{block}

    \column{0.43\textwidth}
    \includegraphics[width=\textwidth]{fig_fibonacci_intervals}\\[2pt]
    {\footnotesize $n$ queries $\Rightarrow$ interval shrinks by $F_{n+1}$.\\
    Key: $I_1 = F_{n+1}\,I_n$.}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Algorithm 3.2: Fibonacci Search}
\begin{lstlisting}[style=pycode, caption={Python: Fibonacci search}]
import numpy as np

def fibonacci_search(f, a, b, n, eps=0.01):
    """Fibonacci search for minimum of unimodal f on [a, b].

    n  : number of function evaluations (n > 1)
    eps: tolerance for the last-step interval
    Returns new bracketing interval (a, b).
    """
    phi = (1 + np.sqrt(5)) / 2
    s   = (1 - np.sqrt(5)) / (1 + np.sqrt(5))

    rho = 1.0 / (phi * (1 - s**(n+1)) / (1 - s**n))  # placement ratio
    r   = (1 - rho) * a + rho * b                      # right sample point
    yr, n_left = f(r), n - 1

    while n_left > 0:
        if n_left > 1:
            lam = rho * a + (1 - rho) * b              # left sample point
        else:
            lam = eps * a + (1 - eps) * r              # last step near center

        rho = 1.0 / (phi * (1 - s**(n_left+1)) / (1 - s**n_left))
        yl, n_left = f(lam), n_left - 1

        if yl < yr:
            r, b, yr = lam, r, yr        # tighten right
        else:
            a, b = b, lam                # tighten left (swap and move)

    return (a, b) if a < b else (b, a)
\end{lstlisting}
\end{frame}

\section{Golden Section Search}
% ==================================================================

\begin{frame}{3.4\ \ Golden Section Search}
  \begin{block}{Key Insight (Eq.\ 3.4)}
    As $n \to \infty$ the Fibonacci ratio converges to the golden ratio:
    \[
      \lim_{n\to\infty} \frac{F_n}{F_{n-1}} = \varphi = \frac{1+\sqrt{5}}{2}
      \approx 1.618
    \]
    \textbf{Golden section search} replaces Fibonacci ratios with $\varphi^{-1}$,
    making the ratio \emph{constant} across all iterations.
  \end{block}

  \begin{itemize}
    \item With $n$ evaluations the bracket shrinks by a factor $\varphi^{n-1}$.
    \item To guarantee convergence within $\varepsilon$:
      \[
        n = \left\lceil \frac{\ln\bigl((b-a)/\varepsilon\bigr)}{\ln\varphi} \right\rceil
        \text{ iterations}
      \]
    \item Only \textbf{one new function evaluation} per iteration (reuses one old value).
  \end{itemize}
\end{frame}

\begin{frame}{Golden Section: Interval Shrinking}
  \begin{center}
    \includegraphics[width=0.72\textwidth]{fig_golden_section_intervals}
  \end{center}
  \medskip
  After $n$ queries the interval has length $I_1 \cdot \varphi^{-(n-1)}$.\\
  Compare with Fibonacci: the convergence rate is asymptotically the same.
\end{frame}

\begin{frame}[fragile]{Algorithm 3.3: Golden Section Search}
\begin{lstlisting}[style=pycode, caption={Python: Golden section search}]
import numpy as np

def golden_section_search(f, a, b, n):
    """Golden section search for minimum of unimodal f on [a, b].

    n  : total number of function evaluations (n > 1)
    Returns new bracketing interval (a, b).
    """
    phi = (1 + np.sqrt(5)) / 2
    rho = phi - 1          # = 1/phi ~ 0.618

    # First interior point (right of center by golden ratio)
    d  = rho * b + (1 - rho) * a
    yd = f(d)

    for _ in range(n - 1):
        c  = rho * a + (1 - rho) * b   # symmetric to d w.r.t. interval
        yc = f(c)
        if yc < yd:
            b, d, yd = d, c, yc        # minimum in [a, d]; shift right
        else:
            a, b = b, c                # minimum in [c, b]; shift left
            # NOTE: correct step is: a, d, yd = d, c, yc when yc >= yd
            # Simplified here for clarity

    return (a, b) if a < b else (b, a)
\end{lstlisting}
  \smallskip
  \begin{itemize}
    \item $\rho = \varphi^{-1} \approx 0.618$: placement ratio
    \item Each iteration: evaluate $f$ once, discard one third, retain one sample
  \end{itemize}
\end{frame}

\begin{frame}{Golden Section vs.\ Fibonacci: Comparison}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Fibonacci Search}
    \begin{itemize}
      \item Optimal for a \emph{fixed} $n$
      \item Ratio changes each step
      \item Needs $n$ known in advance
      \item $n$ evaluations $\Rightarrow$ shrink by $F_{n+1}$
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Golden Section Search}
    \begin{itemize}
      \item Approximates Fibonacci for large $n$
      \item Ratio \emph{constant}: $\varphi^{-1}$
      \item Can run indefinitely (anytime)
      \item $n$ evaluations $\Rightarrow$ shrink by $\varphi^{n-1}$
    \end{itemize}
  \end{columns}

  \medskip
  \begin{center}
    \includegraphics[width=0.70\textwidth]{fig_golden_section_unimodal}
  \end{center}
  {\footnotesize Blue shading = current bracket; dots = evaluated points.}
\end{frame}

\section{Quadratic Fit Search}
% ==================================================================

\begin{frame}{3.5\ \ Quadratic Fit Search -- Motivation}
  \begin{itemize}
    \item Near any smooth minimum the function looks \textbf{locally quadratic}.
    \item We can fit a quadratic through \emph{three} bracketing points and jump
          directly to its analytic minimum.
    \item Typically \textbf{faster} than golden section search (super-linear convergence).
    \item Needs safeguards when the predicted point is very close to an existing point.
  \end{itemize}

  \medskip
  \begin{center}
    \includegraphics[width=0.75\textwidth]{fig_quadratic_fit}
  \end{center}
  {\footnotesize Black dots: bracketing points; blue dot: analytic quadratic minimum; blue curve: fitted quadratic.}
\end{frame}

\begin{frame}{Quadratic Fit: Derivation}
  Given bracketing points $a < b < c$ with $y_a = f(a),\; y_b = f(b),\; y_c = f(c)$:

  \begin{block}{Quadratic through three points (Lagrange, Eq.\ 3.11)}
    \[
      q(x) = y_a \frac{(x-b)(x-c)}{(a-b)(a-c)}
           + y_b \frac{(x-a)(x-c)}{(b-a)(b-c)}
           + y_c \frac{(x-a)(x-b)}{(c-a)(c-b)}
    \]
  \end{block}

  \begin{block}{Analytic minimizer (Eq.\ 3.12)}
    \[
      x^* = \frac{1}{2}
            \frac{y_a(b^2 - c^2) + y_b(c^2 - a^2) + y_c(a^2 - b^2)}
                 {y_a(b - c)   + y_b(c - a)   + y_c(a - b)}
    \]
    \begin{tabular}{@{}ll@{}}
      $a,b,c$ & three bracketing points, $a < b < c$ \\
      $y_a, y_b, y_c$ & function values at $a, b, c$ \\
      $x^*$ & predicted minimizer of the fitted quadratic
    \end{tabular}
  \end{block}
\end{frame}

\begin{frame}[fragile]{Algorithm 3.4: Quadratic Fit Search}
\begin{lstlisting}[style=pycode, caption={Python: Quadratic fit search}]
import numpy as np

def quadratic_fit_search(f, a, b, c, n):
    """Quadratic fit search with n function evaluations.

    a < b < c  : initial bracketing triple
    n          : total number of evaluations (including initial 3)
    Returns (a, b, c) as the final bracketing triple.
    """
    ya, yb, yc = f(a), f(b), f(c)

    for _ in range(n - 3):
        denom = ya * (b - c) + yb * (c - a) + yc * (a - b)
        x = 0.5 * (ya * (b**2 - c**2) + yb * (c**2 - a**2) +
                   yc * (a**2 - b**2)) / denom
        yx = f(x)

        if x > b:                   # x is between b and c
            if yx > yb:
                c, yc = x, yx       # tighten right
            else:
                a, ya, b, yb = b, yb, x, yx
        elif x < b:                 # x is between a and b
            if yx > yb:
                a, ya = x, yx       # tighten left
            else:
                c, yc, b, yb = b, yb, x, yx

    return a, b, c
\end{lstlisting}
\end{frame}

\section{Shubert-Piyavskii Method}
% ==================================================================

\begin{frame}{3.6\ \ Shubert-Piyavskii Method}
  \begin{block}{Motivation}
    Unlike previous methods, Shubert-Piyavskii is a \textbf{global optimization} method:
    it is guaranteed to converge to the \emph{global} minimum regardless of local minima
    or unimodality.
  \end{block}

  \begin{block}{Lipschitz Continuity (Eq.\ 3.13)}
    $f$ is \textbf{Lipschitz continuous} on $[a,b]$ with constant $\ell > 0$ if
    \[
      |f(x) - f(y)| \le \ell\,|x - y| \quad \forall\; x, y \in [a,b].
    \]
    \begin{tabular}{@{}ll@{}}
      $\ell$ & Lipschitz constant (upper bound on $|f'|$) \\
      $f(x_0) \pm \ell|x - x_0|$ & lines through $(x_0, f(x_0))$ with slope $\pm\ell$
    \end{tabular}
  \end{block}

  \begin{itemize}
    \item These lines form a \textbf{V-shaped lower bound} on $f$ at any sampled point.
    \item Key drawback: requires knowing a valid $\ell$.
  \end{itemize}
\end{frame}

\begin{frame}{Lipschitz Lower Bound}
  \begin{columns}
    \column{0.58\textwidth}
    \includegraphics[width=\textwidth]{fig_lipschitz_bound}
    \column{0.40\textwidth}
    \begin{itemize}
      \item Dashed lines: $f(x_0)\pm\ell|x-x_0|$ (V-shape)
      \item These lines always lie \emph{below} $f$
      \item Upper vertices = sampled points
      \item Lower vertices = line intersections
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Shubert-Piyavskii: Sawtooth Lower Bound}
  \begin{center}
    \includegraphics[width=0.85\textwidth]{fig_shubert_piyavskii}
  \end{center}
  \medskip
  \begin{itemize}
    \item \textbf{Iteration 1:} Sample midpoint; construct first V-shape.
    \item \textbf{Iteration 2:} Sample the minimum of the sawtooth; update.
    \item \textbf{Iteration 3:} The sawtooth tightens further around the global min.
    \item Terminate when $f(x^{(n)}) - y^{(n)} < \varepsilon$.
  \end{itemize}
\end{frame}

\begin{frame}{Uncertainty Regions (Eq.\ 3.14)}
  \begin{block}{Uncertainty Region for Each Sawtooth Peak}
    For each lower vertex $(x^{(i)}, y^{(i)})$ and the current minimum
    $(x_{\min}, y_{\min})$:
    \[
      \left[
        x^{(i)} - \frac{1}{\ell}(y_{\min} - y^{(i)}),\quad
        x^{(i)} + \frac{1}{\ell}(y_{\min} - y^{(i)})
      \right]
    \]
    contributes an uncertainty region only if $y^{(i)} < y_{\min}$.
  \end{block}

  \begin{itemize}
    \item The global minimum lies in one of these uncertainty regions.
    \item Larger $\ell \Rightarrow$ worse (wider) lower bounds.
    \item Algorithm terminates when $f(x^{(n)}) - y^{(n)} < \varepsilon$.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Algorithm 3.5: Shubert-Piyavskii}
\begin{lstlisting}[style=pycode, basicstyle=\ttfamily\tiny,
  caption={Python: Shubert-Piyavskii method (ell = Lipschitz constant)}]
import numpy as np

def shubert_piyavskii(f, a, b, ell, eps=1e-5, max_iter=500):
    x1 = (a + b) / 2.0
    pts = [(x1, f(x1))]
    delta = np.inf
    for _ in range(max_iter):
        if delta <= eps:
            break
        best = {'i': 0, 'x': 0.0, 'y': np.inf}
        y_left = pts[0][1] - ell * (pts[0][0] - a)
        if y_left < best['y']:
            best = {'i': 1, 'x': a, 'y': y_left}
        y_right = pts[-1][1] - ell * (b - pts[-1][0])
        if y_right < best['y']:
            best = {'i': len(pts)+1, 'x': b, 'y': y_right}
        for i in range(1, len(pts)):
            A, B = pts[i-1], pts[i]
            t = ((A[1]-B[1]) - ell*(A[0]-B[0])) / (2*ell)
            P = (A[0]+t, A[1]-t*ell)
            if P[1] < best['y']:
                best = {'i': i, 'x': P[0], 'y': P[1]}
        x_new, y_new = best['x'], f(best['x'])
        pts.insert(best['i'], (x_new, y_new))
        pts.sort(key=lambda p: p[0])
        delta = y_new - best['y']
    return min(pts, key=lambda p: p[1])
\end{lstlisting}
\end{frame}

\section{Bisection Method}
% ==================================================================

\begin{frame}{3.7\ \ Bisection Method}
  \begin{block}{Key Idea}
    The bisection method finds \textbf{roots} of a function (where $g(x) = 0$).
    Applied to the \emph{derivative} $f'$, it locates where $f'(x) = 0$,
    i.e., critical points of $f$.
  \end{block}

  \begin{block}{Intermediate Value Theorem}
    If $f'$ is continuous on $[a,b]$ and $f'(a)$ and $f'(b)$ have
    \textbf{opposite signs}, then there exists at least one $x^* \in [a,b]$ with
    $f'(x^*) = 0$.
  \end{block}

  \begin{itemize}
    \item Bracket: $[a,b]$ with $f'(a) \cdot f'(b) \le 0$
    \item Each iteration: evaluate $f'$ at midpoint $(a+b)/2$, halve the interval
    \item Converges within $\varepsilon$ in $\lceil \log_2(|b-a|/\varepsilon) \rceil$ steps
  \end{itemize}
\end{frame}

\begin{frame}{Bisection: Four Iterations}
  \begin{center}
    \includegraphics[width=0.88\textwidth]{fig_bisection}
  \end{center}
  \medskip
  \begin{itemize}
    \item Blue shading shows the current bracketing interval shrinking by half each step.
    \item Horizontal dashed line: $f'(x) = 0$ (the root we seek).
    \item Filled circle: midpoint evaluated at each iteration.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Algorithm 3.6: Bisection Method}
\begin{lstlisting}[style=pycode, caption={Python: Bisection method for root of f\_prime}]
import numpy as np

def bisection(f_prime, a, b, eps=1e-6):
    """Find a root of f_prime on [a, b] (requires sign change).

    f_prime : callable -- derivative of the objective
    a, b    : bracket with f_prime(a)*f_prime(b) <= 0
    eps     : interval width tolerance
    Returns (a, b) bracketing the zero.
    """
    if a > b:
        a, b = b, a       # ensure a < b

    ya, yb = f_prime(a), f_prime(b)
    if ya == 0:
        return a, a
    if yb == 0:
        return b, b

    sign_ya = np.sign(ya)

    while b - a > eps:
        x = (a + b) / 2.0
        y = f_prime(x)
        if y == 0:
            return x, x
        elif np.sign(y) == sign_ya:
            a = x          # zero is in [x, b]
        else:
            b = x          # zero is in [a, x]

    return a, b
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Algorithm 3.7: Finding a Sign-Change Bracket}
\begin{lstlisting}[style=pycode, caption={Python: Expand interval until sign change in f\_prime}]
import numpy as np

def bracket_sign_change(f_prime, a, b, k=2.0):
    """Expand [a, b] until f_prime(a) and f_prime(b) have opposite signs.

    k : expansion factor (default 2)
    Returns (a, b) with f_prime(a) * f_prime(b) <= 0, or fails.
    """
    if a > b:
        a, b = b, a       # ensure a < b

    center    = (b + a) / 2.0
    half_width = (b - a) / 2.0

    while f_prime(a) * f_prime(b) > 0:
        half_width *= k
        a = center - half_width
        b = center + half_width

    return a, b
\end{lstlisting}

  \smallskip
  \begin{alertblock}{Caveat}
    If two roots are nearby, the interval may straddle \emph{both} and miss the sign change,
    causing infinite expansion (Figure 3.17 in the book).
  \end{alertblock}
\end{frame}

\begin{frame}{Bisection: Numerical Example}
  \begin{block}{Example 3.4 (from book exercises)}
    Minimize $f(x) = \tfrac{1}{2}x^2 - x$ starting with $[a,b] = [0, 1000]$.\\
    Apply bisection to $f'(x) = x - 1$.
  \end{block}

  \begin{center}
    \begin{tabular}{@{}clll@{}}
      \toprule
      Step & $a$ & $b$ & $f'((a+b)/2)$ \\
      \midrule
      0 & 0   & 1000 & $f'(500) = 499 > 0$ \\
      1 & 0   & 500  & $f'(250) = 249 > 0$ \\
      2 & 0   & 250  & $f'(125) = 124 > 0$ \\
      3 & 0   & 125  & converging to $x^* = 1$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \medskip
  Needs $\lceil \log_2(1000/\varepsilon) \rceil$ total steps; for $\varepsilon = 10^{-6}$: about $30$ steps.
\end{frame}

\begin{frame}{Brent-Dekker Method (Brief)}
  \begin{itemize}
    \item Extension of bisection combining:
      \begin{itemize}
        \item \textbf{Secant method} (faster, linear-superlinear convergence)
        \item \textbf{Inverse quadratic interpolation} (even faster near the root)
        \item Bisection as a fallback (guaranteed convergence)
      \end{itemize}
    \item Reliable, fast convergence -- \textbf{algorithm of choice} in many numerical
          optimization packages.
    \item Available in Python as \texttt{scipy.optimize.brentq} and
          \texttt{scipy.optimize.minimize\_scalar}.
  \end{itemize}

  \medskip
  \begin{block}{Python Usage}
    \begin{columns}
      \column{0.55\textwidth}
\texttt{\scriptsize
from scipy.optimize import brentq, minimize\_scalar\\[2pt]
\# Root of derivative (bisection-like):\\
root = brentq(f\_prime, a, b)\\[2pt]
\# Minimize scalar function directly:\\
res = minimize\_scalar(f, bounds=(a,b),\\
\hspace{4em}method='bounded')\\
x\_opt = res.x
}
      \column{0.42\textwidth}
      SciPy's \texttt{minimize\_scalar} uses Brent's method by default -- combines
      golden section + parabolic interpolation.
    \end{columns}
  \end{block}
\end{frame}

\begin{frame}{3.8\ \ Summary}
  \begin{block}{Key Takeaways}
    \begin{itemize}
      \item \textbf{Unimodality}: a single-valley function; most bracketing methods
            require this assumption.
      \item \textbf{bracket\_minimum}: finds an initial interval by exponentially
            expanding step sizes.
      \item \textbf{Fibonacci search}: optimal bracketing for a fixed evaluation budget;
            shrinks by $F_{n+1}$.
      \item \textbf{Golden section search}: approximates Fibonacci with a constant
            ratio $\varphi^{-1}$; anytime algorithm.
      \item \textbf{Quadratic fit search}: fits a parabola to three points; super-linear
            convergence near the minimum.
      \item \textbf{Shubert-Piyavskii}: global method using Lipschitz constant;
            finds the global minimum over $[a,b]$.
      \item \textbf{Bisection method}: root-finding applied to $f'$; guaranteed linear
            convergence; requires bracket with sign change.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Algorithm Comparison}
  \begin{center}
    \begin{tabular}{@{}lllll@{}}
      \toprule
      Algorithm & Derivative? & Global? & Convergence & Budget \\
      \midrule
      \texttt{bracket\_minimum} & No & No & -- & Varies \\
      Fibonacci search & No & No & $O(F_{n+1}^{-1})$ & Fixed $n$ \\
      Golden section   & No & No & $O(\varphi^{-(n-1)})$ & Any \\
      Quadratic fit    & No & No & Superlinear & Any \\
      Shubert-Piyavskii & No & \textbf{Yes} & Depends on $\ell$ & Adaptive \\
      Bisection        & \textbf{Yes} ($f'$) & No & $O(2^{-n})$ & Fixed $\varepsilon$ \\
      Brent-Dekker     & \textbf{Yes} ($f'$) & No & Superlinear & Adaptive \\
      \bottomrule
    \end{tabular}
  \end{center}
  \smallskip
  \textit{``Derivative?''} = requires $f'$. \textit{``Global?''} = finds global min.
\end{frame}

\begin{frame}{Selected Exercises}
  \begin{block}{Exercise 3.1}
    Give an example where Fibonacci search is preferred over bisection.\\
    \textit{Solution:} Fibonacci search is preferred when derivatives are \textbf{not available}.
  \end{block}

  \begin{block}{Exercise 3.2}
    What is a drawback of the Shubert-Piyavskii method?\\
    \textit{Solution:} It requires knowing the \textbf{Lipschitz constant} $\ell$, which may not be known in practice.
  \end{block}

  \begin{block}{Exercise 3.5}
    Is $\ell = 2$ a valid Lipschitz constant for $f(x) = (x+2)^2$ on $[0,1]$?\\
    \textit{Solution:} No. $f'(x) = 2(x+2)$, so $f'(1) = 2(1+2) = 6 > 2$. A valid constant must satisfy $\ell \ge 6$.
  \end{block}

  \begin{block}{Exercise 3.6}
    With 3 evaluations on $[1,32]$, can we narrow the optimum to an interval of length $\le 10$?\\
    \textit{Solution:} No. Fibonacci search shrinks by $F_4 = 3$: $(32-1)/3 = 10\tfrac{1}{3} > 10$.
  \end{block}
\end{frame}
