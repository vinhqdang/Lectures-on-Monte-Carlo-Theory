\section{Introduction}
% ============================================================================

\begin{frame}{What is Multidisciplinary Design Optimization?}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Multidisciplinary Design Optimization (MDO)} addresses problems that:
    \begin{itemize}
      \item Span \emph{multiple interacting disciplines} (structures, aerodynamics, controls, economics, \ldots)
      \item Cannot be solved optimally by optimizing each discipline separately
      \item Require handling \emph{interdependencies} among disciplines
    \end{itemize}

    \medskip
    \textbf{Core challenges:}
    \begin{itemize}
      \item Circular dependencies between analyses
      \item Heterogeneous solvers owned by different teams
      \item Large numbers of variables and constraints
      \item Expensive function evaluations (simulations)
    \end{itemize}

    \column{0.42\textwidth}
    \begin{block}{General MDO Formulation}
      \[
        \min_{\mathbf{x}}\; f(\mathcal{A}) \quad \text{s.t.}\;\; \mathcal{A} \in \mathcal{X}
      \]
      \[
        \mathcal{F}_i(\mathcal{A}) = \mathbf{0} \quad \forall\, i = 1,\ldots,m
      \]
      where $\mathcal{A}$ is the \textbf{assignment} of all design and response variables.
    \end{block}
  \end{columns}

  \begin{center}
    \includegraphics[width=0.62\textwidth]{fig_mdo_structure.pdf}
  \end{center}
\end{frame}

\section{Disciplinary Analyses}
% ============================================================================

\begin{frame}{Disciplinary Analyses}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \begin{block}{Key Definitions}
      \begin{itemize}
        \item \textbf{Assignment} $\mathcal{A}$: a mapping from variable names to their values.\\[2pt]
        Contains design variables $\mathbf{x}$ and all response variables $\mathbf{y}^{(1)},\ldots,\mathbf{y}^{(m)}$.
        \medskip
        \item \textbf{Disciplinary analysis} $\mathcal{F}_i$: takes an assignment, runs a discipline-specific computation (e.g.\ FEA, CFD, simulation), and \emph{updates} the assignment with new response values.
        \medskip
        \item \textbf{Response variables} $\mathbf{y}^{(i)}$: outputs of discipline $i$ that may feed into other disciplines.
      \end{itemize}
    \end{block}

    \column{0.45\textwidth}
    \begin{block}{Formal Setup}
      \[
        \mathbf{y}^{(i)} \leftarrow \mathcal{F}_i\!\left(\mathbf{x},\;\mathbf{y}^{(-i)}\right)
      \]
      where $\mathbf{y}^{(-i)}$ denotes responses from all other disciplines.

      \bigskip
      With $n$ design variables and $m$ disciplines each producing $n_i$ responses:
      \[
        |\mathcal{A}| = n + \sum_{i=1}^{m} n_i
      \]
    \end{block}

    \medskip
    \textbf{Example 24.1} (3-discipline system with $x$ as single design variable):
    \[
      y^{(1)} = y^{(2)} - x, \quad
      y^{(2)} = \sin\!\bigl(y^{(1)}+y^{(3)}\bigr), \quad
      y^{(3)} = \cos\!\bigl(x+y^{(1)}+y^{(2)}\bigr)
    \]
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Python: Disciplinary Analyses (Example 24.1)}
  \begin{lstlisting}[style=python]
import numpy as np

# Each disciplinary analysis takes and updates the assignment dict
def F1(A):
    A['y1'] = A['y2'] - A['x']
    return A

def F2(A):
    A['y2'] = np.sin(A['y1'] + A['y3'])
    return A

def F3(A):
    A['y3'] = np.cos(A['x'] + A['y2'] + A['y1'])
    return A

# Initial assignment for x=1
A = {'x': 1.0, 'y1': 1.0, 'y2': 1.0, 'y3': 1.0}

# Run one pass with ordering F1, F2, F3
A = F1(A); A = F2(A); A = F3(A)
print(A)
  \end{lstlisting}

  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{alertblock}{Key Point}
      Each function modifies the \emph{shared assignment} dictionary.
      The ordering of disciplinary analyses affects convergence.
    \end{alertblock}
    \column{0.48\textwidth}
    \begin{block}{Variable Interdependencies}
      \[
        F_1 \leftarrow y^{(2)},\; F_2 \leftarrow y^{(1)}, y^{(3)},\; F_3 \leftarrow y^{(1)}, y^{(2)}
      \]
      Circular: $F_1 \to F_2 \to F_3 \to F_1$
    \end{block}
  \end{columns}
\end{frame}

\section{Interdisciplinary Compatibility}
% ============================================================================

\begin{frame}{Interdisciplinary Compatibility}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \begin{block}{Feasibility Condition}
      An assignment $\mathcal{A}$ is \textbf{interdisciplinarily compatible} if, for all disciplines $i$:
      \[
        \mathcal{F}_i(\mathcal{A}) = \mathcal{A}
      \]
      i.e., running any disciplinary analysis \emph{does not change} the assignment.
    \end{block}

    \medskip
    \begin{block}{Dependency Graph}
      \begin{itemize}
        \item Nodes = disciplines (or variables)
        \item Directed edge $i \to j$: discipline $j$ requires output of $i$
        \item \textbf{Acyclic} graph: analyses can be sequenced
        \item \textbf{Cyclic} graph: requires iterative solution (\emph{MDA})
      \end{itemize}
    \end{block}

    \column{0.47\textwidth}
    \begin{center}
      \includegraphics[width=0.9\textwidth]{fig_dependency_graph.pdf}\\
      \small Ride-sharing dependency graph with cycles (vehicle $\leftrightarrow$ sensor, routing $\leftrightarrow$ demand)
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}{Gauss-Seidel MDA Algorithm}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Algorithm 24.1: Gauss-Seidel MDA}
      \textbf{Input:} List of disciplinary analyses $[F_1,\ldots,F_m]$, initial assignment $\mathcal{A}$, $k_{\max}$, tolerance $\varepsilon$\\[4pt]
      \textbf{Repeat} until convergence or $k > k_{\max}$:
      \begin{enumerate}
        \item $k \leftarrow k+1$
        \item $\mathcal{A}_{\text{old}} \leftarrow \text{copy}(\mathcal{A})$
        \item For each $F_i$: run $F_i(\mathcal{A})$ in order
        \item Check: $\|v(\mathcal{A}) - v(\mathcal{A}_{\text{old}})\|_\infty \leq \varepsilon$ for all $v$
      \end{enumerate}
      \textbf{Return} $(\mathcal{A},\;\text{converged})$
    \end{block}

    \begin{alertblock}{Ordering Matters!}
      A topological sort of the dependency graph (if acyclic) gives an optimal ordering. For cyclic graphs, ordering affects convergence.
    \end{alertblock}

    \column{0.49\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_gauss_seidel.pdf}\\
      \small Ordering $F_1,F_2,F_3$ converges; $F_1,F_3,F_2$ oscillates (Example 24.2, $x=1$)
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Python: Gauss-Seidel MDA}
  \begin{lstlisting}[style=python]
import copy, numpy as np

def gauss_seidel_mda(Fs, A, k_max=100, eps=1e-4):
    """
    Fs      : list of disciplinary analysis functions (each mutates A in-place)
    A       : assignment dict {'x': ..., 'y1': ..., ...}
    k_max   : maximum iterations
    eps     : relative tolerance
    Returns : (A, converged)
    """
    k, converged = 0, False
    while not converged and k <= k_max:
        k += 1
        A_old = copy.deepcopy(A)
        for F in Fs:
            F(A)                          # update assignment in place
        # Check convergence on all numeric values
        converged = all(
            abs(A[v] - A_old[v]) / (abs(A_old[v]) + 1e-12) <= eps
            for v in A if isinstance(A[v], (int, float, np.floating))
        )
    return A, converged

# Run Example 24.2: x=1, ordering F1->F2->F3
A = {'x': 1.0, 'y1': 1.0, 'y2': 1.0, 'y3': 1.0}
A_sol, ok = gauss_seidel_mda([F1, F2, F3], A)
print(f"Converged: {ok}, y1={A_sol['y1']:.4f}, y2={A_sol['y2']:.4f}, y3={A_sol['y3']:.4f}")
  \end{lstlisting}

  \begin{block}{Complexity}
    Each outer iteration costs $O(m \cdot C_{\text{analysis}})$ where $C_{\text{analysis}}$ is the cost of one disciplinary analysis call. Convergence rate depends on spectral radius of the Jacobian of the coupled system.
  \end{block}
\end{frame}

\begin{frame}{MDO Architectures Overview}
  The general MDO problem can be written as:
  \[
    \min_{\mathbf{x}} f(\mathcal{A}) \quad \text{s.t.} \quad
    \mathcal{F}_i(\mathcal{A}) = \mathbf{0},\; i=1,\ldots,m, \quad \mathcal{A} \in \mathcal{X}
  \]
  Different \textbf{architectures} differ in \emph{how} they handle the coupling:

  \end{center}

  \begin{columns}[T]

    \begin{itemize}
      \item MDF, IDF, SAND — one optimizer
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{Hierarchical / distributed:}
    \begin{itemize}
      \item Sequential, CO — multiple levels

\end{frame}

\section{Multidisciplinary Design Feasible (MDF)}
% ============================================================================

\begin{frame}{Multidisciplinary Design Feasible (MDF)}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{MDF Formulation}
      \[
        \min_{\mathbf{x}}\; f\!\bigl(\mathrm{MDA}(\mathbf{x})\bigr)
        \quad\text{s.t.}\quad \mathbf{x} \in \mathcal{X}
      \]
      Interdisciplinary compatibility is satisfied \emph{automatically} by running a full MDA at every optimizer call.
    \end{block}

    \medskip
    \textbf{Procedure:}
    \begin{enumerate}
      \item Optimizer proposes design $\mathbf{x}$
      \item Inner MDA loop (Gauss-Seidel) solves for compatible $\mathbf{y}$
      \item Evaluate objective $f$ on converged $\mathcal{A}$
      \item Optimizer updates $\mathbf{x}$; repeat
    \end{enumerate}

    \begin{block}{Properties}
      \begin{itemize}
        \item[\checkmark] Small variable space (only $\mathbf{x}$)
        \item[\checkmark] Always returns a feasible design
        \item[$\times$] MDA at every function evaluation (expensive)
        \item[$\times$] Gradient computation requires MDA differentiation
      \end{itemize}
    \end{block}

    \column{0.49\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_mdf_architecture.pdf}\\
      \small MDF architecture: single optimizer calls MDA as a black box
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Python: MDF Implementation}
  \begin{lstlisting}[style=python]
import copy, numpy as np
from scipy.optimize import minimize

def mda(x_val, Fs, y_init=None, k_max=200, eps=1e-6):
    """Run a full MDA for design variable x and return converged assignment."""
    A = {'x': x_val}
    A.update(y_init or {'y1': 0.0, 'y2': 0.0, 'y3': 0.0})
    A, converged = gauss_seidel_mda(Fs, A, k_max=k_max, eps=eps)
    if not converged:
        raise RuntimeError('MDA did not converge')
    return A

def objective_mdf(x_val):
    """Objective function under MDF: run MDA, then evaluate f."""
    A = mda(x_val[0], [F1, F2, F3])
    return A['y1']**2 + A['y2']**2 + A['y3']**2   # example objective

# MDF optimization (design variable x in [-5, 5])
result = minimize(objective_mdf, x0=[0.0],
                  bounds=[(-5.0, 5.0)],
                  method='L-BFGS-B',
                  options={'ftol': 1e-9})
print(f"x* = {result.x[0]:.4f},  f* = {result.fun:.4f}")
  \end{lstlisting}

  \begin{alertblock}{Cost}
    If the MDA requires $K$ Gauss-Seidel iterations and the optimizer makes $N_{\text{eval}}$ function evaluations, total cost is $O(K \cdot N_{\text{eval}} \cdot m \cdot C_{\text{analysis}})$.
  \end{alertblock}
\end{frame}

\begin{frame}{MDF Applied to Ride-Sharing (Example 24.4)}
  \begin{columns}[T]
    \column{0.50\textwidth}
    For the ride-sharing problem, the MDF formulation is:
    \[
      \min_{\mathbf{v},\mathbf{s},\mathbf{r},\mathbf{p}}\;
      f\!\bigl(\mathrm{MDA}(\mathbf{v},\mathbf{s},\mathbf{r},\mathbf{p})\bigr)
    \]
    \[
      \text{s.t.}\quad
      [\mathbf{v},\mathbf{s},\mathbf{r},\mathbf{p}] \in \mathcal{X}
    \]
    \[
      \mathrm{MDA}(\cdot)\;\text{converged}
    \]

    \medskip
    The optimizer \emph{only} controls global design variables. The MDA handles all interdisciplinary coupling internally.

    \medskip
    \begin{block}{Advantage}
      Conceptually simple; reduces MDO to a conventional single-discipline optimization problem.
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{MDF Optimization Problem (General)}
      \[
        \min_{\mathbf{x}}\; f(\mathbf{x}, \mathbf{y}^*)
      \]
      \[
        \text{s.t.}\quad \mathbf{x} \in \mathcal{X}, \quad
        \mathbf{y}^* = \mathrm{MDA}(\mathbf{x})
      \]
      where $\mathbf{y}^* = [\mathbf{y}^{(1)*},\ldots,\mathbf{y}^{(m)*}]$ is the fully-compatible response vector.
    \end{block}

    \begin{alertblock}{Limitation}
      Weight sharing across disciplines not exploited; MDA convergence check added as hard constraint.
    \end{alertblock}
  \end{columns}
\end{frame}

\section{Sequential Optimization and Reliability Assessment}
% ============================================================================

\begin{frame}{Sequential Optimization Architecture}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Idea}
      Decompose into \textbf{discipline-level subproblems}, each optimizing a local objective $f_i$ subject to discipline-specific constraints, one after another.
    \end{block}

    \medskip
    \textbf{Procedure:}
    \begin{enumerate}
      \item Top-level optimizer updates global $\mathbf{x}_g$
      \item Subproblem 1: $\min_{\mathbf{x}_\ell^{(1)}} f_1(\mathcal{A})$ s.t.\ $[\mathbf{x}^{(1)},\mathbf{y}^{(1)}]\in\mathcal{X}_1$; run Analysis 1
      \item Subproblem 2: $\min_{\mathbf{x}_\ell^{(2)}} f_2(\mathcal{A})$ s.t.\ $[\mathbf{x}^{(2)},\mathbf{y}^{(2)}]\in\mathcal{X}_2$; run Analysis 2
      \item $\vdots$
      \item Subproblem $m$: $\min_{\mathbf{x}_\ell^{(m)}} f_m(\mathcal{A})$ \ldots
      \item Return to step 1 until convergence
    \end{enumerate}

    \begin{block}{Variable Split}
      Each discipline has \emph{local} variables $\mathbf{x}_\ell^{(i)}$ (discipline-specific) and shares \emph{global} variables $\mathbf{x}_g$ (shared across disciplines).
    \end{block}

    \column{0.49\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_sequential_architecture.pdf}
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}{Sequential Optimization: Properties and Example 24.5}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \begin{block}{Properties}
      \begin{itemize}
        \item[\checkmark] Each subproblem uses domain-specific solvers
        \item[\checkmark] Modular decomposition
        \item[\checkmark] Can exploit parallel execution \emph{within} each subproblem
        \item[$\times$] Not guaranteed to find global MDO optimum
        \item[$\times$] Poor handling of tightly-coupled disciplines
        \item[$\times$] No guarantee of interdisciplinary compatibility
      \end{itemize}
    \end{block}

    \medskip
    \begin{alertblock}{Ride-Sharing (Example 24.5)}
      Vehicle and sensor systems are \emph{tightly coupled}: changes by vehicle sub-problem are addressed by sensor sub-problem, but the reverse must wait for the next outer iteration. Local variables such as drive train are in $\mathbf{v}_\ell$; shared vehicle capacity is in $\mathbf{v}_g$.
    \end{alertblock}

    \column{0.47\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_dependency_graph.pdf}\\
      \small Sequential architecture applied to ride-sharing:\\
      Vehicle $\to$ Sensor $\to$ Autonomy $\to$ Routing $\to$ Demand $\to$ Profit
    \end{center}
    \begin{block}{Convergence Condition}
      Outer loop converges when global variables $\mathbf{x}_g$ and objective $f$ change by less than $\varepsilon$ between iterations.
    \end{block}
  \end{columns}
\end{frame}

\section{Individual Discipline Feasible (IDF)}
% ============================================================================

\begin{frame}{Individual Discipline Feasible (IDF) Architecture}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \begin{block}{Key Idea}
      Replace the MDA inner loop with \textbf{coupling variables} $\mathbf{c}^{(i)}$ that are \emph{optimizer-controlled copies} of the response $\mathbf{y}^{(i)}$.
    \end{block}

    \begin{block}{IDF Formulation}
      \[
        \min_{\mathbf{x},\,\mathbf{c}}\; f(\mathbf{v},\mathbf{s},\mathbf{r},\mathbf{p},\mathbf{c})
      \]
      \[
        \text{s.t.}\quad
        \bigl[\mathbf{x}, \mathbf{c}\bigr] \in \mathcal{X}
      \]
      \[
        \mathbf{c}^{(i)} = \mathbf{y}^{(i)}(\mathbf{x}, \mathbf{c}) \quad \forall\, i
      \]
      Coupling equality constraints enforce compatibility.
    \end{block}

    \begin{block}{Properties}
      \begin{itemize}
        \item[\checkmark] Disciplines run \emph{in parallel} (no MDA)
        \item[\checkmark] No iterative convergence check needed
        \item[$\times$] Optimizer dimension: $n + \sum_i n_i$ (large)
        \item[$\times$] Requires handling equality constraints
      \end{itemize}
    \end{block}

    \column{0.47\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_idf_architecture.pdf}\\
      \small IDF applied to ride-sharing: all disciplines receive same $\mathcal{A}$; run in parallel; coupling enforced by optimizer
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Python: IDF Formulation}
  \begin{lstlisting}[style=python]
import numpy as np
from scipy.optimize import minimize

# IDF: coupling variables c = [y1_c, y2_c, y3_c] are decision variables
# x_idf = [x, y1_c, y2_c, y3_c]

def idf_objective(x_idf):
    x, y1c, y2c, y3c = x_idf
    # Objective depends on coupling copies
    return y1c**2 + y2c**2 + y3c**2

def idf_constraints(x_idf):
    x, y1c, y2c, y3c = x_idf
    # Disciplinary residuals: y^(i) - c^(i) = 0
    # Each discipline uses current assignment with coupling copies
    A = {'x': x, 'y1': y1c, 'y2': y2c, 'y3': y3c}
    r1 = (A['y2'] - A['x']) - y1c   # F1 residual
    r2 = np.sin(A['y1'] + A['y3']) - y2c  # F2 residual
    r3 = np.cos(A['x'] + A['y2'] + A['y1']) - y3c  # F3 residual
    return [r1, r2, r3]

constraints = [{'type': 'eq', 'fun': idf_constraints}]
x0 = [1.0, 1.0, 1.0, 1.0]
result = minimize(idf_objective, x0,
                  constraints=constraints,
                  method='SLSQP',
                  options={'ftol': 1e-9, 'maxiter': 1000})
print(f"x*={result.x[0]:.4f}, y1*={result.x[1]:.4f}, "
      f"y2*={result.x[2]:.4f}, y3*={result.x[3]:.4f}")
  \end{lstlisting}
\end{frame}

\section{Collaborative Optimization (CO)}
% ============================================================================

\begin{frame}{Collaborative Optimization (CO)}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \begin{block}{Idea}
      A \textbf{bi-level} architecture:
      \begin{itemize}
        \item \emph{System-level} optimizer controls global targets $\mathcal{A}_g$
        \item Each discipline has its own \emph{sub-optimizer} that minimizes a coupling objective $J^{(i)}$
      \end{itemize}
    \end{block}

    \begin{block}{System-Level Problem}
      \[
        \min_{\mathcal{A}_g}\; f(\mathcal{A}_g) \quad \text{s.t.}\quad J^{(i)}(\mathcal{A}_g) \leq \delta,\;\forall\, i
      \]
    \end{block}

    \begin{block}{Disciplinary Sub-Problem $i$}
      \[
        \min_{\mathbf{x}^{(i)},\,\mathbf{y}^{(i)}}\; J^{(i)}(\mathcal{A}_g)
        \quad \text{s.t.}\quad [\mathbf{x}^{(i)},\mathbf{y}^{(i)}] \in \mathcal{X}_i
      \]
      \[
        J^{(i)} = \sum_{v \in \mathbf{x}_g} \!\bigl(x_v^g - \mathcal{A}_v^{(i)}\bigr)^2
                + \sum_{v \in \mathbf{y}_g} \!\bigl(y_v^g - \mathcal{A}_v^{(i)}\bigr)^2
      \]
      Minimizing $J^{(i)}$ drives local values to match global targets.
    \end{block}

    \column{0.47\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_co_architecture.pdf}\\
      \small CO: system-level optimizer specifies shared targets $\mathcal{A}_g$; sub-optimizers minimize coupling mismatch
    \end{center}

    \begin{center}
      \includegraphics[width=0.75\textwidth]{fig_co_coupling.pdf}\\
      \small Coupling objective $J^{(i)}$ vs.\ global variable $\theta_g$
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Python: Collaborative Optimization Sub-Problem}
  \begin{lstlisting}[style=python]
import numpy as np
from scipy.optimize import minimize

def make_co_subproblem(i, A_g, discipline_fn, local_init):
    """
    Build and solve the CO sub-problem for discipline i.
    A_g   : system-level target assignment dict
    discipline_fn : function A -> A (updates response y^(i))
    local_init    : initial local variables [x_l, y_c]
    """
    x_g = A_g['x']  # shared global variable target

    def J_sub(local_vars):
        A = dict(A_g)           # start from global targets
        A.update({'y%d' % (i+1): local_vars[0]})  # local response
        discipline_fn(A)
        y_local = A['y%d' % (i+1)]
        # Coupling objective: drive local to match global targets
        return (local_vars[0] - A_g.get('y%d_g' % (i+1), 0.0))**2

    result = minimize(J_sub, local_init, method='L-BFGS-B')
    return result.x, result.fun

# System-level: vary A_g to drive J^(i) -> 0
def system_objective(x_g_val, threshold=1e-4):
    A_g = {'x': x_g_val[0], 'y1_g': 0.0, 'y2_g': 0.0, 'y3_g': 0.0}
    # Each discipline sub-optimizes in parallel
    _, J1 = make_co_subproblem(0, A_g, F1, [0.0])
    _, J2 = make_co_subproblem(1, A_g, F2, [0.0])
    _, J3 = make_co_subproblem(2, A_g, F3, [0.0])
    return J1 + J2 + J3          # sum of coupling objectives

result = minimize(system_objective, x0=[1.0],
                  bounds=[(-5.0, 5.0)], method='L-BFGS-B')
print(f"x_g* = {result.x[0]:.4f}")
  \end{lstlisting}

  \begin{block}{CO Properties}
    \begin{itemize}
      \item[\checkmark] Each discipline is fully \textbf{autonomous}: uses own tools and constraints
      \item[\checkmark] Subproblems run \textbf{in parallel}
      \item[$\times$] Bi-level structure introduces numerical difficulties (non-smooth $J^{(i)}$)
      \item[$\times$] Convergence not always guaranteed
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{CO Applied to Ride-Sharing (Example 24.7)}
  \begin{columns}[T]
    \column{0.52\textwidth}
    The CO architecture applied to the ride-sharing problem:
    \begin{itemize}
      \item System targets: $\mathcal{A}_g = \{\mathbf{v}_g, \mathbf{s}_g, \mathbf{r}_g, \mathbf{p}_g, \theta_g, M_g, \ldots\}$
      \item Vehicle sub-optimizer minimizes $J_v$, sensor sub-optimizer minimizes $J_s$, etc.
      \item System optimizer minimizes $-\text{profit}$ subject to $J^{(i)} \leq \delta$
    \end{itemize}

    \medskip
    \begin{block}{Spring-Pendulum CO (Exercise 24.6)}
      Two sub-problems:
      \[
        \min_{k,M} J_{\mathrm{disp}} = (k_g - k)^2 + (M_g - M)^2 + (\theta_{\mathrm{disp}}(k_g,M_g)-\theta)^2
      \]
      \[
        \min_{\theta_g} J_{\mathrm{loads}} = (\theta_g - \theta)^2 + (F_{\mathrm{loads}}(\theta_g) - M)^2
      \]
      System: $\min_{k_g,\theta_g,M_g} k_g$ \; s.t.\ $J_{\mathrm{disp}}=0, J_{\mathrm{loads}}=0$
    \end{block}

    \column{0.45\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_co_architecture.pdf}
    \end{center}
    \begin{alertblock}{Primary Advantage}
      Discipline teams can independently apply domain-specific methods (e.g.\ topology optimization for vehicle, game-theoretic pricing for demand).
    \end{alertblock}
  \end{columns}
\end{frame}

\section{Simultaneous Analysis and Design (SAND)}
% ============================================================================

\begin{frame}{Simultaneous Analysis and Design (SAND)}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \begin{block}{Idea}
      Treat \emph{both} design variables $\mathbf{x}$ and response variables $\mathbf{y}$ as \textbf{free optimization variables}. Disciplinary equations are enforced as equality constraints (residuals).
    \end{block}

    \begin{block}{SAND Formulation}
      \[
        \min_{\mathbf{x},\,\mathbf{y}}\; f(\mathbf{x}, \mathbf{y})
      \]
      \[
        \text{s.t.}\quad R_i(\mathbf{x}, \mathbf{y}) = \mathbf{0},\quad i = 1,\ldots,m
      \]
      \[
        [\mathbf{x}, \mathbf{y}] \in \mathcal{X}
      \]
      where $R_i(\mathbf{x},\mathbf{y}) \equiv \mathcal{F}_i(\mathbf{x},\mathbf{y}) - \mathbf{y}^{(i)}$ is the disciplinary residual.
    \end{block}

    \begin{block}{Properties}
      \begin{itemize}
        \item[\checkmark] Single monolithic optimizer — no inner MDA loop
        \item[\checkmark] Analytic gradients propagated once per iteration
        \item[\checkmark] Disciplinary analyses run \emph{once} per iteration
        \item[$\times$] Optimizer dimension: $n + \sum_i n_i$ (very large)
        \item[$\times$] Requires analytic Jacobians $\partial R_i / \partial \mathbf{x}$, $\partial R_i / \partial \mathbf{y}$
        \item[$\times$] Iterates may be infeasible during optimization
      \end{itemize}
    \end{block}

    \column{0.45\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_sand_architecture.pdf}
    \end{center}
    \begin{alertblock}{Key Distinction from MDF}
      MDF satisfies residuals at \emph{every function evaluation}.
      SAND satisfies residuals only at \emph{convergence}.
      Intermediate iterates are infeasible.
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Python: SAND Formulation}
  \begin{lstlisting}[style=python]
import numpy as np
from scipy.optimize import minimize

# SAND: x_sand = [x, y1, y2, y3]
# Residuals R_i must be zero at solution

def sand_objective(z):
    x, y1, y2, y3 = z
    return y1**2 + y2**2 + y3**2   # minimize sum of squares

def sand_residuals(z):
    x, y1, y2, y3 = z
    # R_i = F_i(A) - y_i  (disciplinary residuals)
    R1 = (y2 - x) - y1                            # F1 residual
    R2 = np.sin(y1 + y3) - y2                     # F2 residual
    R3 = np.cos(x + y2 + y1) - y3                 # F3 residual
    return [R1, R2, R3]

def sand_jacobian_residuals(z):
    """Analytic Jacobian of residuals wrt z = [x, y1, y2, y3]."""
    x, y1, y2, y3 = z
    # dR1/dz = [-1, -1, 1, 0]
    # dR2/dz = [0, cos(y1+y3)-1, -1, cos(y1+y3)]
    # dR3/dz = [-sin(x+y2+y1), -sin(x+y2+y1), -sin(x+y2+y1)-1, -1]
    s2 = np.cos(y1 + y3)
    s3 = -np.sin(x + y2 + y1)
    return np.array([[-1, -1, 1, 0],
                     [0, s2, -1, s2],
                     [s3, s3, s3 - 1, -1]])

constraints = [{'type': 'eq', 'fun': sand_residuals,
                'jac': sand_jacobian_residuals}]
z0 = [1.0, 0.5, 0.5, 0.5]
result = minimize(sand_objective, z0,
                  constraints=constraints,
                  method='SLSQP',
                  options={'ftol': 1e-10, 'maxiter': 2000})
print(f"SAND: x*={result.x[0]:.4f}, y1*={result.x[1]:.4f}, "
      f"y2*={result.x[2]:.4f}, y3*={result.x[3]:.4f}")
  \end{lstlisting}
\end{frame}

\begin{frame}{SAND Applied to Ride-Sharing}
  \begin{columns}[T]
    \column{0.55\textwidth}
    The SAND formulation for the ride-sharing example:
    \[
      \min_{\mathbf{v},\mathbf{s},\mathbf{r},\mathbf{p},\mathbf{y}^{(v)},\ldots,\mathbf{y}^{(p)}}\;
      f(\mathbf{v},\mathbf{s},\mathbf{r},\mathbf{p},\mathbf{y})
    \]
    \[
      \text{s.t.}\quad
      R_{\mathrm{vehicle}}(\mathbf{v},\mathbf{y}) = \mathbf{0}
    \]
    \[
      R_{\mathrm{sensor}}(\mathbf{s},\mathbf{y}) = \mathbf{0}
    \]
    \[
      \vdots
    \]
    \[
      R_{\mathrm{profit}}(\mathbf{p},\mathbf{y}) = \mathbf{0}
    \]

    All design and response variables optimized \emph{simultaneously}.

    \begin{block}{Practical Use}
      SAND is most effective when used with gradient-based solvers (e.g., interior-point, SQP) and adjoint-based sensitivities, e.g., in aerodynamic shape optimization.
    \end{block}

    \column{0.42\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_sand_architecture.pdf}
    \end{center}

    \begin{block}{Residual Form}
      For discipline $i$:
      \[
        R_i(\mathbf{x},\mathbf{y}) \equiv \mathbf{y}^{(i)} - \mathcal{F}_i(\mathbf{x},\mathbf{y}) = \mathbf{0}
      \]
      Total constraints: $\sum_{i=1}^m n_i$ equality constraints.
    \end{block}
  \end{columns}
\end{frame}

\section{Numerical Example: Spring-Pendulum}
% ============================================================================

\begin{frame}{Spring-Pendulum Numerical Example (Exercises 24.4--24.6)}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Problem Setup:}\\
    Minimize spring stiffness $k$ such that displacement $\theta \leq \theta_{\max}$.

    \medskip
    \textbf{Two disciplinary analyses:}
    \begin{itemize}
      \item \textbf{Loads:} $M = mg\ell\cos(\theta)$ (moment from gravity)
      \item \textbf{Displacement:} $\theta = M/k$ (spring restoring)
    \end{itemize}

    \medskip
    Parameters: $m=1\,\mathrm{kg}$, $\ell=1\,\mathrm{m}$, $g=9.81\,\mathrm{m/s}^2$, $\theta_{\max}=10^\circ$

    \medskip
    \begin{block}{MDF Solution}
      \[
        \min_{k>0} f(\mathrm{MDA}(k)) \quad\text{s.t.}\quad \theta \leq \theta_{\max}
      \]
      \textbf{Result:} $k^* \approx 55.353\,\mathrm{N/m}$
    \end{block}

    \begin{block}{IDF Solution (Exercise 24.5)}
      \[
        \min_{k,\theta_c,M_c} k \quad \text{s.t.}\quad k>0,\;
        \theta_c = F_{\mathrm{disp}}(k,M_c),\;
        M_c = F_{\mathrm{loads}}(\theta_c),\;
        \theta_c \leq \theta_{\max}
      \]
      Same $k^*$; disciplines run in parallel.
    \end{block}

    \column{0.43\textwidth}
    \begin{center}
      \includegraphics[width=\textwidth]{fig_spring_pendulum.pdf}
    \end{center}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Python: Spring-Pendulum MDF Solution}
  \begin{lstlisting}[style=python]
import numpy as np
from scipy.optimize import minimize_scalar, minimize

m, ell, g = 1.0, 1.0, 9.81
theta_max = np.deg2rad(10.0)    # 10 degrees in radians

def mda_pendulum(k, tol=1e-10, max_iter=100):
    """Gauss-Seidel MDA for spring-pendulum system."""
    theta = theta_max / 2        # initial guess
    for _ in range(max_iter):
        M     = m * g * ell * np.cos(theta)  # loads analysis
        theta_new = M / k                     # displacement analysis
        if abs(theta_new - theta) < tol:
            return theta_new, M, True
        theta = theta_new
    return theta, M, False

def mdf_objective(k):
    theta, M, converged = mda_pendulum(k)
    penalty = max(0, theta - theta_max)**2 * 1e6
    return k + penalty            # minimize k subject to theta <= theta_max

# Bounded scalar optimization
result = minimize_scalar(mdf_objective, bounds=(1.0, 1000.0),
                         method='bounded', options={'xatol': 1e-8})
k_star = result.x
theta_star, M_star, _ = mda_pendulum(k_star)
print(f"k*       = {k_star:.3f} N/m")
print(f"theta*   = {np.rad2deg(theta_star):.4f} deg  (limit {np.rad2deg(theta_max):.1f} deg)")
print(f"M*       = {M_star:.4f} N·m")
  \end{lstlisting}

  \textbf{Output:} \texttt{k* = 55.353 N/m,  theta* = 10.0000 deg}
\end{frame}

\section{Summary \& Exercises}
% ============================================================================

\begin{frame}[allowframebreaks]{Summary}
  \begin{block}{Chapter 24 Key Takeaways}
    \begin{enumerate}
      \item \textbf{MDO} addresses optimization problems involving multiple interacting disciplines, where solving each discipline in isolation leads to suboptimal system-level solutions.

      \item \textbf{Disciplinary analysis} $\mathcal{F}_i$ takes a shared assignment $\mathcal{A}$ and updates the response variables $\mathbf{y}^{(i)}$. Running analyses in cycles gives the \textbf{Gauss-Seidel MDA}; ordering affects convergence.

      \item \textbf{Interdisciplinary compatibility}: assignment $\mathcal{A}$ is feasible iff all disciplinary residuals $R_i = 0$. Dependency graphs (with cycles $\Rightarrow$ MDA required).

      \item Five main architectures:
        \begin{itemize}
          \item \textbf{MDF}: black-box MDA at every function evaluation; simplest
          \item \textbf{Sequential}: hierarchical, discipline-by-discipline; modular
          \item \textbf{IDF}: coupling variables as optimizer decisions; parallel disciplines
          \item \textbf{CO}: bi-level; each discipline has own sub-optimizer; maximum autonomy
          \item \textbf{SAND}: all variables free; residuals as constraints; one MDA at convergence
        \end{itemize}

      \item Trade-offs:
        \begin{itemize}
          \item MDF: low variable count, expensive (MDA per eval)
          \item IDF/SAND: large variable count, disciplinary parallelism
          \item CO: disciplinary autonomy, but complex bi-level structure
        \end{itemize}
    \end{enumerate}
  \end{block}

  \framebreak

  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Architecture Decision Guide}
      \begin{itemize}
        \item \textbf{Loosely coupled} disciplines + few shared variables $\Rightarrow$ Sequential
        \item \textbf{Tightly coupled} + need for feasible iterates $\Rightarrow$ MDF
        \item \textbf{Expensive MDA} + analytic gradients available $\Rightarrow$ SAND
        \item \textbf{Organizational boundaries} + parallel teams $\Rightarrow$ CO
        \item \textbf{Need parallel disciplines} + medium problem size $\Rightarrow$ IDF
      \end{itemize}
    \end{block}

    \column{0.48\textwidth}
